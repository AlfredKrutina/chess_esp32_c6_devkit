50static int cheb_dist_sq(uint8_t a, uint8_t b) {
51 int ar = (int)(a / 8);
52 int ac = (int)(a % 8);
53 int br = (int)(b / 8);
54 int bc = (int)(b % 8);
55 int dr = ar - br;
56 int dc = ac - bc;
57 if (dr < 0) {
58 dr = -dr;
59 }
60 if (dc < 0) {
61 dc = -dc;
62 }
63 return (dr > dc) ? dr : dc;
64}

66static int build_manhattan_path(uint8_t from, uint8_t to, uint8_t *out, int cap,
67 bool columns_first) {
68 if (cap < 1 || from >= 64 || to >= 64) {
69 return 0;
70 }
71 int len = 0;
72 out[len++] = from;
73 int r = (int)(from / 8);
74 int c = (int)(from % 8);
75 int tr = (int)(to / 8);
76 int tc = (int)(to % 8);
77 if (columns_first) {
78 while (c != tc && len < cap) {
79 c += (tc > c) ? 1 : -1;
80 out[len++] = (uint8_t)(r * 8 + c);
81 }
82 while (r != tr && len < cap) {
83 r += (tr > r) ? 1 : -1;
84 out[len++] = (uint8_t)(r * 8 + c);
85 }
86 } else {
87 while (r != tr && len < cap) {
88 r += (tr > r) ? 1 : -1;
89 out[len++] = (uint8_t)(r * 8 + c);
90 }
91 while (c != tc && len < cap) {
92 c += (tc > c) ? 1 : -1;
93 out[len++] = (uint8_t)(r * 8 + c);
94 }
95 }
96 return len;
97}

99static int build_knight_path(uint8_t from, uint8_t to, uint8_t *out, int cap) {
100 if (cap < 2 || from >= 64 || to >= 64 || from == to) {
101 return 0;
102 }
103 int r = (int)(from / 8);
104 int c = (int)(from % 8);
105 int tr = (int)(to / 8);
106 int tc = (int)(to % 8);
107 int dr = tr - r;
108 int dc = tc - c;
109 int adr = (dr < 0) ? -dr : dr;
110 int adc = (dc < 0) ? -dc : dc;
111 out[0] = from;
112 if (((adr == 2 && adc == 1) || (adr == 1 && adc == 2)) && cap >= 3) {
113 int sr = (dr > 0) ? 1 : ((dr < 0) ? -1 : 0);
114 int sc = (dc > 0) ? 1 : ((dc < 0) ? -1 : 0);
115 if (adr == 2) {
116 out[1] = (uint8_t)((r + 2 * sr) * 8 + c);
117 } else {
118 out[1] = (uint8_t)(r * 8 + (c + 2 * sc));
119 }
120 out[2] = to;
121 return 3;
122 }
123 out[1] = to;
124 return 2;
125}

273 /* Repeated shortest pair. Identical pieces have no left/right identity,
274 * and a full search of eight pawns can stall the game task. */
275 bool expect_used[MAX_PIECES] = {false};
276 bool phys_used[MAX_PIECES] = {false};
277 int pair_n = (e_count < p_count) ? e_count : p_count;
278 for (int n = 0; n < pair_n; n++) {
279 int best_e = -1;
280 int best_p = -1;
281 int best_d = 1 << 30;
282 for (int ei = 0; ei < e_count; ei++) {
283 if (expect_used[ei]) {
284 continue;
285 }
286 for (int pi = 0; pi < p_count; pi++) {
287 if (phys_used[pi]) {
288 continue;
289 }
290 int d = manhattan_steps(phys[pi].phys_sq, expect[ei].home_sq);
291 if (d < best_d) {
292 best_d = d;
293 best_e = ei;
294 best_p = pi;
295 }
296 }
297 }
298 if (best_e < 0 || best_p < 0) {
299 break;
300 }
301 expect_used[best_e] = true;
302 phys_used[best_p] = true;
303 uint8_t from = phys[best_p].phys_sq;
304 uint8_t home = expect[best_e].home_sq;
305 if (from == home || *job_n >= MAX_JOBS) {
306 continue;
307 }
308 jobs[*job_n].from_sq = from;
309 jobs[*job_n].home_sq = home;
310 jobs[*job_n].type = expect[best_e].type;
311 if (expect[best_e].type == HALL_PIECE_TYPE_KNIGHT) {
312 jobs[*job_n].path_len = (uint8_t)build_knight_path(
313 from, home, jobs[*job_n].path, LED_RESTORE_GUIDE_MAX_PATH_LEN);
314 } else {
315 jobs[*job_n].path_len = (uint8_t)build_manhattan_path(
316 from, home, jobs[*job_n].path, LED_RESTORE_GUIDE_MAX_PATH_LEN, false);
317 }
318 jobs[*job_n].swap_corridor = false;
319 (*job_n)++;
320 }
321
322 for (int ei = 0; ei < e_count; ei++) {
323 if (!expect_used[ei] && *missing_n < LED_RESTORE_GUIDE_MAX_MISSING) {
324 missing[(*missing_n)++] = expect[ei].home_sq;
325 }
326 }
327 for (int pi = 0; pi < p_count; pi++) {
328 if (!phys_used[pi] && *extra_n < LED_RESTORE_GUIDE_MAX_EXTRA) {
329 extra[(*extra_n)++] = phys[pi].phys_sq;
330 }
331 }
332}

359 p_idx[pc++] = i;

363 hall_piece_slot_t p_slice[MAX_PIECES];

427static void merge_swap_pairs(restore_job_t *jobs, int *job_n) {
428 for (int i = 0; i < *job_n; i++) {
429 if (jobs[i].swap_corridor) {
430 continue;
431 }
432 for (int j = i + 1; j < *job_n; j++) {
433 if (jobs[j].swap_corridor) {
434 continue;
435 }
436 if (jobs[i].from_sq == jobs[j].home_sq &&
437 jobs[j].from_sq == jobs[i].home_sq) {
438 uint8_t end_a = jobs[i].from_sq;
439 uint8_t end_b = jobs[j].from_sq;
440 bool knight = jobs[i].type == HALL_PIECE_TYPE_KNIGHT ||
441 jobs[j].type == HALL_PIECE_TYPE_KNIGHT;
442 jobs[i].home_sq = end_b;
443 jobs[i].type = knight ? HALL_PIECE_TYPE_KNIGHT : jobs[i].type;
444 if (knight) {
445 jobs[i].path_len = (uint8_t)build_knight_path(
446 end_a, end_b, jobs[i].path, LED_RESTORE_GUIDE_MAX_PATH_LEN);
447 jobs[i].swap_corridor = false;
448 } else {
449 jobs[i].path_len = (uint8_t)build_manhattan_path(
450 end_a, end_b, jobs[i].path, LED_RESTORE_GUIDE_MAX_PATH_LEN, false);
451 jobs[i].swap_corridor = true;
452 }
453 if (j < *job_n - 1) {
454 jobs[j] = jobs[*job_n - 1];
455 }
456 (*job_n)--;
457 j--;
458 }
459 }
460 }

477static void prefer_path_clear_of_missing(restore_job_t *job,
478 const uint8_t *missing,
479 uint8_t missing_n) {
480 if (job->path_len == 0 || job->type == HALL_PIECE_TYPE_KNIGHT) {
481 if (job->type == HALL_PIECE_TYPE_KNIGHT) {
482 job->path_len = (uint8_t)build_knight_path(
483 job->from_sq, job->home_sq, job->path, LED_RESTORE_GUIDE_MAX_PATH_LEN);
484 }
485 return;
486 }
487 uint8_t from = job->path[0];
488 uint8_t to = job->path[job->path_len - 1];
489 if (from >= 64 || to >= 64) {
490 job->path_len = 0;
491 return;
492 }
493 uint8_t row_first[LED_RESTORE_GUIDE_MAX_PATH_LEN];
494 uint8_t col_first[LED_RESTORE_GUIDE_MAX_PATH_LEN];
495 int row_n = build_manhattan_path(from, to, row_first,
496 LED_RESTORE_GUIDE_MAX_PATH_LEN, false);
497 int col_n = build_manhattan_path(from, to, col_first,
498 LED_RESTORE_GUIDE_MAX_PATH_LEN, true);
499 restore_job_t probe = *job;
500 if (row_n > 0) {
501 probe.path_len = (uint8_t)row_n;
502 memcpy(probe.path, row_first, (size_t)row_n);
503 if (!path_crosses_missing_home(&probe, missing, missing_n)) {
504 job->path_len = probe.path_len;
505 memcpy(job->path, probe.path, (size_t)row_n);
506 return;
507 }
508 }
509 if (col_n > 0) {
510 probe.path_len = (uint8_t)col_n;
511 memcpy(probe.path, col_first, (size_t)col_n);
512 if (!path_crosses_missing_home(&probe, missing, missing_n)) {
513 job->path_len = probe.path_len;
514 memcpy(job->path, probe.path, (size_t)col_n);
515 return;
516 }
517 }
518 job->path_len = 0;
519}

521static void pick_display_paths(restore_job_t *jobs, int job_n,
522 const uint8_t *missing, uint8_t missing_n,
523 led_restore_guide_plan_t *plan) {
524 int candidates[MAX_JOBS];
525 int cand_n = 0;
526 for (int i = 0; i < job_n; i++) {
527 prefer_path_clear_of_missing(&jobs[i], missing, missing_n);
528 if (jobs[i].path_len > 0 &&
529 (jobs[i].type == HALL_PIECE_TYPE_KNIGHT ||
530 !path_crosses_missing_home(&jobs[i], missing, missing_n))) {
531 candidates[cand_n++] = i;
532 }
533 }
534
535 int best_jobs[LED_RESTORE_GUIDE_MAX_PATHS];
536 int best_count = 0;
537 int best_score = -1;
538
539 for (int i0 = 0; i0 < cand_n; i0++) {
540 int j0 = candidates[i0];
541 if (!set_paths_conflict(&j0, 1, jobs)) {
542 int sc = score_path_set(&j0, 1, jobs);
543 if (sc > best_score || (sc == best_score && 1 > best_count)) {
544 best_score = sc;
545 best_count = 1;
546 best_jobs[0] = j0;
547 }
548 }
549 for (int i1 = i0 + 1; i1 < cand_n; i1++) {
550 int pair[2] = {candidates[i0], candidates[i1]};
551 if (!set_paths_conflict(pair, 2, jobs)) {
552 int sc = score_path_set(pair, 2, jobs);
553 if (sc > best_score || (sc == best_score && 2 > best_count)) {
554 best_score = sc;
555 best_count = 2;
556 best_jobs[0] = pair[0];
557 best_jobs[1] = pair[1];
558 }
559 }
560 for (int i2 = i1 + 1; i2 < cand_n; i2++) {
561 int triple[3] = {candidates[i0], candidates[i1], candidates[i2]};
562 if (!set_paths_conflict(triple, 3, jobs)) {
563 int sc = score_path_set(triple, 3, jobs);
564 if (sc > best_score || (sc == best_score && 3 > best_count)) {
565 best_score = sc;
566 best_count = 3;
567 best_jobs[0] = triple[0];
568 best_jobs[1] = triple[1];
569 best_jobs[2] = triple[2];
570 }
571 }
572 }
573 }
574 }
575
576 plan->path_count = (uint8_t)best_count;
577 for (int i = 0; i < best_count; i++) {
578 const restore_job_t *job = &jobs[best_jobs[i]];
579 plan->paths[i].len = job->path_len;
580 plan->paths[i].pulse_corridor = job->swap_corridor;
581 memcpy(plan->paths[i].squares, job->path, job->path_len);
582 }
583}

681void game_hall_restore_guide_sync(void) {
682#if HALL_DEBUG_LIGHT_WHERE_PIECE
683 game_hall_restore_guide_stop_leds();
684 return;
685#endif
686 if (!hall_i2c_squares_readings_usable()) {
687 return;
688 }
689 uint8_t occ[64];
690 uint8_t digest[64];
691 hall_piece_slot_t expect[MAX_PIECES];
692 hall_piece_slot_t phys[MAX_PIECES];
693 restore_job_t jobs[MAX_JOBS];
694 uint8_t missing[LED_RESTORE_GUIDE_MAX_MISSING];
695 uint8_t extra[LED_RESTORE_GUIDE_MAX_EXTRA];
696 led_restore_guide_plan_t plan;
697 int expect_n = 0;
698 int phys_n = 0;
699 int job_n = 0;
700 uint8_t missing_n = 0;
701 uint8_t extra_n = 0;
702 bool has_led_work = false;
703
704 if (s_in_sync) {
705 return;
706 }
707 s_in_sync = true;
708
709 if (!chess_policy_squares_guard_enabled() || !chess_policy_squares_guard_led_enabled()) {
710 game_hall_restore_guide_stop_leds();
711 goto out;
712 }
713 if (mode_blocks_guide()) {
714 game_hall_restore_guide_stop_leds();
715 goto out;
716 }
717
718 squares_get_state(occ);
719 digest_sensors(occ, digest);
720 if (s_digest_valid && sensors_unchanged(digest)) {
721 goto out;
722 }
723
724 /* Empty board stays dark. Missing homes are capped at 16, which is both pawn ranks. */
725 if (occupancy_empty(occ)) {
726 game_hall_restore_guide_stop_leds();
727 led_clear_board_only();
728 remember_digest(digest);
729 goto out;
730 }
731
732 if (!collect_board(expect, &expect_n, phys, &phys_n, occ)) {
733 goto out;
734 }
735
736 match_all(expect, expect_n, phys, phys_n, jobs, &job_n, missing, &missing_n,
737 extra, &extra_n);
738
739 if (board_fully_aligned(job_n, missing_n, extra_n)) {
740 game_hall_restore_guide_stop_leds();
741 remember_digest(digest);
742 goto out;
743 }
744
745 if (is_single_lift(job_n, missing_n, extra_n)) {
746 game_hall_restore_guide_stop_leds();
747 remember_digest(digest);
748 goto out;
749 }
750
751 merge_swap_pairs(jobs, &job_n);
752
753 memset(&plan, 0, sizeof(plan));
754 pick_display_paths(jobs, job_n, missing, missing_n, &plan);
755 plan.missing_count = missing_n;
756 memcpy(plan.missing_homes, missing, missing_n);
757 plan.extra_count = extra_n;
758 memcpy(plan.extra_squares, extra, extra_n);
759 fill_setup_marks(jobs, job_n, missing, missing_n, extra, extra_n, &plan);
760
761 has_led_work = plan.mark_count > 0;
762 if (!has_led_work) {
763 game_hall_restore_guide_stop_leds();
764 remember_digest(digest);
765 goto out;
766 }
767
768 plan.active = true;
769 ensure_squares_pause();
770 led_board_restore_guide_apply(&plan);
771 s_guide_led_active = true;
772 remember_digest(digest);
773
774 ESP_LOGI(TAG,
775 "[STAGING] restore guide: paths=%u missing=%u extra=%u jobs=%d",
776 (unsigned)plan.path_count, (unsigned)missing_n, (unsigned)extra_n,
777 job_n);
778out:
779 s_in_sync = false;
780}
