252static void squares_scan_row_internal(uint8_t row) {
253 if (row >= 8)
254 return;
255
256#if CONFIG_CHESS_MATRIX_INPUT_I2C_HALL
257 (void)row;
258 return;
259#else
260
261 // IMPROVED: Switch all lines to INPUT before activating the current line
262 // This is safer than just setting it to HIGH because it prevents conflicts
263 // when switching between lines (as in the reference code)
264 for (int i = 0; i < 8; i++) {
265 gpio_config_t io_conf = {.pin_bit_mask = (1ULL << matrix_row_pins[i]),
266 .mode = GPIO_MODE_INPUT,
267 .pull_up_en = GPIO_PULLUP_DISABLE,
268 .pull_down_en = GPIO_PULLDOWN_DISABLE,
269 .intr_type = GPIO_INTR_DISABLE};
270 gpio_config(&io_conf);
271 }
272
273 // IMPROVED: Use esp_rom_delay_us instead of vTaskDelay (timer callback
274 // context)
275 extern void esp_rom_delay_us(uint32_t us);
276 esp_rom_delay_us(20); // 20us settling time (reduced from 1ms)
277
278 // Set current line to OUTPUT and LOW (activation)
279 // This pulls the column pin LOW if there is a piece (reed
280 // switch closed)
281 gpio_config_t io_conf = {.pin_bit_mask = (1ULL << matrix_row_pins[row]),
282 .mode = GPIO_MODE_OUTPUT,
283 .pull_up_en = GPIO_PULLUP_DISABLE,
284 .pull_down_en = GPIO_PULLDOWN_DISABLE,
285 .intr_type = GPIO_INTR_DISABLE};
286 gpio_config(&io_conf);
287 gpio_set_level(matrix_row_pins[row], 0); // LOW = active
288
289 // GPIO settling time: 200us for signal stabilization (long wires support)
290 // For digital signals with long cables (capacitance), we need more time
291 // for the signal to fall to logic LOW (grounding).
292 esp_rom_delay_us(200);
293
294 // Read all column pins for this row
295 // Column pins have a pull-up, so they will be HIGH without the figure
296 // If the figure is (reed switch closed), the column pin will be LOW
297
298 // Read all column pins for this row
299 // Column pins have a pull-up, so they will be HIGH without the figure
300 // If the figure is (reed switch closed), the column pin will be LOW
301
302 for (int col = 0; col < 8; col++) {
303 // Invert column index to fix GPIO physical wiring (H→A instead of
304 // A→H) GPIO pins are wired in reverse: pin 0 = column H, pin 7 = column A
305 // So we invert: col 0 → index 7 (H), col 1 → index 6 (G), ..., col 7 →
306 // index 0 (A)
307 int inverted_col = 7 - col;
308 int index = row * 8 + inverted_col;
309
310 // DEBUG: Check if this is GPIO17 (column 7)
311 bool is_gpio17 = (matrix_col_pins[col] == GPIO_NUM_17);
312
313 int pin_level = gpio_get_level(matrix_col_pins[col]);
314
315 // In simulation mode, use simulated values
316 if (simulation_mode) {
317 squares_state[index] = simulation_patterns[current_pattern][index];
318 } else {
319 // Real hardware: reed switch closed = piece present
320 // pin_level == 0 means that the column pin is pulled to LOW (the figure is
321 // present)
322 // NOTE: GPIO17 is configured with pull-down for debug testing
323 // Normal logic still applies, but we'll log values for analysis
324 squares_state[index] = (pin_level == 0) ? 1 : 0;
325
326 // DEBUG: Minimal logging for GPIO17 (user testing with multimeter)
327 // Only log first state change to confirm scanning works
328 static int last_gpio17_level = -1;
329 static bool gpio17_logged_once = false;
330 if (is_gpio17 && !gpio17_logged_once) {
331 ESP_LOGI(TAG, "🔍 GPIO17 scan confirmed: row=%d, level=%d (pull-down mode)", row, pin_level);
332 gpio17_logged_once = true;
333 last_gpio17_level = pin_level;
334 } else if (is_gpio17 && last_gpio17_level == -1) {
335 last_gpio17_level = pin_level;
336 }
337 }
338 }
339
340 // IMPROVED: Switch line back to INPUT after scan (safer)
341 io_conf.pin_bit_mask = (1ULL << matrix_row_pins[row]);
342 io_conf.mode = GPIO_MODE_INPUT;
343 gpio_config(&io_conf);
344
345 // A small wait before the next line for stabilization
346 esp_rom_delay_us(20); // 20us delay (reduced from 2ms)
347#endif
348}

350// Public function - scans row WITH mutex protection
351void squares_scan_row(uint8_t row) {
352 if (row >= 8)
353 return;
354
355 // CRITICAL: Protect matrix state with mutex
356 if (squares_mutex != NULL) {
357 if (xSemaphoreTake(squares_mutex, pdMS_TO_TICKS(100)) == pdTRUE) {
358 squares_scan_row_internal(row);
359 xSemaphoreGive(squares_mutex);
360 } else {
361 ESP_LOGW(TAG, "Failed to acquire matrix mutex for scan row");
362 }
363 } else {
364 // Fallback if mutex not available
365 squares_scan_row_internal(row);
366 }

675void squares_detect_moves(void) {
676#if CONFIG_CHESS_MATRIX_INPUT_I2C_HALL
677 /* Per-square wakeups only. The game diffs the snapshot; nothing is tracked. */
678 squares_hall_publish_changes();
679 return;
680#else
681 // Reed: one lifted square, then the drop that completes it.
682 if (last_piece_lifted != 255) {
683 uint32_t current_time = esp_timer_get_time() / 1000;
684 if (current_time > move_detection_timeout) {
685 ESP_LOGW(TAG,
686 "⏰ Move taking longer than 5s - piece from %d still in air",
687 last_piece_lifted);
688 // DON'T reset last_piece_lifted - allow longer moves
689 // Just extend timeout to give player more time
690 move_detection_timeout = current_time + 5000; // Give another 5 seconds
691 }
692 }
693
694 // Collect full delta in this scan to detect ambiguous states.
695 uint8_t piece_lifted = 255;
696 uint8_t piece_placed = 255;
697 uint8_t lift_count = 0;
698 uint8_t drop_count = 0;
699 uint32_t lifted_mask_low = 0;
700 uint32_t lifted_mask_high = 0;
701 uint32_t dropped_mask_low = 0;
702 uint32_t dropped_mask_high = 0;
703 for (int i = 0; i < 64; i++) {
704 if (squares_previous[i] == 1 && squares_state[i] == 0) {
705 lift_count++;
706 if (piece_lifted == 255) {
707 piece_lifted = i;
708 }
709 if (i < 32) {
710 lifted_mask_low |= (1UL << i);
711 } else {
712 lifted_mask_high |= (1UL << (i - 32));
713 }
714 } else if (squares_previous[i] == 0 && squares_state[i] == 1) {
715 drop_count++;
716 if (piece_placed == 255) {
717 piece_placed = i;
718 }
719 if (i < 32) {
720 dropped_mask_low |= (1UL << i);
721 } else {
722 dropped_mask_high |= (1UL << (i - 32));
723 }
724 }
725 }
726
727 // Two UP in a row: normally ambiguous (matrix guard), but guided capture
728 // (victim first, then attacker) and 3-step defense (own → opponent) require it.
729 bool multi_change = (lift_count > 2) || (drop_count > 1) ||
730 (lift_count == 2 && drop_count > 0);
731 bool pending_lift_and_new_up =
732 (last_piece_lifted != 255 && lift_count > 0);
733 bool ambiguous_state = multi_change;
734 if (!ambiguous_state && pending_lift_and_new_up) {
735 if (!(lift_count == 1 && game_squares_allow_second_sequential_lift())) {
736 ambiguous_state = true;
737 }
738 }
739
740 if (ambiguous_state) {
741 if (!squares_guard_mode_active) {
742 if (!game_policy_squares_guard_enabled()) {
743 ESP_LOGW(TAG,
744 "Ambiguous matrix state ignored (matrix guard disabled)");
745 return;
746 }
747 // Snapshot expected board occupancy before anomaly; game_task realigns to
748 // board[] when handling a board-guard command.
749 memcpy(squares_guard_expected_state, squares_previous,
750 sizeof(squares_guard_expected_state));
751 squares_guard_mode_active = true;
752 last_piece_lifted = 255;
753 last_piece_placed = 255;
754 move_detection_timeout = 0;
755 squares_send_guard_command(lifted_mask_low, lifted_mask_high, dropped_mask_low,
756 dropped_mask_high, 1);
757 }
758 return;
759 }
760
761 if (squares_guard_mode_active) {
762 // Keep matrix flow paused until board occupancy is back to expected state.
763 bool back_to_expected =
764 (memcmp(squares_state, squares_guard_expected_state,
765 sizeof(squares_guard_expected_state)) == 0);
766 if (lift_count == 0 && drop_count == 0 && back_to_expected) {
767 squares_guard_mode_active = false;
768 squares_send_guard_command(0, 0, 0, 0, 0);
769 }
770 return;
771 }
772
773 if (lift_count == 2 && drop_count == 0) {
774 for (int i = 0; i < 64; i++) {
775 if (squares_previous[i] == 1 && squares_state[i] == 0) {
776 last_piece_lifted = (uint8_t)i;
777 move_detection_timeout = esp_timer_get_time() / 1000 + 5000;
778 squares_send_pickup_command((uint8_t)i);
779 }
780 }
781 return;
782 }
783
784 // Process move detection
785 if (piece_lifted != 255) {
786 last_piece_lifted = piece_lifted;
787 move_detection_timeout =
788 esp_timer_get_time() / 1000 + 5000; // 5 second timeout
789
790 // Convert square index to row/col
791 uint8_t from_row = piece_lifted / 8;
792 uint8_t from_col = piece_lifted % 8;
793
794 ESP_LOGI(TAG, "Piece lifted from square %d (%c%d)", piece_lifted,
795 'a' + from_col, from_row + 1);
796
797 // Send PICKUP through game_submit (same path as web, BLE, and main)
798 squares_send_pickup_command(piece_lifted);
799 }
800
801 if (piece_placed != 255) {
802 last_piece_placed = piece_placed;
803
804 // Convert square index to row/col
805 uint8_t to_row = piece_placed / 8;
806 uint8_t to_col = piece_placed % 8;
807
808 ESP_LOGI(TAG, "Piece placed on square %d (%c%d)", piece_placed,
809 'a' + to_col, to_row + 1);
810
811 // Check if this completes a move
812 if (last_piece_lifted != 255) {
813 if (last_piece_lifted == piece_placed) {
814 // Return to same square - send with from=to
815 ESP_LOGI(TAG,
816 "Piece returned to same square - sending DROP with from=to");
817 squares_send_drop_command_with_from(last_piece_lifted, piece_placed);
818 } else {
819 // Complete move - send with from/to
820 squares_send_drop_command_with_from(last_piece_lifted, piece_placed);
821 }
822 last_piece_lifted = 255; // Reset
823 } else {
824 ESP_LOGW(TAG, "Piece placed without previous lift (square %d) - ignoring",
825 piece_placed);
826 }
827 }
828#endif
829}

861uint8_t squares_notation_to_square(const char *notation) {
862 if (notation == NULL || strlen(notation) != 2)
863 return 255;
864
865 char file = tolower(notation[0]);
866 char rank = notation[1];
867
868 if (file < 'a' || file > 'h' || rank < '1' || rank > '8') {
869 return 255;
870 }
871
872 int col = file - 'a';
873 int row = rank - '1';
874
875 return row * 8 + col;
876}

984 uint8_t command;
985
986 // Process commands from queue
987 if (squares_command_queue != NULL) {
988 while (xQueueReceive(squares_command_queue, &command, 0) == pdTRUE) {
989 switch (command) {
990 case 0: // Reset matrix
991 squares_reset();
992 break;
993
994 case 1: // Print state
995 squares_print_state();
996 break;
997
998 case 2: // SQUARES_CMD_TEST - Test matrix scanning
999 ESP_LOGI(TAG, "=== Matrix Test Started ===");
1000 squares_test_scanning();
1001 ESP_LOGI(TAG, "=== Matrix Test Complete ===");
1002 break;
1003
1004 case 3: // Change simulation pattern
1005 current_pattern = (current_pattern + 1) % 3;
1006 ESP_LOGI(TAG, "Simulation pattern changed to %d", current_pattern);
1007 break;
1008
1009 case 4: // SQUARES_CMD_DISABLE - Disable matrix scanning
1010 squares_scanning_enabled = false;
1011 ESP_LOGI(TAG, "Matrix scanning DISABLED");
1012 break;
1013
1014 case 5: // SQUARES_CMD_ENABLE - Enable matrix scanning
1015 squares_scanning_enabled = true;
1016 ESP_LOGI(TAG, "Matrix scanning ENABLED");
1017 break;
1018
1019 default:
1020 ESP_LOGW(TAG, "Unknown matrix command: %d", command);
1021 break;
1022 }
1023 }
1024 }

1241 ESP_LOGI(TAG, " • Matrix event generation");
1242 ESP_LOGI(TAG, " • Simulation mode (no HW required)");
1243 ESP_LOGI(TAG, " • 20ms scan cycle");
1244

1246

1252

1274 // Matrix scanning is now handled by FreeRTOS timer callback
1275 // No need to call squares_scan_all() here to avoid race conditions
