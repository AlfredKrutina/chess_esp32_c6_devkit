55static void swap_corridor_color(uint32_t now_ms, uint8_t *r, uint8_t *g,
56 uint8_t *b) {
57 uint32_t phase = now_ms % RESTORE_SWAP_PULSE_MS;
58 uint32_t tri = phase * 2U;
59 if (tri > RESTORE_SWAP_PULSE_MS) {
60 tri = RESTORE_SWAP_PULSE_MS * 2U - tri;
61 }
62 uint8_t bright = (uint8_t)((tri * 255U) / RESTORE_SWAP_PULSE_MS);
63 *r = (uint8_t)(RESTORE_SWAP_FLOOR_R +
64 ((uint16_t)(RESTORE_ORANGE_R - RESTORE_SWAP_FLOOR_R) * bright) /
65 255U);
66 *g = (uint8_t)(RESTORE_SWAP_FLOOR_G +
67 ((uint16_t)(RESTORE_ORANGE_G - RESTORE_SWAP_FLOOR_G) * bright) /
68 255U);
69 *b = (uint8_t)(RESTORE_SWAP_FLOOR_B +
70 ((uint16_t)(RESTORE_ORANGE_B - RESTORE_SWAP_FLOOR_B) * bright) /
71 255U);
72}

110void led_board_restore_guide_tick(void) {

115 s_plan.extra_count == 0) {

120 uint32_t now = esp_timer_get_time() / 1000U;
121
122 if (now - s_last_pip_ms >= RESTORE_PIP_MS) {
123 s_last_pip_ms = now;

125 if (s_plan.paths[p].len > 0 && !s_plan.paths[p].pulse_corridor) {
126 s_pip_phase[p] =

128 }

130 }

133 s_last_missing_ms = now;
134 s_missing_bright = !s_missing_bright;
135 }
136

138 s_last_extra_ms = now;
139 s_extra_bright = !s_extra_bright;
140 }
141
142 if (!s_anim_active || !s_plan.active) {
143 return;
