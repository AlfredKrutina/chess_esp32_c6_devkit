41static uint32_t guide_hash(uint8_t sq) {
42 return (uint32_t)sq * 37U + ((uint32_t)sq >> 3) * 11U;
43}

45static uint16_t guide_period_ms(uint8_t sq) {
46 uint32_t span = GUIDE_BREATHE_JITTER_MS;
47 uint32_t base = GUIDE_BREATHE_PERIOD_MS - span / 2U;
48 return (uint16_t)(base + (guide_hash(sq) % (span + 1U)));
49}

51static uint16_t guide_phase_ms(uint8_t sq, uint16_t period) {
52 if (period == 0U) {
53 return 0U;
54 }
55 return (uint16_t)(guide_hash(sq) % period);
56}

58static guide_rgb_t color_for_piece(uint8_t type, uint8_t color) {
59 /* Hue stays saturated. The fourth number is that color's own brightness. */
60 static const guide_rgb_t white_pieces[7] = {
61 {0, 0, 0, 0},
62 {0, 255, 0, 120},
63 {255, 80, 0, 170},
64 {0, 90, 255, 200},
65 {255, 0, 160, 150},
66 {255, 255, 230, 90},
67 {255, 220, 0, 140},
68 };
69 static const guide_rgb_t black_pieces[7] = {
70 {0, 0, 0, 0},
71 {0, 40, 255, 90},
72 {130, 0, 255, 80},
73 {255, 0, 50, 70},
74 {0, 170, 40, 48},
75 {30, 0, 140, 110},
76 {255, 50, 0, 64},
77 };
78 if (type < 1 || type > 6) {
79 guide_rgb_t red = {EXTRA_R, EXTRA_G, EXTRA_B, EXTRA_LEVEL};
80 return red;
81 }
82 if (color == 2) {
83 return black_pieces[type];
84 }
85 return white_pieces[type];
86}

95static void push_plan(void) {
96 uint8_t i;
97 s_mask = 0;
98 led_target_begin_fade(GUIDE_FADE_MS);
99 for (i = 0; i < s_plan.mark_count; i++) {
100 const led_restore_guide_mark_t *mark = &s_plan.marks[i];
101 guide_rgb_t rgb;
102 uint8_t level;
103 uint8_t led;
104 uint8_t r, g, b;
105 if (mark->square >= 64) {
106 continue;
107 }
108 rgb = color_for_piece(mark->type, mark->color);
109 level = rgb.level;
110 if (mark->role == LED_RESTORE_MARK_EXTRA) {
111 rgb.r = EXTRA_R;
112 rgb.g = EXTRA_G;
113 rgb.b = EXTRA_B;
114 level = EXTRA_LEVEL;
115 }
116 led = square_to_led(mark->square);
117 if (led >= 64) {
118 continue;
119 }
120 scale_rgb(rgb, level, &r, &g, &b);
121 s_mask |= (uint64_t)1 << led;
122 if (mark->role == LED_RESTORE_MARK_HOME) {
123 uint16_t period = guide_period_ms(mark->square);
124 uint8_t floor =
125 (mark->color == 2U) ? GUIDE_BLACK_FLOOR : GUIDE_WHITE_FLOOR;
126 led_target_set_breathe_sq(led, r, g, b, floor, period,
127 guide_phase_ms(mark->square, period),
128 GUIDE_FADE_MS);
129 } else {
130 led_target_set_breathe_sq(led, r, g, b, 0, 0, 0, GUIDE_FADE_MS);
131 }
132 }
133 led_target_end();
134 s_need_push = false;
135}

137void led_board_restore_guide_apply(const led_restore_guide_plan_t *plan) {
138 if (plan == NULL || !plan->active || plan->mark_count == 0) {
139 led_board_restore_guide_clear();
140 return;
141 }
142 bool overlay_blocks = led_gesture_anim_busy() || led_lamp_board_held();
143 if (!overlay_blocks && s_plan.active && s_plan.mark_count == plan->mark_count &&
144 memcmp(s_plan.marks, plan->marks,
145 (size_t)plan->mark_count * sizeof(plan->marks[0])) == 0 &&
146 s_pushed_gen == led_board_picture_gen()) {
147 return;
148 }
149 memcpy(&s_plan, plan, sizeof(s_plan));
150 if (overlay_blocks) {
151 s_need_push = true;
152 return;
153 }
154 push_plan();
155 s_pushed_gen = led_board_picture_gen();
156}

167void led_board_restore_guide_fade_out(void) {
168 uint64_t mask;
169 uint8_t i;
170 if (!s_plan.active && s_mask == 0U) {
171 return;
172 }
173 s_plan.active = false;
174 s_plan.mark_count = 0;
175 s_need_push = false;
176 /* A caller picture (lift, idle) owns the frame. Do not fade under it. */
177 if (led_target_is_open()) {
178 s_mask = 0U;
179 return;
180 }
181 mask = s_mask;
182 s_mask = 0U;
183 for (i = 0; i < 64U; i++) {
184 if ((mask & ((uint64_t)1 << i)) == 0U) {
185 continue;
186 }
187 led_target_set_breathe_sq(i, 0, 0, 0, 0, 0, 0, GUIDE_FADE_MS);
188 }
189}
