64static guide_rgb_t color_for_piece(uint8_t type, uint8_t color) {
65 /* White hues stay saturated so pawn, knight and bishop stay apart.
66 * Black uses another hue and stays dark. */
67 static const guide_rgb_t white_pieces[7] = {
68 {0, 0, 0}, {0, 255, 0}, {255, 80, 0}, {0, 90, 255},
69 {255, 0, 160}, {255, 255, 230}, {255, 220, 0},
70 };
71 static const guide_rgb_t black_pieces[7] = {
72 {0, 0, 0}, {0, 40, 255}, {130, 0, 255}, {255, 0, 50},
73 {0, 170, 40}, {30, 0, 140}, {255, 50, 0},
74 };
75 if (type < 1 || type > 6) {
76 guide_rgb_t red = {EXTRA_R, EXTRA_G, EXTRA_B};
77 return red;
78 }
79 if (color == 2) {
80 return black_pieces[type];
81 }
82 return white_pieces[type];
83}

85static uint8_t smoothstep_u8(uint8_t x) {
86 uint32_t v = (uint32_t)x;
87 uint32_t s = (v * v * (765U - 2U * v)) / (255U * 255U);
88 return (uint8_t)s;
89}

91static uint8_t breathe_level(uint32_t now_ms, uint32_t period_ms, uint32_t phase_ms,
92 uint8_t floor_lv, uint8_t peak_lv) {
93 if (period_ms < 2U) {
94 return peak_lv;
95 }
96 uint32_t t = (now_ms + phase_ms) % period_ms;
97 uint32_t half = period_ms / 2U;
98 uint32_t up = (t < half) ? t : (period_ms - t);
99 uint8_t x = (uint8_t)((up * 255U) / half);
100 uint8_t s = smoothstep_u8(x);
101 return (uint8_t)(floor_lv + ((uint16_t)(peak_lv - floor_lv) * s) / 255U);
102}

111static void paint_square(uint8_t sq) {
112 uint8_t gain = s_gain[sq];
113 uint8_t r = (uint8_t)(((uint16_t)s_hold[sq].r * gain) / 255U);
114 uint8_t g = (uint8_t)(((uint16_t)s_hold[sq].g * gain) / 255U);
115 uint8_t b = (uint8_t)(((uint16_t)s_hold[sq].b * gain) / 255U);
116 led_set_pixel_internal(square_to_led(sq), r, g, b);
117 s_on[sq] = (gain > 0U) ? 1U : 0U;

120static unsigned reveal_step(uint32_t now) {
121 uint32_t dt = (s_last_ms == 0U) ? 33U : (now - s_last_ms);
122 s_last_ms = now;
123 if (dt == 0U || dt > 80U) {
124 dt = 33U;
125 }
126 unsigned step = (unsigned)((dt * 255U) / REVEAL_MS);
127 if (step < 1U) {
128 step = 1U;
129 }
130 return step;
131}

133void led_board_restore_guide_apply(const led_restore_guide_plan_t *plan) {
134 if (plan == NULL || !plan->active || plan->mark_count == 0) {
135 led_board_restore_guide_clear();
136 return;
137 }
138 bool fresh = !s_anim_active;
139 memcpy(&s_plan, plan, sizeof(s_plan));
140 s_anim_active = true;
141 if (fresh) {
142 memset(s_gain, 0, sizeof(s_gain));
143 memset(s_on, 0, sizeof(s_on));
144 memset(s_hold, 0, sizeof(s_hold));
145 s_last_ms = 0;
146 }

171void led_board_restore_guide_tick(void) {
172 if (led_gesture_anim_busy() || led_lamp_board_held()) {
173 return;
174 }
175 if (!s_plan.active || s_plan.mark_count == 0) {
176 if (!s_anim_active) {
177 return;
178 }
179 uint32_t now = esp_timer_get_time() / 1000U;
180 unsigned step = reveal_step(now);
181 bool any = false;
182 for (uint8_t sq = 0; sq < 64U; sq++) {
183 if (s_gain[sq] > 0U) {
184 unsigned gain = (unsigned)s_gain[sq];
185 s_gain[sq] = (gain > step) ? (uint8_t)(gain - step) : 0U;
186 paint_square(sq);
187 if (s_gain[sq] > 0U) {
188 any = true;
189 }
190 } else if (s_on[sq]) {
191 led_set_pixel_internal(square_to_led(sq), 0, 0, 0);
192 s_on[sq] = 0U;
193 }
194 }
195 if (!any) {
196 s_anim_active = false;
197 }
198 return;
199 }
200 s_anim_active = true;
201
202 uint32_t now = esp_timer_get_time() / 1000U;
203 unsigned step = reveal_step(now);
204 memset(s_want, 0, sizeof(s_want));
205
206 for (uint8_t i = 0; i < s_plan.mark_count; i++) {
207 const led_restore_guide_mark_t *mark = &s_plan.marks[i];
208 if (mark->square >= 64) {
209 continue;
210 }
211 guide_rgb_t rgb;
212 uint32_t period;
213 uint32_t phase;
214 uint8_t peak;
215 if (mark->role == LED_RESTORE_MARK_EXTRA) {
216 rgb.r = EXTRA_R;
217 rgb.g = EXTRA_G;
218 rgb.b = EXTRA_B;
219 period = 1800U;
220 phase = (uint32_t)mark->square * 70U;
221 peak = 220U;
222 } else {
223 rgb = color_for_piece(mark->type, mark->color);
224 period = period_for_type(mark->type);
225 phase = (uint32_t)mark->square * 90U;
226 if (mark->color == 2) {
227 peak = (mark->role == LED_RESTORE_MARK_SITTING) ? BLACK_SIT_PEAK
228 : BLACK_HOME_PEAK;
229 } else {
230 peak = (mark->role == LED_RESTORE_MARK_SITTING) ? WHITE_SIT_PEAK
231 : WHITE_HOME_PEAK;
232 }
233 if (mark->role == LED_RESTORE_MARK_SITTING) {
234 phase += period / 2U;
235 }
236 }
237 uint8_t floor_lv;
238 if (mark->role == LED_RESTORE_MARK_EXTRA) {
239 floor_lv = 40U;
240 } else if (mark->color == 2) {
241 floor_lv = BLACK_FLOOR;
242 } else {
243 floor_lv = WHITE_FLOOR;
244 }
245 uint8_t level = breathe_level(now, period, phase, floor_lv, peak);
246 uint8_t sq = mark->square;
247 s_want[sq] = 1U;
248 scale_rgb(rgb, level, &s_hold[sq].r, &s_hold[sq].g, &s_hold[sq].b);
249 }
250
251 for (uint8_t sq = 0; sq < 64U; sq++) {
252 if (s_want[sq]) {
253 unsigned gain = (unsigned)s_gain[sq] + step;
254 s_gain[sq] = (gain > 255U) ? 255U : (uint8_t)gain;
255 paint_square(sq);
256 } else if (s_gain[sq] > 0U) {
257 unsigned gain = (unsigned)s_gain[sq];
258 s_gain[sq] = (gain > step) ? (uint8_t)(gain - step) : 0U;
259 paint_square(sq);
260 } else if (s_on[sq]) {
261 led_set_pixel_internal(square_to_led(sq), 0, 0, 0);
262 s_on[sq] = 0U;
263 }
264 }
