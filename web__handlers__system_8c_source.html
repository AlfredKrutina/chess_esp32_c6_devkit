754static void factory_reset_worker(void *ignored) {
755 (void)ignored;
756 vTaskDelay(pdMS_TO_TICKS(500));
757 ESP_LOGW(TAG,
758 "[STAGING] factory_reset: raw erase default NVS partition + restart");
759
760 const esp_partition_t *nvs_part = esp_partition_find_first(
761 ESP_PARTITION_TYPE_DATA, ESP_PARTITION_SUBTYPE_DATA_NVS, NULL);
762 esp_err_t er = ESP_FAIL;
763 if (nvs_part != NULL) {
764 er = esp_partition_erase_range(nvs_part, 0, nvs_part->size);
765 ESP_LOGW(TAG, "esp_partition_erase_range(label=%s size=%u): %s",
766 nvs_part->label, (unsigned)nvs_part->size, esp_err_to_name(er));
767 } else {
768 er = nvs_flash_erase();
769 ESP_LOGW(TAG, "nvs_flash_erase (no nvs partition label): %s",
770 esp_err_to_name(er));
771 }
772
773 esp_restart();
774}

781 return ESP_FAIL;
782 }
783 return ESP_OK;

963 httpd_resp_send(req, "Type out of range (0-14)", -1);
964 return ESP_FAIL;

969 // Parse custom time values ​​if type is CUSTOM
970 if (type_value == 14) { // TIME_CONTROL_CUSTOM

972 char *increment_str = strstr(content, "\"custom_increment\":");

977 if (minutes >= 1 && minutes <= 180) {
978 cmd.u.timer.custom_minutes = (uint32_t)minutes;
979 } else {
980 httpd_resp_set_status(req, "400 Bad Request");
981 httpd_resp_send(req, "Minutes must be 1-180", -1);
982 return ESP_FAIL;

984 }
985 }
986
987 if (increment_str) {
988 int increment;
989 if (sscanf(increment_str, "\"custom_increment\":%d", &increment) == 1) {
990 if (increment >= 0 && increment <= 60) {
991 cmd.u.timer.custom_increment = (uint32_t)increment;
992 } else {
993 httpd_resp_set_status(req, "400 Bad Request");
994 httpd_resp_send(req, "Increment must be 0-60", -1);

1002 httpd_resp_set_status(req, "400 Bad Request");
1003 httpd_resp_send(req, "Custom time control requires minutes and increment",
1004 -1);
1005 return ESP_FAIL;
1006 }
1007 }
1008
1009 // Send the command to the game task
1010 if (!game_submit(&cmd)) {
1011 cmd_wire_http_send(req, "503 Service Unavailable",
1012 "{\"ok\":false,\"error\":\"busy\"}", content, false,
1013 NULL);
1014 return ESP_OK;
1015 }

1017 cmd_wire_http_send(req, "200 OK",
1018 "{\"success\":true,\"message\":\"Time control set\"}",
1019 content, true, "timer_config");
1020 return ESP_OK;
