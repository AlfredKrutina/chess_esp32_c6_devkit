66bool game_parse_fen(const char *fen, game_fen_position_t *out) {
67 position_t *parsed = game_logic_scratch_position();
68 int row;
69 int col;
70 int victim;
71
72 if (fen == NULL || out == NULL || parsed == NULL ||
73 !position_from_fen(parsed, fen)) {
74 return false;
75 }
76 memset(out, 0, sizeof(*out));
77 for (row = 0; row < 8; row++) {
78 for (col = 0; col < 8; col++) {
79 out->squares[row][col] = (piece_t)parsed->board[row][col];
80 }
81 }
82 out->side = (player_t)parsed->side;
83 out->white_rook_h_moved = (parsed->castling & POS_CASTLE_WK) == 0;
84 out->white_rook_a_moved = (parsed->castling & POS_CASTLE_WQ) == 0;
85 out->white_king_moved = out->white_rook_h_moved && out->white_rook_a_moved;
86 out->black_rook_h_moved = (parsed->castling & POS_CASTLE_BK) == 0;
87 out->black_rook_a_moved = (parsed->castling & POS_CASTLE_BQ) == 0;
88 out->black_king_moved = out->black_rook_h_moved && out->black_rook_a_moved;
89 if (parsed->ep_square >= 64) {
90 out->ep = false;
91 } else {
92 out->ep = true;
93 out->ep_row = (uint8_t)(parsed->ep_square / 8);
94 out->ep_col = (uint8_t)(parsed->ep_square % 8);
95 victim = (parsed->side == 0) ? ((int)out->ep_row - 1) : ((int)out->ep_row + 1);
96 if (victim >= 0 && victim <= 7) {
97 out->victim_row = (uint8_t)victim;
98 out->victim_col = out->ep_col;
99 }
100 }
101 out->halfmove = parsed->halfmove;
102 out->move_count = (parsed->fullmove == 0)
103 ? 0
104 : (uint32_t)(parsed->fullmove - 1) * 2U +
105 (parsed->side == 1 ? 1U : 0U);
106 return true;
107}

203void game_logic_sync_globals_from_position(const position_t *p) {
204 uint8_t ep_row;
205 uint8_t ep_col;
206 int victim_row;
207
208 if (p == NULL) {
209 return;
210 }
211 current_player = (player_t)p->side;
212 white_rook_h_moved = (p->castling & POS_CASTLE_WK) == 0;
213 white_rook_a_moved = (p->castling & POS_CASTLE_WQ) == 0;
214 /* Castling rights imply the king has not moved. If both rights are gone
215 * and the king left e1, it has moved. If both rooks moved but the king
216 * is still on e1, keep the previous flag. */
217 if ((p->castling & (POS_CASTLE_WK | POS_CASTLE_WQ)) != 0) {
218 white_king_moved = false;
219 } else if (p->board[0][4] != POS_WHITE_KING) {
220 white_king_moved = true;
221 }
222 black_rook_h_moved = (p->castling & POS_CASTLE_BK) == 0;
223 black_rook_a_moved = (p->castling & POS_CASTLE_BQ) == 0;
224 if ((p->castling & (POS_CASTLE_BK | POS_CASTLE_BQ)) != 0) {
225 black_king_moved = false;
226 } else if (p->board[7][4] != POS_BLACK_KING) {
227 black_king_moved = true;
228 }
229 if (p->ep_square >= 64) {
230 en_passant_available = false;
231 en_passant_target_row = 0;
232 en_passant_target_col = 0;
233 en_passant_victim_row = 0;
234 en_passant_victim_col = 0;
235 } else {
236 game_square_to_row_col(p->ep_square, &ep_row, &ep_col);
237 victim_row =
238 (p->side == PLAYER_BLACK) ? ((int)ep_row + 1) : ((int)ep_row - 1);
239 if (victim_row < 0 || victim_row > 7) {
240 en_passant_available = false;
241 } else {
242 en_passant_available = true;
243 en_passant_target_row = ep_row;
244 en_passant_target_col = ep_col;
245 en_passant_victim_row = (uint8_t)victim_row;
246 en_passant_victim_col = ep_col;
247 }
248 }
249}

251static void game_logic_fill(position_t *p, bool clear_hist) {
252 uint8_t castle = 0;
253 uint32_t half;
254
255 if (p == NULL) {
256 return;
257 }
258 if (clear_hist) {
259 memset(p, 0, sizeof(*p));
260 }
261 p->side = (uint8_t)current_player;
262 if (!white_rook_h_moved) {
263 castle |= POS_CASTLE_WK;
264 }
265 if (!white_rook_a_moved) {
266 castle |= POS_CASTLE_WQ;
267 }
268 if (!black_rook_h_moved) {
269 castle |= POS_CASTLE_BK;
270 }
271 if (!black_rook_a_moved) {
272 castle |= POS_CASTLE_BQ;
273 }
274 p->castling = castle;
275 if (!en_passant_available || en_passant_target_row > 7 || en_passant_target_col > 7) {
276 p->ep_square = POSITION_SQUARE_NONE;
277 } else {
278 p->ep_square = (uint8_t)(en_passant_target_row * 8 + en_passant_target_col);
279 }
280 half = game_get_fifty_move_counter();
281 p->halfmove = (uint16_t)(half > 65535u ? 65535u : half);
282 p->fullmove = (uint16_t)(move_count / 2u + 1u);
283 if (p->fullmove == 0) {
284 p->fullmove = 1;
285 }
286 if (clear_hist) {
287 p->hist_len = 0;
288 }
289}

318bool game_logic_play(uint8_t from_row, uint8_t from_col, uint8_t to_row,
319 uint8_t to_col, int promo) {
320 position_move_t mv;
321
322 if (castling_state.in_progress && from_row == castling_state.rook_from_row &&
323 from_col == castling_state.rook_from_col &&
324 to_row == castling_state.rook_to_row &&
325 to_col == castling_state.rook_to_col) {
326 game_session_snap_refresh();
327 game_logic_sync_clock(&s_logic);
328 return true;
329 }
330 if (castling_state.in_progress && s_logic.hist_len > 0) {
331 (void)position_undo(&s_logic);
332 }
333 game_logic_fill(&s_logic, false);
334 mv.from = (uint8_t)(from_row * 8u + from_col);
335 mv.to = (uint8_t)(to_row * 8u + to_col);
336 mv.promo = (position_promo_t)promo;
337 if (mv.from >= 64 || mv.to >= 64 || !position_apply(&s_logic, mv)) {
338 return false;
339 }
340 game_logic_sync_globals_from_position(&s_logic);
341 game_session_snap_refresh();
342 game_logic_sync_clock(&s_logic);
343 return true;
344}
