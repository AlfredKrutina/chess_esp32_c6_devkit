39static size_t stm32_hex_to_bin(const char *hex, uint8_t *out, size_t out_max) {
40 size_t n = strlen(hex);
41 if ((n % 2u) != 0u || n / 2u > out_max) {
42 return 0;
43 }
44 for (size_t i = 0; i < n / 2u; i++) {
45 unsigned v = 0;
46 if (sscanf(hex + 2 * i, "%2x", &v) != 1) {
47 return 0;
48 }
49 out[i] = (uint8_t)v;
50 }
51 return n / 2u;
52}

170 if (le != ESP_OK) {

176 while (eff > 0 && buf[eff - 1] == 0xFF) {
177 eff--;
178 }
179 if (eff == 0) {

182 return CMD_ERROR_INVALID_PARAMETER;
183 }

185 esp_err_t fe =

187 free(buf);
188 if (fe != ESP_OK) {
189 uart_send_formatted("FLASH_PART: %s", esp_err_to_name(fe));
190 return CMD_ERROR_SYSTEM_ERROR;
191 }
192 uart_send_success("FLASH_PART done");
193 return CMD_SUCCESS;
194 }
