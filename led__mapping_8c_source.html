70void led_index_to_chess_pos(uint8_t led_index, uint8_t* row, uint8_t* col)
71{
72 if (led_index >= 64 || row == NULL || col == NULL) {
73 ESP_LOGE(TAG, "Invalid LED index: %d", led_index);
74 if (row) *row = 0;

79 *row = led_row[led_index];

92uint8_t chess_notation_to_led_index(const char* notation)
93{
94 if (!notation || strlen(notation) < 2) {
95 ESP_LOGE(TAG, "Invalid notation: %s", notation ? notation : "NULL");
96 return 0;
97 }
98
99 char file = tolower(notation[0]);
100 char rank = notation[1];
101
102 if (file < 'a' || file > 'h' || rank < '1' || rank > '8') {
103 ESP_LOGE(TAG, "Invalid notation: %s", notation);
104 return 0;
105 }
106
107 uint8_t col = file - 'a'; // a=0, b=1, ..., h=7
108 uint8_t row = rank - '1'; // 1=0, 2=1, ..., 8=7
109
110 return chess_pos_to_led_index(row, col);
111}
