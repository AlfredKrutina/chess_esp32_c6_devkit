30bool game_is_king_in_check(player_t player) {
31 // Find king position
32 int king_row = -1, king_col = -1;
33 piece_t king_piece =
34 (player == PLAYER_WHITE) ? PIECE_WHITE_KING : PIECE_BLACK_KING;
35
36 for (int row = 0; row < 8; row++) {
37 for (int col = 0; col < 8; col++) {
38 if (game_get_piece(row, col) == king_piece) {
39 king_row = row;
40 king_col = col;
41 break;
42 }
43 }
44 if (king_row != -1)
45 break;
46 }
47
48 if (king_row == -1)
49 return false; // King not found
50
51 // Check if any opponent piece can capture the king
52 piece_t opponent_pieces_start =
53 (player == PLAYER_WHITE) ? PIECE_BLACK_PAWN : PIECE_WHITE_PAWN;
54 piece_t opponent_pieces_end =
55 (player == PLAYER_WHITE) ? PIECE_BLACK_KING : PIECE_WHITE_KING;
56
57 for (int row = 0; row < 8; row++) {
58 for (int col = 0; col < 8; col++) {
59 piece_t piece = game_get_piece(row, col);
60 if (piece >= opponent_pieces_start && piece <= opponent_pieces_end) {
61 // Create temporary move to check if it can capture king
62 chess_move_t temp_move = {.from_row = row,
63 .from_col = col,
64 .to_row = king_row,
65 .to_col = king_col,
66 .piece = piece,
67 .captured_piece = king_piece,
68 .timestamp = 0};
69
70 // Check if this move is valid (would capture king)
71 if (game_validate_piece_move_enhanced(&temp_move, piece) ==
72 MOVE_ERROR_NONE) {
73 return true; // King is in check
74 }
75 }
76 }
77 }
78
79 return false;
80}

108bool game_is_insufficient_material(void) {
109 int white_pieces = 0, black_pieces = 0;
110 int white_minors = 0, black_minors = 0; // Knights and bishops
111 bool white_has_bishop = false, black_has_bishop = false;
112 bool white_has_knight = false, black_has_knight = false;
113 bool white_bishop_color = false,
114 black_bishop_color = false; // false = light, true = dark
115
116 // Count pieces and analyze material
117 for (int row = 0; row < 8; row++) {
118 for (int col = 0; col < 8; col++) {
119 piece_t piece = game_get_piece(row, col);
120
121 switch (piece) {
122 case PIECE_WHITE_PAWN:
123 case PIECE_WHITE_ROOK:
124 case PIECE_WHITE_QUEEN:
125 white_pieces++;
126 break;
127 case PIECE_WHITE_BISHOP:
128 white_pieces++;
129 white_minors++;
130 white_has_bishop = true;
131 white_bishop_color = ((row + col) % 2 == 0); // Light squares = false
132 break;
133 case PIECE_WHITE_KNIGHT:
134 white_pieces++;
135 white_minors++;
136 white_has_knight = true;
137 break;
138 case PIECE_BLACK_PAWN:
139 case PIECE_BLACK_ROOK:
140 case PIECE_BLACK_QUEEN:
141 black_pieces++;
142 break;
143 case PIECE_BLACK_BISHOP:
144 black_pieces++;
145 black_minors++;
146 black_has_bishop = true;
147 black_bishop_color = ((row + col) % 2 == 0); // Light squares = false
148 break;
149 case PIECE_BLACK_KNIGHT:
150 black_pieces++;
151 black_minors++;
152 black_has_knight = true;
153 break;
154 default:
155 break;
156 }
157 }
158 }
159
160 // King vs King - always insufficient
161 if (white_pieces == 0 && black_pieces == 0) {
162 return true;
163 }
164
165 // King + minor vs King - insufficient
166 if ((white_pieces == 1 && black_pieces == 0) ||
167 (white_pieces == 0 && black_pieces == 1)) {
168 return true;
169 }
170
171 // King + Bishop vs King + Bishop (same color) - insufficient
172 if (white_pieces == 1 && black_pieces == 1 && white_has_bishop &&
173 black_has_bishop && white_bishop_color == black_bishop_color) {
174 return true;
175 }
176
177 // King + Knight vs King + Knight - insufficient
178 if (white_pieces == 1 && black_pieces == 1 && white_has_knight &&
179 black_has_knight) {
180 return true;
181 }
182
183 // King + 2 Knights vs King - insufficient (very rare, but technically
184 // insufficient)
185 if ((white_pieces == 2 && black_pieces == 0 && white_minors == 2 &&
186 white_has_knight) ||
187 (white_pieces == 0 && black_pieces == 2 && black_minors == 2 &&
188 black_has_knight)) {
189 return true;
190 }
191
192 return false; // Sufficient material for checkmate
193}

223game_state_t game_check_end_game_conditions(void) {
224 const position_t *logic = game_logic_position();
225 bool use_core = (logic != NULL && logic->side == (uint8_t)current_player);
226 position_status_t st = use_core ? position_status(logic) : (position_status_t){0};
227 bool in_check = use_core ? st.in_check : game_is_king_in_check(current_player);
228 bool has_moves = use_core ? (!st.checkmate && !st.stalemate)
229 : game_has_legal_moves(current_player);
230
231 if (in_check && !has_moves) {
232 // Checkmate - determine the type according to the last move
233 (*game_state_game_result()) = GAME_STATE_FINISHED;
234 (*game_state_current_result_type()) = (current_player == PLAYER_WHITE) ? RESULT_BLACK_WINS
235 : RESULT_WHITE_WINS;
236
237 // Distinguish the type of chess according to the last move
238 switch ((*game_state_last_move_type())) {
239 case LAST_MOVE_EN_PASSANT:
240 (*game_state_current_endgame_reason()) = ENDGAME_REASON_CHECKMATE_EN_PASSANT;
241 ESP_LOGI(TAG, "🎯 CHECKMATE BY EN PASSANT! %s wins in %" PRIu32 " moves!",
242 (current_player == PLAYER_WHITE) ? "Black" : "White",
243 move_count);
244 break;
245 case LAST_MOVE_CASTLING:
246 (*game_state_current_endgame_reason()) = ENDGAME_REASON_CHECKMATE_CASTLING;
247 ESP_LOGI(TAG, "🎯 CHECKMATE BY CASTLING! %s wins in %" PRIu32 " moves!",
248 (current_player == PLAYER_WHITE) ? "Black" : "White",
249 move_count);
250 break;
251 case LAST_MOVE_PROMOTION:
252 (*game_state_current_endgame_reason()) = ENDGAME_REASON_CHECKMATE_PROMOTION;
253 ESP_LOGI(TAG, "🎯 CHECKMATE BY PROMOTION! %s wins in %" PRIu32 " moves!",
254 (current_player == PLAYER_WHITE) ? "Black" : "White",
255 move_count);
256 break;
257 case LAST_MOVE_DISCOVERED:
258 (*game_state_current_endgame_reason()) = ENDGAME_REASON_CHECKMATE_DISCOVERED;
259 ESP_LOGI(
260 TAG,
261 "🎯 CHECKMATE BY DISCOVERED CHECK! %s wins in %" PRIu32 " moves!",
262 (current_player == PLAYER_WHITE) ? "Black" : "White", move_count);
263 break;
264 default:
265 (*game_state_current_endgame_reason()) = ENDGAME_REASON_CHECKMATE;
266 ESP_LOGI(TAG, "🎯 CHECKMATE! %s wins in %" PRIu32 " moves!",
267 (current_player == PLAYER_WHITE) ? "Black" : "White",
268 move_count);
269 break;
270 }
271
272 game_update_endgame_statistics((*game_state_current_result_type()));
273 (*game_state_endgame_report_requested()) = true; // Report request
274
275 // Stop timer (game over)
276 timer_pause();
277
278 // UNIFIED VICTORY ANIMATION
279 // Checkmate = Victory for current_player (who delivered mate? No wait.)
280 // Logic: if current_player is in check & no moves, they LOST.
281 // Winner is the OPPONENT.
282 player_t winner =
283 (current_player == PLAYER_WHITE) ? PLAYER_BLACK : PLAYER_WHITE;
284 game_trigger_victory_animation(winner);
285
286 return GAME_STATE_FINISHED;
287 } else if (!in_check && !has_moves) {
288 // Stalemate
289 (*game_state_game_result()) = GAME_STATE_FINISHED;
290 (*game_state_current_result_type()) = RESULT_DRAW_STALEMATE;
291 (*game_state_current_endgame_reason()) = ENDGAME_REASON_STALEMATE; // Mark as stalemate
292 game_update_endgame_statistics((*game_state_current_result_type()));
293 (*game_state_endgame_report_requested()) = true; // Report request
294
295 // Stop timer (game over)
296 timer_pause();
297 game_cmd_reply("Stalemate. The game is drawn.", false, NULL);
298 game_trigger_victory_animation(current_player);
299
300 ESP_LOGI(TAG, "🤝 STALEMATE! Game drawn in %" PRIu32 " moves", move_count);
301 return GAME_STATE_FINISHED;
302 }
303
304 // BUG #13: Check for draw conditions
305 // 50-move rule: 50 full moves = 100 half-moves/ply
306 // moves_without_capture counts half moves so >= 100
307 if (st.fifty_move || game_get_fifty_move_counter() >= 100) {
308 (*game_state_game_result()) = GAME_STATE_FINISHED;
309 (*game_state_current_result_type()) = RESULT_DRAW_50_MOVE;
310 (*game_state_current_endgame_reason()) =
311 ENDGAME_REASON_50_MOVE; // Mark as 50-move rule
312 game_update_endgame_statistics((*game_state_current_result_type()));
313 (*game_state_endgame_report_requested()) = true; // Report request
314
315 // Stop timer (game over)
316 timer_pause();
317 game_cmd_reply("Draw by the 50-move rule.", false, NULL);
318 game_trigger_victory_animation(current_player);
319
320 ESP_LOGI(TAG, "🤝 DRAW! 50 moves without capture (50-move rule)");
321 return GAME_STATE_FINISHED;
322 }
323
324 if (st.repetition || game_is_position_repeated()) {
325 (*game_state_game_result()) = GAME_STATE_FINISHED;
326 (*game_state_current_result_type()) = RESULT_DRAW_REPETITION;
327 (*game_state_current_endgame_reason()) =
328 ENDGAME_REASON_REPETITION; // Mark as threefold repetition
329 game_update_endgame_statistics((*game_state_current_result_type()));
330 (*game_state_endgame_report_requested()) = true; // Report request
331
332 // Stop timer (game over)
333 timer_pause();
334 game_cmd_reply("Draw by repetition.", false, NULL);
335 game_trigger_victory_animation(current_player);
336
337 ESP_LOGI(TAG, "🤝 DRAW! Position repeated (draw by repetition)");
338 return GAME_STATE_FINISHED;
339 }
340
341 // Check for insufficient material
342 if (st.insufficient_material || game_is_insufficient_material()) {
343 (*game_state_game_result()) = GAME_STATE_FINISHED;
344 (*game_state_current_result_type()) = RESULT_DRAW_INSUFFICIENT;
345 (*game_state_current_endgame_reason()) =
346 ENDGAME_REASON_INSUFFICIENT; // Mark as insufficient material
347 game_update_endgame_statistics((*game_state_current_result_type()));
348 (*game_state_endgame_report_requested()) = true; // Report request
349
350 // Stop timer (game over)
351 timer_pause();
352 game_cmd_reply("Draw by insufficient material.", false, NULL);
353 game_trigger_victory_animation(current_player);
354
355 ESP_LOGI(TAG, "🤝 DRAW! Insufficient material to checkmate");
356 return GAME_STATE_FINISHED;
357 }
358
359 return GAME_STATE_ACTIVE;
360}
