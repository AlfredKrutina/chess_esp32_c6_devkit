28void game_update_endgame_statistics(game_result_type_t result_type) {
29 switch (result_type) {
30 case RESULT_WHITE_WINS:
31 white_wins++;
32 ESP_LOGI(TAG, "📊 Statistics: White wins! (Total: %" PRIu32 ")",

35 case RESULT_BLACK_WINS:
36 black_wins++;

39 break;
40 case RESULT_DRAW_STALEMATE:
41 case RESULT_DRAW_50_MOVE:
42 case RESULT_DRAW_REPETITION:
43 case RESULT_DRAW_INSUFFICIENT:
44 case RESULT_DRAW_AGREEMENT:
45 draws++;
46 ESP_LOGI(TAG, "📊 Statistics: Draw! (Total: %" PRIu32 ")", draws);
47 break;

49 total_games++;
50 ESP_LOGI(TAG, "📊 Total games: %" PRIu32, total_games);
51
52 uint8_t fact = HA_GAME_DRAW;
53 uint8_t reason = HA_REASON_NONE;
54 if (result_type == RESULT_WHITE_WINS) {
55 fact = HA_GAME_WHITE_WIN;
56 } else if (result_type == RESULT_BLACK_WINS) {
57 fact = HA_GAME_BLACK_WIN;
58 }
59 switch (current_endgame_reason) {
60 case ENDGAME_REASON_CHECKMATE:
61 case ENDGAME_REASON_CHECKMATE_EN_PASSANT:
62 case ENDGAME_REASON_CHECKMATE_CASTLING:
63 case ENDGAME_REASON_CHECKMATE_PROMOTION:
64 case ENDGAME_REASON_CHECKMATE_DISCOVERED:

68 reason = HA_REASON_STALEMATE;
69 break;
70 case ENDGAME_REASON_50_MOVE:
71 reason = HA_REASON_FIFTY;
72 break;
73 case ENDGAME_REASON_REPETITION:
74 reason = HA_REASON_REPETITION;
75 break;
76 case ENDGAME_REASON_INSUFFICIENT:
77 reason = HA_REASON_INSUFFICIENT;

170void game_print_endgame_report_uart(game_result_type_t result_type) {
171 uint32_t current_time = esp_timer_get_time() / 1000;
172 uint32_t game_duration =
173 (game_start_time > 0) ? (current_time - game_start_time) : 0;
174 uint32_t minutes = game_duration / 60;
175 uint32_t seconds = game_duration % 60;
176
177 // Print endgame announcement
178 print_report_header("🏆 ENDGAME REPORT");
179
180 // Determine outcome and end type by current_endgame_reason
181 const char *winner = "";
182 const char *loser = "";
183 const char *end_reason = "";
184
185 // First we determine the winner from the result_type
186 if (result_type == RESULT_WHITE_WINS) {
187 winner = "White";
188 loser = "Black";
189 } else if (result_type == RESULT_BLACK_WINS) {
190 winner = "Black";
191 loser = "White";
192 } else {
193 winner = "Draw";
194 loser = "Draw";
195 }
196
197 // Then we determine the reason for the end of the game and print the result
198 switch (current_endgame_reason) {
199 // === CHECKMATE VARIATIONS ===
200 case ENDGAME_REASON_CHECKMATE:
201 printf("🎯 Game Result: %s WINS BY CHECKMATE\r\n",
202 (result_type == RESULT_WHITE_WINS) ? "WHITE" : "BLACK");
203 end_reason = "Checkmate";
204 break;
205 case ENDGAME_REASON_CHECKMATE_EN_PASSANT:
206 printf("⚔️ Game Result: %s WINS BY EN PASSANT CHECKMATE!\r\n",
207 (result_type == RESULT_WHITE_WINS) ? "WHITE" : "BLACK");
208 end_reason = "Checkmate (En Passant)";
209 break;
210 case ENDGAME_REASON_CHECKMATE_CASTLING:
211 printf("🏰 Game Result: %s WINS BY CASTLING CHECKMATE!\r\n",
212 (result_type == RESULT_WHITE_WINS) ? "WHITE" : "BLACK");
213 end_reason = "Checkmate (Castling)";
214 break;
215 case ENDGAME_REASON_CHECKMATE_PROMOTION:
216 printf("👑 Game Result: %s WINS BY PROMOTION CHECKMATE!\r\n",
217 (result_type == RESULT_WHITE_WINS) ? "WHITE" : "BLACK");
218 end_reason = "Checkmate (Promotion)";
219 break;
220 case ENDGAME_REASON_CHECKMATE_DISCOVERED:
221 printf("💥 Game Result: %s WINS BY DISCOVERED CHECK CHECKMATE!\r\n",
222 (result_type == RESULT_WHITE_WINS) ? "WHITE" : "BLACK");
223 end_reason = "Checkmate (Discovered Check)";
224 break;
225
226 // === PLAYER ACTIONS ===
227 case ENDGAME_REASON_RESIGNATION:
228 printf("🏳️ Game Result: %s WINS BY RESIGNATION\r\n",
229 (result_type == RESULT_WHITE_WINS) ? "WHITE" : "BLACK");
230 end_reason = "Resignation";
231 break;
232 case ENDGAME_REASON_TIMEOUT:
233 printf("⏰ Game Result: %s WINS BY TIMEOUT\r\n",
234 (result_type == RESULT_WHITE_WINS) ? "WHITE" : "BLACK");
235 end_reason = "Timeout";
236 break;
237
238 // === DRAW CONDITIONS ===
239 case ENDGAME_REASON_STALEMATE:
240 printf("🤝 Game Result: DRAW - Stalemate\r\n");
241 end_reason = "Stalemate";
242 break;
243 case ENDGAME_REASON_50_MOVE:
244 printf("🤝 Game Result: DRAW - 50-move rule\r\n");
245 end_reason = "50-move rule";
246 break;
247 case ENDGAME_REASON_REPETITION:
248 printf("🤝 Game Result: DRAW - Three-fold repetition\r\n");
249 end_reason = "Threefold repetition";
250 break;
251 case ENDGAME_REASON_INSUFFICIENT:
252 printf("🤝 Game Result: DRAW - Insufficient material\r\n");
253 end_reason = "Insufficient material";
254 break;
255 case ENDGAME_REASON_AGREEMENT:
256 printf("🤝 Game Result: DRAW - Agreement\r\n");
257 end_reason = "Agreement";
258 break;
259 }
260
261 print_separator();
262 printf("📋 Game Summary:\r\n");
263 print_report_line("Winner", "%s", winner);
264 if (result_type != RESULT_DRAW_STALEMATE &&
265 result_type != RESULT_DRAW_50_MOVE &&
266 result_type != RESULT_DRAW_REPETITION &&
267 result_type != RESULT_DRAW_INSUFFICIENT &&
268 result_type != RESULT_DRAW_AGREEMENT) {
269 print_report_line("Loser", "%s", loser);
270 }
271 print_report_line("End Reason", "%s", end_reason);
272 print_report_line("Total Moves", "%" PRIu32, move_count);
273 print_report_line("Game Duration", "%" PRIu32 ":%02" PRIu32 " (mm:ss)",
274 minutes, seconds);
275
276 // Calculate material advantage
277 int white_material = 0, black_material = 0;
278 for (uint32_t i = 0; i < white_captured_count; i++) {
279 piece_t p = white_captured_pieces[i];
280 if (p == PIECE_BLACK_PAWN)
281 white_material += 1;
282 else if (p == PIECE_BLACK_KNIGHT || p == PIECE_BLACK_BISHOP)
283 white_material += 3;
284 else if (p == PIECE_BLACK_ROOK)
285 white_material += 5;
286 else if (p == PIECE_BLACK_QUEEN)
287 white_material += 9;
288 }
289 for (uint32_t i = 0; i < black_captured_count; i++) {
290 piece_t p = black_captured_pieces[i];
291 if (p == PIECE_WHITE_PAWN)
292 black_material += 1;
293 else if (p == PIECE_WHITE_KNIGHT || p == PIECE_WHITE_BISHOP)
294 black_material += 3;
295 else if (p == PIECE_WHITE_ROOK)
296 black_material += 5;
297 else if (p == PIECE_WHITE_QUEEN)
298 black_material += 9;
299 }
300 int material_diff = white_material - black_material;
301
302 print_separator();
303 printf("📊 Game Statistics:\r\n");
304 print_report_line("White Moves", "%" PRIu32, (move_count + 1) / 2);
305 print_report_line("Black Moves", "%" PRIu32, move_count / 2);
306 if (game_duration > 0 && move_count > 0) {
307 print_report_line("Avg Time per Move", "%" PRIu32 "s",
308 game_duration / move_count);
309 }
310
311 print_separator();
312 printf("⚔️ Combat Statistics:\r\n");
313 print_report_line("White Captures", "%" PRIu32 " pieces (%d points)",
314 white_captures, white_material);
315 print_report_line("Black Captures", "%" PRIu32 " pieces (%d points)",
316 black_captures, black_material);
317 if (material_diff > 0) {
318 print_report_line("Material Advantage", "White +%d", material_diff);
319 } else if (material_diff < 0) {
320 print_report_line("Material Advantage", "Black +%d", -material_diff);
321 } else {
322 print_report_line("Material Advantage", "Equal (0)");
323 }
324 print_report_line("White Checks", "%" PRIu32, white_checks);
325 print_report_line("Black Checks", "%" PRIu32, black_checks);
326
327 // Captured pieces visually
328 if (white_captured_count > 0) {
329 printf(" • White Captured: ");
330 for (uint32_t i = 0; i < white_captured_count && i < GAME_TASK_MAX_CAPTURED_PIECES;
331 i++) {
332 char piece_char = piece_to_char(white_captured_pieces[i]);
333 printf("%c ", piece_char);
334 }
335 printf("\r\n");
336 }
337 if (black_captured_count > 0) {
338 printf(" • Black Captured: ");
339 for (uint32_t i = 0; i < black_captured_count && i < GAME_TASK_MAX_CAPTURED_PIECES;
340 i++) {
341 char piece_char = piece_to_char(black_captured_pieces[i]);
342 printf("%c ", piece_char);
343 }
344 printf("\r\n");
345 }
346
347 // Castling info
348 print_report_line("White Castling", "%s", white_castles > 0 ? "Yes" : "No");
349 print_report_line("Black Castling", "%s", black_castles > 0 ? "Yes" : "No");
350
351 print_separator();
352 printf("📈 Overall Statistics:\r\n");
353 print_report_line("Total Games Played", "%" PRIu32, total_games);
354 print_report_line("White Wins", "%" PRIu32 " (%.0f%%)", white_wins,
355 total_games > 0 ? (white_wins * 100.0f / total_games)
356 : 0.0f);
357 print_report_line("Black Wins", "%" PRIu32 " (%.0f%%)", black_wins,
358 total_games > 0 ? (black_wins * 100.0f / total_games)
359 : 0.0f);
360 print_report_line("Draws", "%" PRIu32 " (%.0f%%)", draws,
361 total_games > 0 ? (draws * 100.0f / total_games) : 0.0f);
362
363 printf("═══════════════════════════════════════════════════════════════\r\n");
364 printf("\r\n");
365
366 // Flush stdout for immediate display
367 fflush(stdout);
368}
