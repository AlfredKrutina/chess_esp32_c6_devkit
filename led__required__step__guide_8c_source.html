20static void push_plan(void) {

48void led_required_step_guide_apply(const led_required_step_plan_t *plan) {
49 if (plan == NULL || !plan->active || plan->count == 0) {
50 led_required_step_guide_clear();
51 return;
52 }
53 bool overlay_blocks = led_gesture_anim_busy() || led_lamp_board_held();
54 if (!overlay_blocks && s_plan.active && s_plan.count == plan->count &&
55 s_plan.fade_ms == plan->fade_ms &&
56 memcmp(s_plan.px, plan->px,
57 (size_t)plan->count * sizeof(plan->px[0])) == 0 &&
58 s_pushed_gen == led_board_picture_gen()) {
59 return;
60 }
61 memcpy(&s_plan, plan, sizeof(s_plan));
62 if (overlay_blocks) {
63 s_need_push = true;
64 return;
65 }
66 push_plan();
67}
