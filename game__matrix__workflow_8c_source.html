44void game_process_promotion_command(const game_cmd_t *cmd) {
45 if (!cmd) {
46 ESP_LOGE(TAG, "❌ Invalid promotion command");
47 return;
48 }
49
50 ESP_LOGI(TAG, "👑 Processing promotion command");
51
52 // #2 & #3: Atomic check with mutex to prevent double execution
53 // Use a recursive mutex function ((*game_state_promotion_mutex()) is a recursive mutex)
54 if (xSemaphoreTakeRecursive((*game_state_promotion_mutex()), pdMS_TO_TICKS(100)) != pdTRUE) {
55 ESP_LOGE(TAG, "❌ Failed to acquire promotion mutex");
56 return;
57 }
58
59 // Check if promotion is still pending (might have been processed by another
60 // task)
61 if (!(*game_state_promotion_state()).pending) {
62 xSemaphoreGiveRecursive((*game_state_promotion_mutex()));
63 ESP_LOGW(TAG, "⚠️ Promotion already processed by another task");
64 return;
65 }
66
67 // CRITICAL: Keep mutex held during ENTIRE execution to prevent race!
68 // DO NOT release here - release only after completion or on error
69
70 // Determine promotion choice (default to Queen if invalid or not specified)
71 promotion_choice_t choice = PROMOTION_QUEEN;
72
73 {
74 char piece = (cmd->kind == GCMD_PROMOTION_CHOICE) ? cmd->u.promo.piece : 0;
75 if (piece == 'r' || piece == 'R') {
76 choice = PROMOTION_ROOK;
77 } else if (piece == 'b' || piece == 'B') {
78 choice = PROMOTION_BISHOP;
79 } else if (piece == 'n' || piece == 'N') {
80 choice = PROMOTION_KNIGHT;
81 } else {
82 choice = PROMOTION_QUEEN;
83 if (piece != 0 && piece != 'q' && piece != 'Q') {
84 ESP_LOGW(TAG, "Invalid promotion piece '%c', defaulting to Queen",
85 piece);
86 }
87 }
88 ESP_LOGI(TAG, "Promotion choice received: %d (%s)", choice,
89 choice == PROMOTION_QUEEN ? "Queen"
90 : choice == PROMOTION_ROOK ? "Rook"
91 : choice == PROMOTION_BISHOP ? "Bishop"
92 : "Knight");
93 }
94
95 // Execute the promotion with the selected piece
96 if (game_execute_promotion(choice)) {
97 ESP_LOGI(TAG, "✅ Promotion executed successfully");
98
99#if CONFIG_CHESS_MATRIX_INPUT_I2C_HALL
100 {
101 uint8_t promo_row = (*game_state_promotion_state()).square_row;
102 uint8_t promo_col = (*game_state_promotion_state()).square_col;
103 piece_t promoted_on_board =
104 game_board_piece_at(promo_row, promo_col);
105 if (!game_hall_validate_drop_square(promo_row, promo_col,
106 promoted_on_board)) {
107 (void)game_board_undo_promotion_choice();
108 ESP_LOGW(TAG,
109 "❌ Hall type/color mismatch on promo square %c%d after choice",
110 'a' + promo_col, promo_row + 1);
111 game_cmd_reply(
112 "❌ Wrong piece on promotion square (Hall type/color) — fix "
113 "physical piece and choose promotion again",
114 true, (QueueHandle_t)cmd->response_queue);
115 xSemaphoreGiveRecursive((*game_state_promotion_mutex()));
116 return;
117 }
118 }
119#endif
120
121 (*game_state_promotion_state()).pending = false;
122 game_move_undo_meta_on_promotion_completed(game_board_piece_at(
123 (*game_state_promotion_state()).square_row,
124 (*game_state_promotion_state()).square_col));
125#if CONFIG_CHESS_MATRIX_INPUT_I2C_HALL
126 game_hall_snapshot_last_valid();
127 game_hall_position_arm_idle_handover();
128#endif
129 game_print_board();
130 game_promotion_finish_turn();
131
132 } else {
133 ESP_LOGE(TAG, "❌ Failed to execute promotion");
134 }
135
136 // CRITICAL: Release mutex at the end of the function to ensure it's always
137 // released Use recursive mutex functions ((*game_state_promotion_mutex()) is a recursive mutex)
138 xSemaphoreGiveRecursive((*game_state_promotion_mutex()));
139}

164bool game_execute_promotion(promotion_choice_t choice) {
165 ESP_LOGI(TAG, "👑 Executing pawn promotion: %d", choice);
166
167 // Find the pawn that needs to be promoted
168 // This should be set when a pawn reaches the promotion rank
169 // For now, we'll implement a basic version
170
171 // Convert promotion choice to piece type
172 piece_t promoted_piece;
173 // Use (*game_state_promotion_state()).player instead of current_player
174 // current_player has already been switched to the opponent by
175 // game_execute_move() so we must use the player stored in the promotion
176 // state
177 if ((*game_state_promotion_state()).player == PLAYER_WHITE) {
178 switch (choice) {
179 case PROMOTION_QUEEN:
180 promoted_piece = PIECE_WHITE_QUEEN;
181 break;
182 case PROMOTION_ROOK:
183 promoted_piece = PIECE_WHITE_ROOK;
184 break;
185 case PROMOTION_BISHOP:
186 promoted_piece = PIECE_WHITE_BISHOP;
187 break;
188 case PROMOTION_KNIGHT:
189 promoted_piece = PIECE_WHITE_KNIGHT;
190 break;
191 default:
192 return false;
193 }
194 } else {
195 switch (choice) {
196 case PROMOTION_QUEEN:
197 promoted_piece = PIECE_BLACK_QUEEN;
198 break;
199 case PROMOTION_ROOK:
200 promoted_piece = PIECE_BLACK_ROOK;
201 break;
202 case PROMOTION_BISHOP:
203 promoted_piece = PIECE_BLACK_BISHOP;
204 break;
205 case PROMOTION_KNIGHT:
206 promoted_piece = PIECE_BLACK_KNIGHT;
207 break;
208 default:
209 return false;
210 }
211 }
212
213 // BUG #3: Use stored position instead of scanning entire board
214 // The position was already found and stored in (*game_state_promotion_state()) by
215 // game_check_promotion_needed()
216 uint8_t row = (*game_state_promotion_state()).square_row;
217 uint8_t col = (*game_state_promotion_state()).square_col;
218 piece_t piece = game_get_piece(row, col);
219
220 // Verify it's actually a pawn (safety check)
221 if (piece != PIECE_WHITE_PAWN && piece != PIECE_BLACK_PAWN) {
222 ESP_LOGE(TAG, "❌ Promotion failed: No pawn at stored position %c%d",
223 'a' + col, row + 1);
224 return false;
225 }
226
227 char promo_ch = 'q';
228 switch (choice) {
229 case PROMOTION_ROOK:
230 promo_ch = 'r';
231 break;
232 case PROMOTION_BISHOP:
233 promo_ch = 'b';
234 break;
235 case PROMOTION_KNIGHT:
236 promo_ch = 'n';
237 break;
238 default:
239 promo_ch = 'q';
240 break;
241 }
242 if (!game_board_apply_promotion_piece(promo_ch)) {
243 return false;
244 }
245 game_move_undo_meta_on_promotion_completed(promoted_piece);
246 current_game_state = GAME_STATE_ACTIVE; // Restore game state to ACTIVE
247 ESP_LOGI(TAG, "✅ Promoted %s pawn at %c%d to %s",
248 current_player == PLAYER_WHITE ? "white" : "black", 'a' + col,
249 row + 1,
250 choice == PROMOTION_QUEEN ? "Queen"
251 : choice == PROMOTION_ROOK ? "Rook"
252 : choice == PROMOTION_BISHOP ? "Bishop"
253 : "Knight");
254
255 // Start promotion animation
256 uint8_t promotion_led = chess_pos_to_led_index(row, col);
257 game_led_flush();
258 game_event_post_board_anim(GAME_EVT_ANIM_PROMOTE, promotion_led, 0);
259
260 return true;
261}

266void game_highlight_movable_pieces(void) {
267#if HALL_DEBUG_LIGHT_WHERE_PIECE
268 return;
269#endif
270 if (game_led_lamp_held() || game_led_gesture_busy()) {
271 return;
272 }

274 return;
275 }
276 // Do not highlight pieces if system is booting
277 if (game_led_boot_active()) {
278 return;
279 }
280
281 // Do not highlight movable pieces if a piece is already
282 // lifted This prevents overwriting the "valid moves" (green) with "movable
283 // pieces" (yellow)
284 if (piece_lifted) {
285 ESP_LOGD(TAG, "Highlight skipped - piece is lifted");
286 return;
287 }
288
289 // Do not run the highlight during casting - casting has its own LED indication
290 if (castling_state.in_progress) {
291 ESP_LOGD(TAG, "Highlight skipped - castling in progress");
292 return;

294
295 if (!game_led_guidance_show_movable_yellow()) {
296 ESP_LOGD(TAG, "Movable piece highlights skipped (LED guidance level %u)",
297 (unsigned)game_get_led_guidance_level());
298 return;
299 }
300
301#if CONFIG_CHESS_MATRIX_INPUT_I2C_HALL
302 /* Logical pieces must not light an empty or not-yet-locked Hall board. */
303 if (!hall_i2c_squares_readings_usable()) {
304 return;
305 }
306 {
307 uint8_t occ[64] = {0};
308 bool any_piece = false;
309 squares_get_state(occ);
310 for (int i = 0; i < 64; i++) {
311 if (occ[i] != 0) {
312 any_piece = true;
313 break;
314 }
315 }
316 if (!any_piece) {
317 return;
318 }
319 }
320#endif
321
322 ESP_LOGI(TAG, "🟡 Highlighting movable pieces for %s player",
323 current_player == PLAYER_WHITE ? "white" : "black");
324 const bool nested_frame = game_led_frame_is_open();
325 if (!nested_frame) {
326 game_led_frame_begin(0, false);
327 }
328
329 // Endgame detection is handled in game_analyze_position() called from
330 // game_execute_move()
331
332 // LED command queue removed - using direct LED calls now
333
334 uint32_t highlighted_count = 0;
335
336 // Scan all squares for current player's pieces
337 for (uint8_t row = 0; row < 8; row++) {
338 for (uint8_t col = 0; col < 8; col++) {
339 piece_t piece = game_get_piece(row, col);
340
341 // Check if piece belongs to current player
342 bool is_current_player_piece = false;
343 if (current_player == PLAYER_WHITE) {
344 is_current_player_piece =
345 (piece >= PIECE_WHITE_PAWN && piece <= PIECE_WHITE_KING);
346 } else {
347 is_current_player_piece =
348 (piece >= PIECE_BLACK_PAWN && piece <= PIECE_BLACK_KING);
349 }
350
351 if (is_current_player_piece) {
352 // Check if piece has valid moves
353 // OPTIMIZATION: Reduced buffer size from 64 to 1 to prevent stack
354 // overflow in Timer Task We only need to know if count > 0, so asking
355 // for 1 move is enough
356 move_suggestion_t suggestions[1];
357 uint32_t valid_moves =
358 game_get_available_moves(row, col, suggestions, 1);
359
360 if (valid_moves > 0) {
361 game_led_frame_sq(row, col, 255, 255, 0);
362 highlighted_count++;
363 }
364 }
365 }
366 }
367
368 if (!nested_frame) {
369 game_led_frame_end();
370 }
371 ESP_LOGI(TAG, "🟡 Highlighted %lu movable pieces", highlighted_count);
