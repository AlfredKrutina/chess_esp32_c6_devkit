35void game_process_promotion_command(const chess_move_command_t *cmd) {
36 if (!cmd) {
37 ESP_LOGE(TAG, "❌ Invalid promotion command");
38 return;
39 }
40
41 ESP_LOGI(TAG, "👑 Processing promotion command");
42
43 // #2 & #3: Atomic check with mutex to prevent double execution
44 // Use a recursive mutex function (promotion_mutex is a recursive mutex)
45 if (xSemaphoreTakeRecursive(promotion_mutex, pdMS_TO_TICKS(100)) != pdTRUE) {
46 ESP_LOGE(TAG, "❌ Failed to acquire promotion mutex");
47 return;
48 }
49
50 // Check if promotion is still pending (might have been processed by another
51 // task)
52 if (!promotion_state.pending) {
53 xSemaphoreGiveRecursive(promotion_mutex);
54 ESP_LOGW(TAG, "⚠️ Promotion already processed by another task");
55 return;
56 }
57
58 // CRITICAL: Keep mutex held during ENTIRE execution to prevent race!
59 // DO NOT release here - release only after completion or on error
60
61 // Determine promotion choice (default to Queen if invalid or not specified)
62 promotion_choice_t choice = PROMOTION_QUEEN;
63
64 if (cmd->promotion_choice <= PROMOTION_KNIGHT) {
65 choice = (promotion_choice_t)cmd->promotion_choice;
66 ESP_LOGI(TAG, "👑 Promotion choice received: %d (%s)", choice,
67 choice == PROMOTION_QUEEN ? "Queen"
68 : choice == PROMOTION_ROOK ? "Rook"
69 : choice == PROMOTION_BISHOP ? "Bishop"
70 : "Knight");
71 } else {
72 ESP_LOGW(TAG,
73 "⚠️ Invalid or missing promotion choice (%d), defaulting to Queen",
74 cmd->promotion_choice);
75 }
76
77 // Execute the promotion with the selected piece
78 if (game_execute_promotion(choice)) {
79 ESP_LOGI(TAG, "✅ Promotion executed successfully");
80
81#if CONFIG_CHESS_MATRIX_INPUT_I2C_HALL
82 {
83 uint8_t promo_row = promotion_state.square_row;
84 uint8_t promo_col = promotion_state.square_col;
85 piece_t promoted_on_board = board[promo_row][promo_col];
86 if (!game_hall_validate_drop_square(promo_row, promo_col,
87 promoted_on_board)) {
88 piece_t revert_pawn = (promotion_state.player == PLAYER_WHITE)
89 ? PIECE_WHITE_PAWN
90 : PIECE_BLACK_PAWN;
91 board[promo_row][promo_col] = revert_pawn;
92 ESP_LOGW(TAG,
93 "❌ Hall type/color mismatch on promo square %c%d after choice",
94 'a' + promo_col, promo_row + 1);
95 game_send_response_to_uart(
96 "❌ Wrong piece on promotion square (Hall type/color) — fix "
97 "physical piece and choose promotion again",
98 true, (QueueHandle_t)cmd->response_queue);
99 xSemaphoreGiveRecursive(promotion_mutex);
100 return;
101 }
102 }
103#endif
104
105 promotion_state.pending = false;
106 game_print_board();
107 game_promotion_finish_turn();
108
109 } else {
110 ESP_LOGE(TAG, "❌ Failed to execute promotion");
111 }
112
113 // CRITICAL: Release mutex at the end of the function to ensure it's always
114 // released Use recursive mutex functions (promotion_mutex is a recursive mutex)
115 xSemaphoreGiveRecursive(promotion_mutex);
116}

141bool game_execute_promotion(promotion_choice_t choice) {
142 ESP_LOGI(TAG, "👑 Executing pawn promotion: %d", choice);
143
144 // Find the pawn that needs to be promoted
145 // This should be set when a pawn reaches the promotion rank
146 // For now, we'll implement a basic version
147
148 // Convert promotion choice to piece type
149 piece_t promoted_piece;
150 // Use promotion_state.player instead of current_player
151 // current_player has already been switched to the opponent by
152 // game_execute_move() so we must use the player stored in the promotion
153 // state
154 if (promotion_state.player == PLAYER_WHITE) {
155 switch (choice) {
156 case PROMOTION_QUEEN:
157 promoted_piece = PIECE_WHITE_QUEEN;
158 break;
159 case PROMOTION_ROOK:
160 promoted_piece = PIECE_WHITE_ROOK;
161 break;
162 case PROMOTION_BISHOP:
163 promoted_piece = PIECE_WHITE_BISHOP;
164 break;
165 case PROMOTION_KNIGHT:
166 promoted_piece = PIECE_WHITE_KNIGHT;
167 break;
168 default:
169 return false;
170 }
171 } else {
172 switch (choice) {
173 case PROMOTION_QUEEN:
174 promoted_piece = PIECE_BLACK_QUEEN;
175 break;
176 case PROMOTION_ROOK:
177 promoted_piece = PIECE_BLACK_ROOK;
178 break;
179 case PROMOTION_BISHOP:
180 promoted_piece = PIECE_BLACK_BISHOP;
181 break;
182 case PROMOTION_KNIGHT:
183 promoted_piece = PIECE_BLACK_KNIGHT;
184 break;
185 default:
186 return false;
187 }
188 }
189
190 // BUG #3: Use stored position instead of scanning entire board
191 // The position was already found and stored in promotion_state by
192 // game_check_promotion_needed()
193 uint8_t row = promotion_state.square_row;
194 uint8_t col = promotion_state.square_col;
195 piece_t piece = board[row][col];
196
197 // Verify it's actually a pawn (safety check)
198 if (piece != PIECE_WHITE_PAWN && piece != PIECE_BLACK_PAWN) {
199 ESP_LOGE(TAG, "❌ Promotion failed: No pawn at stored position %c%d",
200 'a' + col, row + 1);
201 return false;
202 }
203
204 // Promote the pawn
205 board[row][col] = promoted_piece;
206 game_move_undo_meta_on_promotion_completed(promoted_piece);
207 current_game_state = GAME_STATE_ACTIVE; // Restore game state to ACTIVE
208 ESP_LOGI(TAG, "✅ Promoted %s pawn at %c%d to %s",
209 current_player == PLAYER_WHITE ? "white" : "black", 'a' + col,
210 row + 1,
211 choice == PROMOTION_QUEEN ? "Queen"
212 : choice == PROMOTION_ROOK ? "Rook"
213 : choice == PROMOTION_BISHOP ? "Bishop"
214 : "Knight");
215
216 // Start promotion animation
217 uint8_t promotion_led = chess_pos_to_led_index(row, col);
218 led_command_t promote_cmd = {.type = LED_CMD_ANIM_PROMOTE,
219 .led_index = promotion_led,
220 .red = 255,
221 .green = 215,
222 .blue = 0, // Gold
223 .duration_ms = 2000,
224 .data = NULL};
225 led_execute_command_new(&promote_cmd);
226
227 return true;
228}

234#if HALL_DEBUG_LIGHT_WHERE_PIECE

237 if (led_lamp_board_held() || led_gesture_anim_busy()) {

287#endif

289 ESP_LOGI(TAG, "🟡 Highlighting movable pieces for %s player",
290 current_player == PLAYER_WHITE ? "white" : "black");
291
292 // Endgame detection is handled in game_analyze_position() called from
293 // game_execute_move()
294
295 // LED command queue removed - using direct LED calls now
296
297 uint32_t highlighted_count = 0;
298
299 // Scan all squares for current player's pieces
300 for (uint8_t row = 0; row < 8; row++) {
301 for (uint8_t col = 0; col < 8; col++) {
302 piece_t piece = board[row][col];
303
304 // Check if piece belongs to current player
305 bool is_current_player_piece = false;
306 if (current_player == PLAYER_WHITE) {
307 is_current_player_piece =
308 (piece >= PIECE_WHITE_PAWN && piece <= PIECE_WHITE_KING);
309 } else {
310 is_current_player_piece =
311 (piece >= PIECE_BLACK_PAWN && piece <= PIECE_BLACK_KING);
312 }
313
314 if (is_current_player_piece) {
315 // Check if piece has valid moves
316 // OPTIMIZATION: Reduced buffer size from 64 to 1 to prevent stack
317 // overflow in Timer Task We only need to know if count > 0, so asking
318 // for 1 move is enough
319 move_suggestion_t suggestions[1];
320 uint32_t valid_moves =
321 game_get_available_moves(row, col, suggestions, 1);
322
323 if (valid_moves > 0) {
324 // Highlight movable piece in yellow
325 led_set_pixel_safe(chess_pos_to_led_index(row, col), 255, 255,
326 0); // Yellow
327 highlighted_count++;
328 }
329 }
330 }
331 }
332
333 ESP_LOGI(TAG, "🟡 Highlighted %lu movable pieces", highlighted_count);
334}
