44static uint8_t game_count_mask_bits(uint32_t low, uint32_t high) {
45 uint8_t count = 0;
46 while (low != 0) {
47 count += (uint8_t)(low & 1U);
48 low >>= 1;
49 }
50 while (high != 0) {
51 count += (uint8_t)(high & 1U);
52 high >>= 1;
53 }
54 return count;
55}

135void game_squares_guard_render_leds(void) {
136 if (game_demo_show_active()) {
137 return;
138 }
139#if HALL_DEBUG_LIGHT_WHERE_PIECE
140 return;
141#endif
142 if (!squares_guard_pause_state.active) {
143 return;
144 }
145 if (!chess_policy_squares_guard_led_enabled()) {
146 return;
147 }
148
149#if CONFIG_CHESS_MATRIX_INPUT_I2C_HALL
150 game_hall_restore_guide_sync();
151 if (game_hall_restore_guide_is_active()) {
152 return;
153 }
154#endif
155
156 led_clear_board_only();
157 uint8_t phys[64] = {0};
158 squares_get_state(phys);
159
160 bool any_piece = false;
161 for (uint8_t i = 0; i < 64; i++) {
162 if (phys[i] != 0) {
163 any_piece = true;
164 break;
165 }
166 }
167 if (!any_piece) {
168 return;
169 }
170
171 for (uint8_t square = 0; square < 64; square++) {
172 uint8_t row = square / 8;
173 uint8_t col = square % 8;
174 piece_t piece = board[row][col];
175 if (game_squares_sensor_matches_engine(square, phys)) {
176 continue;
177 }
178
179 if (piece >= PIECE_WHITE_PAWN && piece <= PIECE_WHITE_KING) {
180 if (chess_policy_mg_led_white_yellow()) {
181 led_set_pixel_safe(chess_pos_to_led_index(row, col), 255, 255, 0);
182 }
183 } else if (piece >= PIECE_BLACK_PAWN && piece <= PIECE_BLACK_KING) {
184 if (chess_policy_mg_led_black_blue()) {
185 led_set_pixel_safe(chess_pos_to_led_index(row, col), 0, 0, 255);
186 }
187 } else if (phys[square] != 0) {
188 if (chess_policy_mg_led_ghost_orange()) {
189 led_set_pixel_safe(chess_pos_to_led_index(row, col), 255, 140, 0);
190 }
191 } else if (chess_policy_mg_led_missing_white()) {
192 led_set_pixel_safe(chess_pos_to_led_index(row, col), 220, 220, 220);
193 }
194 }

197void game_squares_guard_handle_command(const chess_move_command_t *cmd) {
198 if (cmd == NULL || cmd->type != GAME_CMD_SQUARES_GUARD) {
199 return;
200 }
201
202 uint8_t action = cmd->timer_data.squares_guard.action;
203 if (action == 0) {
204 game_squares_guard_restore_after_clear(true);
205 STAGING_LOGI(TAG, "Matrix guard cleared");
206 return;
207 }
208
209 if (!chess_policy_squares_guard_enabled()) {
210 STAGING_LOGI(TAG, "Matrix guard command ignored (disabled in menuconfig)");
211 squares_abort_ambiguous_guard_baseline();
212 return;
213 }
214
215 if (game_task_squares_guard_mode_conflict_active()) {
216 STAGING_LOGI(TAG,
217 "Matrix guard ignored (special mode) — clearing matrix local "
218 "guard so UP/DN continues");
219 if (squares_guard_pause_state.active) {
220 memset(&squares_guard_pause_state, 0, sizeof(squares_guard_pause_state));
221 resync_required_after_restore = false;
222 }
223 squares_abort_ambiguous_guard_baseline();
224 return;
225 }
226
227 squares_guard_pause_state.active = true;
228 squares_guard_pause_state.lifted_mask_low =
229 cmd->timer_data.squares_guard.lifted_mask_low;
230 squares_guard_pause_state.lifted_mask_high =
231 cmd->timer_data.squares_guard.lifted_mask_high;
232 squares_guard_pause_state.dropped_mask_low =
233 cmd->timer_data.squares_guard.dropped_mask_low;
234 squares_guard_pause_state.dropped_mask_high =
235 cmd->timer_data.squares_guard.dropped_mask_high;
236 squares_guard_pause_state.conflict_count = game_count_mask_bits(
237 squares_guard_pause_state.lifted_mask_low |
238 squares_guard_pause_state.dropped_mask_low,
239 squares_guard_pause_state.lifted_mask_high |
240 squares_guard_pause_state.dropped_mask_high);
241
242 if (chess_policy_squares_guard_should_freeze()) {
243 game_task_squares_guard_freeze_move_flow();
244 }
245
246 uint8_t expected[64];
247 game_board_to_occupancy(expected);
248 squares_guard_apply_expected_occupancy(expected);
249
250 STAGING_LOGI(TAG, "Matrix guard active, conflict_count=%u",
251 squares_guard_pause_state.conflict_count);
252 game_squares_guard_render_leds();
253}

255void game_squares_guard_check_resync_after_restore(void) {
256#if CONFIG_CHESS_MATRIX_INPUT_I2C_HALL
257 if (!hall_i2c_squares_readings_usable()) {
258 return;
259 }
260#endif
261 if (!chess_policy_squares_guard_enabled() ||
262 !chess_policy_squares_guard_nvs_resync_enabled()) {
263 return;
264 }
265 if (game_is_opening_trainer_active() || game_is_opening_trainer_setup_active()) {
266 return;
267 }
268 uint8_t squares_state_now[64] = {0};
269 squares_get_state(squares_state_now);
270
271 uint32_t lifted_low = 0, lifted_high = 0;
272 uint8_t conflicts = 0;
273 for (uint8_t i = 0; i < 64; i++) {
274 if (game_squares_sensor_matches_engine(i, squares_state_now)) {
275 continue;
276 }
277 if (i < 32) {
278 lifted_low |= (1UL << i);
279 } else {
280 lifted_high |= (1UL << (i - 32));
281 }
282 conflicts++;
283 }
284
285 if (conflicts == 0) {
286 resync_required_after_restore = false;
287 return;
288 }
289
290 squares_guard_pause_state.active = true;
291 squares_guard_pause_state.lifted_mask_low = lifted_low;
292 squares_guard_pause_state.lifted_mask_high = lifted_high;
293 squares_guard_pause_state.dropped_mask_low = 0;
294 squares_guard_pause_state.dropped_mask_high = 0;
295 squares_guard_pause_state.conflict_count = conflicts;
296 resync_required_after_restore = true;
297
298 uint8_t expected[64];
299 game_board_to_occupancy(expected);
300 squares_guard_apply_expected_occupancy(expected);
301
302 game_squares_guard_render_leds();
303}
