91static uint8_t game_count_mask_bits(uint32_t low, uint32_t high) {
92 uint8_t count = 0;
93 while (low != 0) {
94 count += (uint8_t)(low & 1U);
95 low >>= 1;
96 }
97 while (high != 0) {
98 count += (uint8_t)(high & 1U);
99 high >>= 1;
100 }
101 return count;
102}

183void game_squares_guard_render_leds(void) {
184 if (game_demo_show_active()) {
185 return;
186 }
187#if HALL_DEBUG_LIGHT_WHERE_PIECE
188 return;
189#endif
190 if (!squares_guard_pause_state.active) {
191 return;
192 }
193 if (!game_policy_squares_guard_led_enabled()) {
194 return;
195 }
196
197#if CONFIG_CHESS_MATRIX_INPUT_I2C_HALL
198 /* Yellow and blue here mean "a piece should stand on this square".
199 * On Hall that locked the board after boot. The opening guide draws
200 * those homes instead, and only before the first move. */
201 game_hall_restore_guide_sync();
202 return;
203#else
204 game_led_clear_board();
205 uint8_t phys[64] = {0};
206 squares_get_state(phys);
207
208 bool any_piece = false;
209 for (uint8_t i = 0; i < 64; i++) {
210 if (phys[i] != 0) {
211 any_piece = true;
212 break;
213 }
214 }
215 if (!any_piece) {
216 return;
217 }
218
219 for (uint8_t square = 0; square < 64; square++) {
220 uint8_t row = square / 8;
221 uint8_t col = square % 8;
222 piece_t piece = game_get_piece(row, col);
223 if (game_squares_sensor_matches_engine(square, phys)) {
224 continue;
225 }
226
227 if (piece >= PIECE_WHITE_PAWN && piece <= PIECE_WHITE_KING) {
228 if (game_policy_mg_led_white_yellow()) {
229 game_led_frame_sq(row, col, 255, 255, 0);
230 }
231 } else if (piece >= PIECE_BLACK_PAWN && piece <= PIECE_BLACK_KING) {
232 if (game_policy_mg_led_black_blue()) {
233 game_led_frame_sq(row, col, 0, 0, 255);
234 }
235 } else if (phys[square] != 0) {
236 if (game_policy_mg_led_ghost_orange()) {
237 game_led_frame_sq(row, col, 255, 140, 0);
238 }
239 } else if (game_policy_mg_led_missing_white()) {
240 game_led_frame_sq(row, col, 220, 220, 220);
241 }
242 }
243#endif
244}

246void game_squares_guard_handle_command(const game_cmd_t *cmd) {
247 if (cmd == NULL || cmd->kind != GCMD_BOARD_GUARD) {
248 return;
249 }
250
251 uint8_t action = cmd->u.guard.action;
252 if (action == 0) {
253 game_squares_guard_restore_after_clear(true);
254 STAGING_LOGI(TAG, "Matrix guard cleared");
255 return;
256 }
257
258 if (!game_policy_squares_guard_enabled()) {
259 STAGING_LOGI(TAG, "Matrix guard command ignored (disabled in menuconfig)");
260 squares_abort_ambiguous_guard_baseline();
261 return;
262 }
263
264#if CONFIG_CHESS_MATRIX_INPUT_I2C_HALL
265 /* In play, a multi-square anomaly must not freeze occupancy. The Hall
266 * guide walks the pieces back to the last legal position. */
267 if (hall_i2c_squares_readings_usable() &&
268 game_hall_restore_guide_in_play() &&
269 !game_task_squares_guard_mode_conflict_active() &&
270 (current_game_state == GAME_STATE_ACTIVE ||
271 current_game_state == GAME_STATE_PLAYING)) {
272 squares_abort_ambiguous_guard_baseline();
273 game_hall_position_enter_chaos_restore();
274 if (game_hall_position_in_chaos_restore()) {
275 game_squares_guard_enter_pause_from_hall_guide();
276 }
277 return;
278 }
279 if (hall_i2c_squares_readings_usable() &&
280 game_hall_position_mismatch_count() <= 4) {
281 STAGING_LOGI(TAG, "Matrix guard left to Hall restore (mismatch<=4)");
282 squares_abort_ambiguous_guard_baseline();
283 return;
284 }
285#endif
286
287 if (game_task_squares_guard_mode_conflict_active()) {
288 STAGING_LOGI(TAG,
289 "Matrix guard ignored (special mode) — clearing matrix local "
290 "guard so UP/DN continues");
291 if (squares_guard_pause_state.active) {
292 memset(&squares_guard_pause_state, 0, sizeof(squares_guard_pause_state));
293 resync_required_after_restore = false;
294 }
295 squares_abort_ambiguous_guard_baseline();
296 return;
297 }
298
299 squares_guard_pause_state.active = true;
300 squares_guard_pause_state.lifted_mask_low =
301 cmd->u.guard.lifted_low;
302 squares_guard_pause_state.lifted_mask_high =
303 cmd->u.guard.lifted_high;
304 squares_guard_pause_state.dropped_mask_low =
305 cmd->u.guard.dropped_low;
306 squares_guard_pause_state.dropped_mask_high =
307 cmd->u.guard.dropped_high;
308 squares_guard_pause_state.conflict_count = game_count_mask_bits(
309 squares_guard_pause_state.lifted_mask_low |
310 squares_guard_pause_state.dropped_mask_low,
311 squares_guard_pause_state.lifted_mask_high |
312 squares_guard_pause_state.dropped_mask_high);
313
314 if (game_policy_squares_guard_should_freeze()) {
315 game_task_squares_guard_freeze_move_flow();
316 }
317
318 uint8_t expected[64];
319 game_board_to_occupancy(expected);
320 squares_guard_apply_expected_occupancy(expected);
321
322 STAGING_LOGI(TAG, "Matrix guard active, conflict_count=%u",
323 squares_guard_pause_state.conflict_count);
324 guard_mode_hold();
325 game_squares_guard_render_leds();
326}

328void game_squares_guard_check_resync_after_restore(void) {
329#if CONFIG_CHESS_MATRIX_INPUT_I2C_HALL
330 /* Restoring a saved game must not paint blue on every unseen black home
331 * and then refuse lifts until those homes read as occupied. */
332 return;
333#endif
334 if (!game_policy_squares_guard_enabled() ||
335 !game_policy_squares_guard_nvs_resync_enabled()) {
336 return;
337 }
338 if (game_is_opening_trainer_active() || game_is_opening_trainer_setup_active()) {
339 return;
340 }
341 uint8_t squares_state_now[64] = {0};
342 squares_get_state(squares_state_now);
343
344 uint32_t lifted_low = 0, lifted_high = 0;
345 uint8_t conflicts = 0;
346 for (uint8_t i = 0; i < 64; i++) {
347 if (game_squares_sensor_matches_engine(i, squares_state_now)) {
348 continue;
349 }
350 if (i < 32) {
351 lifted_low |= (1UL << i);
352 } else {
353 lifted_high |= (1UL << (i - 32));
354 }
355 conflicts++;
356 }
357
358 if (conflicts == 0) {
359 resync_required_after_restore = false;
360 return;
361 }
362
363 squares_guard_pause_state.active = true;
364 squares_guard_pause_state.lifted_mask_low = lifted_low;
365 squares_guard_pause_state.lifted_mask_high = lifted_high;
366 squares_guard_pause_state.dropped_mask_low = 0;
367 squares_guard_pause_state.dropped_mask_high = 0;
368 squares_guard_pause_state.conflict_count = conflicts;
369 resync_required_after_restore = true;
370
371 uint8_t expected[64];
372 game_board_to_occupancy(expected);
373 squares_guard_apply_expected_occupancy(expected);
374
375 guard_mode_hold();
376 game_squares_guard_render_leds();
377}

379void game_squares_guard_try_clear_from_matrix(void) {

401 STAGING_LOGI(TAG, "Matrix guard cleared (game + matrix layers synced)");

404void game_squares_guard_enter_pause_from_hall_guide(void) {
405 if (!game_policy_squares_guard_enabled()) {
406 return;
407 }
408 if (squares_guard_pause_state.active) {

413 squares_guard_pause_state.lifted_mask_high = 0;
414 squares_guard_pause_state.dropped_mask_low = 0;
415 squares_guard_pause_state.dropped_mask_high = 0;
416 squares_guard_pause_state.conflict_count = 1;
417
418 /* Do not call squares_guard_apply_expected_occupancy(). Occupancy often
419 * already matches, and the scan would send action=0, which clears this
420 * hold and the guide enters it again. */
421 if (game_policy_squares_guard_should_freeze()) {
422 game_task_squares_guard_freeze_move_flow();
423 }
424 guard_mode_hold();
425 STAGING_LOGI(TAG, "Matrix guard entered from Hall restore guide");
426}

430 return false;
431 }
