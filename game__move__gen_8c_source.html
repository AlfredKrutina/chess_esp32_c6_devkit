66bool game_is_square_attacked(uint8_t row, uint8_t col, player_t by_player) {
67 // Check pawn attacks
68 int pawn_dir = (by_player == PLAYER_WHITE) ? 1 : -1;
69 int pawn_row = row - pawn_dir;
70 piece_t attacking_pawn =
71 (by_player == PLAYER_WHITE) ? PIECE_WHITE_PAWN : PIECE_BLACK_PAWN;
72
73 if (game_is_valid_square(pawn_row, col - 1) &&
74 game_get_piece(pawn_row, col - 1) == attacking_pawn)
75 return true;
76 if (game_is_valid_square(pawn_row, col + 1) &&
77 game_get_piece(pawn_row, col + 1) == attacking_pawn)
78 return true;
79
80 // Check knight attacks
81 piece_t attacking_knight =
82 (by_player == PLAYER_WHITE) ? PIECE_WHITE_KNIGHT : PIECE_BLACK_KNIGHT;
83 for (int i = 0; i < 8; i++) {
84 int nr = row + knight_moves[i][0];
85 int nc = col + knight_moves[i][1];
86 if (game_is_valid_square(nr, nc) && game_get_piece(nr, nc) == attacking_knight) {
87 return true;
88 }
89 }
90
91 // Check bishop/queen diagonal attacks
92 piece_t attacking_bishop =
93 (by_player == PLAYER_WHITE) ? PIECE_WHITE_BISHOP : PIECE_BLACK_BISHOP;
94 piece_t attacking_queen =
95 (by_player == PLAYER_WHITE) ? PIECE_WHITE_QUEEN : PIECE_BLACK_QUEEN;
96
97 for (int d = 0; d < 4; d++) {
98 int dr = bishop_dirs[d][0];
99 int dc = bishop_dirs[d][1];
100
101 for (int nr = row + dr, nc = col + dc; game_is_valid_square(nr, nc);
102 nr += dr, nc += dc) {
103 piece_t piece = game_get_piece(nr, nc);
104 if (piece == PIECE_EMPTY)
105 continue;
106
107 if (piece == attacking_bishop || piece == attacking_queen) {
108 return true;
109 }
110 break; // Blocked by any piece
111 }
112 }
113
114 // Check rook/queen orthogonal attacks
115 piece_t attacking_rook =
116 (by_player == PLAYER_WHITE) ? PIECE_WHITE_ROOK : PIECE_BLACK_ROOK;
117
118 for (int d = 0; d < 4; d++) {
119 int dr = rook_dirs[d][0];
120 int dc = rook_dirs[d][1];
121
122 for (int nr = row + dr, nc = col + dc; game_is_valid_square(nr, nc);
123 nr += dr, nc += dc) {
124 piece_t piece = game_get_piece(nr, nc);
125 if (piece == PIECE_EMPTY)
126 continue;
127
128 if (piece == attacking_rook || piece == attacking_queen) {
129 return true;
130 }
131 break; // Blocked by any piece
132 }
133 }
134
135 // Check king attacks
136 piece_t attacking_king =
137 (by_player == PLAYER_WHITE) ? PIECE_WHITE_KING : PIECE_BLACK_KING;
138 for (int i = 0; i < 8; i++) {
139 int nr = row + king_moves[i][0];
140 int nc = col + king_moves[i][1];
141 if (game_is_valid_square(nr, nc) && game_get_piece(nr, nc) == attacking_king) {
142 return true;
143 }
144 }
145
146 return false;
147}

156bool game_simulate_move_check(chess_move_extended_t *move, player_t player) {
157 position_t *scratch = game_logic_scratch_position();
158 position_move_t mv;
159
160 (void)player;
161 if (move == NULL || scratch == NULL) {
162 return false;
163 }
164 game_logic_copy_globals(scratch);
165 if (scratch->board[move->from_row][move->from_col] == POS_EMPTY &&
166 move->piece != PIECE_EMPTY) {
167 scratch->board[move->from_row][move->from_col] = (uint8_t)move->piece;
168 }
169 mv.from = (uint8_t)(move->from_row * 8u + move->from_col);
170 mv.to = (uint8_t)(move->to_row * 8u + move->to_col);
171 mv.promo = POS_PROMO_NONE;
172 if (move->move_type == MOVE_TYPE_PROMOTION) {
173 switch (move->promotion_piece) {
174 case PROMOTION_ROOK:
175 mv.promo = POS_PROMO_R;
176 break;
177 case PROMOTION_BISHOP:
178 mv.promo = POS_PROMO_B;
179 break;
180 case PROMOTION_KNIGHT:
181 mv.promo = POS_PROMO_N;
182 break;
183 default:
184 mv.promo = POS_PROMO_Q;
185 break;
186 }
187 } else if ((move->piece == PIECE_WHITE_PAWN && move->to_row == 7) ||
188 (move->piece == PIECE_BLACK_PAWN && move->to_row == 0)) {
189 mv.promo = POS_PROMO_Q;
190 }
191 return position_apply(scratch, mv);
192}

579uint32_t game_generate_legal_moves(player_t player) {
580 legal_moves_count = 0;
581
582 for (int row = 0; row < 8; row++) {
583 for (int col = 0; col < 8; col++) {
584 piece_t piece = game_get_piece(row, col);
585
586 if (!game_is_own_piece(piece, player))
587 continue;
588
589 switch (piece) {
590 case PIECE_WHITE_PAWN:
591 case PIECE_BLACK_PAWN:
592 game_generate_pawn_moves(row, col, player);
593 break;
594
595 case PIECE_WHITE_KNIGHT:
596 case PIECE_BLACK_KNIGHT:
597 game_generate_knight_moves(row, col, player);
598 break;
599
600 case PIECE_WHITE_BISHOP:
601 case PIECE_BLACK_BISHOP:
602 game_generate_sliding_moves(row, col, player, bishop_dirs, 4);
603 break;
604
605 case PIECE_WHITE_ROOK:
606 case PIECE_BLACK_ROOK:
607 game_generate_sliding_moves(row, col, player, rook_dirs, 4);
608 break;
609
610 case PIECE_WHITE_QUEEN:
611 case PIECE_BLACK_QUEEN:
612 game_generate_sliding_moves(row, col, player, bishop_dirs, 4);
613 game_generate_sliding_moves(row, col, player, rook_dirs, 4);
614 break;
615
616 case PIECE_WHITE_KING:
617 case PIECE_BLACK_KING:
618 game_generate_king_moves(row, col, player);
619 break;
620
621 default:
622 break;
623 }
624 }
625 }
626
627 return legal_moves_count;
628}

636move_error_t game_validate_move_enhanced(uint8_t from_row, uint8_t from_col,
637 uint8_t to_row, uint8_t to_col) {
638 // Basic coordinate validation
639 if (!game_is_valid_square(from_row, from_col) ||
640 !game_is_valid_square(to_row, to_col)) {
641 return MOVE_ERROR_INVALID_COORDINATES;
642 }
643
644 // Check if there's a piece at source
645 piece_t piece = game_get_piece(from_row, from_col);
646 if (piece == PIECE_EMPTY) {
647 return MOVE_ERROR_NO_PIECE;
648 }
649
650 // Check if it's player's piece
651 if (!game_is_own_piece(piece, current_player)) {
652 return MOVE_ERROR_WRONG_COLOR;
653 }
654
655 // Generate all legal moves and check if this move is in the list
656 uint32_t num_legal = game_generate_legal_moves(current_player);
657
658 for (uint32_t i = 0; i < num_legal; i++) {
659 chess_move_extended_t *move = &legal_moves_buffer[i];
660 if (move->from_row == from_row && move->from_col == from_col &&
661 move->to_row == to_row && move->to_col == to_col) {
662 return MOVE_ERROR_NONE; // Move is legal
663 }
664 }
665
666 return MOVE_ERROR_ILLEGAL_MOVE;
