93void game_handle_invalid_move(move_error_t error, const chess_move_t *move) {
94 ESP_LOGI(TAG, "🚨 ENTERING game_handle_invalid_move - error: %d", error);
95
96 if (!game_policy_error_recovery_enabled()) {
97 ESP_LOGW(TAG, "Invalid move ignored (error recovery disabled)");
98 return;
99 }
100
101 // CRITICAL FIX: Move pointer security check
102 if (!move) {
103 ESP_LOGE(TAG, "❌ Critical error: NULL move pointer in error handling");
104 return;
105 }
106
107 // Checking the validity of coordinates before use
108 if (move->from_row >= 8 || move->from_col >= 8 || move->to_row >= 8 ||
109 move->to_col >= 8) {
110 ESP_LOGE(TAG, "❌ Critical error: Invalid coordinates in move structure");
111 return;
112 }
113
114 ESP_LOGI(TAG, "🚨 Invalid move detected - smart error handling");
115 ESP_LOGI(TAG, " Move: %c%d -> %c%d", 'a' + move->from_col,
116 move->from_row + 1, 'a' + move->to_col, move->to_row + 1);
117
118 // Remote / deprecated UI can send a move from an empty field. The simulation “figure is
119 // physically to it" copies board[from] to board[to]; when from is empty,
120 // the figure on the target is deleted (logically it "disappears"). HW recovery in this case
121 // it doesn't make sense — don't change the board.
122 piece_t src_piece = game_get_piece(move->from_row, move->from_col);
123 if (src_piece == PIECE_EMPTY) {
124 ESP_LOGW(TAG,
125 "[STAGING] game_handle_invalid_move: empty source square — "
126 "skipping board mutation and LED recovery (error=%d)",
127 (int)error);
128 return;
129 }
130
131 // Smart error handling according to requirements
132
133 // 1. Set the red field to an invalid position
134 error_recovery_state.has_invalid_piece = true;
135 error_recovery_state.invalid_row = move->to_row;
136 error_recovery_state.invalid_col = move->to_col;
137 error_recovery_state.piece_type = game_get_piece(move->from_row, move->from_col);
138
139 // 2. Remember the original valid position (from where the move came from)
140 error_recovery_state.original_valid_row = move->from_row;
141 error_recovery_state.original_valid_col = move->from_col;
142
143 // 3. Legal board[] unchanged; recovery tracks invalid square in state.
144
145 // 4. CLEAR VISUAL ALERT - red field + flashing for attention
146 // pozornosti
147 if (game_policy_error_recovery_led_red_blink() ||
148 game_policy_error_recovery_led_red_persist()) {
149 game_event_post_led_ctrl(GAME_EVT_CTRL_CLEAR_ALL, 0, 0, 0, 0);
150 uint8_t invalid_led = chess_pos_to_led_index(move->to_row, move->to_col);
151
152 ESP_LOGI(TAG, "🚨 STARTING ERROR ANIMATION - RED BLINKING");
153
154 if (game_policy_error_recovery_led_red_blink()) {
155 for (int blink = 0; blink < 4; blink++) {
156 game_led_paint_led(invalid_led, 255, 0, 0);
157 game_led_flush(); vTaskDelay(pdMS_TO_TICKS(150));
158 game_led_paint_led(invalid_led, 0, 0, 0);
159 game_led_flush(); vTaskDelay(pdMS_TO_TICKS(150));
160 }
161 }
162
163 if (game_policy_error_recovery_led_red_persist()) {
164 game_led_paint_led(invalid_led, 255, 0, 0);
165 }
166 }
167
168 ESP_LOGI(TAG, "🔴 RED SQUARE ACTIVE: %c%d (piece must be lifted from here)",
169 'a' + move->to_col, move->to_row + 1);
170 ESP_LOGI(TAG,
171 "📍 RECOVERY TARGET: Valid moves will show from %c%d when piece is "
172 "lifted",
173 'a' + move->from_col, move->from_row + 1);
174 ESP_LOGI(TAG,
175 "💡 USER ACTION REQUIRED: Lift piece from red square to continue");
176
177 // CRITICAL FIX: NON-BLOCKING error handling - no loops!
178 // The red field will remain lit until the figure is picked up
179 // Recovery is handled in game_process_pickup_command()
180
181 ESP_LOGI(TAG, "💡 Error recovery active - red square will stay lit until "
182 "piece is lifted");
183
184 game_policy_error_recovery_enter_lock();
185 game_show_invalid_move_error_with_blink(move->to_row, move->to_col);

397 (*game_state_black_moves_count())++;
398 } else {
399 // White just moved
400 (*game_state_white_time_total()) += move_time;
401 (*game_state_white_moves_count())++;
402 }

408 // 6612) If game finished, it was already handled with endgame animation
409
410 // Check if king is in check
411 if (game_is_king_in_check(current_player)) {
412 if (current_player == PLAYER_WHITE) {
413 (*game_state_white_checks())++;
414 } else {
415 (*game_state_black_checks())++;
416 }
417 ESP_LOGI(TAG, "⚠️ CHECK! %s king is under attack!",
418 (current_player == PLAYER_WHITE) ? "White" : "Black");
419 }
420
421 // Update last move time for next move
422 (*game_state_last_move_time()) = current_time;
423
424 // Display updated board (check auto-display setting)
425 // Note: Auto-display control is in UART task, board is always shown for now
426 game_print_board();
427
428 // Check for check/checkmate conditions
429 game_check_game_conditions();
