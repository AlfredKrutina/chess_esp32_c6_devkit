133esp_err_t game_process_timer_command(const game_cmd_t *cmd) {
134 if (cmd == NULL) {
135 ESP_LOGE(TAG, "Invalid command parameter");
136 return ESP_ERR_INVALID_ARG;
137 }
138
139 esp_err_t ret = ESP_OK;
140
141 switch (cmd->kind) {
142 case GCMD_TIMER_SET: {
143 time_control_config_t config;
144 time_control_type_t type =
145 (time_control_type_t)cmd->u.timer.time_control_type;
146
147 if (type == TIME_CONTROL_CUSTOM) {
148 ret = timer_set_custom_time_control(cmd->u.timer.custom_minutes,
149 cmd->u.timer.custom_increment);
150 } else {
151 ret = timer_get_config_by_type(type, &config);
152 if (ret == ESP_OK) {
153 ret = timer_set_time_control(&config);
154 }
155 }
156
157 if (ret == ESP_OK) {
158 ESP_LOGI(TAG, "Time control command processed successfully");
159 } else {
160 ESP_LOGE(TAG, "Failed to process time control command");
161 }
162 } break;
163
164 case GCMD_TIMER_PAUSE:
165 ret = game_pause_timer();
166 break;
167
168 case GCMD_TIMER_RESUME:
169 ret = game_resume_timer();
170 break;
171
172 case GCMD_TIMER_RESET:
173 ret = game_reset_timer();
174 break;
175
176 default:
177 ESP_LOGW(TAG, "Unknown timer command: %d", (int)cmd->kind);
178 ret = ESP_ERR_INVALID_ARG;
179 break;
180 }
181
182 return ret;
183}

185esp_err_t game_handle_time_expiration(void) {
186 ESP_LOGW(TAG, "⏰ Time expired! Handling timeout...");
187
188 // Get the timer status
189 chess_timer_t timer_data;
190 esp_err_t ret = timer_get_state(&timer_data);
191 if (ret != ESP_OK) {
192 ESP_LOGE(TAG, "Failed to get timer state");
193 return ret;
194 }
195
196 // Determine the winner
197 player_t winner = timer_data.is_white_turn ? PLAYER_BLACK : PLAYER_WHITE;
198 const char *winner_name = (winner == PLAYER_WHITE) ? "White" : "Black";
199 const char *loser_name = (winner == PLAYER_WHITE) ? "Black" : "White";
200
201 ESP_LOGW(TAG, "🏆 %s wins by timeout! %s ran out of time.", winner_name,
202 loser_name);
203
204 // End the game
205 current_game_state = GAME_STATE_FINISHED;
206 (*game_state_game_result()) = GAME_STATE_FINISHED;
207 (*game_state_current_result_type()) =
208 (winner == PLAYER_WHITE) ? RESULT_WHITE_WINS : RESULT_BLACK_WINS;
209 (*game_state_current_endgame_reason()) = ENDGAME_REASON_TIMEOUT; // Mark as timeout
210
211 // Endgame report request (same fix as resignation)
212 game_update_endgame_statistics((*game_state_current_result_type()));
213 (*game_state_endgame_report_requested()) = true;
214
215 // UNIFIED VICTORY ANIMATION
216 // Timeout = Winner determined above
217 game_trigger_victory_animation(winner);
218
219 // Display result on LED (use existing LED functions)
220 // led_show_game_result((*game_state_game_result())); // The function does not exist
221
222 // Pozastavit timer
223 timer_pause();
224
225 ESP_LOGI(TAG, "Game ended due to timeout. Winner: %s", winner_name);
226 game_snapshot_persist_after_valid_move();
227
228 return ESP_OK;
229}
