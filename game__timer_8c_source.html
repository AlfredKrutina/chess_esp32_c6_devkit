130esp_err_t game_process_timer_command(const chess_move_command_t *cmd) {
131 if (cmd == NULL) {
132 ESP_LOGE(TAG, "Invalid command parameter");
133 return ESP_ERR_INVALID_ARG;
134 }
135
136 esp_err_t ret = ESP_OK;
137
138 switch (cmd->type) {
139 case GAME_CMD_SET_TIME_CONTROL: {
140 time_control_config_t config;
141 time_control_type_t type =
142 (time_control_type_t)cmd->timer_data.timer_config.time_control_type;
143
144 if (type == TIME_CONTROL_CUSTOM) {
145 // Custom time control
146 ret = timer_set_custom_time_control(
147 cmd->timer_data.timer_config.custom_minutes,
148 cmd->timer_data.timer_config.custom_increment);
149 } else {
150 // Predefined time control
151 ret = timer_get_config_by_type(type, &config);
152 if (ret == ESP_OK) {
153 ret = timer_set_time_control(&config);
154 }
155 }
156
157 if (ret == ESP_OK) {
158 ESP_LOGI(TAG, "Time control command processed successfully");
159 } else {
160 ESP_LOGE(TAG, "Failed to process time control command");
161 }
162 } break;
163
164 case GAME_CMD_PAUSE_TIMER:
165 ret = game_pause_timer();
166 break;
167
168 case GAME_CMD_RESUME_TIMER:
169 ret = game_resume_timer();
170 break;
171
172 case GAME_CMD_RESET_TIMER:
173 ret = game_reset_timer();
174 break;
175
176 case GAME_CMD_GET_TIMER_STATE:
177 // Timer state is retrieved via JSON API
178 ret = ESP_OK;
179 break;
180
181 case GAME_CMD_TIMER_TIMEOUT:
182 ret = game_handle_time_expiration();
183 break;
184
185 default:
186 ESP_LOGW(TAG, "Unknown timer command: %d", cmd->type);
187 ret = ESP_ERR_INVALID_ARG;
188 break;
189 }
190
191 return ret;
192}

194esp_err_t game_handle_time_expiration(void) {
195 ESP_LOGW(TAG, "⏰ Time expired! Handling timeout...");
196
197 // Get the timer status
198 chess_timer_t timer_data;
199 esp_err_t ret = timer_get_state(&timer_data);
200 if (ret != ESP_OK) {
201 ESP_LOGE(TAG, "Failed to get timer state");
202 return ret;
203 }
204
205 // Determine the winner
206 player_t winner = timer_data.is_white_turn ? PLAYER_BLACK : PLAYER_WHITE;
207 const char *winner_name = (winner == PLAYER_WHITE) ? "White" : "Black";
208 const char *loser_name = (winner == PLAYER_WHITE) ? "Black" : "White";
209
210 ESP_LOGW(TAG, "🏆 %s wins by timeout! %s ran out of time.", winner_name,
211 loser_name);
212
213 // End the game
214 current_game_state = GAME_STATE_FINISHED;
215 game_result = GAME_STATE_FINISHED;
216 current_result_type =
217 (winner == PLAYER_WHITE) ? RESULT_WHITE_WINS : RESULT_BLACK_WINS;
218 current_endgame_reason = ENDGAME_REASON_TIMEOUT; // Mark as timeout
219
220 // Endgame report request (same fix as resignation)
221 game_update_endgame_statistics(current_result_type);
222 endgame_report_requested = true;
223
224 // UNIFIED VICTORY ANIMATION
225 // Timeout = Winner determined above
226 game_trigger_victory_animation(winner);
227
228 // Display result on LED (use existing LED functions)
229 // led_show_game_result(game_result); // The function does not exist
230
231 // Pozastavit timer
232 timer_pause();
233
234 ESP_LOGI(TAG, "Game ended due to timeout. Winner: %s", winner_name);
235 game_snapshot_persist_after_valid_move();
236
237 return ESP_OK;
238}
