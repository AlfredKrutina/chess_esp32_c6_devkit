19static void apply_restore(const game_event_t *ev) {
20 switch (ev->restore_op) {
21 case GAME_EVT_RESTORE_OP_APPLY:
22 led_board_restore_guide_apply((const led_restore_guide_plan_t *)&ev->restore);
23 game_led_board_picture_bump();
24 break;
25 case GAME_EVT_RESTORE_OP_CLEAR:
26 led_board_restore_guide_clear();
27 break;
28 case GAME_EVT_RESTORE_OP_CANCEL:
29 led_board_restore_guide_cancel();
30 break;
31 case GAME_EVT_RESTORE_OP_FADE_OUT:
32 led_board_restore_guide_fade_out();
33 break;
34 case GAME_EVT_RESTORE_OP_BLANK:
35 led_board_restore_guide_blank();
36 break;
37 case GAME_EVT_RESTORE_OP_TAKEBACK: {
38 uint8_t leds[8];
39 uint8_t n = 0;
40 uint8_t i;
41 if (ev->takeback_count == 0) {
42 led_takeback_clear();
43 return;
44 }
45 led_target_begin();
46 for (i = 0; i < ev->takeback_count && n < 8; i++) {
47 uint8_t sq = ev->takeback_squares[i];
48 uint8_t led;
49 if (sq >= 64) {
50 continue;
51 }
52 led = chess_pos_to_led_index(sq / 8, sq % 8);
53 leds[n++] = led;
54 led_target_set(led, 0, 255, 255, LED_VIS_BREATHE);
55 }
56 led_target_end();
57 led_takeback_arm(leds, n);
58 break;
59 }
60 default:
61 break;
62 }
63}

77static void apply_hints(const game_event_t *ev) {
78 uint8_t i;
79 bool required = (ev->flags & GAME_EVT_HINT_REQUIRED_STEP) != 0;
80 if (required && ev->hint_lift_sq == 254U && ev->square_count == 0) {
81 led_required_step_guide_clear();
82 return;
83 }
84 bool patch = (ev->flags & GAME_EVT_HINT_PATCH) != 0;
85 if (!patch && !required) {
86 led_required_step_guide_yield();
87 }
88 if (!patch) {
89 led_move_trail_cancel();
90 if (ev->fade_ms != 0) {
91 led_target_begin_fade(ev->fade_ms);
92 } else {
93 led_target_begin();
94 }
95 }
96 if (ev->hint_lift_sq < 64) {
97 paint_square(ev->hint_lift_sq, 255, 255, 0, LED_VIS_SNAP);
98 }
99 for (i = 0; i < ev->square_count && i < GAME_EVT_HINT_MAX; i++) {
100 const game_evt_hint_pixel_t *h = &ev->hints[i];
101 paint_square(h->square, h->r, h->g, h->b, LED_VIS_SNAP);
102 }
103 for (i = 0; i < ev->breathe_count && i < GAME_EVT_BREATHE_MAX; i++) {
104 const game_evt_breathe_t *b = &ev->breathes[i];
105 uint8_t row;
106 uint8_t col;
107 uint8_t led;
108 if (b->square >= 64) {
109 continue;
110 }
111 row = (uint8_t)(b->square / 8);
112 col = (uint8_t)(b->square % 8);
113 led = chess_pos_to_led_index(row, col);
114 if (b->mode == GAME_EVT_BREATHE_SQ) {
115 led_target_set_breathe_sq(led, b->r, b->g, b->b, b->floor, b->period_ms,
116 b->phase_ms, ev->fade_ms);
117 } else {
118 paint_square(b->square, b->r, b->g, b->b, LED_VIS_BREATHE);
119 }
120 }
121 if (!patch) {
122 led_target_end();
123 if (ev->hint_lift_sq < 64) {
124 led_move_collapse_capture();
125 }
126 }
127 if (required) {
128 led_required_step_plan_t plan;
129 plan.active = true;
130 plan.count = ev->square_count;
131 plan.fade_ms = ev->fade_ms;
132 if (plan.count > LED_REQUIRED_STEP_MAX_PX) {
133 plan.count = LED_REQUIRED_STEP_MAX_PX;
134 }
135 for (i = 0; i < plan.count; i++) {
136 plan.px[i] = ev->hints[i];
137 }
138 led_required_step_guide_apply(&plan);
139 }
140}

236static void apply_move_rejected(const game_event_t *ev) {
237 if (ev->flags & 2U) {
238 if (s_error_led < 64U) {
239 led_target_pulse_cancel(s_error_led);
240 s_error_led = 255U;
241 }
242 led_target_begin();
243 led_target_end();
244 return;
245 }
246 if (ev->from >= 64) {
247 return;
248 }
249 uint8_t led = chess_pos_to_led_index((uint8_t)(ev->from / 8),
250 (uint8_t)(ev->from % 8));
251 if (s_error_led < 64U && s_error_led != led) {
252 led_target_pulse_cancel(s_error_led);
253 }
254 s_error_led = led;
255 led_target_begin();
256 if (ev->flags & 1U) {
257 led_target_set(led, 255, 0, 0, LED_VIS_SNAP);
258 }
259 led_target_end();
260 if (!(ev->flags & 1U)) {
261 led_target_pulse(led, 255, 0, 0);
262 }
263}

282void led_game_events_process(void) {
283 game_event_msg_t msg;
284 while (game_event_queue != NULL &&
285 xQueueReceive(game_event_queue, &msg, 0) == pdTRUE) {
286 game_event_resolve(&msg, &s_led_game_ev);
287 const game_event_t *ev = &s_led_game_ev;
288 switch (ev->kind) {
289 case GAME_EVT_MOVE_REJECTED:
290 apply_move_rejected(ev);
291 break;
292 case GAME_EVT_CHECK:
293 apply_check(ev);
294 break;
295 case GAME_EVT_GAME_END:
296 apply_game_end(ev);
297 break;
298 case GAME_EVT_RESTORE_BOARD:
299 apply_restore(ev);
300 break;
301 case GAME_EVT_HINT_SQUARES:
302 apply_hints(ev);
303 break;
304 case GAME_EVT_BOARD_ANIM:
305 apply_board_anim(ev);
306 break;
307 case GAME_EVT_LED_CTRL:
308 apply_led_ctrl(ev);
309 break;
310 case GAME_EVT_MOVE_ACCEPTED:
311 default:
312 break;
313 }
314 }
315}
