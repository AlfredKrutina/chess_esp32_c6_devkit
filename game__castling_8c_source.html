42void game_detect_and_handle_castling(const chess_move_t *move) {
43 piece_t piece = move->piece;
44
45 // Is it a king and moves 2 squares?
46 if ((piece == PIECE_WHITE_KING || piece == PIECE_BLACK_KING) &&
47 abs((int)move->to_col - (int)move->from_col) == 2) {
48
49 ESP_LOGI(TAG, "🏰 CASTLING DETECTED! King moved 2 squares");
50
51 // Set cast status
52 castling_state.in_progress = true;
53 castling_state.player = current_player;
54
55 // Calculate tower positions
56 bool is_kingside = (move->to_col == 6);
57 castling_state.rook_from_row = move->to_row;
58 castling_state.rook_from_col = is_kingside ? 7 : 0;
59 castling_state.rook_to_row = move->to_row;
60 castling_state.rook_to_col = is_kingside ? 5 : 3;
61
62 // Show LED guidance for tower
63 game_show_castling_rook_guidance();
64
65 // DO NOT CHANGE PLAYERS - wait for casting to complete
66 ESP_LOGI(TAG, "⏳ Waiting for rook move to complete castling...");
67 }
68}

96bool game_check_castling_completion(const chess_move_t *move) {
97 if (!castling_state.in_progress)
98 return false;
99
100 // Is this the correct move of the tower?
101 if (move->from_row == castling_state.rook_from_row &&
102 move->from_col == castling_state.rook_from_col &&
103 move->to_row == castling_state.rook_to_row &&
104 move->to_col == castling_state.rook_to_col) {
105
106 ESP_LOGI(TAG, "✅ CASTLING COMPLETED! Rook moved correctly");
107
108 // We will move the tower in the board[][]
109 game_set_piece(move->to_row, move->to_col, move->piece);
110 game_set_piece(castling_state.rook_from_row, castling_state.rook_from_col,
111 PIECE_EMPTY);
112
113 // Update flags for king and rook
114 if (castling_state.player == PLAYER_WHITE) {
115 white_king_moved = true;
116 piece_moved[0][4] = true;
117 if (castling_state.rook_from_col == 7) {
118 white_rook_h_moved = true;
119 piece_moved[0][7] = true;
120 } else {
121 white_rook_a_moved = true;
122 piece_moved[0][0] = true;
123 }
124 } else {
125 black_king_moved = true;
126 piece_moved[7][4] = true;
127 if (castling_state.rook_from_col == 7) {
128 black_rook_h_moved = true;
129 piece_moved[7][7] = true;
130 } else {
131 black_rook_a_moved = true;
132 piece_moved[7][0] = true;
133 }
134 }
135
136 // Golden cast completion animation
137 show_castling_completion_animation();
138
139 // Change players RIGHT NOW
140 current_player =
141 (current_player == PLAYER_WHITE) ? PLAYER_BLACK : PLAYER_WHITE;
142
143 // CRITICAL: Endgame check BEFORE player change animations!
144 // If endgame, player change will NOT start
145 game_state_t end_game_result = game_check_end_game_conditions();
146 if (end_game_result == GAME_STATE_FINISHED) {
147 current_game_state = GAME_STATE_FINISHED;
148 game_active = false;
149
150 // Find the position of the winning king for the endgame animation
151 player_t winner =
152 (current_player == PLAYER_WHITE) ? PLAYER_BLACK : PLAYER_WHITE;
153 uint8_t king_pos = 28; // default e4
154 for (int i = 0; i < 64; i++) {
155 piece_t piece = game_get_piece(i / 8, i % 8);
156 if ((winner == PLAYER_WHITE && piece == PIECE_WHITE_KING) ||
157 (winner == PLAYER_BLACK && piece == PIECE_BLACK_KING)) {
158 king_pos = i;
159 break;
160 }
161 }
162
163 ESP_LOGI(TAG,
164 "🎯 Game finished after castling completion! Starting endgame "
165 "animation at position %d",
166 king_pos);
167
168 // Start the endgame animation (wave from king winner)
169 game_led_flush();
170 game_event_post_game_end(king_pos, 0);
171
172 ESP_LOGI(
173 TAG,
174 "✅ Endgame animation started - player change animation SKIPPED");
175 } else {
176 memset(&castling_state, 0, sizeof(castling_state));
177
178 // Show moveable figures for new player
179 game_led_clear_board();
180 game_policy_highlight_movable_if_enabled();
181
182 ESP_LOGI(TAG, "🏰 Castling completed! %s to move",
183 (current_player == PLAYER_WHITE) ? "White" : "Black");
184
185 // Check if the new player is in check
186 bool in_check = game_is_king_in_check(current_player);
187 if (in_check) {
188 // Find the position of the king
189 int king_row = -1, king_col = -1;
190 piece_t king_piece = (current_player == PLAYER_WHITE)
191 ? PIECE_WHITE_KING
192 : PIECE_BLACK_KING;
193 for (int row = 0; row < 8; row++) {
194 for (int col = 0; col < 8; col++) {
195 if (game_get_piece(row, col) == king_piece) {
196 king_row = row;
197 king_col = col;
198 break;
199 }
200 }
201 if (king_row != -1)
202 break;
203 }
204
205 if (king_row != -1 && king_col != -1) {
206 if (game_led_guidance_show_check_anim()) {
207 uint8_t king_led_index = chess_pos_to_led_index(king_row, king_col);
208 // Start the check animation - pink lighting on the king
209 game_led_flush();
210 game_event_post_check(king_led_index);
211 ESP_LOGI(TAG, "⚠️ CHECK! %s is in check",
212 current_player == PLAYER_WHITE ? "White" : "Black");
213 }
214 }
215 }
216 }
217
218 return true;
219 } else {
220 // Incorrect tower move - show guidance again
221 ESP_LOGI(TAG, "❌ Incorrect rook move for castling");
222 game_show_castling_rook_guidance();
223 return false;
224 }
225}

236void game_start_castle_animation(const chess_move_extended_t *move) {
237 if (!move)
238 return;
239
240 ESP_LOGI(TAG, "🏰 Starting castle animation for %s",
241 (move->move_type == MOVE_TYPE_CASTLE_KING) ? "kingside"
242 : "queenside");
243
244 // Calculate rook positions
245 uint8_t king_row = (current_player == PLAYER_WHITE) ? 0 : 7;
246
247 if (move->move_type == MOVE_TYPE_CASTLE_KING) {
248 // Kingside: rook from h-file to f-file
249 rook_from_row = king_row;
250 rook_from_col = 7; // h-file
251 rook_to_row = king_row;
252 rook_to_col = 5; // f-file
253 } else {
254 // Queenside: rook from a-file to d-file
255 rook_from_row = king_row;
256 rook_from_col = 0; // a-file
257 rook_to_row = king_row;
258 rook_to_col = 3; // d-file
259 }
260
261 // Store castle move for completion
262 pending_castle_move = *move;
263 castle_animation_active = true;
264
265 // Clear board and start repeating rook move animation
266 game_led_clear_board();
267
268 // Start a repeating tower movement animation
269 game_start_repeating_rook_animation();
270
271 // Show initial animation immediately
272 uint8_t from_led = chess_pos_to_led_index(rook_from_row, rook_from_col);
273 uint8_t to_led = chess_pos_to_led_index(rook_to_row, rook_to_col);
274 game_led_paint_led(from_led, 255, 255, 0); // Yellow for rook position
275 game_led_paint_led(to_led, 0, 255, 0); // Green for destination
276
277 ESP_LOGI(TAG, "🏰 Rook at %c%d needs to move to %c%d", 'a' + rook_from_col,
278 rook_from_row + 1, 'a' + rook_to_col, rook_to_row + 1);
279}

390bool game_complete_castle_animation(uint8_t from_row, uint8_t from_col,
391 uint8_t to_row, uint8_t to_col) {
392 if (!castle_animation_active) {
393 return false;
394 }
395
396 // Check if rook was moved to correct position
397 if (from_row == rook_from_row && from_col == rook_from_col &&
398 to_row == rook_to_row && to_col == rook_to_col) {
399
400 ESP_LOGI(TAG, "✅ Castle animation completed! Rook moved correctly.");
401
402 // Only move the tower when the player makes the correct move
403 piece_t rook_piece = game_get_piece(rook_from_row, rook_from_col);
404 game_set_piece(rook_from_row, rook_from_col, PIECE_EMPTY);
405 game_set_piece(rook_to_row, rook_to_col, rook_piece);
406
407 // Stop repeating rook animation
408 game_stop_repeating_rook_animation();
409
410 // Complete the castle move
411 castle_animation_active = false;
412
413 // Switch players now that castle is complete
414 current_player =
415 (current_player == PLAYER_WHITE) ? PLAYER_BLACK : PLAYER_WHITE;
416
417 // Clear board and show movable pieces for new player
418 game_led_clear_board();
419 game_policy_highlight_movable_if_enabled();
420
421 ESP_LOGI(TAG, "🏰 Castling completed! %s to move",
422 (current_player == PLAYER_WHITE) ? "White" : "Black");
423
424 return true;
425 } else {
426 ESP_LOGW(TAG,
427 "❌ Invalid rook move for castling: %c%d -> %c%d (expected: %c%d "
428 "-> %c%d)",
429 'a' + from_col, from_row + 1, 'a' + to_col, to_row + 1,
430 'a' + rook_from_col, rook_from_row + 1, 'a' + rook_to_col,
431 rook_to_row + 1);
432
433 // Keep castle animation active, wait for correct move
434 return false;
435 }

444void game_refresh_leds(void) {
445 ESP_LOGI(TAG, "🔄 Refreshing LED state based on game status...");
446
447 // If game is not active, just show the board pattern
448 if (!game_active && current_game_state != GAME_STATE_PROMOTION) {
449 // If we are in "setup" or "idle" mode
450 game_event_post_led_ctrl(GAME_EVT_CTRL_SHOW_BOARD, 0, 0, 0, 0);
451 return;
452 }
453
454 // Clear board first (to remove HA colors)
455 game_led_clear_board();
456
457 if (game_is_squares_guard_active()) {
458 ESP_LOGI(TAG, " - Restoring MATRIX GUARD LEDs (physical vs logic)");
459 game_squares_guard_render_leds();
460 game_check_promotion_needed();
461 return;
462 }
463
464 // 1. Check for Error Recovery State
465 if ((*game_state_error_recovery_state()).waiting_for_move_correction) {
466 ESP_LOGI(TAG, " - Restoring ERROR state LEDs");
467 // Show validation error
468 game_show_invalid_move_error_with_blink((*game_state_error_recovery_state()).invalid_row,
469 (*game_state_error_recovery_state()).invalid_col);
470
471 // Also highlight the original valid position (Blue)
472 if (game_policy_error_recovery_led_valid_blue()) {
473 game_led_frame_sq((*game_state_error_recovery_state()).original_valid_row,
474 (*game_state_error_recovery_state()).original_valid_col, 0, 0, 255);
475 }
476
477 return;
478 }
479
480 // 2. Check for Lifted Piece State
481 if (piece_lifted) {
482 ESP_LOGI(TAG, " - Restoring LIFTED piece state LEDs at %c%d",
483 'a' + lifted_piece_col, lifted_piece_row + 1);
484
485 // Highlight source square (Yellow)
486 game_led_frame_sq(lifted_piece_row, lifted_piece_col, 255, 255, 0);
487
488 // Show valid moves for this piece
489 move_suggestion_t *suggestions = game_move_suggestions_work();
490 uint32_t valid_moves = game_get_available_moves(
491 lifted_piece_row, lifted_piece_col, suggestions, 64);
492
493 if (valid_moves > 0) {
494 for (uint32_t i = 0; i < valid_moves; i++) {
495 uint8_t dest_row = suggestions[i].to_row;
496 uint8_t dest_col = suggestions[i].to_col;
497 uint8_t led_index = chess_pos_to_led_index(dest_row, dest_col);
498
499 // Check if destination has opponent's piece
500 piece_t dest_piece = game_get_piece(dest_row, dest_col);
501 bool is_opponent_piece =
502 (current_player == PLAYER_WHITE && dest_piece >= PIECE_BLACK_PAWN &&
503 dest_piece <= PIECE_BLACK_KING) ||
504 (current_player == PLAYER_BLACK && dest_piece >= PIECE_WHITE_PAWN &&
505 dest_piece <= PIECE_WHITE_KING);
506
507 if (is_opponent_piece) {
508 game_led_paint_led(led_index, 255, 165, 0); // Orange for capture
509 } else {
510 game_led_paint_led(led_index, 0, 255, 0); // Green for move
511 }
512 }
513 }
514 return;
515 }
516
517 // 3. Check for Check state
518 if (game_is_king_in_check(current_player)) {
519 // Find King
520 int king_row = -1, king_col = -1;
521 piece_t king_piece =
522 (current_player == PLAYER_WHITE) ? PIECE_WHITE_KING : PIECE_BLACK_KING;
523 for (int row = 0; row < 8; row++) {
524 for (int col = 0; col < 8; col++) {
525 if (game_get_piece(row, col) == king_piece) {
526 king_row = row;
527 king_col = col;
528 break;
529 }
530 }
531 if (king_row != -1)
532 break;
533 }
534
535 if (king_row != -1) {
536 if (game_led_guidance_show_check_anim()) {
537 ESP_LOGI(TAG, " - Restoring CHECK state LEDs");
538 uint8_t king_led_index = chess_pos_to_led_index(king_row, king_col);
539 game_led_flush();
540 game_event_post_check(king_led_index);
541 }
542 }
543 }
544
545 // 4. Default Active State - Highlight movable pieces
546 ESP_LOGI(TAG, " - Restoring ACTIVE state LEDs (movable pieces)");
547 game_policy_highlight_movable_if_enabled();
548
549 // Update button LEDs (64-71) to match current game state (promotion green/blue)
550 // so they are correct after switching back to game mode (e.g. from HA).
551 game_check_promotion_needed();
552}
