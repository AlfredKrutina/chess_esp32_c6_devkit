22bool position_sq_attacked(const uint8_t board[8][8], int row, int col, int by_white) {
23 static const int knight[8][2] = {{-2, -1}, {-2, 1}, {-1, -2}, {-1, 2},
24 {1, -2}, {1, 2}, {2, -1}, {2, 1}};
25 static const int king[8][2] = {{-1, -1}, {-1, 0}, {-1, 1}, {0, -1},
26 {0, 1}, {1, -1}, {1, 0}, {1, 1}};
27 static const int bishop[4][2] = {{-1, -1}, {-1, 1}, {1, -1}, {1, 1}};
28 static const int rook[4][2] = {{-1, 0}, {1, 0}, {0, -1}, {0, 1}};
29 uint8_t pawn = by_white ? POS_WHITE_PAWN : POS_BLACK_PAWN;
30 uint8_t knight_p = by_white ? POS_WHITE_KNIGHT : POS_BLACK_KNIGHT;
31 uint8_t bishop_p = by_white ? POS_WHITE_BISHOP : POS_BLACK_BISHOP;
32 uint8_t rook_p = by_white ? POS_WHITE_ROOK : POS_BLACK_ROOK;
33 uint8_t queen_p = by_white ? POS_WHITE_QUEEN : POS_BLACK_QUEEN;
34 uint8_t king_p = by_white ? POS_WHITE_KING : POS_BLACK_KING;
35 int pawn_dir = by_white ? 1 : -1;
36 int pawn_row = row - pawn_dir;
37 int i;
38
39 if (on_board(pawn_row, col - 1) && board[pawn_row][col - 1] == pawn) {
40 return true;
41 }
42 if (on_board(pawn_row, col + 1) && board[pawn_row][col + 1] == pawn) {
43 return true;
44 }
45
46 for (i = 0; i < 8; i++) {
47 int nr = row + knight[i][0];
48 int nc = col + knight[i][1];
49 if (on_board(nr, nc) && board[nr][nc] == knight_p) {
50 return true;
51 }
52 }
53
54 for (i = 0; i < 4; i++) {
55 int nr = row + bishop[i][0];
56 int nc = col + bishop[i][1];
57 while (on_board(nr, nc)) {
58 uint8_t piece = board[nr][nc];
59 if (piece != POS_EMPTY) {
60 if (piece == bishop_p || piece == queen_p) {
61 return true;
62 }
63 break;
64 }
65 nr += bishop[i][0];
66 nc += bishop[i][1];
67 }
68 }
69
70 for (i = 0; i < 4; i++) {
71 int nr = row + rook[i][0];
72 int nc = col + rook[i][1];
73 while (on_board(nr, nc)) {
74 uint8_t piece = board[nr][nc];
75 if (piece != POS_EMPTY) {
76 if (piece == rook_p || piece == queen_p) {
77 return true;
78 }
79 break;
80 }
81 nr += rook[i][0];
82 nc += rook[i][1];
83 }
84 }
85
86 for (i = 0; i < 8; i++) {
87 int nr = row + king[i][0];
88 int nc = col + king[i][1];
89 if (on_board(nr, nc) && board[nr][nc] == king_p) {
90 return true;
91 }
92 }
93
94 return false;
95}

97bool position_mover_in_check(const position_t *p, uint8_t side) {
98 uint8_t king = (side == 0) ? POS_WHITE_KING : POS_BLACK_KING;
99 int by_white = (side == 0) ? 0 : 1;
100 int row;
101 int col;
102
103 if (p == NULL) {
104 return false;
105 }
106 for (row = 0; row < 8; row++) {
107 for (col = 0; col < 8; col++) {
108 if (p->board[row][col] == king) {
109 return position_sq_attacked(p->board, row, col, by_white);
110 }
111 }
112 }
113 return false;
