113static int hall_count_play(piece_t piece) {
114 bool held_origin[64] = {0};
115 int n = 0;
116 for (int i = 0; i < s_count; i++) {
117 held_origin[s_hand[i].row * 8 + s_hand[i].col] = true;
118 if (s_hand[i].piece == piece) {
119 n++;
120 }
121 }
122 for (uint8_t sq = 0; sq < 64; sq++) {
123 if (held_origin[sq]) {
124 continue;
125 }
126 if (board[sq / 8][sq % 8] == piece) {

152static int hall_type_limit(piece_t piece) {
153 hall_piece_type_t type;
154 hall_piece_color_t color;
155 if (!game_hall_engine_to_sensor(piece, &type, &color)) {
156 return 0;
157 }
158 if (type == HALL_PIECE_TYPE_KING) {
159 return 1;
160 }
161 if (type == HALL_PIECE_TYPE_PAWN) {
162 return 8;
163 }
164 int missing_pawns = 8 - hall_accounted(hall_pawn_of(color));
165 if (missing_pawns < 0) {
166 missing_pawns = 0;
167 }
168 int base = (type == HALL_PIECE_TYPE_QUEEN) ? 1 : 2;
169 return base + missing_pawns;
170}

177/* 2 = exact or same-color K/Q band, 1 = pawn promotion wildcard, 0 = no. */
178static int hall_match_rank(piece_t piece, hall_piece_type_t ht,
179 hall_piece_color_t hc, uint8_t to_row) {
180 hall_piece_type_t want;
181 hall_piece_color_t want_color;
182 if (!game_hall_engine_to_sensor(piece, &want, &want_color) || hc != want_color) {
183 return 0;
184 }
185 if (ht == want || (hall_is_royal(ht) && hall_is_royal(want))) {
186 return 2;
187 }
188 bool promo = (piece == PIECE_WHITE_PAWN && to_row == 7) ||
189 (piece == PIECE_BLACK_PAWN && to_row == 0);
190 if (!promo) {
191 return 0;
192 }
193 if (ht == HALL_PIECE_TYPE_QUEEN || ht == HALL_PIECE_TYPE_KING ||
194 ht == HALL_PIECE_TYPE_ROOK || ht == HALL_PIECE_TYPE_BISHOP ||
195 ht == HALL_PIECE_TYPE_KNIGHT) {
196 return 1;
197 }
198 return 0;
199}

201static bool hall_legal_landing(piece_t piece, uint8_t from_row, uint8_t from_col,
202 uint8_t to_row, uint8_t to_col) {
203 if (from_row == to_row && from_col == to_col) {
204 return false;
205 }
206 chess_move_t move = {.from_row = from_row,
207 .from_col = from_col,
208 .to_row = to_row,
209 .to_col = to_col,
210 .piece = piece,
211 .captured_piece = board[to_row][to_col]};
212 return game_is_valid_move(&move) == MOVE_ERROR_NONE;
213}

215/* Prefer the piece's own square, then a legal destination, then the newest lift. */
216static int hall_fallback_index(const int *cands, int nc, uint8_t to_row,
217 uint8_t to_col) {
218 int best = -1;
219 int best_score = -1;
220 for (int n = 0; n < nc; n++) {
221 int i = cands[n];
222 int score = 0;
223 if (s_hand[i].row == to_row && s_hand[i].col == to_col) {
224 score = 1000;
225 } else if (hall_legal_landing(s_hand[i].piece, s_hand[i].row, s_hand[i].col,
226 to_row, to_col)) {
227 score = 200;
228 }
229 if (score > best_score || (score == best_score && i > best)) {
230 best = i;
231 best_score = score;
232 }
233 }
234 return best;
235}

243static hall_map_t hall_map_reading(hall_piece_type_t ht, hall_piece_color_t hc,
244 piece_t *out) {
245 if (hc != HALL_PIECE_COLOR_WHITE && hc != HALL_PIECE_COLOR_BLACK) {
246 return HALL_MAP_AMBIGUOUS;
247 }
248 piece_t piece = PIECE_EMPTY;
249 if (hc == HALL_PIECE_COLOR_WHITE) {
250 switch (ht) {
251 case HALL_PIECE_TYPE_PAWN:
252 piece = PIECE_WHITE_PAWN;
253 break;
254 case HALL_PIECE_TYPE_KNIGHT:
255 piece = PIECE_WHITE_KNIGHT;
256 break;
257 case HALL_PIECE_TYPE_BISHOP:
258 piece = PIECE_WHITE_BISHOP;
259 break;
260 case HALL_PIECE_TYPE_ROOK:
261 piece = PIECE_WHITE_ROOK;
262 break;
263 default:
264 break;
265 }
266 } else {
267 switch (ht) {
268 case HALL_PIECE_TYPE_PAWN:
269 piece = PIECE_BLACK_PAWN;
270 break;
271 case HALL_PIECE_TYPE_KNIGHT:
272 piece = PIECE_BLACK_KNIGHT;
273 break;
274 case HALL_PIECE_TYPE_BISHOP:
275 piece = PIECE_BLACK_BISHOP;
276 break;
277 case HALL_PIECE_TYPE_ROOK:
278 piece = PIECE_BLACK_ROOK;
279 break;
280 default:
281 break;
282 }
283 }
284 if (hall_is_royal(ht)) {
285 piece_t king =
286 hc == HALL_PIECE_COLOR_WHITE ? PIECE_WHITE_KING : PIECE_BLACK_KING;
287 piece_t queen =
288 hc == HALL_PIECE_COLOR_WHITE ? PIECE_WHITE_QUEEN : PIECE_BLACK_QUEEN;
289 bool king_ok = hall_accounted(king) < hall_type_limit(king);
290 bool queen_ok = hall_accounted(queen) < hall_type_limit(queen);
291 if (king_ok && !queen_ok) {
292 *out = king;
293 return HALL_MAP_PIECE;
294 }
295 if (queen_ok && !king_ok) {
296 *out = queen;
297 return HALL_MAP_PIECE;
298 }
299 if (!king_ok && !queen_ok) {
300 return HALL_MAP_IMPOSSIBLE;
301 }
302 return HALL_MAP_AMBIGUOUS;
303 }
304 if (piece == PIECE_EMPTY) {
305 return HALL_MAP_AMBIGUOUS;
306 }
307 if (hall_accounted(piece) >= hall_type_limit(piece)) {
308 return HALL_MAP_IMPOSSIBLE;
309 }
310 *out = piece;
311 return HALL_MAP_PIECE;
312}

340game_hall_pick_status_t game_hall_hand_pick_drop(uint8_t to_row, uint8_t to_col,
341 game_hall_pick_t *out) {
342 if (out == NULL || s_count <= 0 || to_row > 7 || to_col > 7) {
343 return GAME_HALL_PICK_NONE;
344 }
345 uint8_t sq = (uint8_t)(to_row * 8 + to_col);
346 hall_piece_type_t ht = HALL_PIECE_TYPE_NONE;
347 hall_piece_color_t hc = HALL_PIECE_COLOR_NONE;
348 hall_read_square(sq, false, &ht, &hc);
349 bool known = hall_reading_known(ht, hc);
350
351 int all[GAME_HALL_HAND_MAX];
352 int exact[GAME_HALL_HAND_MAX];
353 int wild[GAME_HALL_HAND_MAX];
354 int n_exact = 0;
355 int n_wild = 0;
356 for (int i = 0; i < s_count; i++) {
357 all[i] = i;
358 if (!known) {
359 continue;
360 }
361 int rank = hall_match_rank(s_hand[i].piece, ht, hc, to_row);
362 if (rank == 2) {
363 exact[n_exact++] = i;
364 } else if (rank == 1) {
365 wild[n_wild++] = i;
366 }
367 }
368
369 const int *cands = all;
370 int nc = s_count;
371 bool contradicted = false;
372 bool corrected = false;
373 if (known && n_exact == 1) {
374 hall_fill_pick(exact[0], false, false, true, out);
375 ESP_LOGI(TAG, "drop %c%d: sensor matched held piece %d", 'a' + to_col,
376 to_row + 1, (int)out->piece);
377 return GAME_HALL_PICK_OK;
378 }
379 if (known && n_exact > 1) {
380 cands = exact;
381 nc = n_exact;
382 } else if (known && n_exact == 0 && n_wild == 1) {
383 hall_fill_pick(wild[0], false, false, true, out);
384 ESP_LOGI(TAG, "drop %c%d: promotion reading, held pawn kept", 'a' + to_col,
385 to_row + 1);
386 return GAME_HALL_PICK_OK;
387 } else if (known && n_exact == 0 && n_wild > 1) {
388 cands = wild;
389 nc = n_wild;
390 } else if (known && n_exact == 0 && n_wild == 0) {
391 piece_t seen = PIECE_EMPTY;
392 hall_map_t mapped = hall_map_reading(ht, hc, &seen);
393 if (mapped == HALL_MAP_PIECE) {
394 contradicted = true;
395 ESP_LOGW(TAG,
396 "drop %c%d: Hall says piece %d, engine identity kept",
397 'a' + to_col, to_row + 1, (int)seen);
398 } else if (mapped == HALL_MAP_IMPOSSIBLE) {
399 contradicted = true;
400 ESP_LOGW(TAG, "drop %c%d: Hall [%s %s] matches no plausible piece",
401 'a' + to_col, to_row + 1, squares_get_hall_piece_color_name(hc),
402 squares_get_hall_piece_type_name(ht));
403 }
404 }
405
406 int idx = hall_fallback_index(cands, nc, to_row, to_col);
407 if (idx < 0) {
408 return GAME_HALL_PICK_NONE;
409 }
410 hall_fill_pick(idx, corrected, contradicted, false, out);
411 ESP_LOGI(TAG, "drop %c%d: rules chose piece %d from %c%d (contradict=%d)",
412 'a' + to_col, to_row + 1, (int)out->piece, 'a' + out->col,
413 out->row + 1, contradicted ? 1 : 0);
414 return GAME_HALL_PICK_OK;
415}
