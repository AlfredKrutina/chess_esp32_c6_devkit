90static uint32_t visible_rgb(uint32_t state) {
91 unsigned w = (state >> 24) & 0xFFu;
92 unsigned r = ((state >> 16) & 0xFFu) + w;
93 unsigned g = ((state >> 8) & 0xFFu) + w;
94 unsigned b = (state & 0xFFu) + w;
95 if (r > 255u) {
96 r = 255u;
97 }
98 if (g > 255u) {
99 g = 255u;
100 }
101 if (b > 255u) {
102 b = 255u;
103 }
104 return (r << 16) | (g << 8) | b;
105}

139static void ensure_black_order(void) {
140 if (s_black_ready) {
141 return;
142 }
143 uint8_t n = 0;
144 for (uint8_t row = 6; row <= 7; row++) {
145 for (uint8_t col = 0; col < 8; col++) {
146 s_black_leds[n++] = chess_pos_to_led_index(row, col);
147 }
148 }
149 for (int i = 1; i < 16; i++) {
150 uint8_t v = s_black_leds[i];
151 int j = i;
152 while (j > 0 && s_black_leds[j - 1] > v) {
153 s_black_leds[j] = s_black_leds[j - 1];
154 j--;
155 }
156 s_black_leds[j] = v;
157 }
158 s_black_ready = true;
159}

162static int dist2_half(uint8_t row, uint8_t col) {
163 int dx = (int)col * 2 - 14;
164 int dy = (int)row * 2;
165 return dx * dx + dy * dy;
166}

168static uint8_t smoothstep_u8(uint8_t x) {
169 uint32_t v = (uint32_t)x;
170 uint32_t s = (v * v * (765U - 2U * v)) / (255U * 255U);
171 return (uint8_t)s;
172}

174static uint8_t lerp_u8(uint8_t a, uint8_t b, uint8_t t) {
175 return (uint8_t)(((uint32_t)a * (255U - t) + (uint32_t)b * t) / 255U);
176}

178static uint32_t lerp_rgb(uint32_t a, uint32_t b, uint8_t t) {
179 return ((uint32_t)lerp_u8((uint8_t)((a >> 16) & 0xFFu),
180 (uint8_t)((b >> 16) & 0xFFu), t)
181 << 16) |
182 ((uint32_t)lerp_u8((uint8_t)((a >> 8) & 0xFFu),
183 (uint8_t)((b >> 8) & 0xFFu), t)
184 << 8) |
185 (uint32_t)lerp_u8((uint8_t)(a & 0xFFu), (uint8_t)(b & 0xFFu), t);
186}

188static void tick_wave(void) {
189 if (!s_have_old) {
190 capture_board();
191 }
192 /* Farthest square from h1 is a8 at 14^2+14^2 = 392.
193 * Outward radius 20 covers it. Inward starts past that (23) and shrinks
194 * through h1 (radius -3) so the same crest runs backwards. */
195 const int radius = s_wave_in ? (23 - (int)s_step) : (int)s_step;
196 if (s_wave_in && radius <= -3) {
197 for (int i = 0; i < 64; i++) {
198 paint((uint8_t)i, s_new[i]);
199 }
200 ESP_LOGI(TAG, "[STAGING] wave uncovered the board");
201 post_result(LED_GESTURE_WAVE_DONE);
202 return;
203 }
204 const int radius2 = radius * radius;
205 const int crest_lo = (radius > 2) ? (radius - 2) * (radius - 2) : -1;
206 bool covered = true;
207 for (uint8_t row = 0; row < 8; row++) {
208 for (uint8_t col = 0; col < 8; col++) {
209 int d2 = dist2_half(row, col);
210 uint8_t led = chess_pos_to_led_index(row, col);
211 /* Disk interior is the spreading color. Inward, that disk is the lamp
212 * shrinking toward h1 and the outside is the picture being revealed. */
213 uint32_t outside = s_wave_in ? s_new[led] : s_old[led];
214 uint32_t inside = s_wave_in ? s_old[led] : s_new[led];
215 uint32_t crest = s_cover;
216 int outer = radius + 3;
217 int outer2 = outer * outer;
218 uint32_t color;
219 if (d2 >= outer2) {
220 color = outside;
221 covered = false;
222 } else if (d2 > radius2) {
223 int span = outer2 - radius2;
224 int t = (span > 0) ? ((outer2 - d2) * 255) / span : 255;
225 if (t < 0) {
226 t = 0;
227 } else if (t > 255) {
228 t = 255;
229 }
230 color = lerp_rgb(outside, crest, smoothstep_u8((uint8_t)t));
231 covered = false;
232 } else if (radius > 2 && d2 > crest_lo) {
233 int span = radius2 - crest_lo;
234 int t = (span > 0) ? ((radius2 - d2) * 255) / span : 255;
235 if (t < 0) {
236 t = 0;
237 } else if (t > 255) {
238 t = 255;
239 }
240 color = lerp_rgb(crest, inside, smoothstep_u8((uint8_t)t));
241 } else {
242 color = inside;
243 }
244 paint(led, color);
245 }
246 }
247 if (!s_wave_in && covered && radius >= 20) {
248 for (int i = 0; i < 64; i++) {
249 paint((uint8_t)i, s_new[i]);
250 }
251 ESP_LOGI(TAG, "[STAGING] wave covered the board");
252 post_result(LED_GESTURE_WAVE_DONE);
253 return;
254 }
255 if (s_step < 250) {
256 s_step++;
257 }
258}

261static void snake_square(uint8_t step, uint8_t *row, uint8_t *col) {
262 uint8_t rank = (uint8_t)(step / 8u);
263 uint8_t file = (uint8_t)(step % 8u);
264 if ((rank & 1u) == 0u) {
265 file = (uint8_t)(7u - file);
266 }
267 *row = rank;
268 *col = file;
269}

271/* Same crawl as led_boot_animation_step (164/25 ms). Hold runs a bit faster:
272 * 300 counts per 33 ms frame, about 1.8 s across the board. */
273#define SNAKE_Q8_END (64u * 256u)
274#define SNAKE_Q8_FWD 300u
275#define SNAKE_Q8_REV 420u
276
277static void paint_boot_snake(uint16_t q8) {
278 unsigned head_sq = (unsigned)(q8 >> 8);
279 uint8_t lead = smoothstep_u8((uint8_t)(q8 & 0xFFu));
280 for (unsigned i = 0; i < 64u; i++) {
281 uint8_t row;
282 uint8_t col;
283 snake_square((uint8_t)i, &row, &col);
284 uint8_t g = 0;
285 if (head_sq >= 64u || i + 2u < head_sq) {
286 g = 128;
287 } else if (i + 1u < head_sq) {
288 g = 200;
289 } else if (i < head_sq) {
290 g = 255;
291 } else if (i == head_sq && head_sq < 64u) {
292 g = lead;
293 }
294 paint(chess_pos_to_led_index(row, col), (uint32_t)g << 8);
295 }
296}

298static void tick_snake_forward(void) {
299 if (!s_have_old) {
300 capture_board();
301 }
302 bool released = false;
303 taskENTER_CRITICAL(&s_mux);
304 released = s_release;
305 s_release = false;
306 taskEXIT_CRITICAL(&s_mux);
307 if (released) {
308 s_phase = PH_SNAKE_REV;
309 ESP_LOGI(TAG, "[STAGING] snake reverse (released early)");
310 return;
311 }
312 if (s_snake_q8 >= SNAKE_Q8_END) {
313 ESP_LOGI(TAG, "[STAGING] snake reached the end");
314 post_result(LED_GESTURE_SNAKE_DONE);
315 return;
316 }
317 uint32_t next = (uint32_t)s_snake_q8 + SNAKE_Q8_FWD;
318 if (next > SNAKE_Q8_END) {
319 next = SNAKE_Q8_END;
320 }
321 s_snake_q8 = (uint16_t)next;
322 paint_boot_snake(s_snake_q8);
323}

325static void tick_snake_reverse(void) {
326 taskENTER_CRITICAL(&s_mux);
327 s_release = false;
328 taskEXIT_CRITICAL(&s_mux);
329 if (s_snake_q8 <= SNAKE_Q8_REV) {
330 s_snake_q8 = 0;
331 restore_board();
332 ESP_LOGI(TAG, "[STAGING] snake reverse finished, game not started");
333 post_result(LED_GESTURE_SNAKE_CANCELLED);
334 return;
335 }
336 s_snake_q8 = (uint16_t)(s_snake_q8 - SNAKE_Q8_REV);
337 paint_boot_snake(s_snake_q8);
338}

340static void tick_shutdown(void) {
341 if (!s_have_old) {
342 capture_board();
343 ensure_black_order();
344 }
345 bool released = false;
346 taskENTER_CRITICAL(&s_mux);
347 released = s_release;
348 s_release = false;
349 taskEXIT_CRITICAL(&s_mux);
350 if (released) {
351 restore_board();
352 ESP_LOGI(TAG, "[STAGING] shutdown cancelled, lights restored");
353 post_result(LED_GESTURE_SHUTDOWN_CANCELLED);
354 return;
355 }
356 /* Each black-player square eases white, then out, over two frames.
357 * After the last square the board stays dark for a short beat. */
358 if (s_step >= 16) {
359 for (int i = 0; i < 64; i++) {
360 paint((uint8_t)i, 0);
361 }
362 s_snake_pace++;
363 if (s_snake_pace >= 6) {
364 s_btn_leave_dark = true;
365 ESP_LOGI(TAG, "[STAGING] shutdown chase finished");
366 post_result(LED_GESTURE_SHUTDOWN_DONE);
367 }
368 return;
369 }
370 uint8_t fade = (s_snake_pace == 0) ? 70 : 210;
371 for (int i = 0; i < 64; i++) {
372 paint((uint8_t)i, s_old[i]);
373 }
374 for (uint8_t i = 0; i < s_step; i++) {
375 paint(s_black_leds[i], 0);
376 }
377 {
378 uint8_t led = s_black_leds[s_step];
379 uint32_t mid = lerp_rgb(s_old[led], 0xDCDCDCu, (uint8_t)(255u - fade));
380 paint(led, lerp_rgb(mid, 0, fade));
381 }
382 s_snake_pace++;
383 if (s_snake_pace >= 2) {
384 paint(s_black_leds[s_step], 0);
385 s_snake_pace = 0;
386 s_step++;
387 }
388}

392 unsigned level = 80u + ((175u * (unsigned)amount) / 140u);

401static void tick_pair_hold(void) {
402 if (!s_pair_captured) {
403 capture_board();
404 s_pair_captured = true;
405 s_pair_step = 0;
406 return;
407 }
408 /* 0..47 rise, 48..95 fall. Smoothstep keeps the turn at the top soft. */
409 uint8_t tri = s_pair_step < 48 ? s_pair_step : (uint8_t)(96 - s_pair_step);
410 uint8_t shaped = smoothstep_u8((uint8_t)((unsigned)tri * 255u / 48u));
411 uint8_t amount = (uint8_t)((unsigned)shaped * 140u / 255u);
412 s_pair_amount = amount;
413 uint32_t color = pair_rgb(amount);
414 for (int i = 0; i < 64; i++) {
415 paint((uint8_t)i, color);
416 }
417 s_pair_step++;
418 if (s_pair_step >= 96) {
419 s_pair_step = 0;
420 }
421}

429 if (s_hold_level < target) {

458static void hold_preview(void) {
459 if (s_phase != PH_IDLE) {
460 return;
461 }
462 /* Boot still owns the board. A floating gesture pin reads as held and
463 * would turn the button LED orange on top of the boot snake. */
464 if (led_is_booting()) {
465 if (s_hold_lit) {
466 s_hold_level = 0;
467 hold_restore();
468 }
469 return;
470 }
471 if (!s_button_held) {
472 if (!s_hold_lit) {
473 return;
474 }
475 hold_step(0);
476 if (s_hold_level == 0) {
477 hold_restore();
478 }
479 return;
480 }
481 if (!s_hold_lit) {
482 s_hold_lit = true;
483 s_hold_level = 0;
484 }
485 hold_step(255);
486}

488static void paint_gesture_button(void) {
489 if (s_btn_leave_dark) {
490 paint(GESTURE_BUTTON_LED, 0);
491 s_btn_leave_dark = false;
492 s_btn_owned = false;
493 s_btn_saved = false;
494 s_pair_amount = 0;
495 return;
496 }
497
498 bool own = false;
499 uint32_t color = 0;
500 if (s_phase == PH_WAVE) {
501 unsigned span = s_wave_in ? 26u : 20u;
502 unsigned num = (unsigned)s_step * 255u / span;
503 if (num > 255u) {
504 num = 255u;
505 }
506 uint8_t t = smoothstep_u8((uint8_t)num);
507 color = s_wave_in ? lerp_rgb(s_cover, 0, t) : lerp_rgb(0, s_cover, t);
508 own = true;
509 } else if (s_phase == PH_SNAKE_FWD || s_phase == PH_SNAKE_REV) {
510 uint8_t t = smoothstep_u8(
511 (uint8_t)(((unsigned)s_snake_q8 * 255u) / SNAKE_Q8_END));
512 uint8_t g = (s_phase == PH_SNAKE_FWD)
513 ? (uint8_t)(40u + ((unsigned)t * 215u) / 255u)
514 : (uint8_t)(((unsigned)t * 180u) / 255u);
515 color = (uint32_t)g << 8;
516 own = true;
517 } else if (s_phase == PH_SHUTDOWN) {
518 uint8_t t = (s_step >= 16) ? 255 : (uint8_t)((unsigned)s_step * 255u / 16u);
519 color = lerp_rgb(0xDCDCDCu, 0, smoothstep_u8(t));
520 own = true;
521 } else if (s_pair_hold && s_pair_captured && s_phase == PH_IDLE) {
522 color = pair_rgb(s_pair_amount);
523 own = true;
524 } else if (s_hold_level > 0) {
525 color = lerp_rgb(s_btn_rgb, 0xFF6600u, smoothstep_u8(s_hold_level));
526 own = true;
527 }
528
529 if (own) {
530 if (!s_btn_saved) {
531 uint32_t all[73];
532 memset(all, 0, sizeof(all));
533 led_get_all_states(all, 73);
534 s_btn_rgb = all[GESTURE_BUTTON_LED];
535 s_btn_saved = true;
536 }
537 paint(GESTURE_BUTTON_LED, color);
538 s_btn_owned = true;
539 return;
540 }
541 if (s_btn_owned) {
542 paint(GESTURE_BUTTON_LED, s_btn_rgb);
543 s_btn_owned = false;
544 s_btn_saved = false;
545 s_pair_amount = 0;
546 }

549void led_gesture_anim_tick(void) {
550#if CONFIG_CHESS_GESTURE_BUTTON
551 /* Follow the pin here too. The button task owns the gesture timing, but
552 * the button LED must turn orange even if that task has not published yet. */
553 s_button_held = led_gesture_button_down();
554#endif
555 if (s_phase != PH_IDLE && s_hold_lit) {
556 s_hold_level = 0;
557 hold_restore();
558 }
559 bool yield = false;
560 bool stop = false;
561 taskENTER_CRITICAL(&s_mux);
562 yield = s_pair_yield;
563 s_pair_yield = false;
564 stop = s_pair_stop;
565 s_pair_stop = false;
566 taskEXIT_CRITICAL(&s_mux);
567 if (yield && s_pair_captured) {
568 restore_board();
569 s_pair_captured = false;
570 s_have_old = false;
571 }
572 if (stop && s_pair_captured && s_phase == PH_IDLE) {
573 restore_board();
574 s_pair_captured = false;
575 }
576 switch (s_phase) {
577 case PH_WAVE:
578 tick_wave();
579 break;
580 case PH_SNAKE_FWD:
581 tick_snake_forward();
582 break;
583 case PH_SNAKE_REV:
584 tick_snake_reverse();
585 break;
586 case PH_SHUTDOWN:
587 tick_shutdown();
588 break;
589 case PH_PULSE:
590 case PH_IDLE:
591 default:
592 break;
593 }
594 if (s_pair_hold && s_phase == PH_IDLE && !stop) {
595 tick_pair_hold();
596 }
597 hold_preview();
598 paint_gesture_button();
599 if (s_phase != PH_IDLE || s_hold_lit || s_btn_owned) {
600 led_force_immediate_update();
601 }
602}
