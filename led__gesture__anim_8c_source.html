91static void ensure_black_order(void) {
92 if (s_black_ready) {
93 return;
94 }
95 uint8_t n = 0;
96 for (uint8_t row = 6; row <= 7; row++) {
97 for (uint8_t col = 0; col < 8; col++) {
98 s_black_leds[n++] = chess_pos_to_led_index(row, col);
99 }
100 }
101 for (int i = 1; i < 16; i++) {
102 uint8_t v = s_black_leds[i];
103 int j = i;
104 while (j > 0 && s_black_leds[j - 1] > v) {
105 s_black_leds[j] = s_black_leds[j - 1];
106 j--;
107 }
108 s_black_leds[j] = v;
109 }
110 s_black_ready = true;
111}

128static void tick_wave(void) {
129 if (!s_have_old) {
130 capture_board();
131 }
132 /* 19^2 = 361 does not cover h8 (394). 20^2 = 400 does. */
133 const int radius = (int)s_step;
134 const int radius2 = radius * radius;
135 const int crest_lo = (radius > 2) ? (radius - 2) * (radius - 2) : -1;
136 bool covered = true;
137 for (uint8_t row = 0; row < 8; row++) {
138 for (uint8_t col = 0; col < 8; col++) {
139 int d2 = dist2_half(row, col);
140 uint8_t led = chess_pos_to_led_index(row, col);
141 uint32_t color;
142 if (d2 > radius2) {
143 color = s_old[led];
144 covered = false;
145 } else if (d2 > crest_lo) {
146 uint32_t base = s_new[led];
147 color = ((uint32_t)crest_channel((uint8_t)((base >> 16) & 0xFFu))
148 << 16) |
149 ((uint32_t)crest_channel((uint8_t)((base >> 8) & 0xFFu))
150 << 8) |
151 (uint32_t)crest_channel((uint8_t)(base & 0xFFu));
152 } else {
153 color = s_new[led];
154 }
155 paint(led, color);
156 }
157 }
158 if (covered && radius >= 20) {
159 for (int i = 0; i < 64; i++) {
160 paint((uint8_t)i, s_new[i]);
161 }
162 ESP_LOGI(TAG, "[STAGING] wave covered the board");
163 post_result(LED_GESTURE_WAVE_DONE);
164 return;
165 }
166 if (s_step < 250) {
167 s_step++;
168 }
169}

182static void tick_snake_forward(void) {
183 if (!s_have_old) {
184 capture_board();
185 }
186 bool released = false;
187 taskENTER_CRITICAL(&s_mux);
188 released = s_release;
189 s_release = false;
190 taskEXIT_CRITICAL(&s_mux);
191 if (released) {
192 s_phase = PH_SNAKE_REV;
193 s_rev_alt = 0;
194 ESP_LOGI(TAG, "[STAGING] snake reverse (released early)");
195 return;
196 }
197 if (s_step >= 64) {
198 ESP_LOGI(TAG, "[STAGING] snake reached the end");
199 post_result(LED_GESTURE_SNAKE_DONE);
200 return;
201 }
202 s_step++;
203 paint_snake(s_step);
204}

206static void tick_snake_reverse(void) {
207 taskENTER_CRITICAL(&s_mux);
208 s_release = false;
209 taskEXIT_CRITICAL(&s_mux);
210 uint8_t back = (uint8_t)(2u + (s_rev_alt & 1u));
211 s_rev_alt++;
212 if (s_step <= back) {
213 s_step = 0;
214 } else {
215 s_step = (uint8_t)(s_step - back);
216 }
217 if (s_step == 0) {
218 restore_board();
219 ESP_LOGI(TAG, "[STAGING] snake reverse finished, game not started");
220 post_result(LED_GESTURE_SNAKE_CANCELLED);
221 return;
222 }
223 paint_snake(s_step);
224}

226static void tick_shutdown(void) {
227 if (!s_have_old) {
228 capture_board();
229 ensure_black_order();
230 }
231 bool released = false;
232 taskENTER_CRITICAL(&s_mux);
233 released = s_release;
234 s_release = false;
235 taskEXIT_CRITICAL(&s_mux);
236 if (released) {
237 restore_board();
238 ESP_LOGI(TAG, "[STAGING] shutdown cancelled, lights restored");
239 post_result(LED_GESTURE_SHUTDOWN_CANCELLED);
240 return;
241 }
242 /* 0..15 chase, 16..17 whole board dark, then done. */
243 if (s_step >= 18) {
244 for (int i = 0; i < 64; i++) {
245 paint((uint8_t)i, 0);
246 }
247 ESP_LOGI(TAG, "[STAGING] shutdown chase finished");
248 post_result(LED_GESTURE_SHUTDOWN_DONE);
249 return;
250 }
251 if (s_step < 16) {
252 for (int i = 0; i < 64; i++) {
253 paint((uint8_t)i, s_old[i]);
254 }
255 for (uint8_t i = 0; i < s_step; i++) {
256 paint(s_black_leds[i], 0);
257 }
258 paint(s_black_leds[s_step], 0xDCDCDCu);
259 } else {
260 for (int i = 0; i < 64; i++) {
261 paint((uint8_t)i, 0);
262 }
263 }
264 s_step++;
265}

267static void tick_pulse(void) {

286void led_gesture_anim_tick(void) {
287 switch (s_phase) {
288 case PH_WAVE:
289 tick_wave();
290 break;
291 case PH_SNAKE_FWD:
292 tick_snake_forward();
293 break;
294 case PH_SNAKE_REV:
295 tick_snake_reverse();
296 break;
297 case PH_SHUTDOWN:
298 tick_shutdown();
299 break;
300 case PH_PULSE:
301 tick_pulse();
302 break;
303 case PH_IDLE:
304 default:
305 break;
306 }
307}
