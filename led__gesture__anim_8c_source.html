90static uint32_t visible_rgb(uint32_t state) {
91 unsigned w = (state >> 24) & 0xFFu;
92 unsigned r = ((state >> 16) & 0xFFu) + w;
93 unsigned g = ((state >> 8) & 0xFFu) + w;
94 unsigned b = (state & 0xFFu) + w;
95 if (r > 255u) {
96 r = 255u;
97 }
98 if (g > 255u) {
99 g = 255u;
100 }
101 if (b > 255u) {
102 b = 255u;
103 }
104 return (r << 16) | (g << 8) | b;
105}

140static int dist2_half(uint8_t row, uint8_t col) {
141 int dx = (int)col * 2 - 14;
142 int dy = (int)row * 2;
143 return dx * dx + dy * dy;

146static uint8_t smoothstep_u8(uint8_t x) {
147 uint32_t v = (uint32_t)x;
148 uint32_t s = (v * v * (765U - 2U * v)) / (255U * 255U);
149 return (uint8_t)s;
150}

152static uint8_t lerp_u8(uint8_t a, uint8_t b, uint8_t t) {
153 return (uint8_t)(((uint32_t)a * (255U - t) + (uint32_t)b * t) / 255U);
154}

156static uint32_t lerp_rgb(uint32_t a, uint32_t b, uint8_t t) {
157 return ((uint32_t)lerp_u8((uint8_t)((a >> 16) & 0xFFu),
158 (uint8_t)((b >> 16) & 0xFFu), t)
159 << 16) |
160 ((uint32_t)lerp_u8((uint8_t)((a >> 8) & 0xFFu),
161 (uint8_t)((b >> 8) & 0xFFu), t)
162 << 8) |
163 (uint32_t)lerp_u8((uint8_t)(a & 0xFFu), (uint8_t)(b & 0xFFu), t);

166static void tick_wave(void) {
167 if (!s_have_old) {
168 capture_board();
169 }
170 /* Farthest square from h1 is a8 at 14^2+14^2 = 392.
171 * Outward radius 20 covers it. Inward starts past that (23) and shrinks
172 * through h1 (radius -3) so the same crest runs backwards. */
173 const int radius = s_wave_in ? (23 - (int)s_step) : (int)s_step;
174 if (s_wave_in && radius <= -3) {
175 for (int i = 0; i < 64; i++) {
176 paint((uint8_t)i, s_new[i]);
177 }
178 ESP_LOGI(TAG, "[STAGING] wave uncovered the board");
179 post_result(LED_GESTURE_WAVE_DONE);
180 return;
181 }
182 const int radius2 = radius * radius;
183 const int crest_lo = (radius > 2) ? (radius - 2) * (radius - 2) : -1;
184 bool covered = true;
185 for (uint8_t row = 0; row < 8; row++) {
186 for (uint8_t col = 0; col < 8; col++) {
187 int d2 = dist2_half(row, col);
188 uint8_t led = chess_pos_to_led_index(row, col);
189 /* Disk interior is the spreading color. Inward, that disk is the lamp
190 * shrinking toward h1 and the outside is the picture being revealed. */
191 uint32_t outside = s_wave_in ? s_new[led] : s_old[led];
192 uint32_t inside = s_wave_in ? s_old[led] : s_new[led];
193 uint32_t crest = s_cover;
194 int outer = radius + 3;
195 int outer2 = outer * outer;
196 uint32_t color;
197 if (d2 >= outer2) {
198 color = outside;
199 covered = false;
200 } else if (d2 > radius2) {
201 int span = outer2 - radius2;
202 int t = (span > 0) ? ((outer2 - d2) * 255) / span : 255;
203 if (t < 0) {
204 t = 0;
205 } else if (t > 255) {
206 t = 255;
207 }
208 color = lerp_rgb(outside, crest, smoothstep_u8((uint8_t)t));
209 covered = false;
210 } else if (radius > 2 && d2 > crest_lo) {
211 int span = radius2 - crest_lo;
212 int t = (span > 0) ? ((radius2 - d2) * 255) / span : 255;
213 if (t < 0) {
214 t = 0;
215 } else if (t > 255) {
216 t = 255;
217 }
218 color = lerp_rgb(crest, inside, smoothstep_u8((uint8_t)t));
219 } else {
220 color = inside;
221 }
222 paint(led, color);
223 }
224 }
225 if (!s_wave_in && covered && radius >= 20) {
226 for (int i = 0; i < 64; i++) {
227 paint((uint8_t)i, s_new[i]);
228 }
229 ESP_LOGI(TAG, "[STAGING] wave covered the board");
230 post_result(LED_GESTURE_WAVE_DONE);
231 return;
232 }
233 if (s_step < 250) {
234 s_step++;
235 }
236}

239static void snake_square(uint8_t step, uint8_t *row, uint8_t *col) {
240 uint8_t rank = (uint8_t)(step / 8u);
241 uint8_t file = (uint8_t)(step % 8u);
242 if ((rank & 1u) == 0u) {
243 file = (uint8_t)(7u - file);
244 }
245 *row = rank;
246 *col = file;
247}

249/* Head position is in 1/256 of a square. The crest eases in over 3 squares
250 * and the tail fades out over 5, so the wipe is a soft brush. The head must
251 * travel past a8 until that tail leaves the board black. ~3.4 s at 33 ms. */
252#define SWEEP_AHEAD_Q8 768
253#define SWEEP_BEHIND_Q8 1280
254#define SWEEP_CREST 0xFFE6B0u
255#define SNAKE_Q8_END (64u * 256u + SWEEP_BEHIND_Q8)
256#define SNAKE_Q8_FWD 170u
257
258static uint8_t sweep_glow(int dist_q8) {
259 int reach = (dist_q8 >= 0) ? SWEEP_AHEAD_Q8 : SWEEP_BEHIND_Q8;
260 int ad = dist_q8 < 0 ? -dist_q8 : dist_q8;
261 if (ad >= reach) {
262 return 0;
263 }
264 uint8_t t = (uint8_t)((ad * 255) / reach);
265 uint8_t eased = smoothstep_u8(smoothstep_u8(t));
266 return (uint8_t)(255u - eased);
267}

270static void paint_new_game_sweep(uint16_t q8) {
271 int head = (int)q8;
272 for (unsigned i = 0; i < 64u; i++) {
273 uint8_t row;
274 uint8_t col;
275 snake_square((uint8_t)i, &row, &col);
276 uint8_t led = chess_pos_to_led_index(row, col);
277 int dist = (int)i * 256 + 128 - head;
278 uint32_t base = (dist > 0) ? s_old[led] : 0;
279 paint(led, lerp_rgb(base, SWEEP_CREST, sweep_glow(dist)));
280 }
281}

283static uint32_t sweep_button_rgb(uint16_t q8, bool reverse) {
284 unsigned board = 64u * 256u;
285 unsigned level;
286 if (reverse) {
287 level = (q8 >= board) ? 255u : ((unsigned)q8 * 255u) / board;
288 } else if (q8 <= board) {
289 level = 255u;
290 } else {
291 unsigned tail = (unsigned)SNAKE_Q8_END - board;
292 unsigned gone = (unsigned)q8 - board;
293 level = (gone >= tail) ? 0u : 255u - (gone * 255u) / tail;
294 }
295 return lerp_rgb(0, SWEEP_CREST, smoothstep_u8((uint8_t)level));
296}

319 }

321 uint32_t next = (uint32_t)s_snake_q8 + SNAKE_Q8_FWD;

328static void tick_snake_reverse(void) {
329 taskENTER_CRITICAL(&s_mux);
330 s_release = false;
331 taskEXIT_CRITICAL(&s_mux);
332 /* Same step as the way out. At the button the saved picture comes back
333 * and the new game does not start. */
334 if (s_snake_q8 == 0) {
335 restore_board();
336 ESP_LOGI(TAG, "[STAGING] snake reverse finished, game not started");
337 post_result(LED_GESTURE_SNAKE_CANCELLED);
338 return;
339 }
340 if (s_snake_q8 <= SNAKE_Q8_FWD) {
341 s_snake_q8 = 0;
342 } else {
343 s_snake_q8 = (uint16_t)(s_snake_q8 - SNAKE_Q8_FWD);
344 }
345 paint_new_game_sweep(s_snake_q8);
346}

376static uint8_t shut_trail(int head_q8, int index) {
377 int d = head_q8 - index * 256;
378 if (d <= 0) {
379 return 255;
380 }
381 if (d >= SHUT_TAIL_Q8) {
382 return 0;
383 }
384 uint8_t t = (uint8_t)((d * 255) / SHUT_TAIL_Q8);
385 return (uint8_t)(255u - smoothstep_u8(t));
386}

389static uint8_t shut_glyph_level(uint8_t rc, uint8_t frame) {
390 int ring_i = -1;
391 int stem_i = -1;
392 for (int i = 0; i < SHUT_RING_N; i++) {
393 if (s_shut_ring[i] == rc) {
394 ring_i = i;
395 }
396 }
397 for (int i = 0; i < SHUT_STEM_N; i++) {
398 if (s_shut_stem[i] == rc) {
399 stem_i = i;
400 }
401 }
402 if (ring_i < 0 && stem_i < 0) {
403 return 0;
404 }
405 if (frame < SHUT_HOLD_END) {
406 return 255;
407 }
408 if (frame < SHUT_RING_END) {
409 if (stem_i >= 0) {
410 return 255;
411 }
412 int span = SHUT_RING_N * 256 + SHUT_TAIL_Q8;
413 int head = ((int)frame - SHUT_HOLD_END) * span / SHUT_RING_FRAMES;
414 return shut_trail(head, ring_i);
415 }
416 if (frame < SHUT_STEM_END && stem_i >= 0) {
417 int span = SHUT_STEM_N * 256 + SHUT_TAIL_Q8;
418 int head = ((int)frame - SHUT_RING_END) * span / SHUT_STEM_FRAMES;
419 return shut_trail(head, stem_i);
420 }
421 return 0;
422}

431static void paint_shutdown(uint8_t frame) {
432 uint8_t intro = shut_intro(frame);
433 for (uint8_t row = 0; row < 8; row++) {
434 for (uint8_t col = 0; col < 8; col++) {
435 uint8_t led = chess_pos_to_led_index(row, col);
436 uint8_t rc = (uint8_t)((row << 3) | col);
437 uint32_t bg = (frame >= SHUT_FADE) ? 0 : lerp_rgb(s_old[led], 0, intro);
438 uint8_t level = shut_glyph_level(rc, frame);
439 if (level == 0) {
440 paint(led, bg);
441 } else {
442 uint8_t t = (uint8_t)(((unsigned)level * intro) / 255u);
443 paint(led, lerp_rgb(bg, SHUT_RED, t));
444 }
445 }
446 }

461static void tick_shutdown(void) {
462 if (!s_have_old) {
463 capture_board();
464 }
465 bool released = false;
466 taskENTER_CRITICAL(&s_mux);
467 released = s_release;
468 s_release = false;
469 taskEXIT_CRITICAL(&s_mux);
470 if (released && !s_shut_back) {
471 s_shut_back = true;
472 }
473 /* Release runs the same frames backwards and does not power off. */
474 if (s_shut_back) {

477 s_shut_back = false;
478 ESP_LOGI(TAG, "[STAGING] shutdown reversed, power stays on");
479 post_result(LED_GESTURE_SHUTDOWN_CANCELLED);
480 return;
481 }
482 s_step--;
483 paint_shutdown(s_step);
484 return;
485 }
486 if (s_step >= SHUT_DONE) {

488 paint((uint8_t)i, 0);

492 post_result(LED_GESTURE_SHUTDOWN_DONE);
493 return;
494 }
495 paint_shutdown(s_step);
496 s_step++;
497}

526static void tick_pair_hold(void) {
527 if (!s_pair_captured) {
528 capture_board();
529 s_pair_captured = true;
530 s_pair_step = 0;
531 return;
532 }
533 /* 0..47 rise, 48..95 fall. Smoothstep keeps the turn at the top soft. */
534 uint8_t tri = s_pair_step < 48 ? s_pair_step : (uint8_t)(96 - s_pair_step);
535 uint8_t shaped = smoothstep_u8((uint8_t)((unsigned)tri * 255u / 48u));
536 uint8_t amount = (uint8_t)((unsigned)shaped * 140u / 255u);
537 s_pair_amount = amount;

539 uint32_t logo = pair_logo_rgb(amount);
540 for (int i = 0; i < 64; i++) {
541 paint((uint8_t)i, color);

543 for (unsigned i = 0; i < sizeof(s_ble_logo); i++) {
544 uint8_t rc = s_ble_logo[i];
545 paint(chess_pos_to_led_index((uint8_t)(rc >> 3), (uint8_t)(rc & 7u)), logo);
546 }
547 s_pair_step++;
548 if (s_pair_step >= 96) {
549 s_pair_step = 0;

618static void paint_gesture_button(void) {

620 paint(GESTURE_BUTTON_LED, 0);
621 s_btn_leave_dark = false;
622 s_btn_owned = false;
623 s_btn_saved = false;

636 uint8_t t = smoothstep_u8((uint8_t)num);
637 color = s_wave_in ? lerp_rgb(s_cover, 0, t) : lerp_rgb(0, s_cover, t);
638 own = true;
639 } else if (s_phase == PH_SNAKE_FWD || s_phase == PH_SNAKE_REV) {
640 color = sweep_button_rgb(s_snake_q8, s_phase == PH_SNAKE_REV);
641 own = true;
642 } else if (s_phase == PH_SHUTDOWN) {
643 uint8_t frame = (s_shut_back || s_step == 0) ? s_step : (uint8_t)(s_step - 1);
644 color = shutdown_button_rgb(frame);
645 own = true;
646 } else if (s_pair_hold && s_pair_captured && s_phase == PH_IDLE) {
647 color = pair_rgb(s_pair_amount);
648 own = true;
649 } else if (s_hold_level > 0) {
650 color = lerp_rgb(s_btn_rgb, 0xFF6600u, smoothstep_u8(s_hold_level));
651 own = true;
652 }
653
654 if (own) {
655 if (!s_btn_saved) {
656 uint32_t all[73];
657 memset(all, 0, sizeof(all));
658 led_get_all_states(all, 73);
659 s_btn_rgb = all[GESTURE_BUTTON_LED];
660 s_btn_saved = true;
661 }
662 paint(GESTURE_BUTTON_LED, color);
663 s_btn_owned = true;
664 return;
665 }
666 if (s_btn_owned) {
667 paint(GESTURE_BUTTON_LED, s_btn_rgb);
668 s_btn_owned = false;
669 s_btn_saved = false;
670 s_pair_amount = 0;
671 }

674void led_gesture_anim_tick(void) {
675#if CONFIG_CHESS_GESTURE_BUTTON
676 /* Follow the pin here too. The button task owns the gesture timing, but
677 * the button LED must turn orange even if that task has not published yet. */
678 s_button_held = led_gesture_button_down();
679#endif
680 if (s_phase != PH_IDLE && s_hold_lit) {
681 s_hold_level = 0;
682 hold_restore();
683 }
684 bool yield = false;
685 bool stop = false;
686 taskENTER_CRITICAL(&s_mux);
687 yield = s_pair_yield;
688 s_pair_yield = false;
689 stop = s_pair_stop;
690 s_pair_stop = false;
691 taskEXIT_CRITICAL(&s_mux);
692 if (yield && s_pair_captured) {
693 restore_board();
694 s_pair_captured = false;
695 s_have_old = false;
696 }
697 if (stop && s_pair_captured && s_phase == PH_IDLE) {
698 restore_board();
699 s_pair_captured = false;
700 }
701 switch (s_phase) {
702 case PH_WAVE:
703 tick_wave();
704 break;
705 case PH_SNAKE_FWD:
706 tick_snake_forward();
707 break;
708 case PH_SNAKE_REV:
709 tick_snake_reverse();
710 break;
711 case PH_SHUTDOWN:
712 tick_shutdown();
713 break;
714 case PH_PULSE:
715 case PH_IDLE:
716 default:
717 break;
718 }
719 if (s_pair_hold && s_phase == PH_IDLE && !stop) {
720 tick_pair_hold();
721 }
722 hold_preview();
723 paint_gesture_button();
724 if (s_phase != PH_IDLE || s_hold_lit || s_btn_owned) {
725 led_force_immediate_update();
726 }
727}
