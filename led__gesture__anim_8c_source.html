84static void gesture_sample_dt(void) {
85 uint32_t now = (uint32_t)(esp_timer_get_time() / 1000);
86 uint32_t dt;
87 if (s_anim_last == 0U) {
88 s_anim_last = now;
89 s_dt_ms = GESTURE_FRAME_MS;
90 return;
91 }
92 dt = (uint32_t)(now - s_anim_last);
93 s_anim_last = now;
94 if (dt == 0U || dt > 200U) {
95 dt = GESTURE_FRAME_MS;
96 }
97 s_dt_ms = dt;
98}

100static uint32_t frame_steps(uint32_t *acc) {
101 uint32_t n;
102 *acc += s_dt_ms;
103 n = *acc / GESTURE_FRAME_MS;
104 *acc %= GESTURE_FRAME_MS;
105 return n;

140static uint32_t visible_rgb(uint32_t state) {
141 unsigned w = (state >> 24) & 0xFFu;
142 unsigned r = ((state >> 16) & 0xFFu) + w;
143 unsigned g = ((state >> 8) & 0xFFu) + w;
144 unsigned b = (state & 0xFFu) + w;
145 if (r > 255u) {
146 r = 255u;
147 }
148 if (g > 255u) {
149 g = 255u;
150 }
151 if (b > 255u) {
152 b = 255u;
153 }
154 return (r << 16) | (g << 8) | b;
155}

196static int dist2_half(uint8_t row, uint8_t col) {

202static uint8_t smoothstep_u8(uint8_t x) {
203 uint32_t v = (uint32_t)x;
204 uint32_t s = (v * v * (765U - 2U * v)) / (255U * 255U);
205 return (uint8_t)s;
206}

208static uint8_t lerp_u8(uint8_t a, uint8_t b, uint8_t t) {
209 return (uint8_t)(((uint32_t)a * (255U - t) + (uint32_t)b * t) / 255U);
210}

212static uint32_t lerp_rgb(uint32_t a, uint32_t b, uint8_t t) {
213 return ((uint32_t)lerp_u8((uint8_t)((a >> 16) & 0xFFu),
214 (uint8_t)((b >> 16) & 0xFFu), t)
215 << 16) |
216 ((uint32_t)lerp_u8((uint8_t)((a >> 8) & 0xFFu),
217 (uint8_t)((b >> 8) & 0xFFu), t)
218 << 8) |
219 (uint32_t)lerp_u8((uint8_t)(a & 0xFFu), (uint8_t)(b & 0xFFu), t);
220}

222static void tick_wave(void) {
223 if (!s_have_old) {
224 capture_board();
225 }
226 /* Farthest square from h1 is a8 at 14^2+14^2 = 392.
227 * Outward radius 20 covers it. Inward starts past that (23) and shrinks
228 * through h1 (radius -3) so the same crest runs backwards. */
229 const int radius = s_wave_in ? (23 - (int)s_step) : (int)s_step;
230 if (s_wave_in && radius <= -3) {
231 paint_squares(s_new);
232 ESP_LOGI(TAG, "[STAGING] wave uncovered the board");
233 post_result(LED_GESTURE_WAVE_DONE);
234 return;
235 }
236 const int radius2 = radius * radius;
237 const int crest_lo = (radius > 2) ? (radius - 2) * (radius - 2) : -1;
238 bool covered = true;
239 for (uint8_t row = 0; row < 8; row++) {
240 for (uint8_t col = 0; col < 8; col++) {
241 int d2 = dist2_half(row, col);
242 uint8_t sq = sq_of(row, col);
243 /* Disk interior is the spreading color. Inward, that disk is the lamp
244 * shrinking toward h1 and the outside is the picture being revealed. */
245 uint32_t outside = s_wave_in ? s_new[sq] : s_old[sq];
246 uint32_t inside = s_wave_in ? s_old[sq] : s_new[sq];
247 uint32_t crest = s_cover;
248 int outer = radius + 3;
249 int outer2 = outer * outer;
250 uint32_t color;
251 if (d2 >= outer2) {
252 color = outside;
253 covered = false;
254 } else if (d2 > radius2) {
255 int span = outer2 - radius2;
256 int t = (span > 0) ? ((outer2 - d2) * 255) / span : 255;
257 if (t < 0) {
258 t = 0;
259 } else if (t > 255) {
260 t = 255;
261 }
262 color = lerp_rgb(outside, crest, smoothstep_u8((uint8_t)t));
263 covered = false;
264 } else if (radius > 2 && d2 > crest_lo) {
265 int span = radius2 - crest_lo;
266 int t = (span > 0) ? ((radius2 - d2) * 255) / span : 255;
267 if (t < 0) {
268 t = 0;
269 } else if (t > 255) {
270 t = 255;
271 }
272 color = lerp_rgb(crest, inside, smoothstep_u8((uint8_t)t));
273 } else {
274 color = inside;
275 }
276 paint_square(row, col, color);
277 }
278 }
279 if (!s_wave_in && covered && radius >= 20) {
280 paint_squares(s_new);
281 ESP_LOGI(TAG, "[STAGING] wave covered the board");
282 post_result(LED_GESTURE_WAVE_DONE);
283 return;
284 }
285 if (s_step < 250) {
286 uint32_t add = frame_steps(&s_wave_acc);
287 if (add > (uint32_t)(250 - s_step)) {
288 add = (uint32_t)(250 - s_step);
289 }
290 s_step = (uint8_t)(s_step + add);
291 }
292}

295static void snake_square(uint8_t step, uint8_t *row, uint8_t *col) {
296 uint8_t rank = (uint8_t)(step / 8u);
297 uint8_t file = (uint8_t)(step % 8u);
298 if ((rank & 1u) == 0u) {
299 file = (uint8_t)(7u - file);
300 }
301 *row = rank;
302 *col = file;
303}

305/* Head position is in 1/256 of a square. The crest eases in over 3 squares
306 * and the tail fades out over 5, so the wipe is a soft brush. The head must
307 * travel past a8 until that tail leaves the board black. One pass, ~0.85 s. */
308#define SWEEP_AHEAD_Q8 768
309#define SWEEP_BEHIND_Q8 1280
310#define SWEEP_CREST 0xFFE6B0u
311#define SNAKE_Q8_END (64u * 256u + SWEEP_BEHIND_Q8)
312#define SNAKE_Q8_FWD 680u
313
314static uint32_t snake_q8_delta(void) {
315 uint32_t num = s_snake_acc + s_dt_ms * SNAKE_Q8_FWD;
316 uint32_t step = num / GESTURE_FRAME_MS;
317 s_snake_acc = num % GESTURE_FRAME_MS;
318 return step;
319}

321static uint8_t sweep_glow(int dist_q8) {
322 int reach = (dist_q8 >= 0) ? SWEEP_AHEAD_Q8 : SWEEP_BEHIND_Q8;
323 int ad = dist_q8 < 0 ? -dist_q8 : dist_q8;
324 if (ad >= reach) {
325 return 0;
326 }
327 uint8_t t = (uint8_t)((ad * 255) / reach);
328 uint8_t eased = smoothstep_u8(smoothstep_u8(t));
329 return (uint8_t)(255u - eased);
330}

333static void paint_new_game_sweep(uint16_t q8) {
334 int head = (int)q8;
335 for (unsigned i = 0; i < 64u; i++) {
336 uint8_t row;
337 uint8_t col;
338 snake_square((uint8_t)i, &row, &col);
339 int dist = (int)i * 256 + 128 - head;
340 uint32_t base = (dist > 0) ? s_old[sq_of(row, col)] : 0;
341 paint_square(row, col, lerp_rgb(base, SWEEP_CREST, sweep_glow(dist)));
342 }
343}

345static uint32_t sweep_button_rgb(uint16_t q8, bool reverse) {
346 unsigned board = 64u * 256u;
347 unsigned level;
348 if (reverse) {
349 level = (q8 >= board) ? 255u : ((unsigned)q8 * 255u) / board;
350 } else if (q8 <= board) {
351 level = 255u;
352 } else {
353 unsigned tail = (unsigned)SNAKE_Q8_END - board;
354 unsigned gone = (unsigned)q8 - board;
355 level = (gone >= tail) ? 0u : 255u - (gone * 255u) / tail;
356 }
357 return lerp_rgb(0, SWEEP_CREST, smoothstep_u8((uint8_t)level));
358}

360static void tick_snake_forward(void) {
361 if (!s_have_old) {
362 capture_board();
363 }
364 bool released = false;
365 taskENTER_CRITICAL(&s_mux);
366 released = s_release;
367 s_release = false;
368 taskEXIT_CRITICAL(&s_mux);
369 if (released) {
370 s_phase = PH_SNAKE_REV;
371 ESP_LOGI(TAG, "[STAGING] snake reverse (released early)");
372 return;
373 }
374 if (s_snake_q8 >= SNAKE_Q8_END) {
375 ESP_LOGI(TAG, "[STAGING] new-game sweep finished");
376 post_result(LED_GESTURE_SNAKE_DONE);
377 return;
378 }
379 paint_new_game_sweep(s_snake_q8);
380 {
381 uint32_t next = (uint32_t)s_snake_q8 + snake_q8_delta();
382 if (next > SNAKE_Q8_END) {
383 next = SNAKE_Q8_END;
384 }
385 s_snake_q8 = (uint16_t)next;
386 }
387}

389static void tick_snake_reverse(void) {
390 if (s_button_held) {
391 taskENTER_CRITICAL(&s_mux);
392 s_release = false;
393 taskEXIT_CRITICAL(&s_mux);
394 s_phase = PH_SNAKE_FWD;
395 tick_snake_forward();
396 return;
397 }
398 taskENTER_CRITICAL(&s_mux);
399 s_release = false;
400 taskEXIT_CRITICAL(&s_mux);
401 /* Same step as the way out. At the button the saved picture comes back
402 * and the new game does not start. */
403 if (s_snake_q8 == 0) {
404 restore_board();
405 ESP_LOGI(TAG, "[STAGING] snake reverse finished, game not started");
406 post_result(LED_GESTURE_SNAKE_CANCELLED);
407 return;
408 }
409 {
410 uint32_t delta = snake_q8_delta();
411 if ((uint32_t)s_snake_q8 <= delta) {
412 s_snake_q8 = 0;
413 } else {
414 s_snake_q8 = (uint16_t)(s_snake_q8 - delta);
415 }
416 }
417 paint_new_game_sweep(s_snake_q8);
418}

435/* Chess {row, col}. Row 0 is rank 1, col 0 is the a-file. Clockwise from
436 * the right shoulder of the gap (f7), then the bar, tip first (d8, e8). */
437static const uint8_t s_shut_ring[SHUT_RING_N][2] = {
438 {6, 5}, {5, 6}, {4, 7}, {3, 7}, {2, 7}, {1, 6}, {0, 5}, {0, 4},
439 {0, 3}, {0, 2}, {1, 1}, {2, 0}, {3, 0}, {4, 0}, {5, 1}, {6, 2},
440};
441static const uint8_t s_shut_stem[SHUT_STEM_N][2] = {
442 {7, 3}, {7, 4}, {5, 3}, {5, 4}, {4, 3}, {4, 4}, {3, 3}, {3, 4},
443};

446static uint8_t shut_trail(int head_q8, int index) {
447 int d = head_q8 - index * 256;
448 if (d <= 0) {
449 return 255;
450 }
451 if (d >= SHUT_TAIL_Q8) {
452 return 0;
453 }
454 uint8_t t = (uint8_t)((d * 255) / SHUT_TAIL_Q8);
455 return (uint8_t)(255u - smoothstep_u8(t));
456}

459static uint8_t shut_glyph_level(uint8_t row, uint8_t col, uint8_t frame) {
460 int ring_i = -1;
461 int stem_i = -1;
462 for (int i = 0; i < SHUT_RING_N; i++) {
463 if (s_shut_ring[i][0] == row && s_shut_ring[i][1] == col) {
464 ring_i = i;
465 }
466 }
467 for (int i = 0; i < SHUT_STEM_N; i++) {
468 if (s_shut_stem[i][0] == row && s_shut_stem[i][1] == col) {
469 stem_i = i;
470 }
471 }
472 if (ring_i < 0 && stem_i < 0) {
473 return 0;
474 }
475 if (frame < SHUT_HOLD_END) {
476 return 255;
477 }
478 if (frame < SHUT_RING_END) {
479 if (stem_i >= 0) {
480 return 255;
481 }
482 int span = SHUT_RING_N * 256 + SHUT_TAIL_Q8;
483 int head = ((int)frame - SHUT_HOLD_END) * span / SHUT_RING_FRAMES;
484 return shut_trail(head, ring_i);
485 }
486 if (frame < SHUT_STEM_END && stem_i >= 0) {
487 int span = SHUT_STEM_N * 256 + SHUT_TAIL_Q8;
488 int head = ((int)frame - SHUT_RING_END) * span / SHUT_STEM_FRAMES;
489 return shut_trail(head, stem_i);
490 }
491 return 0;
492}

530static void tick_shutdown(void) {
531 if (!s_have_old) {
532 capture_board();
533 }
534 bool released = false;
535 taskENTER_CRITICAL(&s_mux);
536 released = s_release;
537 s_release = false;
538 taskEXIT_CRITICAL(&s_mux);
539 if (released && !s_shut_back) {
540 s_shut_back = true;
541 }
542 /* Release runs the same frames backwards and does not power off. */
543 if (s_shut_back) {
544 if (s_step == 0) {
545 restore_board();
546 s_shut_back = false;
547 ESP_LOGI(TAG, "[STAGING] shutdown reversed, power stays on");
548 post_result(LED_GESTURE_SHUTDOWN_CANCELLED);
549 return;
550 }
551 {
552 uint32_t add = frame_steps(&s_shut_acc);
553 if (add >= s_step) {
554 s_step = 0;
555 } else {
556 s_step = (uint8_t)(s_step - add);
557 }
558 }
559 paint_shutdown(s_step);
560 return;
561 }
562 if (s_step >= SHUT_DONE) {
563 for (uint8_t row = 0; row < 8; row++) {
564 for (uint8_t col = 0; col < 8; col++) {
565 paint_square(row, col, 0);
566 }
567 }
568 s_btn_leave_dark = true;
569 ESP_LOGI(TAG, "[STAGING] shutdown power symbol finished");
570 post_result(LED_GESTURE_SHUTDOWN_DONE);
571 return;
572 }
573 paint_shutdown(s_step);
574 {
575 uint32_t add = frame_steps(&s_shut_acc);
576 if ((uint32_t)s_step + add > 255U) {
577 s_step = 255;
578 } else {
579 s_step = (uint8_t)(s_step + add);
580 }
581 }
582}

608static void tick_pair_hold(void) {
609 if (!s_pair_captured) {
610 capture_board();
611 s_pair_captured = true;
612 s_pair_step = 0;
613 return;
614 }
615 /* 0..47 rise, 48..95 fall. Smoothstep keeps the turn at the top soft. */
616 uint8_t tri = s_pair_step < 48 ? s_pair_step : (uint8_t)(96 - s_pair_step);
617 uint8_t shaped = smoothstep_u8((uint8_t)((unsigned)tri * 255u / 48u));
618 uint8_t amount = (uint8_t)((unsigned)shaped * 140u / 255u);
619 s_pair_amount = amount;
620 uint32_t color = pair_rgb(amount);
621 uint32_t logo = pair_logo_rgb(amount);
622 for (uint8_t row = 0; row < 8; row++) {
623 for (uint8_t col = 0; col < 8; col++) {
624 paint_square(row, col, color);
625 }
626 }
627 for (unsigned i = 0; i < sizeof(s_ble_logo) / sizeof(s_ble_logo[0]); i++) {
628 paint_square(s_ble_logo[i][0], s_ble_logo[i][1], logo);
629 }
630 {
631 uint32_t add = frame_steps(&s_pair_acc);
632 s_pair_step = (uint8_t)((s_pair_step + add) % 96U);
633 }

644static void hold_step(uint8_t target) {
645 uint32_t num = s_hold_acc + s_dt_ms * 48U;
646 uint32_t step = num / GESTURE_FRAME_MS;
647 s_hold_acc = num % GESTURE_FRAME_MS;
648 if (step > 255U) {
649 step = 255U;
650 }
651 if (s_hold_level < target) {
652 uint8_t room = (uint8_t)(target - s_hold_level);
653 if (step > room) {
654 step = room;
655 }
656 s_hold_level = (uint8_t)(s_hold_level + step);
657 } else if (s_hold_level > target) {
658 uint8_t room = (uint8_t)(s_hold_level - target);
659 if (step > room) {
660 step = room;
661 }
662 s_hold_level = (uint8_t)(s_hold_level - step);
663 }
664}

716static void paint_gesture_button(void) {
717 if (s_btn_leave_dark) {
718 paint(GESTURE_BUTTON_LED, 0);
719 s_btn_leave_dark = false;
720 s_btn_owned = false;
721 s_btn_saved = false;
722 s_pair_amount = 0;
723 return;
724 }
725
726 bool own = false;
727 uint32_t color = 0;
728 if (s_phase == PH_WAVE) {
729 unsigned span = s_wave_in ? 26u : 20u;
730 unsigned num = (unsigned)s_step * 255u / span;
731 if (num > 255u) {
732 num = 255u;
733 }
734 uint8_t t = smoothstep_u8((uint8_t)num);
735 color = s_wave_in ? lerp_rgb(s_cover, 0, t) : lerp_rgb(0, s_cover, t);
736 own = true;
737 } else if (s_phase == PH_SNAKE_FWD || s_phase == PH_SNAKE_REV) {
738 color = sweep_button_rgb(s_snake_q8, s_phase == PH_SNAKE_REV);
739 own = true;
740 } else if (s_phase == PH_SHUTDOWN) {
741 uint8_t frame = (s_shut_back || s_step == 0) ? s_step : (uint8_t)(s_step - 1);
742 color = shutdown_button_rgb(frame);
743 own = true;
744 } else if (s_pair_hold && s_pair_captured && s_phase == PH_IDLE) {
745 color = pair_rgb(s_pair_amount);
746 own = true;
747 } else if (s_hold_level > 0) {
748 color = lerp_rgb(s_btn_rgb, 0xFF6600u, smoothstep_u8(s_hold_level));
749 own = true;
750 }
751
752 if (own) {
753 if (!s_btn_saved) {
754 uint32_t all[73];
755 memset(all, 0, sizeof(all));
756 led_get_all_states(all, 73);
757 s_btn_rgb = all[GESTURE_BUTTON_LED];
758 s_btn_saved = true;
759 }
760 paint(GESTURE_BUTTON_LED, color);
761 s_btn_owned = true;
762 return;
763 }
764 if (s_btn_owned) {
765 paint(GESTURE_BUTTON_LED, s_btn_rgb);
766 s_btn_owned = false;
767 s_btn_saved = false;
768 s_pair_amount = 0;
769 }
770}

772void led_gesture_anim_tick(void) {
773 gesture_sample_dt();
774#if CONFIG_CHESS_GESTURE_BUTTON
775 /* Follow the pin here too. The button task owns the gesture timing, but
776 * the button LED must turn orange even if that task has not published yet. */
777 s_button_held = led_gesture_button_down();
778#endif
779 if (s_phase != PH_IDLE && s_hold_lit) {
780 s_hold_level = 0;
781 hold_restore();
782 }
783 bool yield = false;
784 bool stop = false;
785 taskENTER_CRITICAL(&s_mux);
786 yield = s_pair_yield;
787 s_pair_yield = false;
788 stop = s_pair_stop;
789 s_pair_stop = false;
790 taskEXIT_CRITICAL(&s_mux);
791 if (yield && s_pair_captured) {
792 restore_board();
793 s_pair_captured = false;
794 s_have_old = false;
795 }
796 if (stop && s_pair_captured && s_phase == PH_IDLE) {
797 restore_board();
798 s_pair_captured = false;
799 }
800 switch (s_phase) {
801 case PH_WAVE:
802 tick_wave();
803 break;
804 case PH_SNAKE_FWD:
805 tick_snake_forward();
806 break;
807 case PH_SNAKE_REV:
808 tick_snake_reverse();
809 break;
810 case PH_SHUTDOWN:
811 tick_shutdown();
812 break;
813 case PH_PULSE:
814 case PH_IDLE:
815 default:
816 break;
817 }
818 if (s_pair_hold && s_phase == PH_IDLE && !stop) {
819 tick_pair_hold();
820 }
821 hold_preview();
822 paint_gesture_button();
823 if (s_phase != PH_IDLE || s_hold_lit || s_btn_owned) {
824 led_force_immediate_update();
825 }
826}
