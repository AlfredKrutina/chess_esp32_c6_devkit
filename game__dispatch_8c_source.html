176/*
177 * Command queue carries game_cmd_t. Only game_submit() calls xQueueSend.
178 *
179 * kind | sender | receiver
180 * GCMD_NEW_GAME main.c initialize_chess_game (skip if snapshot or boot
181 * already started), execute_demo_move; game_task.c auto
182 * start; web/BLE new | game_start_new_game
183 * GCMD_NEW_GAME_FEN web/BLE new + fen | game_start_new_game_from_fen
184 * GCMD_UNDO BLE cmd undo (no HTTP undo route) | game_undo_rewind_last_move
185 * GCMD_MOVE web_game_enqueue_move; BLE move | game_process_chess_move
186 * GCMD_BOARD_LIFT squares pickup; virtual pickup; demo | game_process_pickup_command
187 * GCMD_BOARD_DROP reed drop (from and to); Hall place (empty from);
188 * virtual drop; demo | game_process_drop_command
189 * GCMD_BOARD_GUARD squares_send_guard_command (four masks and action) |
190 * game_squares_guard_handle_command
191 * GCMD_RESIGN web_resign_dispatch | game_resign_from_app
192 * GCMD_AGREE_DRAW web_resign_dispatch draw | game_agree_draw
193 * GCMD_TIMER_SET HTTP/BLE timer config | game_process_timer_command
194 * GCMD_TIMER_PAUSE HTTP/BLE timer pause | game_process_timer_command
195 * GCMD_TIMER_RESUME HTTP/BLE timer resume | game_process_timer_command
196 * GCMD_TIMER_RESET HTTP/BLE timer reset | game_process_timer_command
197 * GCMD_MODE_PUZZLE web puzzle parser (210/211 -> apply/start staged) | puzzle acts
198 * GCMD_MODE_OPENING web_opening_dispatch cancel/start | opening acts
199 * GCMD_MODE_TUTORIAL tutorial start/cancel/finish | tutorial functions
200 * GCMD_PROMOTION_CHOICE virtual promote HTTP/BLE | game_process_promotion_command
201 *
202 * Not this queue:
203 * - button_event_queue in this function: id 0-3 -> game_process_promotion_button;
204 * id 8 -> game_reset_game
205 * - timer expiry: game_update_timer_display in the game_task loop
206 * - uart_task does not enqueue; BLE calls web_server_ble_command_dispatch
207 * - status, board, history: game_get_*_json
208 */
209void game_process_commands(void) {
210 game_led_flush();
211 game_physical_undo_tick();
212 resignation_tick();
213
214 if (game_command_queue != NULL) {
215 game_cmd_t cmd;
216 uint32_t commands_processed = 0;
217 const uint32_t MAX_COMMANDS_PER_CYCLE = 15;
218 TickType_t wait = pdMS_TO_TICKS(25);
219
220 while (xQueueReceive(game_command_queue, &cmd, wait) == pdTRUE) {
221 wait = 0;
222 commands_processed++;
223 ESP_LOGI(TAG, "Received command kind: %d (processed: %lu/%lu)",
224 (int)cmd.kind, commands_processed, MAX_COMMANDS_PER_CYCLE);
225 ha_light_report_activity("command");
226
227 switch (cmd.kind) {
228 case GCMD_NEW_GAME:
229 ESP_LOGI(TAG, "Processing NEW GAME");
230 game_start_new_game();
231 if (current_game_state == GAME_STATE_WAITING_FOR_BOARD_SETUP) {
232 game_cmd_reply("Waiting for pieces on start squares",
233 false, cmd.response_queue);
234 } else {
235 game_cmd_reply("New game started, white to move", false,
236 cmd.response_queue);
237 }
238 break;
239
240 case GCMD_NEW_GAME_FEN:
241 ESP_LOGI(TAG, "Processing NEW_GAME_FEN");
242 if (!game_start_new_game_from_fen(cmd.u.fen.fen)) {
243 game_cmd_reply("FEN rejected", true, cmd.response_queue);
244 } else if (current_player == PLAYER_BLACK) {
245 game_cmd_reply("New game from FEN, black to move", false,
246 cmd.response_queue);
247 } else {
248 game_cmd_reply("New game from FEN, white to move", false,
249 cmd.response_queue);
250 }
251 break;
252
253 case GCMD_UNDO:
254 ESP_LOGI(TAG, "Processing UNDO");
255 if (game_undo_rewind_last_move()) {
256 game_cmd_reply("Last move undone", false,
257 cmd.response_queue);
258 } else {
259 game_cmd_reply("Nothing to undo", true, cmd.response_queue);
260 }
261 break;
262
263 case GCMD_MOVE:
264 ESP_LOGI(TAG, "Processing MOVE: %s -> %s", game_cmd_from(&cmd),
265 game_cmd_to(&cmd));
266 game_process_chess_move(&cmd);
267 break;
268
269 case GCMD_BOARD_LIFT:
270 ESP_LOGI(TAG, "Processing BOARD_LIFT: %s", game_cmd_from(&cmd));
271 game_process_pickup_command(&cmd);
272 break;
273
274 case GCMD_BOARD_DROP:
275 ESP_LOGI(TAG, "Processing BOARD_DROP: %s", game_cmd_to(&cmd));
276 game_process_drop_command(&cmd);
277 break;
278
279 case GCMD_BOARD_GUARD:
280 ESP_LOGW(TAG, "Processing BOARD_GUARD");
281 game_squares_guard_handle_command(&cmd);
282 break;
283
284 case GCMD_RESIGN:
285 ESP_LOGI(TAG, "Processing RESIGN (side=%u)",
286 (unsigned)cmd.u.resign.side);
287 if (!game_resign_from_app(cmd.u.resign.side)) {
288 game_cmd_reply("Resign refused", true, cmd.response_queue);
289 }
290 break;
291
292 case GCMD_AGREE_DRAW:
293 ESP_LOGI(TAG, "Processing DRAW");
294 if (!game_agree_draw()) {
295 game_cmd_reply("Draw refused", true, cmd.response_queue);
296 } else {
297 game_cmd_reply("Draw agreed", false, cmd.response_queue);
298 }
299 break;
300
301 case GCMD_TIMER_SET:
302 case GCMD_TIMER_PAUSE:
303 case GCMD_TIMER_RESUME:
304 case GCMD_TIMER_RESET:
305 ESP_LOGI(TAG, "Processing timer kind %d", (int)cmd.kind);
306 game_process_timer_command(&cmd);
307 break;
308
309 case GCMD_MODE_TUTORIAL:
310 ESP_LOGI(TAG, "TUTORIAL act=%u", (unsigned)cmd.u.mode.act);
311 switch (cmd.u.mode.act) {
312 case TUTORIAL_ACT_START:
313 game_enter_board_setup_tutorial();
314 break;
315 case TUTORIAL_ACT_CANCEL:
316 game_exit_board_setup_tutorial(false);
317 break;
318 case TUTORIAL_ACT_FINISH:
319 if (!game_finish_board_setup_tutorial_from_web()) {
320 game_cmd_reply("Tutorial finish refused", true,
321 cmd.response_queue);
322 } else {
323 game_cmd_reply("New game started", false,
324 cmd.response_queue);
325 }
326 break;
327 default:
328 ESP_LOGW(TAG, "TUTORIAL unknown action");
329 break;
330 }
331 break;
332
333 case GCMD_MODE_PUZZLE: {
334 uint8_t act = cmd.u.mode.act;
335 uint8_t id = cmd.u.mode.builtin_id;
336 ESP_LOGI(TAG, "PUZZLE act=%u id=%u", (unsigned)act, (unsigned)id);
337 if (act == PUZZLE_ACT_CANCEL) {
338 game_puzzle_cancel();
339 } else if (act == PUZZLE_ACT_APPLY_SETUP) {
340 if (!game_puzzle_apply_staged_setup()) {
341 ESP_LOGW(TAG, "PUZZLE apply setup failed (no staged line)");
342 }
343 } else if (act == PUZZLE_ACT_START_STAGED) {
344 if (!game_puzzle_try_start_staged()) {
345 ESP_LOGW(TAG, "PUZZLE staged start refused");
346 }
347 } else if (act == PUZZLE_ACT_PREPARE) {
348 if (!game_puzzle_enter_setup(id)) {
349 ESP_LOGW(TAG, "PUZZLE prepare failed id=%u", (unsigned)id);
350 }
351 } else if (act == PUZZLE_ACT_START) {
352 if (!game_puzzle_start(id)) {
353 ESP_LOGW(TAG, "PUZZLE start failed for id=%u", (unsigned)id);
354 }
355 } else {
356 ESP_LOGW(TAG, "PUZZLE unknown act=%u", (unsigned)act);
357 }
358 break;
359 }
360
361 case GCMD_MODE_OPENING:
362 ESP_LOGI(TAG, "OPENING act=%u", (unsigned)cmd.u.mode.act);
363 switch (cmd.u.mode.act) {
364 case OPENING_ACT_CANCEL:
365 game_opening_cancel();
366 break;
367 case OPENING_ACT_START:
368 if (!game_opening_start()) {
369 ESP_LOGW(TAG, "OPENING start failed");
370 }
371 break;
372 case OPENING_ACT_HINT:
373 game_opening_hint();
374 break;
375 case OPENING_ACT_CHECKPOINT:
376 if (!game_opening_checkpoint_ack()) {
377 ESP_LOGW(TAG, "OPENING checkpoint_ack failed");
378 }
379 break;
380 default:
381 ESP_LOGW(TAG, "OPENING unknown action");
382 break;
383 }
384 break;
385
386 case GCMD_PROMOTION_CHOICE:
387 ESP_LOGI(TAG, "Processing PROMOTION_CHOICE");
388 game_process_promotion_command(&cmd);
389 break;
390
391 case GCMD_WEB_HINT:
392 game_process_web_hint(&cmd);
393 break;
394
395 default:
396 ESP_LOGW(TAG, "Unknown game command kind: %d", (int)cmd.kind);
397 break;
398 }
399
400 game_session_snap_refresh();
401
402 if (commands_processed >= MAX_COMMANDS_PER_CYCLE) {
403 ESP_LOGW(TAG,
404 "Reached max commands per cycle (%lu), processing rest in "
405 "next cycle",
406 MAX_COMMANDS_PER_CYCLE);
407 break;
408 }
409 }
410
411 if (commands_processed > 1) {
412 ESP_LOGI(TAG, "Processed %lu commands in this cycle", commands_processed);
413 }
414 } else {
415 game_led_flush();
416 game_led_flush(); vTaskDelay(pdMS_TO_TICKS(25));
417 }
418 game_led_flush();
419
420 if (button_event_queue != NULL) {
421 button_event_t button_event;
422 while (xQueueReceive(button_event_queue, &button_event, 0) == pdTRUE) {
423 if (button_event.type == BUTTON_EVENT_PRESS) {
424 ESP_LOGI(TAG, "Button pressed: ID=%d", button_event.button_id);
425 if (button_event.button_id == 8) {
426 ESP_LOGI(TAG, "Reset button pressed - resetting game");
427 game_reset_game();
428 game_check_promotion_needed();
429 continue;
430 }
431 if (button_event.button_id <= 3) {
432 ESP_LOGI(TAG, "Promotion button %d pressed", button_event.button_id);
433 game_process_promotion_button(button_event.button_id);
434 continue;
435 }
436 ESP_LOGW(TAG, "Unknown button ID: %d", button_event.button_id);
437 }
438 }
439 }
440
441 game_update_promotion_anchor_led();
442 game_gesture_poll();
443}
