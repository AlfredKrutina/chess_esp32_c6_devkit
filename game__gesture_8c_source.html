173static void demo_show_tick(void) {
174 if (s_demo_phase == DEMO_OFF) {
175 return;
176 }
177 TickType_t now = xTaskGetTickCount();
178 switch (s_demo_phase) {
179 case DEMO_MOVE:
180 if ((int32_t)(now - s_demo_until) < 0) {
181 break;
182 }
183 led_set_lamp_hold(false);
184 if (s_demo_ply >= (uint8_t)(sizeof(k_demo_game) / sizeof(k_demo_game[0]))) {
185 s_demo_sent = false;
186 s_demo_phase = DEMO_END;
187 break;
188 }
189 {
190 const uint8_t *step = k_demo_game[s_demo_ply];
191 uint32_t hold_ms = 1100;
192 if (step[4] == DEMO_KIND_CASTLE) {
193 demo_play_castle(step[0], step[1], step[3]);
194 hold_ms = 1600;
195 } else if (step[4] == DEMO_KIND_PROMOTE) {
196 demo_play_move(step[0], step[1], step[2], step[3]);
197 demo_play_promote(step[2], step[3]);
198 hold_ms = 3400;
199 } else {
200 demo_play_move(step[0], step[1], step[2], step[3]);
201 }
202 s_demo_ply++;
203 s_demo_until = xTaskGetTickCount() + pdMS_TO_TICKS(hold_ms);
204 }
205 break;
206 case DEMO_END:
207 if (!s_demo_sent) {
208 demo_play_end();
209 s_demo_sent = true;
210 s_demo_until = now + pdMS_TO_TICKS(5500);
211 break;
212 }
213 if ((int32_t)(now - s_demo_until) < 0) {
214 break;
215 }
216 led_stop_endgame_animation();
217 s_demo_sent = false;
218 s_demo_phase = DEMO_LAMP;
219 break;
220 case DEMO_LAMP:
221 if (!s_demo_sent) {
222 if (!demo_wave(false, 0)) {
223 break;
224 }
225 s_demo_sent = true;
226 break;
227 }
228 if (led_gesture_anim_busy()) {
229 break;
230 }
231 (void)led_gesture_take_result();
232 demo_paint(0);
233 s_demo_color = 1;
234 s_demo_sent = false;
235 s_demo_until = now + pdMS_TO_TICKS(1500);
236 s_demo_phase = DEMO_COLOR;
237 break;
238 case DEMO_COLOR:
239 if ((int32_t)(now - s_demo_until) < 0) {
240 break;
241 }
242 if (s_demo_color < (uint8_t)(sizeof(k_demo_colors) / sizeof(k_demo_colors[0]))) {
243 demo_paint(s_demo_color);
244 s_demo_color++;
245 s_demo_until = now + pdMS_TO_TICKS(1500);
246 break;
247 }
248 if (!led_gesture_start_pair_pulse()) {
249 break;
250 }
251 s_demo_until = now + pdMS_TO_TICKS(3500);
252 s_demo_phase = DEMO_PAIR;
253 break;
254 case DEMO_PAIR:
255 if ((int32_t)(now - s_demo_until) < 0) {
256 break;
257 }
258 led_gesture_stop_pair_pulse();
259 s_demo_sent = false;
260 s_demo_phase = DEMO_CLOSE;
261 break;
262 case DEMO_CLOSE:
263 if (!s_demo_sent) {
264 if (!demo_wave(true, (uint8_t)(sizeof(k_demo_colors) / sizeof(k_demo_colors[0]) - 1))) {
265 break;
266 }
267 s_demo_sent = true;
268 break;
269 }
270 if (led_gesture_anim_busy()) {
271 break;
272 }
273 (void)led_gesture_take_result();
274 led_set_lamp_hold(false);
275 s_demo_sent = false;
276 s_demo_ply = 0;
277 s_demo_until = 0;
278 s_demo_phase = DEMO_MOVE;
279 break;
280 default:
281 s_demo_phase = DEMO_OFF;
282 break;
283 }
284}

372static void gesture_power_off(void) {
373 gpio_num_t pin = (gpio_num_t)CONFIG_CHESS_GESTURE_BUTTON_GPIO;
374 led_set_forced_blackout(true);
375 for (int i = 0; i < CHESS_LED_COUNT_TOTAL; i++) {
376 led_set_pixel_internal((uint8_t)i, 0, 0, 0);
377 }
378 led_force_immediate_update();
379 ha_light_persist_mode();
380
381 /* The hold that finished the chase must be released first, otherwise a
382 * level-low wake fires immediately. */
383 while (led_gesture_button_down()) {
384 (void)esp_task_wdt_reset();
385 vTaskDelay(pdMS_TO_TICKS(25));
386 }
387 vTaskDelay(pdMS_TO_TICKS(25));
388
389 gpio_wakeup_disable(pin);
390 gpio_wakeup_enable(pin, GPIO_INTR_LOW_LEVEL);
391 if ((int)BUTTON_RESET != (int)pin &&
392 !chess_gpio_pin_is_stm32_nrst_output((int)BUTTON_RESET)) {
393 gpio_wakeup_disable(BUTTON_RESET);
394 gpio_wakeup_enable(BUTTON_RESET, GPIO_INTR_LOW_LEVEL);
395 }
396 esp_sleep_enable_gpio_wakeup();
397
398#if ((SOC_GPIO_DEEP_SLEEP_WAKE_VALID_GPIO_MASK) & \
399 (1ULL << CONFIG_CHESS_GESTURE_BUTTON_GPIO))
400 ESP_LOGI(TAG, "[STAGING] power off: deep sleep, wake GPIO%d low",
401 CONFIG_CHESS_GESTURE_BUTTON_GPIO);
402 esp_deep_sleep_enable_gpio_wakeup(1ULL << CONFIG_CHESS_GESTURE_BUTTON_GPIO,
403 ESP_GPIO_WAKEUP_GPIO_LOW);
404 esp_deep_sleep_start();
405#else
406 /* ESP32-C6 deep-sleep GPIO wake is only GPIO0-7. IO23 is not one of them
407 * (and GPIO7 is the LED data pin). The 5 V rail has no load switch, so
408 * "off" is LEDs dark plus light sleep. The next press wakes the CPU and
409 * esp_restart() boots like a power-on. USB power stays connected. */
410 ESP_LOGI(TAG,
411 "[STAGING] power off: light sleep (GPIO%d cannot deep-sleep-wake), "
412 "USB 5 V stays on",
413 CONFIG_CHESS_GESTURE_BUTTON_GPIO);
414 (void)esp_wifi_stop();
415 esp_err_t sl = esp_light_sleep_start();
416 /* USB-Serial/JTAG wakes light sleep at once while the button is still
417 * released. That used to fall through into esp_restart(), so "off" never
418 * stuck and the next press was not a power-on. */
419 bool button_woke = (sl == ESP_OK && led_gesture_button_down());
420 if (!button_woke) {
421 if (sl != ESP_OK) {
422 ESP_LOGW(TAG, "light sleep rejected (%s) — staying dark until SW1",
423 esp_err_to_name(sl));
424 } else {
425 ESP_LOGW(TAG, "spurious wake — staying dark until SW1");
426 }
427 gpio_wakeup_disable(pin);
428 for (;;) {
429 (void)esp_task_wdt_reset();
430 if (led_gesture_button_down()) {
431 vTaskDelay(pdMS_TO_TICKS(25));
432 if (led_gesture_button_down()) {
433 while (led_gesture_button_down()) {
434 (void)esp_task_wdt_reset();
435 vTaskDelay(pdMS_TO_TICKS(25));
436 }
437 vTaskDelay(pdMS_TO_TICKS(25));
438 ESP_LOGI(TAG, "[STAGING] power on from GPIO%d, restart",
439 CONFIG_CHESS_GESTURE_BUTTON_GPIO);
440 esp_restart();
441 }
442 }
443 vTaskDelay(pdMS_TO_TICKS(25));
444 }
445 }
446 while (led_gesture_button_down()) {
447 (void)esp_task_wdt_reset();
448 vTaskDelay(pdMS_TO_TICKS(25));
449 }
450 ESP_LOGI(TAG, "[STAGING] wake on GPIO%d, restarting as power-on",
451 CONFIG_CHESS_GESTURE_BUTTON_GPIO);
452 esp_restart();
453#endif
454}

511static void handle_event(gesture_event_t ev) {
512 if (gesture_in_puzzle() && ev == GESTURE_SNAKE_START) {
513 gesture_next_puzzle();
514 return;
515 }
516 if (gesture_in_puzzle() && ev == GESTURE_COLOR) {
517 gesture_begin_opening();
518 return;
519 }
520 if (ev == GESTURE_DEMO) {
521 game_demo_show_set(!game_demo_show_active());
522 return;
523 }
524 if (game_demo_show_active() && ev != GESTURE_NONE) {
525 game_demo_show_set(false);
526 return;
527 }
528 if (ev == GESTURE_SHUTDOWN_START) {
529 /* Button task already started this. A second start after the player lets
530 * go replays the symbol and then powers off. */
531 if (!led_gesture_anim_busy() && button_gesture_is_down()) {
532 if (!led_gesture_start_shutdown()) {
533 ESP_LOGW(TAG, "shutdown did not start");
534 }
535 }
536 return;
537 }
538 if (ev == GESTURE_SNAKE_START) {
539 /* Button task already started this. A second start after release would
540 * play the sweep again. */
541 if (!led_gesture_anim_busy() && button_gesture_is_down()) {
542 if (!led_gesture_start_snake()) {
543 ESP_LOGW(TAG, "snake did not start");
544 }
545 }
546 return;
547 }
548 if (led_gesture_anim_busy() && ev != GESTURE_NONE) {
549 ESP_LOGI(TAG, "[STAGING] gesture dropped, animation busy");
550 return;
551 }
552 switch (ev) {
553 case GESTURE_TOGGLE: {
554 if (hall_magnet_bench_active()) {
555 hall_magnet_bench_capture();
556 break;
557 }
558 /* One click while pairing is open closes the window. The lamp wave must
559 * not start, or it fights the blue pulse. */
560 if (ble_task_pairing_window_is_open() &&
561 ble_task_claim_pairing_success()) {
562 ESP_LOGI(TAG, "[STAGING] single click closes pairing");
563 ble_task_finish_pairing_window();
564 break;
565 }
566 ha_mode_t next =
567 (ha_light_get_mode() == HA_MODE_HA) ? HA_MODE_GAME : HA_MODE_HA;
568 if (!gesture_begin_mode_wave(next)) {
569 ESP_LOGW(TAG, "wave did not start");
570 return;
571 }
572 break;
573 }
574 case GESTURE_COLOR: {
575 if (ha_light_get_mode() != HA_MODE_HA) {
576 ESP_LOGI(TAG, "[STAGING] double click ignored outside lamp");
577 break;
578 }
579 static const uint8_t k_lamp_colors[][3] = {
580 {255, 255, 255}, /* daylight white */
581 {255, 214, 170}, /* warm white */
582 {0, 0, 255}, /* blue */
583 {255, 0, 0}, /* red */
584 };
585 uint8_t r = 0, g = 0, b = 0, bright = 255;
586 bool on = true;
587 ha_light_get_state(&r, &g, &b, &bright, &on);
588 unsigned next = 0;
589 for (unsigned i = 0; i < 4u; i++) {
590 if (r == k_lamp_colors[i][0] && g == k_lamp_colors[i][1] &&
591 b == k_lamp_colors[i][2]) {
592 next = (i + 1u) % 4u;
593 break;
594 }
595 }
596 if (!ha_light_request_web_lamp(true, k_lamp_colors[next][0],
597 k_lamp_colors[next][1],
598 k_lamp_colors[next][2])) {
599 ESP_LOGW(TAG, "lamp color queue full");
600 break;
601 }
602 ESP_LOGI(TAG, "[STAGING] lamp color %u RGB(%u,%u,%u)", next,
603 k_lamp_colors[next][0], k_lamp_colors[next][1],
604 k_lamp_colors[next][2]);
605 break;
606 }
607 case GESTURE_PAIR:
608 ble_task_request_pairing();
609 break;
610 case GESTURE_SNAKE_START:
611 if (!led_gesture_start_snake()) {
612 ESP_LOGW(TAG, "snake did not start");
613 }
614 break;
615 case GESTURE_SHUTDOWN_START:
616 if (!led_gesture_start_shutdown()) {
617 ESP_LOGW(TAG, "shutdown chase did not start");
618 }
619 break;
620 case GESTURE_NONE:
621 default:
622 break;
623 }
624}

626static void handle_result(led_gesture_result_t result) {
627 switch (result) {
628 case LED_GESTURE_WAVE_DONE:
629 if (!s_wave_wait) {
630 break;
