152static void demo_show_tick(void) {
153 if (s_demo_phase == DEMO_OFF) {
154 return;
155 }
156 TickType_t now = xTaskGetTickCount();
157 switch (s_demo_phase) {
158 case DEMO_MOVE:
159 if ((int32_t)(now - s_demo_until) < 0) {
160 break;
161 }
162 game_event_post_led_ctrl(GAME_EVT_CTRL_LAMP, 0, 0, 0, 0);
163 if (s_demo_ply >= (uint8_t)(sizeof(k_demo_game) / sizeof(k_demo_game[0]))) {
164 s_demo_sent = false;
165 s_demo_phase = DEMO_END;
166 break;
167 }
168 {
169 const uint8_t *step = k_demo_game[s_demo_ply];
170 uint32_t hold_ms = 1100;
171 if (step[4] == DEMO_KIND_CASTLE) {
172 demo_play_castle(step[0], step[1], step[3]);
173 hold_ms = 1600;
174 } else if (step[4] == DEMO_KIND_PROMOTE) {
175 demo_play_move(step[0], step[1], step[2], step[3]);
176 demo_play_promote(step[2], step[3]);
177 hold_ms = 3400;
178 } else {
179 demo_play_move(step[0], step[1], step[2], step[3]);
180 }
181 s_demo_ply++;
182 s_demo_until = xTaskGetTickCount() + pdMS_TO_TICKS(hold_ms);
183 }
184 break;
185 case DEMO_END:
186 if (!s_demo_sent) {
187 demo_play_end();
188 s_demo_sent = true;
189 s_demo_until = now + pdMS_TO_TICKS(5500);
190 break;
191 }
192 if ((int32_t)(now - s_demo_until) < 0) {
193 break;
194 }
195 game_event_post_led_ctrl(GAME_EVT_CTRL_STOP_ENDGAME, 0, 0, 0, 0);
196 s_demo_sent = false;
197 s_demo_phase = DEMO_LAMP;
198 break;
199 case DEMO_LAMP:
200 if (!s_demo_sent) {
201 if (!demo_wave(false, 0)) {
202 break;
203 }
204 s_demo_sent = true;
205 break;
206 }
207 if (led_gesture_anim_busy()) {
208 break;
209 }
210 (void)led_gesture_take_result();
211 demo_paint(0);
212 s_demo_color = 1;
213 s_demo_sent = false;
214 s_demo_until = now + pdMS_TO_TICKS(1500);
215 s_demo_phase = DEMO_COLOR;
216 break;
217 case DEMO_COLOR:
218 if ((int32_t)(now - s_demo_until) < 0) {
219 break;
220 }
221 if (s_demo_color < (uint8_t)(sizeof(k_demo_colors) / sizeof(k_demo_colors[0]))) {
222 demo_paint(s_demo_color);
223 s_demo_color++;
224 s_demo_until = now + pdMS_TO_TICKS(1500);
225 break;
226 }
227 if (!led_gesture_start_pair_pulse()) {
228 break;
229 }
230 s_demo_until = now + pdMS_TO_TICKS(3500);
231 s_demo_phase = DEMO_PAIR;
232 break;
233 case DEMO_PAIR:
234 if ((int32_t)(now - s_demo_until) < 0) {
235 break;
236 }
237 led_gesture_stop_pair_pulse();
238 s_demo_sent = false;
239 s_demo_phase = DEMO_CLOSE;
240 break;
241 case DEMO_CLOSE:
242 if (!s_demo_sent) {
243 if (!demo_wave(true, (uint8_t)(sizeof(k_demo_colors) / sizeof(k_demo_colors[0]) - 1))) {
244 break;
245 }
246 s_demo_sent = true;
247 break;
248 }
249 if (led_gesture_anim_busy()) {
250 break;
251 }
252 (void)led_gesture_take_result();
253 game_event_post_led_ctrl(GAME_EVT_CTRL_LAMP, 0, 0, 0, 0);
254 s_demo_sent = false;
255 s_demo_ply = 0;
256 s_demo_until = 0;
257 s_demo_phase = DEMO_MOVE;
258 break;
259 default:
260 s_demo_phase = DEMO_OFF;
261 break;
262 }
263}

310/* One-shot handoff: sweep cycle ended, game start waits 2 s on the game task. */
311#define NEW_GAME_DWELL_US (2000LL * 1000LL)
312static esp_timer_handle_t s_dwell_timer;
313static volatile uint32_t s_dwell_seq;
314static volatile uint32_t s_dwell_expect;
315static volatile uint32_t s_dwell_due;
316static bool s_drop_snake_done;

318static void new_game_dwell_cb(void *arg) {
319 uint32_t expect = s_dwell_expect;
320 (void)arg;
321 if (expect != 0 && expect == s_dwell_seq) {
322 s_dwell_due = expect;
323 }
324}

326static void new_game_dwell_ensure(void) {
327 if (s_dwell_timer != NULL) {
328 return;
329 }
330 const esp_timer_create_args_t args = {
331 .callback = &new_game_dwell_cb,
332 .name = "ng_dwell",
333 };
334 if (esp_timer_create(&args, &s_dwell_timer) != ESP_OK) {
335 s_dwell_timer = NULL;
336 ESP_LOGE(TAG, "new-game dwell timer create failed");
337 }
338}

340static void new_game_dwell_cancel(void) {
341 uint32_t seq = s_dwell_seq + 1u;
342 if (seq == 0u) {
343 seq = 1u;
344 }
345 s_dwell_seq = seq;
346 s_dwell_expect = 0;
347 s_dwell_due = 0;
348 if (s_dwell_timer != NULL) {
349 (void)esp_timer_stop(s_dwell_timer);
350 }
351}
352
353static void new_game_dwell_arm(void) {
354 uint32_t seq = s_dwell_seq + 1u;
355 new_game_dwell_ensure();
356 if (seq == 0u) {
357 seq = 1u;
358 }
359 s_dwell_seq = seq;
360 s_dwell_due = 0;
361 s_dwell_expect = 0;
362 if (s_dwell_timer == NULL) {
363 return;
364 }
365 (void)esp_timer_stop(s_dwell_timer);
366 s_dwell_expect = seq;
367 if (esp_timer_start_once(s_dwell_timer, NEW_GAME_DWELL_US) != ESP_OK) {
368 s_dwell_expect = 0;
369 ESP_LOGE(TAG, "new-game dwell timer start failed");
370 }
371}

373static bool new_game_dwell_take(void) {
374 uint32_t due = s_dwell_due;
375 if (due == 0u || due != s_dwell_seq || due != s_dwell_expect) {
376 return false;
377 }
378 s_dwell_due = 0;
379 s_dwell_expect = 0;
380 return true;
381}

391static void new_game_dwell_watch(void) {
392 if (s_dwell_expect == 0u && !s_new_game_after_boot) {
393 return;
394 }
395 if (game_demo_show_active() ||
396 (led_gesture_anim_busy() && !led_gesture_snake_active())) {
397 s_new_game_after_boot = false;
398 new_game_dwell_cancel();
399 } else if (led_gesture_snake_active() &&
400 (s_dwell_expect != 0u || s_new_game_after_boot)) {
401 /* A new sweep replaced the cycle that armed this dwell. */
402 s_new_game_after_boot = false;
403 new_game_dwell_cancel();
404 }

483static void gesture_power_off(void) {
484 gpio_num_t pin = (gpio_num_t)CONFIG_CHESS_GESTURE_BUTTON_GPIO;
485 game_led_flush();
486 game_event_post_led_ctrl(GAME_EVT_CTRL_BLACKOUT, 0, 0, 0, 0);
487 ha_light_persist_mode();
488
489 /* The hold that finished the chase must be released first, otherwise a
490 * level-low wake fires immediately. */
491 while (led_gesture_button_down()) {
492 (void)esp_task_wdt_reset();
493 game_led_flush(); vTaskDelay(pdMS_TO_TICKS(25));
494 }
495 game_led_flush(); vTaskDelay(pdMS_TO_TICKS(25));
496
497 gpio_wakeup_disable(pin);
498 gpio_wakeup_enable(pin, GPIO_INTR_LOW_LEVEL);
499 if ((int)BUTTON_RESET != (int)pin &&
500 !chess_gpio_pin_is_stm32_nrst_output((int)BUTTON_RESET)) {
501 gpio_wakeup_disable(BUTTON_RESET);
502 gpio_wakeup_enable(BUTTON_RESET, GPIO_INTR_LOW_LEVEL);
503 }
504 esp_sleep_enable_gpio_wakeup();
505
506#if ((SOC_GPIO_DEEP_SLEEP_WAKE_VALID_GPIO_MASK) & \
507 (1ULL << CONFIG_CHESS_GESTURE_BUTTON_GPIO))
508 ESP_LOGI(TAG, "[STAGING] power off: deep sleep, wake GPIO%d low",
509 CONFIG_CHESS_GESTURE_BUTTON_GPIO);
510 esp_deep_sleep_enable_gpio_wakeup(1ULL << CONFIG_CHESS_GESTURE_BUTTON_GPIO,
511 ESP_GPIO_WAKEUP_GPIO_LOW);
512 esp_deep_sleep_start();
513#else
514 /* ESP32-C6 deep-sleep GPIO wake is only GPIO0-7. IO23 is not one of them
515 * (and GPIO7 is the LED data pin). The 5 V rail has no load switch, so
516 * "off" is LEDs dark plus light sleep. The next press wakes the CPU and
517 * esp_restart() boots like a power-on. USB power stays connected. */
518 ESP_LOGI(TAG,
519 "[STAGING] power off: light sleep (GPIO%d cannot deep-sleep-wake), "
520 "USB 5 V stays on",
521 CONFIG_CHESS_GESTURE_BUTTON_GPIO);
522 (void)esp_wifi_stop();
523 esp_err_t sl = esp_light_sleep_start();
524 /* USB-Serial/JTAG wakes light sleep at once while the button is still
525 * released. That used to fall through into esp_restart(), so "off" never
526 * stuck and the next press was not a power-on. */
527 bool button_woke = (sl == ESP_OK && led_gesture_button_down());
528 if (!button_woke) {
529 if (sl != ESP_OK) {
530 ESP_LOGW(TAG, "light sleep rejected (%s) — staying dark until SW1",
531 esp_err_to_name(sl));
532 } else {
533 ESP_LOGW(TAG, "spurious wake — staying dark until SW1");
534 }
535 gpio_wakeup_disable(pin);
536 for (;;) {
537 (void)esp_task_wdt_reset();
538 if (led_gesture_button_down()) {
539 game_led_flush(); vTaskDelay(pdMS_TO_TICKS(25));
540 if (led_gesture_button_down()) {
541 while (led_gesture_button_down()) {
542 (void)esp_task_wdt_reset();
543 game_led_flush(); vTaskDelay(pdMS_TO_TICKS(25));
544 }
545 game_led_flush(); vTaskDelay(pdMS_TO_TICKS(25));
546 ESP_LOGI(TAG, "[STAGING] power on from GPIO%d, restart",
547 CONFIG_CHESS_GESTURE_BUTTON_GPIO);
548 esp_restart();
549 }
550 }
551 game_led_flush(); vTaskDelay(pdMS_TO_TICKS(25));
552 }
553 }
554 while (led_gesture_button_down()) {
555 (void)esp_task_wdt_reset();
556 game_led_flush(); vTaskDelay(pdMS_TO_TICKS(25));
557 }
558 ESP_LOGI(TAG, "[STAGING] wake on GPIO%d, restarting as power-on",
559 CONFIG_CHESS_GESTURE_BUTTON_GPIO);
560 esp_restart();
561#endif
562}

564static bool gesture_begin_mode_wave(ha_mode_t next) {
565 uint32_t colors[64];
566 s_new_game_after_boot = false;
567 new_game_dwell_cancel();
568 /* Opening spreads the last lamp color. Closing shrinks that same color. */
569 bool inward = (next == HA_MODE_GAME);
570 uint32_t lamp = inward ? lamp_rgb() : lamp_rgb_entering();
571 fill_mode_colors(next, colors);
572 if (!led_gesture_start_wave(colors, inward, lamp)) {
573 return false;
574 }
575 s_pending_mode = next;
576 s_opening_wave = false;
577 s_wave_wait = true;
578 ESP_LOGI(TAG, "[STAGING] wave preview %s -> %s (commit after cover)",
579 next == HA_MODE_HA ? "game" : "lamp",
580 next == HA_MODE_HA ? "lamp" : "game");
581 return true;
582}

638static void handle_event(gesture_event_t ev) {
639 if (gesture_in_puzzle() && ev == GESTURE_SNAKE_START) {
640 s_new_game_after_boot = false;
641 new_game_dwell_cancel();
642 gesture_next_puzzle();
643 return;
644 }
645 if (gesture_in_puzzle() && ev == GESTURE_COLOR) {
646 gesture_begin_opening();
647 return;
648 }
649 if (ev == GESTURE_DEMO) {
650 game_demo_show_set(!game_demo_show_active());
651 return;
652 }
653 if (game_demo_show_active() && ev != GESTURE_NONE) {
654 game_demo_show_set(false);
655 return;
656 }
657 if (ev == GESTURE_SHUTDOWN_START) {
658 /* Button task already started this. A second start after the player lets
659 * go replays the symbol and then powers off. */
660 game_gesture_cancel_new_game_dwell();
661 if (!led_gesture_anim_busy() && button_gesture_is_down()) {
662 if (!led_gesture_start_shutdown()) {
663 ESP_LOGW(TAG, "shutdown did not start");
664 }
665 }
666 return;
667 }
668 if (ev == GESTURE_SNAKE_START) {
669 /* Button task already started this. A second start after release would
670 * play the sweep again. Drop any dwell from the previous cycle. */
671 s_drop_snake_done = false;
672 s_new_game_after_boot = false;
673 new_game_dwell_cancel();
674 if (!led_gesture_anim_busy() && button_gesture_is_down()) {
675 if (!led_gesture_start_snake()) {
676 ESP_LOGW(TAG, "snake did not start");
677 }
678 }
679 return;
680 }
681 if (led_gesture_anim_busy() && ev != GESTURE_NONE) {
682 ESP_LOGI(TAG, "[STAGING] gesture dropped, animation busy");
683 return;
684 }
685 switch (ev) {
686 case GESTURE_TOGGLE: {
687 if (hall_magnet_bench_active()) {
688 hall_magnet_bench_capture();
689 break;
690 }
691 /* One click while pairing is open closes the window. The lamp wave must
692 * not start, or it fights the blue pulse. */
693 if (ble_task_pairing_window_is_open() &&
694 ble_task_claim_pairing_success()) {
695 ESP_LOGI(TAG, "[STAGING] single click closes pairing");
696 ble_task_finish_pairing_window();
697 break;
698 }
699 ha_mode_t next =
700 (ha_light_get_mode() == HA_MODE_HA) ? HA_MODE_GAME : HA_MODE_HA;
701 if (!gesture_begin_mode_wave(next)) {
702 ESP_LOGW(TAG, "wave did not start");
703 return;
704 }
705 break;
706 }
707 case GESTURE_COLOR: {
708 if (ha_light_get_mode() != HA_MODE_HA) {
709 ESP_LOGI(TAG, "[STAGING] double click ignored outside lamp");
710 break;
711 }
712 static const uint8_t k_lamp_colors[][3] = {
713 {255, 255, 255}, /* daylight white */
714 {255, 214, 170}, /* warm white */
715 {0, 0, 255}, /* blue */
716 {255, 0, 0}, /* red */
717 };
718 uint8_t r = 0, g = 0, b = 0, bright = 255;
719 bool on = true;
720 ha_light_get_state(&r, &g, &b, &bright, &on);
721 unsigned next = 0;
722 for (unsigned i = 0; i < 4u; i++) {
723 if (r == k_lamp_colors[i][0] && g == k_lamp_colors[i][1] &&
724 b == k_lamp_colors[i][2]) {
725 next = (i + 1u) % 4u;
726 break;
727 }
728 }
729 if (!ha_light_request_web_lamp(true, k_lamp_colors[next][0],
730 k_lamp_colors[next][1],
731 k_lamp_colors[next][2])) {
732 ESP_LOGW(TAG, "lamp color queue full");
733 break;
734 }
735 ESP_LOGI(TAG, "[STAGING] lamp color %u RGB(%u,%u,%u)", next,
736 k_lamp_colors[next][0], k_lamp_colors[next][1],
737 k_lamp_colors[next][2]);
738 break;
739 }
740 case GESTURE_PAIR:
741 ble_task_request_pairing();
742 break;
743 case GESTURE_SNAKE_START:
744 if (!led_gesture_start_snake()) {
745 ESP_LOGW(TAG, "snake did not start");
746 }
747 break;
748 case GESTURE_SHUTDOWN_START:
749 if (!led_gesture_start_shutdown()) {
750 ESP_LOGW(TAG, "shutdown chase did not start");
751 }
752 break;
753 case GESTURE_NONE:
754 default:
755 break;
756 }
757}

759static void handle_result(led_gesture_result_t result) {
760 switch (result) {
761 case LED_GESTURE_WAVE_DONE:
762 if (!s_wave_wait) {
763 break;
764 }
765 s_wave_wait = false;
766 if (s_opening_wave) {
767 s_opening_wave = false;
768 (void)game_opening_ensure_and_start();
769 break;
770 }
771 if (!ha_light_request_mode(s_pending_mode)) {
772 ESP_LOGW(TAG, "mode commit queue full");
773 } else {
774 ESP_LOGI(TAG, "[STAGING] mode committed: %s",
775 s_pending_mode == HA_MODE_HA ? "lamp" : "game");
776 }
777 break;
778 case LED_GESTURE_SNAKE_DONE:
779 if (s_drop_snake_done) {
780 s_drop_snake_done = false;
781 ESP_LOGI(TAG, "[STAGING] sweep finished after game already started");
782 break;
783 }
784 if (!button_gesture_is_down()) {
785 ESP_LOGI(TAG, "[STAGING] snake ended after release, no new game");
786 break;
787 }
788 ESP_LOGI(TAG, "[STAGING] new-game sweep finished, dwell 2000 ms");
789 new_game_dwell_arm();
790 break;
791 case LED_GESTURE_SNAKE_CANCELLED:
792 s_drop_snake_done = false;
793 s_new_game_after_boot = false;
794 new_game_dwell_cancel();
795 ESP_LOGI(TAG, "[STAGING] new game cancelled");
796 break;
797 case LED_GESTURE_SHUTDOWN_DONE:
798 gesture_power_off();
799 break;
800 case LED_GESTURE_SHUTDOWN_CANCELLED:
801 ESP_LOGI(TAG, "[STAGING] power off cancelled");
802 break;
803 case LED_GESTURE_PULSE_DONE:
804 case LED_GESTURE_RESULT_NONE:
805 default:
806 break;
807 }
808}

810void game_gesture_poll(void) {
811 if (s_off_wave_req || s_lamp_wave_req) {
812 s_new_game_after_boot = false;
813 new_game_dwell_cancel();
814 }
815 new_game_dwell_watch();
816 if (new_game_dwell_take()) {
817 /* Demo, wave, or shutdown owns the board: the dwell must not start a game. */
818 bool board_taken = game_demo_show_active() ||
819 (led_gesture_anim_busy() && !led_gesture_snake_active());
820 if (board_taken) {
821 ESP_LOGI(TAG, "[STAGING] new-game dwell dropped, board taken");
822 } else if (game_led_boot_active()) {
823 s_new_game_after_boot = true;
824 ESP_LOGI(TAG, "[STAGING] new game waits until boot animation ends");
825 } else {
826 commit_sweep_new_game();
827 }
828 }
829 if (s_new_game_after_boot && !game_led_boot_active()) {
830 s_new_game_after_boot = false;
831 ESP_LOGI(TAG, "[STAGING] deferred new game after boot");
832 game_start_new_game();
833#if CONFIG_CHESS_MATRIX_INPUT_I2C_HALL
834 if (current_game_state == GAME_STATE_WAITING_FOR_BOARD_SETUP) {
835 game_hall_restore_guide_stop_leds();
836 game_hall_restore_guide_force_resync();
837 } else if (!game_hall_restore_guide_show_opening()) {
838 game_hall_restore_guide_stop_leds();
839 game_policy_highlight_movable_if_enabled();
840 }
841#endif
842 }
843 if (s_off_wave_req && !s_wave_wait && !led_gesture_anim_busy()) {
844 s_off_wave_req = false;
845 if (game_demo_show_active()) {
846 game_demo_show_set(false);
847 }
848 uint32_t colors[64] = {0};
849 if (s_off_cover != 0 &&
850 led_gesture_start_wave(colors, true, s_off_cover)) {
851 s_pending_mode = HA_MODE_HA;
852 s_opening_wave = false;
853 s_wave_wait = true;
854 ESP_LOGI(TAG, "[STAGING] lamp off wave");
855 } else {
856 ESP_LOGW(TAG, "lamp off wave did not start");
857 (void)ha_light_request_mode(HA_MODE_HA);
858 }
859 }
860 if (s_lamp_wave_req && !s_wave_wait && !led_gesture_anim_busy()) {
861 s_lamp_wave_req = false;
862 if (game_demo_show_active()) {
863 game_demo_show_set(false);
864 }
865 if (!gesture_begin_mode_wave(HA_MODE_HA)) {
866 ESP_LOGW(TAG, "lamp wave did not start");
867 (void)ha_light_request_mode(HA_MODE_HA);
868 }
869 }
870 QueueHandle_t q = button_gesture_queue();
871 /* The button task already started this sweep and left the hold in the
872 * queue. Reading that event only after the sweep goes idle plays it again. */
873 if (q != NULL && !s_wave_wait) {
874 gesture_event_t ev;
875 if (xQueuePeek(q, &ev, 0) == pdTRUE) {
876 bool take = !led_gesture_anim_busy() ||
877 (ev == GESTURE_SNAKE_START && led_gesture_snake_active());
878 if (take && xQueueReceive(q, &ev, 0) == pdTRUE) {
879 handle_event(ev);
880 }
881 }
882 }
883 if (!game_demo_show_active()) {
884 bool want_pulse = ble_task_pairing_pulse_wanted();
885 if (want_pulse != s_pair_pulse_on) {
886 if (want_pulse) {
887 (void)led_gesture_start_pair_pulse();
888 ESP_LOGI(TAG, "[STAGING] pairing pulse on");
889 } else {
890 led_gesture_stop_pair_pulse();
891 ESP_LOGI(TAG, "[STAGING] pairing pulse off");
892 }
893 s_pair_pulse_on = want_pulse;
894 }
895 }
896 demo_show_tick();
897 handle_result(led_gesture_take_result());
898}
