121static void demo_show_tick(void) {
122 if (s_demo_phase == DEMO_OFF) {
123 return;
124 }
125 TickType_t now = xTaskGetTickCount();
126 switch (s_demo_phase) {
127 case DEMO_MOVE:
128 if ((int32_t)(now - s_demo_until) < 0) {
129 break;
130 }
131 led_set_lamp_hold(false);
132 if (s_demo_ply >= (uint8_t)(sizeof(k_demo_game) / sizeof(k_demo_game[0]))) {
133 s_demo_sent = false;
134 s_demo_phase = DEMO_END;
135 break;
136 }
137 demo_play_move(k_demo_game[s_demo_ply][0], k_demo_game[s_demo_ply][1],
138 k_demo_game[s_demo_ply][2], k_demo_game[s_demo_ply][3]);
139 s_demo_ply++;
140 s_demo_until = xTaskGetTickCount() + pdMS_TO_TICKS(900);
141 break;
142 case DEMO_END:
143 if (!s_demo_sent) {
144 demo_play_end();
145 s_demo_sent = true;
146 s_demo_until = now + pdMS_TO_TICKS(5500);
147 break;
148 }
149 if ((int32_t)(now - s_demo_until) < 0) {
150 break;
151 }
152 led_stop_endgame_animation();
153 s_demo_sent = false;
154 s_demo_phase = DEMO_LAMP;
155 break;
156 case DEMO_LAMP:
157 if (!s_demo_sent) {
158 if (!demo_wave(false, 0)) {
159 break;
160 }
161 s_demo_sent = true;
162 break;
163 }
164 if (led_gesture_anim_busy()) {
165 break;
166 }
167 (void)led_gesture_take_result();
168 demo_paint(0);
169 s_demo_color = 1;
170 s_demo_sent = false;
171 s_demo_until = now + pdMS_TO_TICKS(1500);
172 s_demo_phase = DEMO_COLOR;
173 break;
174 case DEMO_COLOR:
175 if ((int32_t)(now - s_demo_until) < 0) {
176 break;
177 }
178 if (s_demo_color < (uint8_t)(sizeof(k_demo_colors) / sizeof(k_demo_colors[0]))) {
179 demo_paint(s_demo_color);
180 s_demo_color++;
181 s_demo_until = now + pdMS_TO_TICKS(1500);
182 break;
183 }
184 if (!led_gesture_start_pair_pulse()) {
185 break;
186 }
187 s_demo_until = now + pdMS_TO_TICKS(3500);
188 s_demo_phase = DEMO_PAIR;
189 break;
190 case DEMO_PAIR:
191 if ((int32_t)(now - s_demo_until) < 0) {
192 break;
193 }
194 led_gesture_stop_pair_pulse();
195 s_demo_sent = false;
196 s_demo_phase = DEMO_CLOSE;
197 break;
198 case DEMO_CLOSE:
199 if (!s_demo_sent) {
200 if (!demo_wave(true, (uint8_t)(sizeof(k_demo_colors) / sizeof(k_demo_colors[0]) - 1))) {
201 break;
202 }
203 s_demo_sent = true;
204 break;
205 }
206 if (led_gesture_anim_busy()) {
207 break;
208 }
209 (void)led_gesture_take_result();
210 led_set_lamp_hold(false);
211 s_demo_sent = false;
212 s_demo_ply = 0;
213 s_demo_until = 0;
214 s_demo_phase = DEMO_MOVE;
215 break;
216 default:
217 s_demo_phase = DEMO_OFF;
218 break;
219 }

307static void gesture_power_off(void) {
308 gpio_num_t pin = (gpio_num_t)CONFIG_CHESS_GESTURE_BUTTON_GPIO;
309 led_set_forced_blackout(true);
310 for (int i = 0; i < CHESS_LED_COUNT_TOTAL; i++) {
311 led_set_pixel_internal((uint8_t)i, 0, 0, 0);
312 }
313 led_force_immediate_update();
314 ha_light_persist_mode();
315
316 /* The hold that finished the chase must be released first, otherwise a
317 * level-low wake fires immediately. */
318 while (led_gesture_button_down()) {
319 (void)esp_task_wdt_reset();
320 vTaskDelay(pdMS_TO_TICKS(25));
321 }
322 vTaskDelay(pdMS_TO_TICKS(25));
323
324 gpio_wakeup_disable(pin);
325 gpio_wakeup_enable(pin, GPIO_INTR_LOW_LEVEL);
326 if ((int)BUTTON_RESET != (int)pin &&
327 !chess_gpio_pin_is_stm32_nrst_output((int)BUTTON_RESET)) {
328 gpio_wakeup_disable(BUTTON_RESET);
329 gpio_wakeup_enable(BUTTON_RESET, GPIO_INTR_LOW_LEVEL);
330 }
331 esp_sleep_enable_gpio_wakeup();
332
333#if ((SOC_GPIO_DEEP_SLEEP_WAKE_VALID_GPIO_MASK) & \
334 (1ULL << CONFIG_CHESS_GESTURE_BUTTON_GPIO))
335 ESP_LOGI(TAG, "[STAGING] power off: deep sleep, wake GPIO%d low",
336 CONFIG_CHESS_GESTURE_BUTTON_GPIO);
337 esp_deep_sleep_enable_gpio_wakeup(1ULL << CONFIG_CHESS_GESTURE_BUTTON_GPIO,
338 ESP_GPIO_WAKEUP_GPIO_LOW);
339 esp_deep_sleep_start();
340#else
341 /* ESP32-C6 deep-sleep GPIO wake is only GPIO0-7. IO23 is not one of them
342 * (and GPIO7 is the LED data pin). The 5 V rail has no load switch, so
343 * "off" is LEDs dark plus light sleep. The next press wakes the CPU and
344 * esp_restart() boots like a power-on. USB power stays connected. */
345 ESP_LOGI(TAG,
346 "[STAGING] power off: light sleep (GPIO%d cannot deep-sleep-wake), "
347 "USB 5 V stays on",
348 CONFIG_CHESS_GESTURE_BUTTON_GPIO);
349 (void)esp_wifi_stop();
350 esp_err_t sl = esp_light_sleep_start();
351 /* USB-Serial/JTAG wakes light sleep at once while the button is still
352 * released. That used to fall through into esp_restart(), so "off" never
353 * stuck and the next press was not a power-on. */
354 bool button_woke = (sl == ESP_OK && led_gesture_button_down());
355 if (!button_woke) {
356 if (sl != ESP_OK) {
357 ESP_LOGW(TAG, "light sleep rejected (%s) — staying dark until SW1",
358 esp_err_to_name(sl));
359 } else {
360 ESP_LOGW(TAG, "spurious wake — staying dark until SW1");
361 }
362 gpio_wakeup_disable(pin);
363 for (;;) {
364 (void)esp_task_wdt_reset();
365 if (led_gesture_button_down()) {
366 vTaskDelay(pdMS_TO_TICKS(25));
367 if (led_gesture_button_down()) {
368 while (led_gesture_button_down()) {
369 (void)esp_task_wdt_reset();
370 vTaskDelay(pdMS_TO_TICKS(25));
371 }
372 vTaskDelay(pdMS_TO_TICKS(25));
373 ESP_LOGI(TAG, "[STAGING] power on from GPIO%d, restart",
374 CONFIG_CHESS_GESTURE_BUTTON_GPIO);
375 esp_restart();
376 }
377 }
378 vTaskDelay(pdMS_TO_TICKS(25));
379 }
380 }
381 while (led_gesture_button_down()) {
382 (void)esp_task_wdt_reset();
383 vTaskDelay(pdMS_TO_TICKS(25));
384 }
385 ESP_LOGI(TAG, "[STAGING] wake on GPIO%d, restarting as power-on",
386 CONFIG_CHESS_GESTURE_BUTTON_GPIO);
387 esp_restart();
388#endif
389}

421static void handle_event(gesture_event_t ev) {
422 if (gesture_in_puzzle() && ev == GESTURE_SNAKE_START) {
423 gesture_next_puzzle();
424 return;
425 }
426 if (gesture_in_puzzle() && ev == GESTURE_COLOR) {
427 gesture_begin_opening();
428 return;
429 }
430 if (ev == GESTURE_DEMO) {
431 game_demo_show_set(!game_demo_show_active());
432 return;
433 }
434 if (game_demo_show_active() && ev != GESTURE_NONE) {
435 game_demo_show_set(false);
436 return;
437 }
438 if (ev == GESTURE_SNAKE_START || ev == GESTURE_SHUTDOWN_START) {
439 if (!led_gesture_anim_busy()) {
440 bool started = (ev == GESTURE_SNAKE_START) ? led_gesture_start_snake()
441 : led_gesture_start_shutdown();
442 if (!started) {
443 ESP_LOGW(TAG, "gesture animation did not start");
444 }
445 }
446 return;
447 }
448 if (led_gesture_anim_busy() && ev != GESTURE_NONE) {
449 ESP_LOGI(TAG, "[STAGING] gesture dropped, animation busy");
450 return;
451 }
452 switch (ev) {
453 case GESTURE_TOGGLE: {
454 if (hall_magnet_bench_active()) {
455 hall_magnet_bench_capture();
456 break;
457 }
458 /* One click while pairing is open closes the window. The lamp wave must
459 * not start, or it fights the blue pulse. */
460 if (ble_task_pairing_window_is_open() &&
461 ble_task_claim_pairing_success()) {
462 ESP_LOGI(TAG, "[STAGING] single click closes pairing");
463 ble_task_finish_pairing_window();
464 break;
465 }
466 ha_mode_t next =
467 (ha_light_get_mode() == HA_MODE_HA) ? HA_MODE_GAME : HA_MODE_HA;
468 uint32_t colors[64];
469 uint32_t lamp = lamp_rgb();
470 fill_mode_colors(next, colors);
471 /* Opening spreads the last lamp color. Closing shrinks that same color. */
472 bool inward = (next == HA_MODE_GAME);
473 if (!led_gesture_start_wave(colors, inward, lamp)) {
474 ESP_LOGW(TAG, "wave did not start");
475 return;
476 }
477 s_pending_mode = next;
478 s_wave_wait = true;
479 ESP_LOGI(TAG, "[STAGING] wave preview %s -> %s (commit after cover)",
480 next == HA_MODE_HA ? "game" : "lamp",
481 next == HA_MODE_HA ? "lamp" : "game");
482 break;
483 }
484 case GESTURE_COLOR: {
485 if (ha_light_get_mode() != HA_MODE_HA) {
486 ESP_LOGI(TAG, "[STAGING] double click ignored outside lamp");
487 break;
488 }
489 static const uint8_t k_lamp_colors[][3] = {
490 {255, 255, 255}, /* daylight white */
491 {255, 214, 170}, /* warm white */
492 {0, 0, 255}, /* blue */
493 {255, 0, 0}, /* red */
494 };
495 uint8_t r = 0, g = 0, b = 0, bright = 255;
496 bool on = true;
497 ha_light_get_state(&r, &g, &b, &bright, &on);
498 unsigned next = 0;
499 for (unsigned i = 0; i < 4u; i++) {
500 if (r == k_lamp_colors[i][0] && g == k_lamp_colors[i][1] &&
501 b == k_lamp_colors[i][2]) {
502 next = (i + 1u) % 4u;
503 break;
504 }
505 }
506 if (!ha_light_request_web_lamp(true, k_lamp_colors[next][0],
507 k_lamp_colors[next][1],
508 k_lamp_colors[next][2])) {
509 ESP_LOGW(TAG, "lamp color queue full");
510 break;
511 }
512 ESP_LOGI(TAG, "[STAGING] lamp color %u RGB(%u,%u,%u)", next,
513 k_lamp_colors[next][0], k_lamp_colors[next][1],
514 k_lamp_colors[next][2]);
515 break;
516 }
517 case GESTURE_PAIR:
518 ble_task_request_pairing();
519 break;
520 case GESTURE_SNAKE_START:
521 if (!led_gesture_start_snake()) {
522 ESP_LOGW(TAG, "snake did not start");
523 }
524 break;
525 case GESTURE_SHUTDOWN_START:
526 if (!led_gesture_start_shutdown()) {
527 ESP_LOGW(TAG, "shutdown chase did not start");
528 }
529 break;
530 case GESTURE_NONE:
531 default:
532 break;
533 }
534}

536static void handle_result(led_gesture_result_t result) {
537 switch (result) {
538 case LED_GESTURE_WAVE_DONE:
539 if (!s_wave_wait) {
540 break;
541 }
542 s_wave_wait = false;
543 if (s_opening_wave) {
544 s_opening_wave = false;
545 (void)game_opening_ensure_and_start();
546 break;
547 }
548 if (!ha_light_request_mode(s_pending_mode)) {
549 ESP_LOGW(TAG, "mode commit queue full");
550 } else {
551 ESP_LOGI(TAG, "[STAGING] mode committed: %s",
552 s_pending_mode == HA_MODE_HA ? "lamp" : "game");
553 }
554 break;
555 case LED_GESTURE_SNAKE_DONE:
556 if (!button_gesture_is_down()) {
557 ESP_LOGI(TAG, "[STAGING] snake ended after release, no new game");
558 break;
559 }
560 if (ha_light_get_mode() == HA_MODE_HA) {
561 (void)ha_light_request_mode(HA_MODE_GAME);
562 }
563 ESP_LOGI(TAG, "[STAGING] new game (snake finished while held)");
564 game_start_new_game();
565 break;
566 case LED_GESTURE_SNAKE_CANCELLED:
567 ESP_LOGI(TAG, "[STAGING] new game cancelled");
568 break;
569 case LED_GESTURE_SHUTDOWN_DONE:
570 gesture_power_off();
571 break;
572 case LED_GESTURE_SHUTDOWN_CANCELLED:
573 ESP_LOGI(TAG, "[STAGING] power off cancelled");
574 break;
575 case LED_GESTURE_PULSE_DONE:
576 case LED_GESTURE_RESULT_NONE:
577 default:
578 break;
579 }

583 QueueHandle_t q = button_gesture_queue();
584 if (q != NULL && !s_wave_wait && !led_gesture_anim_busy()) {
585 gesture_event_t ev;
586 if (xQueueReceive(q, &ev, 0) == pdTRUE) {
587 handle_event(ev);

589 }
590 if (!game_demo_show_active()) {
591 bool want_pulse = ble_task_pairing_pulse_wanted();
592 if (want_pulse != s_pair_pulse_on) {
593 if (want_pulse) {
594 (void)led_gesture_start_pair_pulse();
595 ESP_LOGI(TAG, "[STAGING] pairing pulse on");
596 } else {
597 led_gesture_stop_pair_pulse();
598 ESP_LOGI(TAG, "[STAGING] pairing pulse off");
599 }
600 s_pair_pulse_on = want_pulse;
601 }
602 }
603 demo_show_tick();
604 handle_result(led_gesture_take_result());
605}
