58static void fill_game_colors(uint32_t out[64]) {
59 memset(out, 0, 64 * sizeof(uint32_t));
60 if (!game_active) {
61 fill_checker(out);
62 return;
63 }
64 if (piece_lifted) {
65 out[chess_pos_to_led_index(lifted_piece_row, lifted_piece_col)] = 0xFFFF00u;
66 move_suggestion_t suggestions[32];
67 uint32_t n = game_get_available_moves(lifted_piece_row, lifted_piece_col,
68 suggestions, 32);
69 for (uint32_t i = 0; i < n; i++) {
70 piece_t dest = board[suggestions[i].to_row][suggestions[i].to_col];
71 bool enemy = (current_player == PLAYER_WHITE && dest >= PIECE_BLACK_PAWN &&
72 dest <= PIECE_BLACK_KING) ||
73 (current_player == PLAYER_BLACK && dest >= PIECE_WHITE_PAWN &&
74 dest <= PIECE_WHITE_KING);
75 out[chess_pos_to_led_index(suggestions[i].to_row, suggestions[i].to_col)] =
76 enemy ? 0xFFA500u : 0x00FF00u;
77 }
78 return;
79 }
80 if (castling_state.in_progress ||
81 !chess_policy_move_hints_movable_yellow() ||
82 !game_led_guidance_show_movable_yellow()) {
83 return;
84 }
85 (void)esp_task_wdt_reset();
86 for (uint8_t row = 0; row < 8; row++) {
87 for (uint8_t col = 0; col < 8; col++) {
88 piece_t piece = board[row][col];
89 bool mine = (current_player == PLAYER_WHITE && piece >= PIECE_WHITE_PAWN &&
90 piece <= PIECE_WHITE_KING) ||
91 (current_player == PLAYER_BLACK && piece >= PIECE_BLACK_PAWN &&
92 piece <= PIECE_BLACK_KING);
93 if (!mine) {
94 continue;
95 }
96 move_suggestion_t one;
97 if (game_get_available_moves(row, col, &one, 1) > 0) {
98 out[chess_pos_to_led_index(row, col)] = 0xFFFF00u;
99 }
100 }
101 }
102}

104static void fill_mode_colors(ha_mode_t mode, uint32_t out[64]) {
105 if (mode == HA_MODE_HA) {
106 uint8_t r = 255, g = 255, b = 255, bright = 255;
107 bool on = true;
108 ha_light_get_state(&r, &g, &b, &bright, &on);
109 uint32_t color = 0;
110 if (on) {
111 color = rgb_of((uint8_t)((r * bright) / 255),
112 (uint8_t)((g * bright) / 255),
113 (uint8_t)((b * bright) / 255));
114 }
115 for (int i = 0; i < 64; i++) {
116 out[i] = color;
117 }
118 return;
119 }
120 fill_game_colors(out);
121}

123static void gesture_power_off(void) {
124 gpio_num_t pin = (gpio_num_t)CONFIG_CHESS_GESTURE_BUTTON_GPIO;
125 for (int i = 0; i < CHESS_LED_COUNT_TOTAL; i++) {
126 led_set_pixel_internal((uint8_t)i, 0, 0, 0);

131 /* The hold that finished the chase must be released first, otherwise a
132 * level-low wake fires immediately. */
133 while (gpio_get_level(pin) == 0) {
134 (void)esp_task_wdt_reset();
135 vTaskDelay(pdMS_TO_TICKS(25));
136 }
137 vTaskDelay(pdMS_TO_TICKS(25));

144 (1ULL << CONFIG_CHESS_GESTURE_BUTTON_GPIO))

146 CONFIG_CHESS_GESTURE_BUTTON_GPIO);

148 ESP_GPIO_WAKEUP_GPIO_LOW);

150#else
151 /* ESP32-C6 deep-sleep GPIO wake is only GPIO0-7. IO23 is not one of them
152 * (and GPIO7 is the LED data pin). The 5 V rail has no load switch, so
153 * "off" is LEDs dark plus light sleep. The next press wakes the CPU and
154 * esp_restart() boots like a power-on. USB power stays connected. */
155 ESP_LOGI(TAG,
156 "[STAGING] power off: light sleep (GPIO%d cannot deep-sleep-wake), "
157 "USB 5 V stays on",
158 CONFIG_CHESS_GESTURE_BUTTON_GPIO);
159 (void)esp_wifi_stop();
160 esp_err_t sl = esp_light_sleep_start();
161 if (sl != ESP_OK) {
162 ESP_LOGE(TAG, "light sleep rejected: %s", esp_err_to_name(sl));
163 (void)esp_wifi_start();
164 gpio_wakeup_disable(pin);
165 if (ha_light_get_mode() == HA_MODE_HA) {
166 (void)ha_light_request_mode(HA_MODE_HA);
167 } else {
168 game_refresh_leds();
169 }
170 return;
171 }
172 while (gpio_get_level(pin) == 0) {
173 (void)esp_task_wdt_reset();
174 vTaskDelay(pdMS_TO_TICKS(25));
175 }

182static void handle_event(gesture_event_t ev) {
183 if (led_gesture_anim_busy() && ev != GESTURE_NONE) {
184 ESP_LOGI(TAG, "[STAGING] gesture dropped, animation busy");
185 return;
186 }
187 switch (ev) {
188 case GESTURE_TOGGLE: {
189 ha_mode_t next =
190 (ha_light_get_mode() == HA_MODE_HA) ? HA_MODE_GAME : HA_MODE_HA;
191 uint32_t colors[64];
192 fill_mode_colors(next, colors);
193 if (!led_gesture_start_wave(colors)) {
194 ESP_LOGW(TAG, "wave did not start");
195 return;
196 }
197 s_pending_mode = next;
198 s_wave_wait = true;
199 ESP_LOGI(TAG, "[STAGING] wave preview %s -> %s (commit after cover)",
200 next == HA_MODE_HA ? "game" : "lamp",
201 next == HA_MODE_HA ? "lamp" : "game");
202 break;
203 }
204 case GESTURE_PAIR:
205 ble_task_request_pairing();
206 break;
207 case GESTURE_SNAKE_START:
208 if (!led_gesture_start_snake()) {
209 ESP_LOGW(TAG, "snake did not start");
210 }
211 break;
212 case GESTURE_SHUTDOWN_START:
213 if (!led_gesture_start_shutdown()) {
214 ESP_LOGW(TAG, "shutdown chase did not start");
215 }
216 break;
217 case GESTURE_NONE:
218 default:
219 break;
220 }
221}

223static void handle_result(led_gesture_result_t result) {
224 switch (result) {
225 case LED_GESTURE_WAVE_DONE:
226 if (!s_wave_wait) {
227 break;
228 }
229 s_wave_wait = false;
230 if (!ha_light_request_mode(s_pending_mode)) {
231 ESP_LOGW(TAG, "mode commit queue full");
232 } else {
233 ESP_LOGI(TAG, "[STAGING] mode committed: %s",
234 s_pending_mode == HA_MODE_HA ? "lamp" : "game");
235 }
236 break;
237 case LED_GESTURE_SNAKE_DONE:
238 if (!button_gesture_is_down()) {
239 ESP_LOGI(TAG, "[STAGING] snake ended after release, no new game");
240 break;
241 }
242 if (ha_light_get_mode() == HA_MODE_HA) {
243 (void)ha_light_request_mode(HA_MODE_GAME);
244 }
245 ESP_LOGI(TAG, "[STAGING] new game (snake finished while held)");
246 game_start_new_game();
247 break;
248 case LED_GESTURE_SNAKE_CANCELLED:
249 ESP_LOGI(TAG, "[STAGING] new game cancelled");
250 break;
251 case LED_GESTURE_SHUTDOWN_DONE:
252 gesture_power_off();
253 break;
254 case LED_GESTURE_SHUTDOWN_CANCELLED:
255 ESP_LOGI(TAG, "[STAGING] power off cancelled");
256 break;
257 case LED_GESTURE_PULSE_DONE:
258 case LED_GESTURE_RESULT_NONE:
259 default:
260 break;
261 }
262}

264void game_gesture_poll(void) {
265 QueueHandle_t q = button_gesture_queue();
266 if (q != NULL) {
267 gesture_event_t ev;
268 while (xQueueReceive(q, &ev, 0) == pdTRUE) {
269 handle_event(ev);
270 }
271 }
272 bool want_pulse = ble_task_pairing_pulse_wanted();
273 if (want_pulse != s_pair_pulse_on) {
274 if (want_pulse) {
275 (void)led_gesture_start_pair_pulse();
276 ESP_LOGI(TAG, "[STAGING] pairing pulse on");
277 } else {
278 led_gesture_stop_pair_pulse();
279 ESP_LOGI(TAG, "[STAGING] pairing pulse off");
280 }
281 s_pair_pulse_on = want_pulse;
282 }
283 handle_result(led_gesture_take_result());
284}
