126static bool game_fen_expected_occupancy_64(const char *fen, uint8_t exp[64]) {
127 memset(exp, 0, 64);
128 if (fen == NULL) {
129 return false;
130 }
131 int row = 7;
132 int col = 0;
133 const char *p = fen;
134 while (*p != '\0' && *p != ' ') {
135 if (*p == '/') {
136 if (col != 8) {
137 return false;
138 }
139 row--;
140 col = 0;
141 p++;
142 continue;
143 }
144 if (*p >= '1' && *p <= '8') {
145 col += (*p - '0');
146 if (col > 8) {
147 return false;
148 }
149 p++;
150 continue;
151 }
152 if (row < 0 || row > 7 || col < 0 || col > 7) {
153 return false;
154 }
155 exp[row * 8 + col] = 1;
156 col++;
157 p++;
158 }
159 return (row == 0 && col == 8 && *p == ' ');
160}

162bool game_physical_squares_matches_fen(const char *fen) {
163 uint8_t exp[64];
164 if (!game_fen_expected_occupancy_64(fen, exp)) {
165 return false;
166 }
167 uint8_t phys[64];
168 squares_get_state(phys);
169 for (int i = 0; i < 64; i++) {
170 uint8_t want = exp[i] ? (uint8_t)1 : (uint8_t)0;
171 uint8_t got = phys[i] ? (uint8_t)1 : (uint8_t)0;
172 if (want != got) {
173 return false;
174 }
175 }
176 return true;
177}

183static bool puzzle_uci_ok(const char *uci) {
184 if (uci == NULL) {
185 return false;
186 }
187 size_t n = strlen(uci);
188 if (n != 4 && n != 5) {
189 return false;
190 }
191 if (uci[0] < 'a' || uci[0] > 'h' || uci[2] < 'a' || uci[2] > 'h') {
192 return false;
193 }
194 if (uci[1] < '1' || uci[1] > '8' || uci[3] < '1' || uci[3] > '8') {
195 return false;
196 }
197 if (n == 5) {
198 char p = uci[4];
199 if (p != 'q' && p != 'r' && p != 'b' && p != 'n') {
200 return false;
201 }
202 }
203 return true;
204}

291static bool puzzle_paint_target(const uint8_t exp[64]) {
292 uint8_t phys[64];
293 squares_get_state(phys);
294 game_led_clear_board();
295 bool match = true;
296 for (int i = 0; i < 64; i++) {
297 uint8_t want = exp[i] ? (uint8_t)1 : (uint8_t)0;
298 uint8_t row = (uint8_t)(i / 8);
299 uint8_t col = (uint8_t)(i % 8);
300 uint8_t led = chess_pos_to_led_index(row, col);
301 if (!puzzle_square_disagrees((uint8_t)i, want, phys)) {
302 if (want) {
303 game_led_paint_led(led, 0, 90, 0);
304 }
305 continue;
306 }
307 match = false;
308 uint8_t got = phys[i] ? (uint8_t)1 : (uint8_t)0;
309 if (want && !got) {
310 game_led_paint_led(led, 210, 210, 210);
311 } else {
312 game_led_paint_led(led, 255, 0, 0);
313 }
314 }
315 return match;
316}

335bool game_puzzle_stage_line(const char *fen, const char *title,
336 const char solution[][PUZZLE_UCI_MAX],
337 uint8_t count) {
338 uint8_t exp[64];
339 if (fen == NULL || !game_fen_expected_occupancy_64(fen, exp)) {
340 return false;
341 }
342 if (count > PUZZLE_MAX_PLIES) {
343 return false;
344 }
345 if (solution == NULL && count > 0) {
346 return false;
347 }
348 for (uint8_t i = 0; i < count; i++) {
349 if (!puzzle_uci_ok(solution[i])) {
350 return false;
351 }
352 }
353
354 puzzle_active = false;
355 puzzle_setup_active = false;
356 puzzle_setup_id = 0;
357 puzzle_active_id = 0;
358 puzzle_feedback = PUZZLE_FEEDBACK_NONE;
359 puzzle_solved = false;
360 puzzle_awaiting_opponent = false;
361 puzzle_sync_hold = false;
362 puzzle_setup_match_latched = false;
363 puzzle_ply = 0;
364 puzzle_last_uci[0] = '\0';
365 puzzle_clear_expected();
366
367 strncpy(puzzle_fen, fen, sizeof(puzzle_fen) - 1);
368 puzzle_fen[sizeof(puzzle_fen) - 1] = '\0';
369 puzzle_title[0] = '\0';
370 if (title != NULL) {
371 strncpy(puzzle_title, title, sizeof(puzzle_title) - 1);
372 puzzle_title[sizeof(puzzle_title) - 1] = '\0';
373 }
374 memset(puzzle_solution, 0, sizeof(puzzle_solution));
375 puzzle_solution_count = count;
376 for (uint8_t i = 0; i < count; i++) {
377 strncpy(puzzle_solution[i], solution[i], PUZZLE_UCI_MAX - 1);
378 }
379 puzzle_session++;
380 if (puzzle_session == 0) {
381 puzzle_session = 1;
382 }
383 puzzle_session_loaded = true;
384 STAGING_LOGI(TAG, "puzzle: staged session=%u plies=%u",
385 (unsigned)puzzle_session, (unsigned)count);
386 return true;
387}

470bool game_puzzle_try_start_staged(void) {
471 if (!puzzle_session_loaded || puzzle_fen[0] == '\0') {
472 return false;
473 }
474 if (puzzle_active && !puzzle_solved && !puzzle_sync_hold) {
475 return true;
476 }
477 if (!game_physical_squares_matches_fen(puzzle_fen)) {
478 if (!puzzle_setup_active) {
479 game_puzzle_apply_staged_setup();
480 } else {
481 uint8_t exp[64];
482 if (game_fen_expected_occupancy_64(puzzle_fen, exp)) {
483 puzzle_setup_match_latched = puzzle_paint_target(exp);
484 }
485 }
486 STAGING_LOGI(TAG, "puzzle: start blocked session=%u (occupancy)",
487 (unsigned)puzzle_session);
488 return false;
489 }
490 if (puzzle_solution_count == 0) {
491 char fen_copy[PUZZLE_FEN_MAX];
492 strncpy(fen_copy, puzzle_fen, sizeof(fen_copy) - 1);
493 fen_copy[sizeof(fen_copy) - 1] = '\0';
494 if (!game_start_new_game_from_fen(fen_copy)) {
495 STAGING_LOGI(TAG, "puzzle: position-only FEN rejected");
496 return false;
497 }
498 STAGING_LOGI(TAG, "puzzle: position only (no solution line)");
499 return true;
500 }
501
502 uint8_t keep_id = puzzle_active_id != 0 ? puzzle_active_id : puzzle_setup_id;
503 game_reset_game();
504
505 player_t fen_player = PLAYER_WHITE;
506 if (!game_load_position_from_fen(puzzle_fen, &fen_player)) {
507 ESP_LOGE(TAG, "puzzle: failed to load staged FEN");
508 puzzle_session_loaded = false;
509 return false;
510 }
511
512 current_player = fen_player;
513 current_game_state = GAME_STATE_ACTIVE;
514 game_active = true;
515 (*game_state_game_start_time()) = esp_timer_get_time() / 1000;
516 (*game_state_last_move_time()) = (*game_state_game_start_time());
517 (*game_state_board_setup_tutorial_active()) = false;
518 puzzle_setup_active = false;
519 puzzle_setup_id = 0;
520 puzzle_active = true;
521 puzzle_active_id = keep_id;
522 puzzle_solved = false;
523 puzzle_awaiting_opponent = false;
524 puzzle_ply = 0;
525 puzzle_last_uci[0] = '\0';
526 puzzle_feedback = PUZZLE_FEEDBACK_NONE;
527 puzzle_bind_expected_squares();
528 if (puzzle_expected_from[0] == '\0') {
529 ESP_LOGE(TAG, "puzzle: solution square parse failed");
530 puzzle_active = false;
531 return false;
532 }
533
534 puzzle_sync_hold = !puzzle_sensors_match_logic();
535 if (puzzle_sync_hold) {
536 uint8_t exp[64];
537 puzzle_expected_occupancy_from_logic(exp);
538 puzzle_paint_target(exp);
539 STAGING_LOGI(TAG, "puzzle: started but sensor hold session=%u",
540 (unsigned)puzzle_session);
541 } else {
542 puzzle_show_play_leds(false);
543 STAGING_LOGI(TAG, "puzzle: play session=%u plies=%u",
544 (unsigned)puzzle_session, (unsigned)puzzle_solution_count);
545 }
546 game_bump_revision_and_notify();
547 puzzle_mode_bind();
548 return true;
549}

604static void puzzle_on_drop(const position_t *pos, uint8_t from, uint8_t to,
605 bool has_from) {
606 uint8_t fr;
607 uint8_t fc;
608 uint8_t tr;
609 uint8_t tc;
610 position_move_t mv;
611 (void)has_from;
612 if (puzzle_setup_active || puzzle_sync_hold) {
613 game_mode_reject_drop();
614 return;
615 }
616 if (!puzzle_active || puzzle_solved) {
617 return;
618 }
619 fr = from / 8u;
620 fc = from % 8u;
621 tr = to / 8u;
622 tc = to % 8u;
623 if (game_puzzle_drop_is_expected(fr, fc, tr, tc)) {
624 return;
625 }
626 mv.from = from;

741 puzzle_show_play_leds(puzzle_awaiting_opponent);

823void game_puzzle_service_hall_move(void) {
824#if CONFIG_CHESS_MATRIX_INPUT_I2C_HALL
825 static uint32_t s_since;
826 static bool s_armed;
827 uint8_t occ[64];
828 uint8_t fr;
829 uint8_t fc;
830 uint8_t tr;
831 uint8_t tc;
832 uint32_t now;
833 piece_t mover;
834 chess_move_t move;
835 bool castle;
836 bool ep;
837 uint8_t rook_from = 0;
838 uint8_t rook_to = 0;
839 uint8_t sq;
840 if (!puzzle_active || puzzle_solved || puzzle_setup_active ||
841 puzzle_expected_from[0] == '\0' ||
842 !hall_i2c_squares_readings_usable()) {
843 s_armed = false;
844 return;
845 }
846 fr = puzzle_solution_from_row;
847 fc = puzzle_solution_from_col;
848 tr = puzzle_solution_to_row;
849 tc = puzzle_solution_to_col;
850 mover = game_get_piece(fr, fc);
851 if (mover == PIECE_EMPTY) {
852 s_armed = false;
853 return;
854 }
855 castle = (mover == PIECE_WHITE_KING || mover == PIECE_BLACK_KING) &&
856 fr == tr && (fc > tc ? fc - tc : tc - fc) == 2;
857 ep = en_passant_available && tr == en_passant_target_row &&
858 tc == en_passant_target_col &&
859 (mover == PIECE_WHITE_PAWN || mover == PIECE_BLACK_PAWN);
860 if (castle) {
861 rook_from = (uint8_t)(fr * 8 + (tc > fc ? 7 : 0));
862 rook_to = (uint8_t)(fr * 8 + (tc > fc ? 5 : 3));
863 }
864 squares_get_state(occ);
865 for (sq = 0; sq < 64; sq++) {
866 bool want = game_get_piece(sq / 8, sq % 8) != PIECE_EMPTY;
867 if (sq == (uint8_t)(fr * 8 + fc)) {
868 want = false;
869 }
870 if (sq == (uint8_t)(tr * 8 + tc)) {
871 want = true;
872 }
873 if (castle && sq == rook_from) {
874 want = false;
875 }
876 if (castle && sq == rook_to) {
877 want = true;
878 }
879 if (ep &&
880 sq == (uint8_t)(en_passant_victim_row * 8 + en_passant_victim_col)) {
881 want = false;
882 }
883 if ((occ[sq] != 0) != want) {
884 s_armed = false;
885 return;
886 }
887 }
888 now = (uint32_t)(esp_timer_get_time() / 1000);
889 if (!s_armed) {
890 s_armed = true;
891 s_since = now;
892 return;
893 }
894 if ((uint32_t)(now - s_since) < 150U) {
895 return;
896 }
897 s_armed = false;
898 memset(&move, 0, sizeof(move));
899 move.from_row = fr;
900 move.from_col = fc;
901 move.to_row = tr;
902 move.to_col = tc;
903 move.piece = mover;
904 move.captured_piece = game_get_piece(tr, tc);
905 {
906 board_machine_out_t bout = game_board_on_drop_squares(fr, fc, tr, tc, true);
907 if ((bout.kind != BM_OUT_MOVE && bout.kind != BM_OUT_NEED_PROMO) ||
908 !game_board_apply_machine_move(&bout, &move)) {
909 return;
910 }
911 }
912 if (castle) {
913 board_machine_reset(game_board_machine());
914 }
915 game_cmd_reply("Puzzle move accepted.", false, NULL);
916#else
917 (void)0;
918#endif
919}
