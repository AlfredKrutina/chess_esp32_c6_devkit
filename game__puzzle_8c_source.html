87static bool game_fen_expected_occupancy_64(const char *fen, uint8_t exp[64]) {
88 memset(exp, 0, 64);
89 if (fen == NULL) {
90 return false;
91 }
92 int row = 7;
93 int col = 0;
94 const char *p = fen;
95 while (*p != '\0' && *p != ' ') {
96 if (*p == '/') {
97 if (col != 8) {
98 return false;
99 }
100 row--;
101 col = 0;
102 p++;
103 continue;
104 }
105 if (*p >= '1' && *p <= '8') {
106 col += (*p - '0');
107 if (col > 8) {
108 return false;
109 }
110 p++;
111 continue;
112 }
113 if (row < 0 || row > 7 || col < 0 || col > 7) {
114 return false;
115 }
116 exp[row * 8 + col] = 1;
117 col++;
118 p++;
119 }
120 return (row == 0 && col == 8 && *p == ' ');
121}

133 if (want != got) {
134 return false;
135 }
136 }
137 return true;

144static bool puzzle_uci_ok(const char *uci) {
145 if (uci == NULL) {
146 return false;
147 }
148 size_t n = strlen(uci);
149 if (n != 4 && n != 5) {
150 return false;
151 }
152 if (uci[0] < 'a' || uci[0] > 'h' || uci[2] < 'a' || uci[2] > 'h') {
153 return false;
154 }
155 if (uci[1] < '1' || uci[1] > '8' || uci[3] < '1' || uci[3] > '8') {
156 return false;
157 }
158 if (n == 5) {
159 char p = uci[4];
160 if (p != 'q' && p != 'r' && p != 'b' && p != 'n') {
161 return false;
162 }
163 }
164 return true;
165}

252static bool puzzle_paint_target(const uint8_t exp[64]) {
253 uint8_t phys[64];
254 squares_get_state(phys);
255 led_clear_board_only();
256 bool match = true;
257 for (int i = 0; i < 64; i++) {
258 uint8_t want = exp[i] ? (uint8_t)1 : (uint8_t)0;
259 uint8_t row = (uint8_t)(i / 8);
260 uint8_t col = (uint8_t)(i % 8);
261 uint8_t led = chess_pos_to_led_index(row, col);
262 if (!puzzle_square_disagrees((uint8_t)i, want, phys)) {
263 if (want) {
264 led_set_pixel_safe(led, 0, 90, 0);
265 }
266 continue;
267 }
268 match = false;
269 uint8_t got = phys[i] ? (uint8_t)1 : (uint8_t)0;
270 if (want && !got) {
271 led_set_pixel_safe(led, 210, 210, 210);
272 } else {
273 led_set_pixel_safe(led, 255, 0, 0);
274 }
275 }
276 return match;
277}

304 return false;

306 if (solution == NULL && count > 0) {
307 return false;

309 for (uint8_t i = 0; i < count; i++) {
310 if (!puzzle_uci_ok(solution[i])) {
311 return false;
312 }
313 }

315 puzzle_active = false;
316 puzzle_setup_active = false;
317 puzzle_setup_id = 0;
318 puzzle_active_id = 0;
319 puzzle_feedback = PUZZLE_FEEDBACK_NONE;

323 puzzle_setup_match_latched = false;
324 puzzle_ply = 0;
325 puzzle_last_uci[0] = '\0';
326 puzzle_clear_expected();
327

329 puzzle_fen[sizeof(puzzle_fen) - 1] = '\0';
330 puzzle_title[0] = '\0';

346 (unsigned)puzzle_session, (unsigned)count);

430bool game_puzzle_try_start_staged(void) {
431 if (!puzzle_session_loaded || puzzle_fen[0] == '\0') {
432 return false;
433 }
434 if (puzzle_active && !puzzle_solved && !puzzle_sync_hold) {
435 return true;
436 }
437 if (!game_physical_squares_matches_fen(puzzle_fen)) {
438 if (!puzzle_setup_active) {
439 game_puzzle_apply_staged_setup();
440 } else {
441 uint8_t exp[64];
442 if (game_fen_expected_occupancy_64(puzzle_fen, exp)) {
443 puzzle_setup_match_latched = puzzle_paint_target(exp);
444 }
445 }
446 STAGING_LOGI(TAG, "puzzle: start blocked session=%u (occupancy)",
447 (unsigned)puzzle_session);
448 return false;
449 }
450 if (puzzle_solution_count == 0) {
451 char fen_copy[PUZZLE_FEN_MAX];
452 strncpy(fen_copy, puzzle_fen, sizeof(fen_copy) - 1);
453 fen_copy[sizeof(fen_copy) - 1] = '\0';
454 game_puzzle_cancel();
455 game_start_new_game_from_fen(fen_copy);
456 STAGING_LOGI(TAG, "puzzle: position only (no solution line)");
457 return true;
458 }
459
460 uint8_t keep_id = puzzle_active_id != 0 ? puzzle_active_id : puzzle_setup_id;
461 game_reset_game();
462
463 player_t fen_player = PLAYER_WHITE;
464 if (!game_load_position_from_fen(puzzle_fen, &fen_player)) {
465 ESP_LOGE(TAG, "puzzle: failed to load staged FEN");
466 puzzle_session_loaded = false;
467 return false;
468 }
469
470 current_player = fen_player;
471 current_game_state = GAME_STATE_ACTIVE;
472 game_active = true;
473 game_start_time = esp_timer_get_time() / 1000;
474 last_move_time = game_start_time;
475 board_setup_tutorial_active = false;
476 puzzle_setup_active = false;
477 puzzle_setup_id = 0;
478 puzzle_active = true;
479 puzzle_active_id = keep_id;
480 puzzle_solved = false;
481 puzzle_awaiting_opponent = false;
482 puzzle_ply = 0;
483 puzzle_last_uci[0] = '\0';
484 puzzle_feedback = PUZZLE_FEEDBACK_NONE;
485 puzzle_bind_expected_squares();
486 if (puzzle_expected_from[0] == '\0') {
487 ESP_LOGE(TAG, "puzzle: solution square parse failed");
488 puzzle_active = false;
489 return false;
490 }
491
492 puzzle_sync_hold = !puzzle_sensors_match_logic();
493 if (puzzle_sync_hold) {
494 uint8_t exp[64];
495 puzzle_expected_occupancy_from_logic(exp);
496 puzzle_paint_target(exp);
497 STAGING_LOGI(TAG, "puzzle: started but sensor hold session=%u",
498 (unsigned)puzzle_session);
499 } else {
500 puzzle_show_play_leds(false);
501 STAGING_LOGI(TAG, "puzzle: play session=%u plies=%u",
502 (unsigned)puzzle_session, (unsigned)puzzle_solution_count);
503 }
504 game_bump_revision_and_notify();
505 return true;
506}

602void game_puzzle_service_leds(void) {
603 if (error_recovery_state.waiting_for_move_correction || piece_lifted) {
604 return;
605 }
606 if (puzzle_setup_active && puzzle_fen[0] != '\0') {
607 uint8_t exp[64];
608 if (!game_fen_expected_occupancy_64(puzzle_fen, exp)) {
609 return;
610 }
611 bool match = puzzle_paint_target(exp);
612 if (match != puzzle_setup_match_latched) {
613 puzzle_setup_match_latched = match;
614 puzzle_feedback = match ? PUZZLE_FEEDBACK_READY : PUZZLE_FEEDBACK_NONE;
615 STAGING_LOGI(TAG, "puzzle: setup match -> %d", match ? 1 : 0);
616 game_bump_revision_and_notify();
617 }
618 return;
619 }
620 if (!puzzle_active || puzzle_solved) {
621 return;
622 }
623 bool match = puzzle_sensors_match_logic();
624 if (!match) {
625 uint8_t exp[64];
626 puzzle_expected_occupancy_from_logic(exp);
627 puzzle_paint_target(exp);
628 if (!puzzle_sync_hold) {
629 puzzle_sync_hold = true;
630 STAGING_LOGI(TAG, "puzzle: sync hold (board drifted)");
631 game_bump_revision_and_notify();
632 }
633 return;
634 }
635 if (puzzle_sync_hold) {
636 puzzle_sync_hold = false;
