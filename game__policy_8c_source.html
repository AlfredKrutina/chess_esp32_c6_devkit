207void game_policy_refresh_idle_board_leds(void) {
208#if CONFIG_CHESS_MATRIX_INPUT_I2C_HALL
209 if (game_hall_restore_guide_is_active()) {
210 return;
211 }
212 if (game_hall_position_blocks_idle_leds()) {
213 return;
214 }
215#endif
216 if (piece_lifted || castling_state.in_progress ||
217 (*game_state_promotion_state()).pending || game_ep_capture_waiting() ||
218 game_is_castle_animation_active() || game_is_castling_expected() ||
219 game_puzzle_suppress_move_hints()) {
220 return;
221 }
222 if (current_game_state != GAME_STATE_ACTIVE &&
223 current_game_state != GAME_STATE_IDLE) {
224 return;
225 }
226
227 /* Guide tick must not paint over this full frame if its clear event is dropped. */
228 game_event_required_step_disarm();
229 game_led_frame_begin(GAME_EVT_HINT_FADE_MS, false);
230 game_policy_highlight_movable_if_enabled();
231
232 if (!game_is_king_in_check(current_player)) {
233 game_led_frame_end();
234 return;
235 }
236 int king_row = -1;
237 int king_col = -1;
238 piece_t king_piece = (current_player == PLAYER_WHITE) ? PIECE_WHITE_KING
239 : PIECE_BLACK_KING;
240 for (int row = 0; row < 8; row++) {
241 for (int col = 0; col < 8; col++) {
242 if (game_get_piece(row, col) == king_piece) {
243 king_row = row;
244 king_col = col;
245 break;
246 }
247 }
248 if (king_row != -1) {
249 break;
250 }
251 }
252 if (king_row == -1 || king_col == -1 ||
253 !game_led_guidance_show_check_anim()) {
254 game_led_frame_end();
255 return;
256 }
257 game_led_frame_breathe((uint8_t)king_row, (uint8_t)king_col, 255, 192, 203,
258 GAME_EVT_BREATHE_TARGET, 0, 0, 0);
259 game_led_frame_end();
260}
