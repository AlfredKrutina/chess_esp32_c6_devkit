461void led_set_brightness_global(uint8_t brightness) {
462 if (brightness > 100)
463 brightness = 100;
464
465 if (led_unified_mutex != NULL) {
466 if (xSemaphoreTake(led_unified_mutex, LED_TASK_MUTEX_TIMEOUT_TICKS) ==
467 pdTRUE) {
468 global_brightness = brightness;
469
470 led_changes_pending = true;
471 for (int i = 0; i < CHESS_LED_COUNT_TOTAL; i++) {
472 led_pending_changes[i] = led_states[i];
473 led_changed_flags[i] = true;
474 }
475
476 /* Leave the commit on the LED task loop — calling led_commit_pending_changes()
477 * from the HTTP/BLE thread resulted in visible "fading" when changing brightness. */
478 xSemaphoreGive(led_unified_mutex);
479 ESP_LOGI(TAG, "Global brightness set to %d%% (deferred to LED task)",
480 brightness);
481 } else {
482 ESP_LOGW(TAG, "Failed to set brightness: Mutex timeout");
483 }
484 } else {
485 // No mutex (startup?), just set
486 global_brightness = brightness;
487 }
488}

505static void led_set_pixel_channels(uint8_t led_index, uint8_t red, uint8_t green,
506 uint8_t blue, uint8_t white) {
507 if (led_index >= CHESS_LED_COUNT_TOTAL) {
508 ESP_LOGE(TAG, "Invalid LED index: %d (max: %d)", led_index,
509 CHESS_LED_COUNT_TOTAL - 1);
510 return;
511 }
512
513 // Check if LED component is enabled
514 if (!led_component_enabled) {
515 ESP_LOGD(TAG, "LED component disabled - ignoring LED command for LED %d",
516 led_index);
517 return;
518 }
519
520 // If matrix scanning is disabled and this is a board LED (0-63), limit
521 // animations
522 if (!matrix_scanning_enabled && led_index < 64) {
523 // Only allow static colors, no animations
524 if (red > 0 || green > 0 || blue > 0 || white > 0) {
525 ESP_LOGD(
526 TAG,
527 "Matrix scanning disabled - limiting LED animation for board LED %d",
528 led_index);
529 }
530 }
531
532 // Unified mutex protection
533 if (led_unified_mutex != NULL) {
534 if (xSemaphoreTake(led_unified_mutex, LED_TASK_MUTEX_TIMEOUT_TICKS) !=
535 pdTRUE) {
536 ESP_LOGW(TAG,
537 "Failed to take LED unified mutex - skipping LED operation");
538 return;
539 }
540 }
541
542 // BATCH UPDATE SYSTEM - Just collect change, don't commit immediately.
543 // White lives in the top byte. RGB-only callers leave it at 0 and the
544 // budgeted writer extracts a neutral white onto the white die.
545 uint32_t color = ((uint32_t)white << 24) | ((uint32_t)red << 16) |
546 ((uint32_t)green << 8) | blue;
547
548 // Update internal state
549 led_states[led_index] = color;
550
551 // Mark change as pending for next batch commit
552 if (led_initialized && led_strip != NULL && !simulation_mode) {
553 led_pending_changes[led_index] = color;
554 led_changed_flags[led_index] = true; // Mark this LED as changed
555 led_changes_pending = true;
556 }
557
558 if (simulation_mode) {
559 ESP_LOGI(TAG, "LED[%d] = RGBW(%d,%d,%d,%d) = 0x%08" PRIX32, led_index, red,
560 green, blue, white, led_states[led_index]);
561 }
562
563 // Release unified mutex
564 if (led_unified_mutex != NULL) {
565 xSemaphoreGive(led_unified_mutex);
566 }
567}

702static void led_split_lamp_white(uint8_t *red, uint8_t *green, uint8_t *blue,
703 uint8_t *white) {
704 uint8_t lo = *red;
705 uint8_t hi = *red;
706 if (*green < lo) {
707 lo = *green;
708 }
709 if (*blue < lo) {
710 lo = *blue;
711 }
712 if (*green > hi) {
713 hi = *green;
714 }
715 if (*blue > hi) {
716 hi = *blue;
717 }
718 *white = 0;
719 if (lo == 0) {
720 return;
721 }
722 if ((uint16_t)lo * 3u >= (uint16_t)hi * 2u) {
723 *white = lo;
724 *red = (uint8_t)(*red - lo);
725 *green = (uint8_t)(*green - lo);
726 *blue = (uint8_t)(*blue - lo);
727 }

1816 10 * 1000 * 1000, // 10MHz resolution for precise WS2812B timing
1817 .mem_block_symbols = 128, // More buffer for stability
1818 .flags =
1819 {
1820 .with_dma = false, // ESP32-C6 doesn't support DMA for RMT
1821 },
1822 };

1826 led_strip_new_rmt_device(&strip_config, &rmt_config, &led_strip);
