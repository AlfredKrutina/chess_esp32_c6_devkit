668void uart_show_progress_bar(const char *label, uint32_t max_value,
669 uint32_t duration_ms) {
670 // Locking the mutex for the entire progress bar operation
671 if (uart_mutex != NULL) {
672 xSemaphoreTake(uart_mutex, portMAX_DELAY);
673 }
674
675 const int bar_width = 20;
676 uint32_t step_delay = duration_ms / max_value;
677 if (step_delay < 5)
678 step_delay = 5; // Minimum 5ms delay for smooth animation
679
680 if (color_enabled)
681 uart_write_string_immediate("\033[1;32m"); // bold green
682 char buffer[256];
683 snprintf(buffer, sizeof(buffer), "%s: [", label);
684 uart_write_string_immediate(buffer);
685
686 for (int i = 0; i < bar_width; i++) {
687 uart_write_string_immediate(".");
688 }
689 uart_write_string_immediate("] 0%");
690 if (color_enabled)
691 uart_write_string_immediate("\033[0m"); // reset colors
692
693 for (uint32_t i = 0; i <= max_value; i++) {
694 int filled = (i * bar_width) / max_value;
695
696 // Reset watchdog before each progress update (only if registered)
697 esp_err_t wdt_ret = uart_task_wdt_reset_safe();
698 if (wdt_ret != ESP_OK && wdt_ret != ESP_ERR_NOT_FOUND) {
699 // Task not registered with TWDT yet - this is normal during startup
700 }

990void uart_queue_message(uart_msg_type_t type, bool add_newline,
991 const char *format, ...) {
992 if (uart_output_queue == NULL) {
993 // Queue not ready, fall back to direct output
994 va_list args;
995 va_start(args, format);
996 char buffer[512];
997 vsnprintf(buffer, sizeof(buffer), format, args);
998 va_end(args);
999 if (UART_ENABLED) {
1000 uart_write_bytes(UART_PORT_NUM, buffer, strlen(buffer));
1001 if (add_newline)
1002 uart_write_bytes(UART_PORT_NUM, "\n", 1);
1003 } else {
1004 printf("%s", buffer);
1005 if (add_newline)
1006 printf("\n");
1007 }
1008 return;
1009 }
1010
1011 uart_message_t msg;
1012 msg.type = type;
1013 msg.add_newline = add_newline;
1014
1015 va_list args;
1016 va_start(args, format);
1017 vsnprintf(msg.message, sizeof(msg.message), format, args);
1018 va_end(args);
1019
1020 // Send to queue with timeout to prevent blocking
1021 if (xQueueSend(uart_output_queue, &msg, pdMS_TO_TICKS(10)) != pdTRUE) {
1022 // Queue full, fall back to direct output
1023 if (UART_ENABLED) {
1024 uart_write_bytes(UART_PORT_NUM, msg.message, strlen(msg.message));
1025 if (add_newline)
1026 uart_write_bytes(UART_PORT_NUM, "\n", 1);
1027 } else {
1028 printf("%s", msg.message);
1029 if (add_newline)
1030 printf("\n");
1031 }
1032 }
1033}

1038static void uart_process_output_queue(void) {
1039 uart_message_t msg;
1040
1041 // Process all pending messages
1042 while (xQueueReceive(uart_output_queue, &msg, 0) == pdTRUE) {
1043 // Take mutex for entire message output
1044 if (uart_mutex != NULL) {
1045 xSemaphoreTake(uart_mutex, portMAX_DELAY);
1046 }
1047
1048 // Apply color based on message type
1049 const char *color = COLOR_RESET;
1050 switch (msg.type) {
1051 case UART_MSG_ERROR:
1052 color = COLOR_ERROR;
1053 break;
1054 case UART_MSG_WARNING:
1055 color = COLOR_WARNING;
1056 break;
1057 case UART_MSG_SUCCESS:
1058 color = COLOR_SUCCESS;
1059 break;
1060 case UART_MSG_INFO:
1061 color = COLOR_INFO;
1062 break;
1063 case UART_MSG_DEBUG:
1064 color = COLOR_DEBUG;
1065 break;
1066 default:
1067 color = COLOR_RESET;
1068 break;
1069 }
1070
1071 if (UART_ENABLED) {
1072 if (color_enabled && msg.type != UART_MSG_NORMAL) {
1073 char buffer[512];
1074 snprintf(buffer, sizeof(buffer), "%s%s%s", color, msg.message,
1075 COLOR_RESET);
1076 uart_write_bytes(UART_PORT_NUM, buffer, strlen(buffer));
1077 } else {
1078 uart_write_bytes(UART_PORT_NUM, msg.message, strlen(msg.message));
1079 }
1080
1081 if (msg.add_newline) {
1082 uart_write_bytes(UART_PORT_NUM, "\n", 1);
1083 }
1084 } else {
1085 // For USB Serial JTAG, use printf
1086 if (color_enabled && msg.type != UART_MSG_NORMAL) {
1087 printf("%s%s%s", color, msg.message, COLOR_RESET);
1088 } else {
1089 printf("%s", msg.message);
1090 }
1091
1092 if (msg.add_newline) {
1093 printf("\n");
1094 }
1095 }
1096
1097 // No need for fflush with ESP-IDF UART driver
1098
1099 if (uart_mutex != NULL) {
1100 xSemaphoreGive(uart_mutex);
1101 }
1102 }

1203 if (history->count < history->max_size) {
1204 history->count++;
1205 }
1206}

1278static void handle_arrow_up(void) {
1279 if (command_history.count == 0) {
1280 return; // No history
1281 }
1282
1283 // If we are at the beginning of the navigation, we start from the last command
1284 if (history_navigation_index < 0) {
1285 history_navigation_index = command_history.count - 1;
1286 } else if (history_navigation_index > 0) {
1287 history_navigation_index--;
1288 }
1289
1290 // Get command from history (circular buffer)
1291 // The most recent command is at position (current - 1) % max_size
1292 // Older commands are at (current - 2) % max_size, etc.
1293 int start_idx = (command_history.current - command_history.count +
1294 command_history.max_size) %
1295 command_history.max_size;
1296 int idx = (start_idx + history_navigation_index) % command_history.max_size;
1297
1298 const char *cmd = command_history.commands[idx];
1299 if (cmd == NULL || strlen(cmd) == 0) {
1300 return;
1301 }
1302
1303 // Display the command in the input buffer
1304 input_buffer_clear(&input_buffer);
1305 strncpy(input_buffer.buffer, cmd, UART_CMD_BUFFER_SIZE - 1);
1306 input_buffer.buffer[UART_CMD_BUFFER_SIZE - 1] = '\0';
1307 input_buffer.length = strlen(input_buffer.buffer);
1308 input_buffer.pos = input_buffer.length;
1309

1311 refresh_input_display();
1312}

2108 return; // We are waiting for the next sign

2113 esc_state = ESC_STATE_BRACKET;
2114 return; // We are waiting for the final sign

2116 // Invalid escape sequence, reset and continue normal processing
2117 esc_state = ESC_STATE_NONE;

2119 }
2120
2121 if (esc_state == ESC_STATE_BRACKET) {
2122 esc_state = ESC_STATE_NONE; // Resetovat stav

2125 // Arrow Up - previous command from history
2126 handle_arrow_up();

2129 // Arrow Down - next command from history
2130 handle_arrow_down();
2131 return;
2132 }

2134 return;
2135 }

2141 swallow_lf = false;
2142 return;
2143 }
2144 swallow_lf = false;
2145

2147 if (c == '\r') {
2148 swallow_lf = true;
2149 }

2151 if (input_buffer.length > 0) {
2152 uart_send_line(""); // New line after command

2159
2160 // Clear buffer
2161 input_buffer_clear(&input_buffer);
