666void uart_show_progress_bar(const char *label, uint32_t max_value,
667 uint32_t duration_ms) {
668 // Locking the mutex for the entire progress bar operation
669 if (uart_mutex != NULL) {
670 xSemaphoreTake(uart_mutex, portMAX_DELAY);
671 }
672
673 const int bar_width = 20;
674 uint32_t step_delay = duration_ms / max_value;
675 if (step_delay < 5)
676 step_delay = 5; // Minimum 5ms delay for smooth animation
677
678 if (color_enabled)
679 uart_write_string_immediate("\033[1;32m"); // bold green
680 char buffer[256];
681 snprintf(buffer, sizeof(buffer), "%s: [", label);
682 uart_write_string_immediate(buffer);
683
684 for (int i = 0; i < bar_width; i++) {
685 uart_write_string_immediate(".");
686 }
687 uart_write_string_immediate("] 0%");
688 if (color_enabled)
689 uart_write_string_immediate("\033[0m"); // reset colors
690
691 for (uint32_t i = 0; i <= max_value; i++) {
692 int filled = (i * bar_width) / max_value;
693
694 // Reset watchdog before each progress update (only if registered)
695 esp_err_t wdt_ret = uart_task_wdt_reset_safe();
696 if (wdt_ret != ESP_OK && wdt_ret != ESP_ERR_NOT_FOUND) {
697 // Task not registered with TWDT yet - this is normal during startup
698 }
699
700 // Move cursor back to start of progress bar

988void uart_queue_message(uart_msg_type_t type, bool add_newline,
989 const char *format, ...) {
990 if (uart_output_queue == NULL) {
991 // Queue not ready, fall back to direct output
992 va_list args;
993 va_start(args, format);
994 char buffer[512];
995 vsnprintf(buffer, sizeof(buffer), format, args);
996 va_end(args);
997 if (UART_ENABLED) {
998 uart_write_bytes(UART_PORT_NUM, buffer, strlen(buffer));
999 if (add_newline)
1000 uart_write_bytes(UART_PORT_NUM, "\n", 1);
1001 } else {
1002 printf("%s", buffer);
1003 if (add_newline)
1004 printf("\n");
1005 }
1006 return;
1007 }
1008
1009 uart_message_t msg;
1010 msg.type = type;
1011 msg.add_newline = add_newline;
1012
1013 va_list args;
1014 va_start(args, format);
1015 vsnprintf(msg.message, sizeof(msg.message), format, args);
1016 va_end(args);
1017
1018 // Send to queue with timeout to prevent blocking
1019 if (xQueueSend(uart_output_queue, &msg, pdMS_TO_TICKS(10)) != pdTRUE) {
1020 // Queue full, fall back to direct output
1021 if (UART_ENABLED) {
1022 uart_write_bytes(UART_PORT_NUM, msg.message, strlen(msg.message));
1023 if (add_newline)
1024 uart_write_bytes(UART_PORT_NUM, "\n", 1);
1025 } else {
1026 printf("%s", msg.message);
1027 if (add_newline)
1028 printf("\n");
1029 }
1030 }
1031}

1036static void uart_process_output_queue(void) {
1037 uart_message_t msg;
1038
1039 // Process all pending messages
1040 while (xQueueReceive(uart_output_queue, &msg, 0) == pdTRUE) {
1041 // Take mutex for entire message output
1042 if (uart_mutex != NULL) {
1043 xSemaphoreTake(uart_mutex, portMAX_DELAY);
1044 }
1045
1046 // Apply color based on message type
1047 const char *color = COLOR_RESET;
1048 switch (msg.type) {
1049 case UART_MSG_ERROR:
1050 color = COLOR_ERROR;
1051 break;
1052 case UART_MSG_WARNING:
1053 color = COLOR_WARNING;
1054 break;
1055 case UART_MSG_SUCCESS:
1056 color = COLOR_SUCCESS;
1057 break;
1058 case UART_MSG_INFO:
1059 color = COLOR_INFO;
1060 break;
1061 case UART_MSG_DEBUG:
1062 color = COLOR_DEBUG;
1063 break;
1064 default:
1065 color = COLOR_RESET;
1066 break;
1067 }
1068
1069 if (UART_ENABLED) {
1070 if (color_enabled && msg.type != UART_MSG_NORMAL) {
1071 char buffer[512];
1072 snprintf(buffer, sizeof(buffer), "%s%s%s", color, msg.message,
1073 COLOR_RESET);
1074 uart_write_bytes(UART_PORT_NUM, buffer, strlen(buffer));
1075 } else {
1076 uart_write_bytes(UART_PORT_NUM, msg.message, strlen(msg.message));
1077 }
1078
1079 if (msg.add_newline) {
1080 uart_write_bytes(UART_PORT_NUM, "\n", 1);
1081 }
1082 } else {
1083 // For USB Serial JTAG, use printf
1084 if (color_enabled && msg.type != UART_MSG_NORMAL) {
1085 printf("%s%s%s", color, msg.message, COLOR_RESET);
1086 } else {
1087 printf("%s", msg.message);
1088 }
1089
1090 if (msg.add_newline) {
1091 printf("\n");
1092 }
1093 }
1094
1095 // No need for fflush with ESP-IDF UART driver
1096
1097 if (uart_mutex != NULL) {
1098 xSemaphoreGive(uart_mutex);
1099 }
1100 }

1182void command_history_add(command_history_t *history, const char *command) {
1183 if (command == NULL || strlen(command) == 0)
1184 return;
1185
1186 // Don't add duplicate commands
1187 if (history->count > 0) {
1188 int last_idx =
1189 (history->current - 1 + history->max_size) % history->max_size;
1190 if (strcmp(history->commands[last_idx], command) == 0) {
1191 return;
1192 }
1193 }
1194
1195 // Add command to history
1196 strncpy(history->commands[history->current], command,
1197 UART_CMD_BUFFER_SIZE - 1);
1198 history->commands[history->current][UART_CMD_BUFFER_SIZE - 1] = '\0';
1199
1200 history->current = (history->current + 1) % history->max_size;
1201 if (history->count < history->max_size) {
1202 history->count++;
1203 }
1204}

1276static void handle_arrow_up(void) {
1277 if (command_history.count == 0) {
1278 return; // No history
1279 }
1280
1281 // If we are at the beginning of the navigation, we start from the last command
1282 if (history_navigation_index < 0) {
1283 history_navigation_index = command_history.count - 1;
1284 } else if (history_navigation_index > 0) {
1285 history_navigation_index--;
1286 }
1287
1288 // Get command from history (circular buffer)
1289 // The most recent command is at position (current - 1) % max_size
1290 // Older commands are at (current - 2) % max_size, etc.
1291 int start_idx = (command_history.current - command_history.count +
1292 command_history.max_size) %
1293 command_history.max_size;
1294 int idx = (start_idx + history_navigation_index) % command_history.max_size;
1295
1296 const char *cmd = command_history.commands[idx];
1297 if (cmd == NULL || strlen(cmd) == 0) {
1298 return;
1299 }
1300
1301 // Display the command in the input buffer
1302 input_buffer_clear(&input_buffer);
1303 strncpy(input_buffer.buffer, cmd, UART_CMD_BUFFER_SIZE - 1);
1304 input_buffer.buffer[UART_CMD_BUFFER_SIZE - 1] = '\0';
1305 input_buffer.length = strlen(input_buffer.buffer);
1306 input_buffer.pos = input_buffer.length;
1307
1308 // Refresh the display in the terminal
1309 refresh_input_display();
1310}

1315static void handle_arrow_down(void) {

1321 return; // We're at the end

1335 int start_idx = (command_history.current - command_history.count +
1336 command_history.max_size) %
1337 command_history.max_size;
