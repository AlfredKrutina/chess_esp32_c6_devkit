60static void gesture_emit(gesture_event_t id, const char *staging) {
61 if (s_latched) {
62 return;
63 }
64 s_latched = true;
65 s_clicks = 0;
66 ESP_LOGI(TAG, "[STAGING] gesture: %s", staging);
67 if (id == GESTURE_SNAKE_START) {
68 (void)led_gesture_start_snake();
69 } else if (id == GESTURE_SHUTDOWN_START) {
70 (void)led_gesture_start_shutdown();
71 }
72 if (s_queue != NULL && xQueueSend(s_queue, &id, 0) != pdTRUE) {
73 ESP_LOGW(TAG, "gesture queue full, drop %d", (int)id);
74 }
75 s_phase = GS_LOCK;
76 s_lock_ms = gesture_now_ms();
77}

79static void gesture_on_press(uint32_t now) {
80 if (s_phase == GS_LOCK || s_phase == GS_SWALLOW || s_phase == GS_WAIT_UP ||
81 s_phase == GS_DOWN_IDLE || s_phase == GS_DOWN_SEQ) {
82 return;
83 }
84 s_press_ms = now;
85 if (s_phase == GS_GAP) {
86 s_phase = GS_DOWN_SEQ;
87 return;
88 }
89 s_clicks = 0;
90 s_phase = GS_DOWN_IDLE;
91}

93static void gesture_on_release(uint32_t now) {
94 if (led_gesture_anim_busy() || s_phase == GS_LOCK) {
95 led_gesture_note_release();
96 }
97 s_up_ms = now;
98 if (s_phase == GS_DOWN_IDLE) {
99 uint32_t held = now - s_press_ms;
100 if (held < GESTURE_CLICK_MAX_MS) {
101 s_clicks = 1;
102 s_phase = GS_GAP;
103 } else {
104 s_phase = GS_WAIT_UP;
105 }
106 return;
107 }
108 if (s_phase == GS_DOWN_SEQ) {
109 uint32_t held = now - s_press_ms;
110 if (held >= GESTURE_CLICK_MAX_MS) {
111 s_phase = GS_WAIT_UP;
112 return;
113 }
114 if (s_clicks < 255) {
115 s_clicks++;
116 }
117 if (s_clicks >= 4) {
118 gesture_emit(GESTURE_DEMO, "four clicks -> demo");
119 return;
120 }
121 s_phase = GS_GAP;
122 }
123}

160void button_gesture_poll(void) {
161 if (!s_gpio_ready) {
162 return;
163 }
164 uint32_t now = gesture_now_ms();
165 bool raw = led_gesture_button_down();
166 if (!s_change_valid || raw != s_raw_down) {
167 s_raw_down = raw;
168 s_change_ms = now;
169 s_change_valid = true;
170 }
171 if ((now - s_change_ms) >= GESTURE_DEBOUNCE_MS && s_down != s_raw_down) {
172 s_down = s_raw_down;
173 led_gesture_set_button_held(s_down);
174 ESP_LOGI(TAG, "SW1 %s GPIO%d raw=%d", s_down ? "down" : "up",
175 CONFIG_CHESS_GESTURE_BUTTON_GPIO, (int)raw);
176 if (s_down) {
177 gesture_on_press(now);
178 } else {
179 gesture_on_release(now);
180 }
181 }
182
183 if (s_phase == GS_LOCK) {
184 /* One action at a time. Clicks during the action, or in the quiet after
185 * it, do not start a second or third gesture. */
186 if (s_down || led_gesture_anim_busy()) {
187 s_lock_ms = now;
188 return;
189 }
190 if ((now - s_lock_ms) >= GESTURE_GAP_MS) {
191 s_phase = GS_IDLE;
192 s_clicks = 0;
193 s_latched = false;
194 }
195 return;
196 }
197 if (s_phase == GS_WAIT_UP) {
198 if (!s_down) {
199 s_phase = GS_IDLE;
200 s_clicks = 0;
201 }
202 return;
203 }
204 if (s_phase == GS_SWALLOW) {
205 if (!s_down && (now - s_up_ms) >= GESTURE_GAP_MS) {
206 s_phase = GS_IDLE;
207 s_clicks = 0;
208 }
209 return;
210 }
211 if (s_phase == GS_DOWN_IDLE && s_down &&
212 (now - s_press_ms) >= GESTURE_CLICK_MAX_MS) {
213 gesture_emit(GESTURE_SNAKE_START, "hold from idle -> new game snake");
214 return;
215 }
216 if (s_phase == GS_DOWN_SEQ && s_down && s_clicks == 1 &&
217 (now - s_press_ms) >= GESTURE_CLICK_MAX_MS) {
218 gesture_emit(GESTURE_SHUTDOWN_START, "click then hold -> power off");
219 return;
220 }
221 if (s_phase == GS_GAP && !s_down && (now - s_up_ms) >= GESTURE_GAP_MS) {
222 if (s_clicks == 1) {
223 gesture_emit(GESTURE_TOGGLE, "single click -> game/lamp toggle");
224 } else if (s_clicks == 2) {
225 gesture_emit(GESTURE_COLOR, "double click -> next lamp color");
226 } else if (s_clicks == 3) {
227 gesture_emit(GESTURE_PAIR, "triple click -> BLE pairing");
228 } else {
229 s_phase = GS_IDLE;
230 s_clicks = 0;
231 }
232 }
233}
