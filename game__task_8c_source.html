536uint32_t game_calculate_position_hash(void) {
537 uint32_t hash = 0;
538
539 // Hash board pieces
540 for (int row = 0; row < 8; row++) {
541 for (int col = 0; col < 8; col++) {
542 piece_t piece = game_get_piece(row, col);
543 if (piece != PIECE_EMPTY) {
544 // Combine piece type, position, and color
545 uint32_t piece_hash = (piece << 16) | (row << 8) | col;
546 hash = ((hash << 5) + hash) ^ piece_hash; // Simple hash function
547 }
548 }
549 }
550
551 // Include current player in hash
552 hash = ((hash << 5) + hash) ^ (current_player << 24);
553
554 // Include castling rights
555 uint32_t castling_rights = 0;
556 if (!piece_moved[0][4])
557 castling_rights |= 0x01; // White king
558 if (!piece_moved[0][0])
559 castling_rights |= 0x02; // White rook a1
560 if (!piece_moved[0][7])
561 castling_rights |= 0x04; // White rook h1
562 if (!piece_moved[7][4])
563 castling_rights |= 0x08; // Black king
564 if (!piece_moved[7][0])
565 castling_rights |= 0x10; // Black rook a8
566 if (!piece_moved[7][7])
567 castling_rights |= 0x20; // Black rook h8
568 hash = ((hash << 5) + hash) ^ castling_rights;
569
570 // Include en passant state
571 if (en_passant_available) {
572 uint32_t en_passant_hash =
573 (en_passant_target_row << 8) | en_passant_target_col;
574 hash = ((hash << 5) + hash) ^ en_passant_hash;
575 }
576
577 return hash;
578}

585bool game_is_position_repeated(void) {
586 uint32_t current_hash = game_calculate_position_hash();
587 int repetition_count = 0;
588
589 // Count how many times current position appears in history
590 for (int i = 0; i < position_history_count; i++) {
591 if (position_history[i] == current_hash) {
592 repetition_count++;
593 if (repetition_count >= 3) {
594 return true; // Threefold repetition detected
595 }
596 }
597 }
598
599 return false;

744 char from_square[3], to_square[3];
745 game_coords_to_square(move->from_row, move->from_col, from_square);
746 game_coords_to_square(move->to_row, move->to_col, to_square);
747
748 const char *piece_name = game_get_piece_name(move->piece);
749 const char *player_name =
750 (current_player == PLAYER_WHITE) ? "White" : "Black";
751 piece_t source_piece = game_get_piece(move->from_row, move->from_col);
752 piece_t dest_piece = game_get_piece(move->to_row, move->to_col);
753
754 // Enhanced error messages with colors and detailed reasoning
755 printf("\r\n");
756 printf(FMT_ERROR("❌ INVALID MOVE!") "\r\n");
757 printf(FMT_INFO(" • Move: %s → %s") "\r\n", from_square, to_square);
758 printf(FMT_INFO(" • Piece: %s") "\r\n", piece_name);
759
760 switch (error) {
761 case MOVE_ERROR_NO_PIECE:
762 print_error_detail("Reason", "No piece at %s", from_square);
763 print_error_detail("Solution", "Choose a square with your piece");
764 break;
765

767 print_error_detail("Reason", "%s cannot move %s's piece", player_name,
768 (current_player == PLAYER_WHITE) ? "Black" : "White");
769 print_error_detail("Solution", "Move only your own pieces");
770 break;
771
772 case MOVE_ERROR_BLOCKED_PATH:
773 print_error_detail("Reason", "Path from %s to %s is blocked", from_square,
774 to_square);
775 print_error_detail("Hint", "Another piece is blocking the way");
776 print_error_detail("Solution",
777 "Clear the path or choose different destination");
778 break;
779
780 case MOVE_ERROR_INVALID_PATTERN:
781 // IMPROVED REASONING for pawns
782 if (source_piece == PIECE_WHITE_PAWN || source_piece == PIECE_BLACK_PAWN) {
783 int col_diff = abs((int)move->to_col - (int)move->from_col);
784 bool is_diagonal = (col_diff == 1);
785
786 if (is_diagonal && dest_piece == PIECE_EMPTY) {
787 print_error_detail(
788 "Reason",
789 "Pawn can only capture diagonally when enemy piece is present");

794 print_error_detail(
795 "Solution",
796 "Move pawn forward or capture diagonally where enemy piece is");
797 } else if (!is_diagonal && dest_piece != PIECE_EMPTY) {
798 print_error_detail("Reason", "Pawn cannot capture straight ahead");
799 print_error_detail(
800 "Hint", "Square %s is occupied - pawns can only capture diagonally",

802 print_error_detail("Solution",
803 "Capture the piece diagonally, not forward");

805 print_error_detail("Reason", "Pawn cannot move in this pattern");
806 print_error_detail("Hint", "Pawns move forward 1 square (or 2 from "

808 print_error_detail("Solution", "Follow pawn movement rules");
809 }

811 print_error_detail("Reason", "%s cannot move from %s to %s", piece_name,
812 from_square, to_square);
813 print_error_detail("Solution", "Follow the piece's movement rules");
814 }
815 break;
816
817 case MOVE_ERROR_KING_IN_CHECK:
818 print_error_detail("Reason",
819 "This move would leave/keep your king in check");
820 // Find and display king position
821 printf(FMT_HINT(" • Hint: Your king at "));
822 piece_t king_piece =

837 "Solution",

840
841 case MOVE_ERROR_CASTLING_BLOCKED:
842 print_error_detail("Reason", "Castling is not allowed");
843 print_error_detail(
844 "Hint", "King/Rook moved, squares under attack, or path blocked");
845 print_error_detail(

853 print_error_detail("Solution", "Capture normally or make different move");

870 print_error_detail("Reason", "Game is not active");
871 print_error_detail("Solution",
872 "Start a new game first (use GAME_NEW command)");

1148uint32_t game_get_available_moves(uint8_t row, uint8_t col,
1149 move_suggestion_t *suggestions,
1150 uint32_t max_suggestions) {
1151 if (suggestions == NULL || max_suggestions == 0) {
1152 return 0;
1153 }
1154
1155 piece_t piece = game_get_piece(row, col);
1156
1157 // If the piece is empty and the resignation timer is active for that position,
1158 // use saved king position from (*game_state_resignation_state())
1159 if (piece == PIECE_EMPTY) {
1160 if ((*game_state_resignation_state()).active && (*game_state_resignation_state()).king_row == row &&
1161 (*game_state_resignation_state()).king_col == col) {
1162 // King is raised during resignation timer - use king type z
1163 // (*game_state_resignation_state()).player
1164 piece = ((*game_state_resignation_state()).player == PLAYER_WHITE) ? PIECE_WHITE_KING
1165 : PIECE_BLACK_KING;
1166 ESP_LOGI(TAG,
1167 "🏰 game_get_available_moves: Using stored king piece (%d) "
1168 "because king is lifted (resignation timer active)",
1169 piece);
1170 } else {
1171 return 0;
1172 }
1173 }
1174
1175 uint32_t count = 0;
1176
1177 // If the rook is in progress and the rook is being raised from rook_from, add move to
1178 // rook_to jako valid move
1179 if (castling_state.in_progress) {
1180 ESP_LOGI(
1181 TAG,
1182 "🏰 game_get_available_moves: castling_state.in_progress=true, "
1183 "checking row=%d col=%d piece=%d, rook_from_row=%d rook_from_col=%d",
1184 row, col, piece, castling_state.rook_from_row,
1185 castling_state.rook_from_col);
1186
1187 if (row == castling_state.rook_from_row &&
1188 col == castling_state.rook_from_col &&
1189 (piece == PIECE_WHITE_ROOK || piece == PIECE_BLACK_ROOK)) {
1190 // This is a rook for rook - add move to rook_to as a valid move
1191 ESP_LOGI(TAG,
1192 "🏰 Castling rook detected at %c%d - adding valid move to %c%d",
1193 'a' + col, row + 1, 'a' + castling_state.rook_to_col,
1194 castling_state.rook_to_row + 1);
1195
1196 if (count < max_suggestions) {
1197 suggestions[count].from_row = row;
1198 suggestions[count].from_col = col;
1199 suggestions[count].to_row = castling_state.rook_to_row;
1200 suggestions[count].to_col = castling_state.rook_to_col;
1201 suggestions[count].piece = piece;
1202 suggestions[count].is_capture = false;
1203 suggestions[count].is_check = suggestion_gives_check(
1204 row, col, castling_state.rook_to_row, castling_state.rook_to_col,
1205 piece);
1206 suggestions[count].is_castling = false;
1207 suggestions[count].is_en_passant = false;
1208 suggestions[count].score = 0;
1209 count++;
1210 ESP_LOGI(
1211 TAG,
1212 "🏰 Castling rook: Added valid move from %c%d to %c%d (count=%" PRIu32 ")",
1213 'a' + col, row + 1, 'a' + castling_state.rook_to_col,
1214 castling_state.rook_to_row + 1, count);
1215 return count; // Return only this move - other moves are blocked
1216 } else {
1217 ESP_LOGW(
1218 TAG,
1219 "🏰 Castling rook: max_suggestions=%lu reached, cannot add move",
1220 max_suggestions);
1221 }
1222 } else {
1223 ESP_LOGD(TAG,
1224 "🏰 game_get_available_moves: Not castling rook - row=%d col=%d "
1225 "piece=%d vs rook_from_row=%d rook_from_col=%d",
1226 row, col, piece, castling_state.rook_from_row,
1227 castling_state.rook_from_col);
1228 }
1229 }
1230
1231 // Optimized move generation based on piece type
1232 switch (piece) {
1233 case PIECE_WHITE_KNIGHT:
1234 case PIECE_BLACK_KNIGHT:
1235 // Knight moves in L-shape: 8 possible positions
1236 for (int i = 0; i < 8 && count < max_suggestions; i++) {
1237 int to_row = row + (game_state_knight_moves())[i][0];
1238 int to_col = col + (game_state_knight_moves())[i][1];
1239
1240 if (!game_is_valid_square(to_row, to_col))
1241 continue;
1242
1243 piece_t target = game_get_piece(to_row, to_col);
1244 player_t player =
1245 (piece == PIECE_WHITE_KNIGHT) ? PLAYER_WHITE : PLAYER_BLACK;
1246
1247 // Skip if occupied by own piece
1248 if (game_is_own_piece(target, player))
1249 continue;
1250
1251 // Create temporary move for validation
1252 chess_move_t temp_move = {.from_row = row,
1253 .from_col = col,
1254 .to_row = to_row,
1255 .to_col = to_col,
1256 .piece = piece,
1257 .captured_piece = target,
1258 .timestamp = 0};
1259
1260 // Check if move is valid (including check validation)
1261 move_error_t error = game_is_valid_move(&temp_move);
1262 if (error == MOVE_ERROR_NONE) {
1263 suggestions[count].from_row = row;
1264 suggestions[count].from_col = col;
1265 suggestions[count].to_row = to_row;
1266 suggestions[count].to_col = to_col;
1267 suggestions[count].piece = piece;
1268 suggestions[count].is_capture = (target != PIECE_EMPTY);
1269 suggestions[count].is_check = suggestion_gives_check(
1270 row, col, (uint8_t)to_row, (uint8_t)to_col, piece);
1271 suggestions[count].is_castling = false;
1272 suggestions[count].is_en_passant = false;
1273 suggestions[count].score = 0;
1274 count++;
1275 }
1276 }
1277 break;
1278
1279 default:
1280 // For other pieces, use the original method (less efficient but works)
1281 for (int to_row = 0; to_row < 8 && count < max_suggestions; to_row++) {
1282 for (int to_col = 0; to_col < 8 && count < max_suggestions; to_col++) {
1283 // Skip same position
1284 if (to_row == row && to_col == col) {
1285 continue;
1286 }
1287
1288 // Create temporary move for validation
1289 chess_move_t temp_move = {.from_row = row,
1290 .from_col = col,
1291 .to_row = to_row,
1292 .to_col = to_col,
1293 .piece = piece,
1294 .captured_piece = game_get_piece(to_row, to_col),
1295 .timestamp = 0};
1296
1297 // Check if move is valid
1298 move_error_t error = game_is_valid_move(&temp_move);
1299 if (error == MOVE_ERROR_NONE) {
1300 // Add to suggestions
1301 suggestions[count].from_row = row;
1302 suggestions[count].from_col = col;
1303 suggestions[count].to_row = to_row;
1304 suggestions[count].to_col = to_col;
1305 suggestions[count].piece = piece;
1306 suggestions[count].is_capture =
1307 (game_get_piece(to_row, to_col) != PIECE_EMPTY);
1308 suggestions[count].is_check = suggestion_gives_check(
1309 row, col, (uint8_t)to_row, (uint8_t)to_col, piece);
1310
1311 // Detect castling
1312 suggestions[count].is_castling =
1313 (piece == PIECE_WHITE_KING || piece == PIECE_BLACK_KING) &&
1314 (abs((int)to_col - (int)col) == 2);
1315
1316 // Detect en passant
1317 suggestions[count].is_en_passant =
1318 game_is_en_passant_possible(&temp_move);
1319
1320 // Update capture flag for en passant
1321 if (suggestions[count].is_en_passant) {
1322 suggestions[count].is_capture = true;
1323 }
1324
1325 suggestions[count].score = 0; // TODO: Implement move scoring
1326
1327 count++;
1328 }
1329 }
1330 }
1331 break;
1332 }
1333
1334 return count;
1335}
