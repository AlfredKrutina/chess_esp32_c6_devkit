52esp_err_t web_server_resign_dispatch_body(const char *json,
53 const char **out_error) {
54 uint8_t side = GAME_RESIGN_SIDE_TO_MOVE;
55
56 if (web_is_locked()) {
57 return resign_refuse("web_locked", out_error);
58 }
59
60 if (json != NULL && json[0] != '\0') {
61 cJSON *root = cJSON_Parse(json);
62 if (root == NULL) {
63 if (out_error != NULL) {
64 *out_error = "invalid_json";
65 }
66 ESP_LOGW(TAG, "resign: invalid JSON body");
67 return ESP_ERR_INVALID_ARG;
68 }
69 cJSON *side_j = cJSON_GetObjectItemCaseSensitive(root, "side");
70 if (cJSON_IsString(side_j) && side_j->valuestring != NULL) {
71 if (strcasecmp(side_j->valuestring, "white") == 0) {
72 side = GAME_RESIGN_SIDE_WHITE;
73 } else if (strcasecmp(side_j->valuestring, "black") == 0) {
74 side = GAME_RESIGN_SIDE_BLACK;
75 } else {
76 cJSON_Delete(root);
77 if (out_error != NULL) {
78 *out_error = "invalid_side";
79 }
80 ESP_LOGW(TAG, "resign: invalid side");
81 return ESP_ERR_INVALID_ARG;
82 }
83 }
84 cJSON_Delete(root);
85 }
86
87 game_resign_result_t check = game_resign_check(side);
88 ESP_LOGD(TAG, "[STAGING] resign request: side=%u state=%d on_move=%d check=%d",
89 (unsigned)side, (int)game_get_state(),
90 (int)game_get_current_player(), (int)check);
91 if (check != GAME_RESIGN_OK) {
92 return resign_refuse(resign_error_code(check), out_error);
93 }
94
95 if (game_command_queue == NULL) {
96 if (out_error != NULL) {
97 *out_error = "queue_unavailable";
98 }
99 return ESP_FAIL;
100 }
101
102 chess_move_command_t cmd = {0};
103 cmd.type = GAME_CMD_RESIGN;
104 cmd.promotion_choice = side;
105 cmd.response_queue = NULL;
106 if (xQueueSend(game_command_queue, &cmd, pdMS_TO_TICKS(100)) != pdTRUE) {
107 ESP_LOGE(TAG, "resign: failed to send to game queue");
108 if (out_error != NULL) {
109 *out_error = "queue_failed";
110 }
111 return ESP_FAIL;
112 }
113
114 ESP_LOGI(TAG, "resign: queued (side=%u)", (unsigned)side);
115 return ESP_OK;
116}
