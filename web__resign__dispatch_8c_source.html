53esp_err_t web_server_resign_dispatch_body(const char *json,
54 const char **out_error) {
55 uint8_t side = GAME_RESIGN_SIDE_TO_MOVE;
56
57 if (web_is_locked()) {
58 return resign_refuse("web_locked", out_error);
59 }
60
61 if (json != NULL && json[0] != '\0') {
62 cJSON *root = cJSON_Parse(json);
63 if (root == NULL) {
64 if (out_error != NULL) {
65 *out_error = "invalid_json";
66 }
67 ESP_LOGW(TAG, "resign: invalid JSON body");
68 return ESP_ERR_INVALID_ARG;
69 }
70 cJSON *side_j = cJSON_GetObjectItemCaseSensitive(root, "side");
71 if (cJSON_IsString(side_j) && side_j->valuestring != NULL) {
72 if (strcasecmp(side_j->valuestring, "white") == 0) {
73 side = GAME_RESIGN_SIDE_WHITE;
74 } else if (strcasecmp(side_j->valuestring, "black") == 0) {
75 side = GAME_RESIGN_SIDE_BLACK;
76 } else {
77 cJSON_Delete(root);
78 if (out_error != NULL) {
79 *out_error = "invalid_side";
80 }
81 ESP_LOGW(TAG, "resign: invalid side");
82 return ESP_ERR_INVALID_ARG;
83 }
84 }
85 cJSON_Delete(root);
86 }
87
88 game_resign_result_t check = game_resign_check(side);
89 ESP_LOGD(TAG, "[STAGING] resign request: side=%u state=%d on_move=%d check=%d",
90 (unsigned)side, (int)game_get_state(),
91 (int)game_get_current_player(), (int)check);
92 if (check != GAME_RESIGN_OK) {
93 return resign_refuse(resign_error_code(check), out_error);
94 }
95
96 if (game_command_queue == NULL) {
97 if (out_error != NULL) {
98 *out_error = "queue_unavailable";
99 }
100 return ESP_FAIL;
101 }
102
103 game_cmd_t cmd = {0};
104 cmd.kind = GCMD_RESIGN;
105 cmd.u.resign.side = side;
106 if (!game_submit(&cmd)) {
107 ESP_LOGE(TAG, "resign: failed to send to game queue");
108 if (out_error != NULL) {
109 *out_error = "busy";
110 }
111 return ESP_ERR_TIMEOUT;
112 }
113
114 ESP_LOGI(TAG, "resign: queued (side=%u)", (unsigned)side);
115 return ESP_OK;
116}
