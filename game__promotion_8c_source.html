245void game_update_promotion_anchor_led(void) {
246 if (!promotion_state.pending) {
247 return;
248 }
249
250 uint8_t row = promotion_state.square_row;
251 uint8_t col = promotion_state.square_col;
252 uint8_t led_index = chess_pos_to_led_index(row, col);
253
254 // Pulse between purple (pending) and gold (choice/upgrade) for clarity
255 uint32_t now_ms = (uint32_t)(esp_timer_get_time() / 1000ULL);
256 float t = (float)(now_ms % 1000U) / 1000.0f; // 0..1
257 float phase = t * 2.0f * 3.14159f;
258 float mix = 0.5f + 0.5f * sinf(phase); // 0..1
259
260 // Purple base
261 float pr = 128.0f, pg = 0.0f, pb = 128.0f;
262 // Gold target
263 float gr = 255.0f, gg = 215.0f, gb = 0.0f;
264
265 uint8_t r = (uint8_t)(pr * (1.0f - mix) + gr * mix);
266 uint8_t g = (uint8_t)(pg * (1.0f - mix) + gg * mix);
267 uint8_t b = (uint8_t)(pb * (1.0f - mix) + gb * mix);
268
269 led_set_pixel_safe(led_index, r, g, b);
270}

439void game_process_promotion_button(uint8_t button_id) {
440 // #2 & #3: Atomic check with mutex
441 // Use a recursive mutex function (promotion_mutex is a recursive mutex)
442 if (xSemaphoreTakeRecursive(promotion_mutex, pdMS_TO_TICKS(100)) != pdTRUE) {
443 ESP_LOGE(TAG, "❌ Failed to acquire promotion mutex");
444 return;
445 }
446
447 // Check if promotion is pending
448 if (!promotion_state.pending) {
449 xSemaphoreGiveRecursive(promotion_mutex);
450 ESP_LOGW(TAG, "⚠️ Promotion already processed");
451 return;
452 }
453
454 // Map physical button ID (0-3) to promotion choice
455 // Physical buttons are shared between players:
456 // - Button 0: QUEEN (both white and black)
457 // - Button 1: ROOK (both white and black)
458 // - Button 2: BISHOP (both white and black)
459 // - Button 3: KNIGHT (both white and black)
460 promotion_choice_t choice;
461 const char *piece_name;
462
463 if (button_id <= 3) {
464 // Valid physical button (0-3)
465 choice = (promotion_choice_t)button_id;
466 } else {
467 xSemaphoreGiveRecursive(promotion_mutex); // Release mutex on invalid button
468 ESP_LOGW(TAG, "⚠️ Invalid promotion button ID: %d (expected 0-3)",
469 button_id);
470 return;
471 }
472
473 ESP_LOGI(TAG, "👑 Physical promotion button pressed: button=%d, choice=%d",
474 button_id, choice);
475
476 // Get piece name
477 switch (choice) {
478 case PROMOTION_QUEEN:
479 piece_name = "Queen";
480 break;
481 case PROMOTION_ROOK:
482 piece_name = "Rook";
483 break;
484 case PROMOTION_BISHOP:
485 piece_name = "Bishop";
486 break;
487 case PROMOTION_KNIGHT:
488 piece_name = "Knight";
489 break;
490 default:
491 piece_name = "Unknown";
492 break;
493 }
494
495 ESP_LOGI(TAG, "👑 Processing promotion button %d: %s", button_id, piece_name);
496
497 // Execute promotion
498 if (game_execute_promotion(choice)) {
499 promotion_state.pending = false;
500
501 printf("\r\n");
502 printf("═══════════════════════════════════════════════════════════════\r"
503 "\n");
504 printf("👑 PAWN PROMOTION SUCCESSFUL!\r\n");
505 printf(" • Square: %c%d\r\n", 'a' + promotion_state.square_col,
506 promotion_state.square_row + 1);
507 printf(" • Promoted to: %s\r\n", piece_name);
508 printf(" • Player: %s\r\n",
509 promotion_state.player == PLAYER_WHITE ? "White" : "Black");
510 printf("═══════════════════════════════════════════════════════════════\r"
511 "\n");
512 printf("\r\n");
513
514 game_promotion_finish_turn();
515 ESP_LOGI(TAG, "✅ Promotion completed successfully");
516 } else {
517 ESP_LOGE(TAG, "❌ Promotion execution failed");
518 }
519 // Release mutex at the end of the function to ensure it's always released
520 xSemaphoreGiveRecursive(promotion_mutex);
521}
