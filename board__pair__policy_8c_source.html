40int board_pair_book_put(board_pair_book_t *book, uint8_t kind,
41 uint8_t addr_type, const uint8_t addr[6],
42 const uint8_t token[BOARD_PAIR_TOKEN_LEN]) {
43 if (book == NULL || token == NULL || token_empty(token)) {
44 return -1;
45 }
46 if (kind != BOARD_PAIR_KIND_BLE && kind != BOARD_PAIR_KIND_WIFI) {
47 return -1;
48 }
49
50 int dest = -1;
51 if (kind == BOARD_PAIR_KIND_BLE && addr != NULL) {
52 for (int i = 0; i < BOARD_PAIR_SLOT_COUNT; i++) {
53 if (ble_addr_same(&book->slots[i], addr_type, addr)) {
54 dest = i;
55 break;
56 }
57 }
58 }
59 if (dest < 0) {
60 for (int i = 0; i < BOARD_PAIR_SLOT_COUNT; i++) {
61 if (book->slots[i].kind == BOARD_PAIR_KIND_NONE) {
62 dest = i;
63 break;
64 }
65 }
66 }
67 if (dest < 0) {
68 uint32_t oldest = UINT32_MAX;
69 for (int i = 0; i < BOARD_PAIR_SLOT_COUNT; i++) {
70 if (book->slots[i].seq < oldest) {
71 oldest = book->slots[i].seq;
72 dest = i;
73 }
74 }
75 }
76 if (dest < 0) {
77 return -1;
78 }
79
80 board_pair_slot_t *slot = &book->slots[dest];
81 memset(slot, 0, sizeof(*slot));
82 slot->kind = kind;
83 slot->addr_type = addr_type;
84 if (addr != NULL) {
85 memcpy(slot->addr, addr, 6);
86 }
87 memcpy(slot->token, token, BOARD_PAIR_TOKEN_LEN);
88 book->next_seq++;
89 slot->seq = book->next_seq;
90 if (book->wifi_slot == dest) {
91 book->wifi_slot = -1;
92 book->wifi_fd = -1;
93 }
94 return dest;
95}

97int board_pair_book_find_token(const board_pair_book_t *book,
98 const uint8_t token[BOARD_PAIR_TOKEN_LEN]) {
99 if (book == NULL || token == NULL || token_empty(token)) {
100 return -1;

121int board_pair_book_command(board_pair_book_t *book,
122 const uint8_t token[BOARD_PAIR_TOKEN_LEN],
123 bool claim) {
124 if (book == NULL) {
125 return BOARD_PAIR_CMD_NEED_TOKEN;
126 }
127 int slot = board_pair_book_find_token(book, token);
128 if (slot < 0) {
129 return BOARD_PAIR_CMD_NEED_TOKEN;
130 }
131 if (book->ble_up) {
132 if (!slot_is_ble_commander(book, slot)) {
133 return BOARD_PAIR_CMD_BUSY;
134 }
135 if (claim) {
136 book->wifi_slot = slot;
137 }
138 return BOARD_PAIR_CMD_ALLOW;
139 }
140 if (book->wifi_slot < 0) {
141 if (claim) {
142 book->wifi_slot = slot;
143 }
144 return BOARD_PAIR_CMD_ALLOW;
145 }
146 if (book->wifi_slot == slot) {
147 return BOARD_PAIR_CMD_ALLOW;
148 }
149 return BOARD_PAIR_CMD_BUSY;
150}
