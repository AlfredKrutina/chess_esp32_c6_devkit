32board_machine_out_t game_board_on_drop_squares(uint8_t from_row, uint8_t from_col,
33 uint8_t to_row, uint8_t to_col,
34 bool has_from) {
35 position_t *scratch_pos = game_logic_scratch_position();
36 game_logic_copy_globals(scratch_pos);
37 uint8_t from = has_from ? (uint8_t)(from_row * 8u + from_col)
38 : game_board_machine()->lifted_sq;
39 uint8_t to = (uint8_t)(to_row * 8u + to_col);
40 game_mode_call_drop(scratch_pos, from, to, has_from);
41 board_machine_out_t out = board_machine_drop(game_board_machine(), scratch_pos,
42 from, to, has_from);
43 if (game_mode_take_drop_reject() &&
44 (out.kind == BM_OUT_MOVE || out.kind == BM_OUT_NEED_PROMO)) {
45 board_machine_t *machine = game_board_machine();
46 machine->last_reject = BM_REJECT_ILLEGAL;
47 machine->state = BM_STATE_LIFTED;
48 machine->lifted_sq = from;
49 out.kind = BM_OUT_REJECT;
50 out.reason = BM_REJECT_ILLEGAL;
51 }
52 return out;
53}

55game_board_move_class_t game_board_classify_move(const chess_move_t *move) {
56 if (!move) {
57 return GAME_BOARD_MOVE_QUIET;
58 }
59 piece_t sp = move->piece;
60 if ((sp == PIECE_WHITE_KING || sp == PIECE_BLACK_KING) &&
61 ((int)move->to_col - (int)move->from_col == 2 ||
62 (int)move->to_col - (int)move->from_col == -2)) {
63 return GAME_BOARD_MOVE_CASTLE_KING;
64 }
65 if (castling_state.in_progress &&
66 move->from_row == castling_state.rook_from_row &&
67 move->from_col == castling_state.rook_from_col &&
68 move->to_row == castling_state.rook_to_row &&
69 move->to_col == castling_state.rook_to_col) {
70 return GAME_BOARD_MOVE_CASTLE_ROOK;
71 }
72 if ((sp == PIECE_WHITE_PAWN && move->to_row == 7) ||
73 (sp == PIECE_BLACK_PAWN && move->to_row == 0)) {
74 return GAME_BOARD_MOVE_PROMO_ENTRY;
75 }
76 if (move->captured_piece != PIECE_EMPTY) {
77 return GAME_BOARD_MOVE_CAPTURE;
78 }
79 if (en_passant_available && move->to_row == en_passant_target_row &&
80 move->to_col == en_passant_target_col) {
81 return GAME_BOARD_MOVE_EN_PASSANT;
82 }
83 return GAME_BOARD_MOVE_QUIET;
84}

86static void pos_move_to_chess(const position_move_t *pm, chess_move_t *out) {
87 out->from_row = pm->from / 8u;
88 out->from_col = pm->from % 8u;
89 out->to_row = pm->to / 8u;
90 out->to_col = pm->to % 8u;
91 out->piece = game_board_piece_at(out->from_row, out->from_col);
92 out->captured_piece = game_board_piece_at(out->to_row, out->to_col);
93 out->timestamp = esp_timer_get_time() / 1000;
94}

96bool game_board_apply_machine_move(const board_machine_out_t *out,
97 chess_move_t *move_out) {
98 if (!out || !move_out) {
99 return false;
100 }
101 /* Quiet, capture, castle, and promo entry all go through game_execute_move. */
102 if (out->kind != BM_OUT_NEED_PROMO && out->kind != BM_OUT_MOVE) {
103 return false;
104 }
105 pos_move_to_chess(&out->move, move_out);
106 if (!game_execute_move(move_out)) {
107 return false;
108 }
109 game_mode_call_move_applied(game_logic_position(), out->move);
110 return true;
111}

131bool game_board_apply_promotion_piece(char piece_char) {
132 if (!(*game_state_promotion_state()).pending) {
133 return false;
134 }
135 uint8_t row = (*game_state_promotion_state()).square_row;
136 uint8_t col = (*game_state_promotion_state()).square_col;
137 position_t *logic = game_logic_position();
138 position_promo_t pr = char_to_promo(piece_char);
139 uint8_t p = logic->board[row][col];
140 uint8_t promoted = POS_EMPTY;
141 if (p == POS_WHITE_PAWN) {
142 switch (pr) {
143 case POS_PROMO_R:
144 promoted = POS_WHITE_ROOK;
145 break;
146 case POS_PROMO_B:
147 promoted = POS_WHITE_BISHOP;
148 break;
149 case POS_PROMO_N:
150 promoted = POS_WHITE_KNIGHT;
151 break;
152 default:
153 promoted = POS_WHITE_QUEEN;
154 break;
155 }
156 } else if (p == POS_BLACK_PAWN) {
157 switch (pr) {
158 case POS_PROMO_R:
159 promoted = POS_BLACK_ROOK;
160 break;
161 case POS_PROMO_B:
162 promoted = POS_BLACK_BISHOP;
163 break;
164 case POS_PROMO_N:
165 promoted = POS_BLACK_KNIGHT;
166 break;
167 default:
168 promoted = POS_BLACK_QUEEN;
169 break;
170 }
171 } else {
172 return false;
173 }
174 game_set_piece((int)row, (int)col, (piece_t)promoted);
175 logic->side = (uint8_t)(((*game_state_promotion_state()).player == PLAYER_WHITE)
176 ? PLAYER_BLACK
177 : PLAYER_WHITE);
178 game_session_snap_refresh();
179 game_set_fifty_move_counter(logic->halfmove);
180 moves_without_capture = logic->halfmove;
181 return true;
182}

210bool game_board_apply_uart_castle(bool kingside) {
211 uint8_t king_row = (current_player == PLAYER_WHITE) ? 0u : 7u;
212 uint8_t king_to_col = kingside ? 6u : 2u;
213 position_move_t mv = {.from = (uint8_t)(king_row * 8u + 4u),
214 .to = (uint8_t)(king_row * 8u + king_to_col),
215 .promo = POS_PROMO_NONE};
216 position_t *logic = game_logic_position();
217 game_logic_copy_globals(logic);
218 if (!position_apply(logic, mv)) {
219 return false;
220 }
221 game_session_snap_refresh();
222 game_set_fifty_move_counter(logic->halfmove);
223 moves_without_capture = logic->halfmove;
224 current_player =
225 (current_player == PLAYER_WHITE) ? PLAYER_BLACK : PLAYER_WHITE;
226 return true;
227}
