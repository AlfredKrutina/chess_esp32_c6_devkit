53static void resignation_update_button_leds(uint32_t elapsed_ms) {
54#if CONFIG_CHESS_MATRIX_INPUT_I2C_HALL
55 /* This board has no promotion-button LEDs. The Hall path paints a ring. */
56 (void)elapsed_ms;
57 return;
58#else
59 const uint32_t ORANGE_START_MS = 4000; // Color to orange up to 4 s
60 if (elapsed_ms < ORANGE_START_MS)
61 return;
62
63 const uint8_t WHITE_BASE = 64;
64 const uint8_t BLACK_BASE = 68;
65
66 const uint32_t FADE_START_MS = 5000;
67 const uint32_t FADE_DURATION_MS = 1000;
68 const uint32_t LED_OFFSET_MS = 1000;
69
70 for (int i = 0; i < 4; i++) {
71 uint32_t led_start_time = FADE_START_MS + (i * LED_OFFSET_MS);
72 uint8_t brightness = 255;
73
74 if (elapsed_ms >= led_start_time) {
75 uint32_t led_elapsed = elapsed_ms - led_start_time;
76 if (led_elapsed >= FADE_DURATION_MS) {
77 brightness = 0;
78 } else {
79 brightness = 255 - (led_elapsed * 255 / FADE_DURATION_MS);
80 }
81 }
82
83 uint8_t r = brightness;
84 uint8_t g = brightness / 2;
85 uint8_t b = 0;
86 button_set_led_color(WHITE_BASE + i, r, g, b);
87 button_set_led_color(BLACK_BASE + i, r, g, b);
88 }
89#endif
90}

99static void resignation_end_game(player_t loser) {
100 const char *player_name = (loser == PLAYER_WHITE) ? "White" : "Black";
101 const char *winner_name = (loser == PLAYER_WHITE) ? "Black" : "White";
102
103 ESP_LOGI(TAG, "🏳️ %s resigned! %s wins!", player_name, winner_name);
104
105 // End the game
106 current_game_state = GAME_STATE_FINISHED;
107 (*game_state_game_result()) = GAME_STATE_FINISHED;
108 (*game_state_current_result_type()) =
109 (loser == PLAYER_WHITE) ? RESULT_BLACK_WINS : RESULT_WHITE_WINS;
110 (*game_state_current_endgame_reason()) = ENDGAME_REASON_RESIGNATION;
111 game_active = false;
112
113 // Print report later in game_task (stack-safe)
114 game_update_endgame_statistics((*game_state_current_result_type()));
115 (*game_state_endgame_report_requested()) = true;
116
117 timer_pause();
118
119 // Run the victory animation for the winner
120 player_t winner_player = (loser == PLAYER_WHITE) ? PLAYER_BLACK : PLAYER_WHITE;
121 game_trigger_victory_animation(winner_player);
122
123 if (resignation_state.active) {
124 // Finalized cleanup (no cancel messages, do not restore king)
125 resignation_stop(true);
126 }
127#if !CONFIG_CHESS_MATRIX_INPUT_I2C_HALL
128 else {
129 for (int i = 0; i < 8; i++) {
130 button_set_led_color(64 + i, 0, 0, 0);
131 }
132 }
133#endif
134 game_snapshot_persist_after_valid_move();
135}

176game_resign_result_t game_resign_check(uint8_t side) {
177 if (side > GAME_RESIGN_SIDE_BLACK) {
178 return GAME_RESIGN_ERR_SIDE;
179 }
180 if (game_is_board_setup_tutorial_active() || game_is_puzzle_active() ||
181 game_is_puzzle_setup_active() || game_is_opening_trainer_active() ||
182 game_is_opening_trainer_setup_active()) {
183 return GAME_RESIGN_ERR_MODE;
184 }
185 switch (current_game_state) {
186 case GAME_STATE_ACTIVE:
187 case GAME_STATE_PLAYING:
188 case GAME_STATE_PAUSED:
189 case GAME_STATE_PROMOTION:
190 case GAME_STATE_WAITING_FOR_RETURN:
191 break;
192 default:
193 return GAME_RESIGN_ERR_NOT_RUNNING;
194 }
195 if ((side == GAME_RESIGN_SIDE_WHITE && current_player != PLAYER_WHITE) ||
196 (side == GAME_RESIGN_SIDE_BLACK && current_player != PLAYER_BLACK)) {
197 return GAME_RESIGN_ERR_SIDE;
198 }
199 return GAME_RESIGN_OK;
200}

202bool game_resign_from_app(uint8_t side) {
203 game_resign_result_t check = game_resign_check(side);
204 ESP_LOGD(TAG,
205 "[STAGING] app resign: side=%u on_move=%s state=%d king_lift=%d "
206 "piece_lifted=%d check=%d",
207 (unsigned)side, (current_player == PLAYER_WHITE) ? "White" : "Black",
208 (int)current_game_state, (int)resignation_state.active,
209 (int)piece_lifted, (int)check);
210 if (check != GAME_RESIGN_OK) {
211 ESP_LOGW(TAG, "App resign refused (reason=%d)", (int)check);
212 return false;
213 }
214
215 player_t loser = current_player;
216 if (resignation_state.active && resignation_state.player != loser) {
217 // Not reachable today (only the side on move can lift its king); keep
218 // the board array consistent if it ever happens.
219 resignation_stop(false);
220 }
221 resignation_end_game(loser);
222 return true;
223}

228void resignation_tick(void) {
229 if (!resignation_state.active) {
230 return;
231 }
232
233 uint64_t now_ms = esp_timer_get_time() / 1000;
234 uint32_t elapsed_ms = (uint32_t)(now_ms - resignation_state.start_time_ms);
235
236 // Update button LEDs (fade effect) in safe task context
237 resignation_update_button_leds(elapsed_ms);
238
239 // Finalize after 10 seconds
240 if (elapsed_ms >= 10000) {
241 resignation_finalize_timeout();
242 }

248void resignation_start(player_t player, uint8_t row, uint8_t col) {
249 // Stop an existing timer if it is running (restart = silent cleanup)
250 if (resignation_state.active) {
251 resignation_stop(true);
252 }
253
254 const char *player_name = (player == PLAYER_WHITE) ? "White" : "Black";
255 printf("\r\n⚠️ %s king lifted - hold for 10+ seconds to resign!\r\n",
256 player_name);
257 printf("👑 Resignation timer started for %s player\r\n", player_name);
258 printf("🎨 Button LED animation started (8 buttons, both sides)\r\n\r\n");
259
260 // Initialize state
261 resignation_state.active = true;
262 resignation_state.player = player;
263 resignation_state.king_row = row;
264 resignation_state.king_col = col;
265 resignation_state.start_time_ms = esp_timer_get_time() / 1000;
266 resignation_state.last_countdown_sec = 10;
267
268 ESP_LOGI(TAG, "✅ King lift tracked for resignation at [%d][%d] (legal board unchanged)",
269 row, col);
270
271 // Combination of red (warning) and yellow (source square)
272 // Orange-red mix instead of just red
273 game_led_frame_sq(row, col, 255, 128, 0);
274
275 // Buttons turn orange only after 4s (in resignation_update_button_leds)
276 // – the first 4 seconds remain in the original color
277
278 // PRODUCTION STABILITY:
279 // Do NOT run resignation logic from FreeRTOS timer callbacks ("Tmr Svc").
280 // Handling is done in `resignation_tick()` from the main game_task loop.
281 resignation_state.main_timer = NULL;
282 resignation_state.animation_timer = NULL;
283}

290void resignation_stop(bool finalize) {
291 if (!resignation_state.active)
292 return;
293
294 if (!finalize) {
295 printf("✅ King placed - resignation cancelled!\r\n");
296 printf("🛑 Resignation timer stopped\r\n");
297 printf("🎨 Button LED animation stopped\r\n\r\n");
298 } else {
299 ESP_LOGI(TAG,
300 "🔄 [STAGING] Resignation finalized (cleanup, no cancel msgs)");
301 }
302
303 if (!finalize) {
304 ESP_LOGI(TAG,
305 "✅ Resignation cancelled at [%d][%d] (legal board unchanged)",
306 resignation_state.king_row, resignation_state.king_col);
307
308 game_led_frame_sq(resignation_state.king_row,
309 resignation_state.king_col, 0, 0, 0);
310 }
311
312 // Zastavit a smazat timery
313 if (resignation_state.main_timer) {
314 xTimerStop(resignation_state.main_timer, 0);
315 xTimerDelete(resignation_state.main_timer, 0);
316 resignation_state.main_timer = NULL;
317 }
318
319 if (resignation_state.animation_timer) {
320 xTimerStop(resignation_state.animation_timer, 0);
321 xTimerDelete(resignation_state.animation_timer, 0);
322 resignation_state.animation_timer = NULL;
323 }
324
325 if (finalize) {
326 // When the resignation is complete (timeout), turn off the button LEDs
327 for (int i = 0; i < 8; i++) {
328 button_set_led_color(64 + i, 0, 0, 0);
329 }
330 } else {
331 // When canceling (king laid back) restore the normal state of the buttons
332 // (green/blue according to graduation, LED 72 green)
333 game_check_promotion_needed();
334 }
335
336 // Status reset
337 resignation_state.active = false;
338}
