65static bool game_board_starting_occupancy_matches(const uint8_t state[64]) {
66 for (int row = 2; row < 6; row++) {
67 for (int col = 0; col < 8; col++) {
68 if (state[row * 8 + col] != 0) {
69 return false;
70 }
71 }
72 }
73 for (int row = 0; row < 2; row++) {
74 for (int col = 0; col < 8; col++) {
75 if (state[row * 8 + col] == 0) {
76 return false;
77 }
78 }
79 }
80 for (int row = 6; row < 8; row++) {
81 for (int col = 0; col < 8; col++) {
82 if (state[row * 8 + col] == 0) {
83 return false;
84 }
85 }
86 }
87 return true;
88}

90static void game_log_first_startpos_mismatch(const uint8_t state[64]) {
91 char sq[4];
92 for (int row = 2; row < 6; row++) {
93 for (int col = 0; col < 8; col++) {
94 int idx = row * 8 + col;
95 if (state[idx] != 0) {
96 squares_square_to_notation((uint8_t)idx, sq);
97 ESP_LOGW(TAG,
98 "starting position: rank 3–6 square %s must be empty "

101 return;
102 }
103 }
104 }
105 for (int row = 0; row < 2; row++) {
106 for (int col = 0; col < 8; col++) {
107 int idx = row * 8 + col;
108 if (state[idx] == 0) {
109 squares_square_to_notation((uint8_t)idx, sq);

116 }
117 }
118 for (int row = 6; row < 8; row++) {
119 for (int col = 0; col < 8; col++) {
120 int idx = row * 8 + col;
121 if (state[idx] == 0) {
122 squares_square_to_notation((uint8_t)idx, sq);
123 ESP_LOGW(TAG,
124 "starting position: rank 7–8 square %s must be occupied "

139bool game_is_board_in_starting_position(void) {
140#if CONFIG_CHESS_MATRIX_INPUT_I2C_HALL
141 if (!hall_i2c_squares_readings_usable()) {
142 return false;
143 }
144#endif
145 uint8_t votes[64] = {0};
146
147 for (int s = 0; s < GAME_START_POS_MATRIX_SAMPLES; s++) {
148 uint8_t snap[64];
149 squares_get_state(snap);
150 for (int i = 0; i < 64; i++) {
151 votes[i] += snap[i];
152 }
153 if (s + 1 < GAME_START_POS_MATRIX_SAMPLES) {
154 game_led_flush(); vTaskDelay(pdMS_TO_TICKS(12));
155 }
156 }
157
158 uint8_t stable[64];
159 for (int i = 0; i < 64; i++) {
160 stable[i] = (votes[i] >= GAME_START_POS_MAJORITY_VOTES) ? 1 : 0;
161 }
162
163 if (!game_board_starting_occupancy_matches(stable)) {
164 game_log_first_startpos_mismatch(stable);
165 return false;
166 }
167
168#if CONFIG_CHESS_MATRIX_INPUT_I2C_HALL
169 if (!game_hall_logic_matches_stable_sensors(stable)) {
170 ESP_LOGW(TAG,
171 "starting position: Hall type/color mismatch vs logic board");
172 return false;
173 }
174#endif
175
176 return true;
177}

185void game_show_missing_pieces_led(void) {
186#if CONFIG_CHESS_MATRIX_INPUT_I2C_HALL

188 return;
189#else
190 uint8_t squares_state[64] = {0};

207
208 // Definition of expected starting position (row-major order)
209 // Rows 0-1 and 6-7 should be occupied, rows 2-5 empty
210 for (int row = 0; row < 2; row++) {
211 for (int col = 0; col < 8; col++) {
212 int idx = row * 8 + col;
213 if (squares_state[idx] == 0) {
214 // A figure is missing on the white baseline
215 game_led_frame_sq(row, col, 255, 0, 0); // Red
216 }
217 }
218 }
219 for (int row = 6; row < 8; row++) {
220 for (int col = 0; col < 8; col++) {
221 int idx = row * 8 + col;
222 if (squares_state[idx] == 0) {
223 // A figure is missing on the black baseline
224 game_led_frame_sq(row, col, 255, 0, 0); // Red
225 }
226 }
227 }
228 for (int row = 2; row < 6; row++) {
229 for (int col = 0; col < 8; col++) {
230 int idx = row * 8 + col;
231 if (squares_state[idx] != 0) {
232 // A random figure in the middle of the board - also in red
233 game_led_frame_sq(row, col, 255, 0, 0); // Red
234 }
235 }
236 }
237
238 ESP_LOGI(TAG, "🔴 LED: Showing missing pieces in red");
239#endif

650 ESP_LOGI(TAG, "✅ Auto new game detection flags reset for new game");

657#else

659#endif

661 game_bump_revision_and_notify();
662 game_snapshot_persist_after_valid_move();
663}
