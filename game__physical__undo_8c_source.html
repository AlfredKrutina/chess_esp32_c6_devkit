177static bool game_physical_undo_lift_matches(const chess_move_t *m,
178 move_type_t kind,
179 const move_undo_meta_t *meta,
180 uint8_t row, uint8_t col,
181 piece_t piece) {
182 piece_t on_board = board[row][col];
183 if (on_board == PIECE_EMPTY && piece == PIECE_EMPTY) {
184 return false;
185 }
186
187 if (row == m->to_row && col == m->to_col) {
188 if (kind == MOVE_TYPE_PROMOTION && meta->promoted_to != PIECE_EMPTY) {
189 return piece == meta->promoted_to || piece == m->piece;
190 }
191 return piece == on_board && piece != PIECE_EMPTY;
192 }
193
194 if ((kind == MOVE_TYPE_CASTLE_KING || kind == MOVE_TYPE_CASTLE_QUEEN) &&
195 row == m->from_row && col == meta->rook_castle_col) {
196 piece_t rook = game_physical_undo_rook_for_king(m->piece);
197 return piece == rook && board[row][col] == rook;
198 }
199 return false;
200}

202static void game_physical_undo_build_target_board(piece_t target[8][8],
203 const chess_move_t *m,
204 move_type_t kind,
205 const move_undo_meta_t *meta) {
206 memcpy(target, board, sizeof(piece_t) * 64);
207
208 bool mover_white =
209 (m->piece >= PIECE_WHITE_PAWN && m->piece <= PIECE_WHITE_KING);
210 uint8_t fr = m->from_row;
211 uint8_t fc = m->from_col;
212 uint8_t tr = m->to_row;
213 uint8_t tc = m->to_col;
214
215 switch (kind) {
216 case MOVE_TYPE_EN_PASSANT:
217 target[tr][tc] = PIECE_EMPTY;
218 target[fr][fc] = m->piece;
219 target[meta->capture_return_row][meta->capture_return_col] =
220 meta->capture_return_piece;
221 break;
222 case MOVE_TYPE_PROMOTION:
223 target[tr][tc] = PIECE_EMPTY;
224 target[fr][fc] = m->piece;
225 break;
226 default:
227 target[tr][tc] = m->captured_piece;
228 target[fr][fc] = m->piece;
229 break;
230 }
231
232 if (kind == MOVE_TYPE_CASTLE_KING || kind == MOVE_TYPE_CASTLE_QUEEN) {
233 piece_t rook = game_physical_undo_rook_for_king(m->piece);
234 target[fr][meta->rook_castle_col] = PIECE_EMPTY;
235 target[fr][meta->rook_home_col] = rook;
236 }
237
238 (void)mover_white;
239}

357bool game_undo_rewind_last_move(void) {
358 if (history_index == 0) {
359 ESP_LOGW(TAG, "Undo: no moves in history");
360 return false;
361 }
362
363 uint32_t idx = history_index - 1;
364 chess_move_t m = move_history[idx];
365 move_type_t kind = move_history_kind[idx];
366 move_undo_meta_t meta = move_history_meta[idx];
367
368 bool mover_white =
369 (m.piece >= PIECE_WHITE_PAWN && m.piece <= PIECE_WHITE_KING);
370 uint8_t fr = m.from_row;
371 uint8_t fc = m.from_col;
372 uint8_t tr = m.to_row;
373 uint8_t tc = m.to_col;
374
375 switch (kind) {
376 case MOVE_TYPE_EN_PASSANT:
377 board[tr][tc] = PIECE_EMPTY;
378 board[fr][fc] = m.piece;
379 board[meta.capture_return_row][meta.capture_return_col] =
380 meta.capture_return_piece;
381 break;
382 case MOVE_TYPE_PROMOTION: {
383 if (promotion_state.pending) {
384 if (xSemaphoreTakeRecursive(promotion_mutex, pdMS_TO_TICKS(100)) ==
385 pdTRUE) {
386 promotion_state.pending = false;
387 promotion_state.square_row = 0;
388 promotion_state.square_col = 0;
389 xSemaphoreGiveRecursive(promotion_mutex);
390 }
391 }
392 board[tr][tc] = PIECE_EMPTY;
393 board[fr][fc] = m.piece;
394 break;
395 }
396 default:
397 board[tr][tc] = m.captured_piece;
398 board[fr][fc] = m.piece;
399 break;
400 }
401
402 if (kind == MOVE_TYPE_CASTLE_KING || kind == MOVE_TYPE_CASTLE_QUEEN) {
403 piece_t rook_piece = game_physical_undo_rook_for_king(m.piece);
404 if (board[fr][meta.rook_castle_col] == rook_piece) {
405 board[fr][meta.rook_castle_col] = PIECE_EMPTY;
406 board[fr][meta.rook_home_col] = rook_piece;
407 }
408 }
409
410 if (kind == MOVE_TYPE_CAPTURE) {
411 undo_pop_last_capture_list(m.captured_piece, mover_white);
412 }
413
414 castling_state.in_progress = false;
415
416 white_king_moved = meta.white_king_moved;
417 white_rook_a_moved = meta.white_rook_a_moved;
418 white_rook_h_moved = meta.white_rook_h_moved;
419 black_king_moved = meta.black_king_moved;
420 black_rook_a_moved = meta.black_rook_a_moved;
421 black_rook_h_moved = meta.black_rook_h_moved;
422 en_passant_available = meta.en_passant_available;
423 en_passant_target_row = meta.en_passant_target_row;
424 en_passant_target_col = meta.en_passant_target_col;
425 en_passant_victim_row = meta.en_passant_victim_row;
426 en_passant_victim_col = meta.en_passant_victim_col;
427
428 history_index--;
429 if (move_count > 0) {
430 move_count--;
431 }
432 if (advantage_history_count > 0) {
433 advantage_history_count--;
434 }
435
436 if (mover_white) {
437 if (white_moves_count > 0) {
438 white_moves_count--;
439 }
440 } else {
441 if (black_moves_count > 0) {
442 black_moves_count--;
443 }
444 }
445
446 current_player = meta.side_before;
447
448 if (history_index > 0) {
449 chess_move_t *p = &move_history[history_index - 1];
450 last_move_from_row = p->from_row;
451 last_move_from_col = p->from_col;
452 last_move_to_row = p->to_row;
453 last_move_to_col = p->to_col;
454 has_last_move = true;
455 } else {
456 has_last_move = false;
457 }
458
459 if (current_game_state == GAME_STATE_FINISHED) {
460 current_game_state = GAME_STATE_ACTIVE;
461 game_active = true;
462 }
463
464 piece_lifted = false;
465 lifted_piece_row = 0;
466 lifted_piece_col = 0;
467 lifted_piece = PIECE_EMPTY;
468
469 game_end_timer_move();
470 game_start_timer_move(current_player == PLAYER_WHITE);
471
472 game_squares_guard_check_resync_after_restore();
473 if (!game_is_squares_guard_active()) {
474 led_clear_board_only();
475 chess_policy_highlight_movable_if_enabled();
476 }
477
478 game_snapshot_persist_after_valid_move();
479 game_bump_revision_and_notify();
480
481 ESP_LOGI(TAG, "Undo applied (kind=%d), side to move: %s", (int)kind,
482 current_player == PLAYER_WHITE ? "White" : "Black");
483 return true;
484}

492bool game_physical_undo_handle_pickup(const chess_move_command_t *cmd,
493 uint8_t from_row, uint8_t from_col,
494 piece_t piece) {
495 (void)cmd;
496
497 if (s_undo.phase != PHYS_UNDO_IDLE && !game_physical_undo_enabled()) {
498 game_physical_undo_reset();
499 return false;
500 }
501
502 if (s_undo.phase == PHYS_UNDO_PENDING) {
503 game_physical_undo_try_complete_engine();
504 if (s_undo.phase == PHYS_UNDO_PENDING) {
505 game_physical_undo_show_pending_leds();
506 }
507 return true;
508 }
509
510 if (!game_physical_undo_enabled() || game_physical_undo_mode_blocks_offer()) {
511 return false;
512 }
513
514 chess_move_t m;
515 move_type_t kind;
516 move_undo_meta_t meta;
517 if (!game_physical_undo_load_last(&m, &kind, &meta)) {
518 return false;
519 }
520
521 if (!game_physical_undo_lift_matches(&m, kind, &meta, from_row, from_col,
522 piece)) {
523 return false;
524 }
525
526 s_undo.phase = PHYS_UNDO_ARMED;
527#if CONFIG_CHESS_MATRIX_INPUT_I2C_HALL
528 game_hall_restore_guide_stop_leds();
529#endif
530 s_undo.move = m;
531 s_undo.kind = kind;
532 s_undo.meta = meta;
533 s_undo.lift_row = from_row;
534 s_undo.lift_col = from_col;
535 s_undo.lift_piece = piece;
536 s_undo.undo_castle_rook_lift =
537 (from_row == m.from_row && from_col == meta.rook_castle_col);
538
539 if (s_undo.undo_castle_rook_lift) {
540 s_undo.blue_row = m.from_row;
541 s_undo.blue_col = meta.rook_home_col;
542 } else {
543 s_undo.blue_row = m.from_row;
544 s_undo.blue_col = m.from_col;
545 }
546
547 piece_lifted = true;
548 lifted_piece_row = from_row;
549 lifted_piece_col = from_col;
550 lifted_piece = piece;
551
552 game_physical_undo_show_armed_leds();
553 ESP_LOGI(TAG, "Takeback armed: lift %c%d blue return %c%d",
554 'a' + from_col, from_row + 1, 'a' + s_undo.blue_col,
555 s_undo.blue_row + 1);
556 return true;
557}

559bool game_physical_undo_handle_drop(const chess_move_command_t *cmd,
560 uint8_t to_row, uint8_t to_col) {
561 (void)cmd;
562
563 if (s_undo.phase != PHYS_UNDO_IDLE && !game_physical_undo_enabled()) {
564 game_physical_undo_reset();
565 return false;
566 }
567
568 if (s_undo.phase == PHYS_UNDO_PENDING) {
569 game_physical_undo_try_complete_engine();
570 return true;
571 }
572
573 if (s_undo.phase != PHYS_UNDO_ARMED || !piece_lifted) {
574 return false;
575 }
576
577 if (to_row == s_undo.lift_row && to_col == s_undo.lift_col) {
578 ESP_LOGI(TAG, "Takeback cancelled — piece returned to lift square");
579 game_physical_undo_reset();
580 piece_lifted = false;
581 lifted_piece = PIECE_EMPTY;
582 lifted_piece_row = 0;
583 lifted_piece_col = 0;
584 led_clear_board_only();
585 chess_policy_highlight_movable_if_enabled();
586#if CONFIG_CHESS_MATRIX_INPUT_I2C_HALL
587 game_hall_restore_guide_request_sync();
588#endif
589 return true;
590 }
591
592 if (to_row == s_undo.blue_row && to_col == s_undo.blue_col) {
593 s_undo.phase = PHYS_UNDO_PENDING;
594#if CONFIG_CHESS_MATRIX_INPUT_I2C_HALL
595 game_hall_restore_guide_stop_leds();
596#endif
597 game_physical_undo_build_target_board(s_undo.target_board, &s_undo.move,
598 s_undo.kind, &s_undo.meta);
599 piece_lifted = false;
600 lifted_piece = PIECE_EMPTY;
601 lifted_piece_row = 0;
602 lifted_piece_col = 0;
603 s_undo.pending_blink_phase = 0;
604
605 if (s_undo.kind == MOVE_TYPE_PROMOTION && promotion_state.pending) {
606 if (xSemaphoreTakeRecursive(promotion_mutex, pdMS_TO_TICKS(100)) ==
607 pdTRUE) {
608 promotion_state.pending = false;
609 promotion_state.square_row = 0;
610 promotion_state.square_col = 0;
611 xSemaphoreGiveRecursive(promotion_mutex);
612 }
613 }
614
615 ESP_LOGI(TAG, "Takeback pending — restore all pieces before engine rewind");
616 game_physical_undo_show_pending_leds();
617 game_physical_undo_try_complete_engine();
618 return true;
619 }
620
621 ESP_LOGI(TAG, "Takeback disarmed — drop on non-undo square (normal move)");
622 game_physical_undo_reset();
623 return false;
624}
