102 meta->black_king_moved = black_king_moved;
103 meta->black_rook_a_moved = black_rook_a_moved;

111 meta->rook_home_col = 0;

113 meta->capture_return_piece = PIECE_EMPTY;
114 meta->capture_return_row = move->to_row;

129 } else if (kind == MOVE_TYPE_CAPTURE) {
130 meta->capture_return_piece = move->captured_piece;
131 } else if (kind == MOVE_TYPE_PROMOTION) {
132 meta->capture_return_row = move->to_row;
133 meta->capture_return_col = move->to_col;
134 meta->capture_return_piece = PIECE_EMPTY;

187/* position_undo already stepped the king and rook back. Give that wing back. */
188static void undo_restore_castle_wing(position_t *logic, const chess_move_t *m,
189 move_type_t kind) {
190 int rook_col;
191 uint8_t king;
192 uint8_t rook;
193 uint8_t bit;
194 if (logic == NULL || m == NULL || m->from_row > 7) {
195 return;
196 }
197 if (kind != MOVE_TYPE_CASTLE_KING && kind != MOVE_TYPE_CASTLE_QUEEN) {
198 return;
199 }
200 rook_col = (kind == MOVE_TYPE_CASTLE_KING) ? 7 : 0;
201 if (m->from_row == 0) {
202 king = POS_WHITE_KING;
203 rook = POS_WHITE_ROOK;
204 bit = (kind == MOVE_TYPE_CASTLE_KING) ? POS_CASTLE_WK : POS_CASTLE_WQ;
205 } else if (m->from_row == 7) {
206 king = POS_BLACK_KING;
207 rook = POS_BLACK_ROOK;
208 bit = (kind == MOVE_TYPE_CASTLE_KING) ? POS_CASTLE_BK : POS_CASTLE_BQ;
209 } else {
210 return;
211 }
212 if (logic->board[m->from_row][4] == king &&
213 logic->board[m->from_row][rook_col] == rook) {
214 logic->castling |= bit;
215 }
216}

227static bool game_physical_undo_lift_matches(const chess_move_t *m,
228 move_type_t kind,
229 const move_undo_meta_t *meta,
230 uint8_t row, uint8_t col,
231 piece_t piece) {
232 piece_t on_board = game_get_piece(row, col);
233 if (on_board == PIECE_EMPTY && piece == PIECE_EMPTY) {
234 return false;
235 }
236
237 if (row == m->to_row && col == m->to_col) {
238 if (kind == MOVE_TYPE_PROMOTION && meta->promoted_to != PIECE_EMPTY) {
239 return piece == meta->promoted_to || piece == m->piece;
240 }
241 return piece == on_board && piece != PIECE_EMPTY;
242 }
243
244 if ((kind == MOVE_TYPE_CASTLE_KING || kind == MOVE_TYPE_CASTLE_QUEEN) &&
245 row == m->from_row && col == meta->rook_castle_col) {
246 piece_t rook = game_physical_undo_rook_for_king(m->piece);
247 return piece == rook && game_get_piece(row, col) == rook;
248 }
249 return false;
250}

252static void game_physical_undo_build_target_board(piece_t target[8][8],
253 const chess_move_t *m,
254 move_type_t kind,
255 const move_undo_meta_t *meta) {
256 const position_t *pos = game_logic_position();
257 for (int row = 0; row < 8; row++) {
258 for (int col = 0; col < 8; col++) {
259 target[row][col] =
260 (pos != NULL) ? (piece_t)pos->board[row][col] : PIECE_EMPTY;
261 }
262 }
263
264 bool mover_white =
265 (m->piece >= PIECE_WHITE_PAWN && m->piece <= PIECE_WHITE_KING);
266 uint8_t fr = m->from_row;
267 uint8_t fc = m->from_col;
268 uint8_t tr = m->to_row;
269 uint8_t tc = m->to_col;
270
271 switch (kind) {
272 case MOVE_TYPE_EN_PASSANT:
273 target[tr][tc] = PIECE_EMPTY;
274 target[fr][fc] = m->piece;
275 target[meta->capture_return_row][meta->capture_return_col] =
276 meta->capture_return_piece;
277 break;
278 case MOVE_TYPE_PROMOTION:
279 target[tr][tc] = PIECE_EMPTY;
280 target[fr][fc] = m->piece;
281 break;
282 default:
283 target[tr][tc] = m->captured_piece;
284 target[fr][fc] = m->piece;
285 break;
286 }
287
288 if (kind == MOVE_TYPE_CASTLE_KING || kind == MOVE_TYPE_CASTLE_QUEEN) {
289 piece_t rook = game_physical_undo_rook_for_king(m->piece);
290 target[fr][meta->rook_castle_col] = PIECE_EMPTY;
291 target[fr][meta->rook_home_col] = rook;
292 }
293
294 (void)mover_white;
295}

302static bool game_physical_undo_target_matches_reed(const piece_t target[8][8],
303 const move_undo_meta_t *meta,

306 squares_get_state(phys);

308 piece_t expect = target[sq / 8][sq % 8];
309 uint8_t want_occ = (expect != PIECE_EMPTY) ? 1 : 0;

314 if (kind == MOVE_TYPE_CAPTURE || kind == MOVE_TYPE_EN_PASSANT) {
315 uint8_t sq = (uint8_t)(meta->capture_return_row * 8 + meta->capture_return_col);
316 if (phys[sq] == 0) {
317 return false;
318 }
319 }
320 (void)meta;
321 return true;

353static void game_physical_undo_show_pending_leds(void) {
354 game_led_clear_board();
355 const chess_move_t *m = &s_undo.move;
356 const move_undo_meta_t *meta = &s_undo.meta;
357
358 if (game_policy_error_recovery_led_valid_blue()) {
359 game_led_frame_sq(m->from_row, m->from_col, 0, 0, 255);
360 }
361
362 if (s_undo.kind == MOVE_TYPE_CASTLE_KING ||
363 s_undo.kind == MOVE_TYPE_CASTLE_QUEEN) {
364 piece_t rook = game_physical_undo_rook_for_king(m->piece);
365 if (game_get_piece(m->from_row, meta->rook_castle_col) == rook) {
366 if (game_policy_error_recovery_led_valid_blue()) {
367 game_led_frame_sq(m->from_row, meta->rook_home_col, 0, 0, 255);
368 }
369 }
370 }
371
372 if (meta->capture_return_piece != PIECE_EMPTY) {
373 uint8_t rr = meta->capture_return_row;
374 uint8_t rc = meta->capture_return_col;
375 piece_t expect = s_undo.target_board[rr][rc];
376 uint8_t phys[64] = {0};
377 squares_get_state(phys);
378 uint8_t sq = (uint8_t)(rr * 8 + rc);
379 bool ok = false;
380#if CONFIG_CHESS_MATRIX_INPUT_I2C_HALL
381 ok = game_hall_square_agrees_with_logic(sq, expect, phys[sq]);
382#else
383 bool need_piece = (expect != PIECE_EMPTY);
384 ok = (!need_piece && phys[sq] == 0) || (need_piece && phys[sq] != 0);
385#endif
386 if (!ok && game_policy_error_recovery_led_red_blink()) {
387 s_undo.pending_blink_phase++;
388 bool on = (s_undo.pending_blink_phase / 8U) % 2U == 0U;
389 if (on) {
390 game_led_frame_sq(rr, rc, 255, 0, 0);
391 }
392 }
393 }
394}

411bool game_undo_rewind_last_move(void) {
412 if (history_index == 0) {
413 ESP_LOGW(TAG, "Undo: no moves in history");
414 return false;
415 }
416
417 uint32_t idx = history_index - 1;
418 chess_move_t m = move_history[idx];
419 move_type_t kind = move_history_kind[idx];
420 move_undo_meta_t meta = move_history_meta[idx];
421 position_t *logic = game_logic_position();
422 bool used_core = false;
423
424 if (logic != NULL && logic->hist_len > 0) {
425 if (!position_undo(logic)) {
426 return false;
427 }
428 /* Castle rights come from the undone position, not a stale king_moved meta. */
429 if (kind == MOVE_TYPE_CASTLE_KING || kind == MOVE_TYPE_CASTLE_QUEEN) {
430 undo_restore_castle_wing(logic, &m, kind);
431 }
432 game_logic_sync_globals_from_position(logic);
433 if (kind != MOVE_TYPE_CASTLE_KING && kind != MOVE_TYPE_CASTLE_QUEEN) {
434 white_king_moved = meta.white_king_moved;
435 black_king_moved = meta.black_king_moved;
436 }
437 current_player = meta.side_before;
438 logic->side = (uint8_t)current_player;
439 game_set_fifty_move_counter(logic->halfmove);
440 moves_without_capture = logic->halfmove;
441 used_core = true;
442 }
443
444 bool mover_white =
445 (m.piece >= PIECE_WHITE_PAWN && m.piece <= PIECE_WHITE_KING);
446 uint8_t fr = m.from_row;
447 uint8_t fc = m.from_col;
448 uint8_t tr = m.to_row;
449 uint8_t tc = m.to_col;
450
451 if (kind == MOVE_TYPE_PROMOTION && (*game_state_promotion_state()).pending) {
452 if (xSemaphoreTakeRecursive((*game_state_promotion_mutex()), pdMS_TO_TICKS(100)) == pdTRUE) {
453 (*game_state_promotion_state()).pending = false;
454 (*game_state_promotion_state()).square_row = 0;
455 (*game_state_promotion_state()).square_col = 0;
456 xSemaphoreGiveRecursive((*game_state_promotion_mutex()));
457 }
458 }
459
460 if (!used_core) {
461 white_king_moved = meta.white_king_moved;
462 white_rook_a_moved = meta.white_rook_a_moved;
463 white_rook_h_moved = meta.white_rook_h_moved;
464 black_king_moved = meta.black_king_moved;
465 black_rook_a_moved = meta.black_rook_a_moved;
466 black_rook_h_moved = meta.black_rook_h_moved;
467 en_passant_available = meta.en_passant_available;
468 en_passant_target_row = meta.en_passant_target_row;
469 en_passant_target_col = meta.en_passant_target_col;
470 en_passant_victim_row = meta.en_passant_victim_row;
471 en_passant_victim_col = meta.en_passant_victim_col;
472 current_player = meta.side_before;
473 }
474
475 if (kind == MOVE_TYPE_CAPTURE) {
476 undo_pop_last_capture_list(m.captured_piece, mover_white);
477 }
478
479 memset(&castling_state, 0, sizeof(castling_state));
480 if (castle_animation_active) {
481 game_stop_repeating_rook_animation();
482 castle_animation_active = false;
483 }
484 undo_sync_piece_moved_flags();
485
486 history_index--;
487 if (move_count > 0) {
488 move_count--;
489 }
490 if ((*game_state_advantage_history_count()) > 0) {
491 (*game_state_advantage_history_count())--;
492 }
493
494 if (mover_white) {
495 if ((*game_state_white_moves_count()) > 0) {
496 (*game_state_white_moves_count())--;
497 }
498 } else {
499 if ((*game_state_black_moves_count()) > 0) {
500 (*game_state_black_moves_count())--;
501 }
502 }
503
504 if (history_index > 0) {
505 chess_move_t *p = &move_history[history_index - 1];
506 last_move_from_row = p->from_row;
507 last_move_from_col = p->from_col;
508 last_move_to_row = p->to_row;
509 last_move_to_col = p->to_col;
510 has_last_move = true;
511 } else {
512 has_last_move = false;
513 }
514
515 if (current_game_state == GAME_STATE_FINISHED) {
516 current_game_state = GAME_STATE_ACTIVE;
517 game_active = true;
518 }
519
520 game_board_set_piece_lifted(false, 0, 0, PIECE_EMPTY);
521
522 game_end_timer_move();
523 game_start_timer_move(current_player == PLAYER_WHITE);
524
525#if CONFIG_CHESS_MATRIX_INPUT_I2C_HALL
526 game_hall_snapshot_last_valid();
527 game_hall_position_arm_idle_handover();
528#endif
529 game_squares_guard_check_resync_after_restore();
530 game_event_post_led_ctrl(GAME_EVT_CTRL_TRAIL_CANCEL, 0, 0, 0, 0);
531 if (!game_is_squares_guard_active()) {
532 /* patch false: full frame, not the castle-partial path. */
533 game_policy_refresh_idle_board_leds();
534 }
535#if CONFIG_CHESS_MATRIX_INPUT_I2C_HALL
536 game_hall_position_clear_castle_wait();
537#endif
538
539 game_session_snap_refresh();
540 game_snapshot_persist_after_valid_move();
541 game_bump_revision_and_notify();
542
543 ESP_LOGI(TAG, "Undo applied (kind=%d), side to move: %s", (int)kind,
544 current_player == PLAYER_WHITE ? "White" : "Black");
545 return true;
546}

559bool game_physical_undo_intercept_pickup(uint8_t row, uint8_t col) {
560 if (s_undo.phase != PHYS_UNDO_ARMED || !s_undo.stray_valid) {
561 return false;
562 }
563 if (row != s_undo.stray_row || col != s_undo.stray_col) {
564 return false;
565 }
566 s_undo.stray_valid = false;
567 piece_lifted = true;

573 return true;

576static void game_physical_undo_arm_from_last(const chess_move_t *m,
577 move_type_t kind,
578 const move_undo_meta_t *meta,
579 uint8_t from_row, uint8_t from_col,
580 piece_t piece) {
581 s_undo.phase = PHYS_UNDO_ARMED;
582 s_undo.stray_valid = false;
583#if CONFIG_CHESS_MATRIX_INPUT_I2C_HALL
584 game_hall_restore_guide_stop_leds();
585#endif
586 s_undo.move = *m;
587 s_undo.kind = kind;
588 s_undo.meta = *meta;
589 s_undo.lift_row = from_row;
590 s_undo.lift_col = from_col;
591 s_undo.lift_piece = piece;
592 s_undo.undo_castle_rook_lift =
593 (from_row == m->from_row && from_col == meta->rook_castle_col);
594
595 if (s_undo.undo_castle_rook_lift) {
596 s_undo.blue_row = m->from_row;
597 s_undo.blue_col = meta->rook_home_col;
598 } else {
599 s_undo.blue_row = m->from_row;
600 s_undo.blue_col = m->from_col;
601 }
602
603 piece_lifted = true;
604 lifted_piece_row = from_row;
605 lifted_piece_col = from_col;
606 lifted_piece = piece;
607}

609static bool game_physical_undo_drop_is_last_reverse(uint8_t to_row,
610 uint8_t to_col,
611 chess_move_t *m,
612 move_type_t *kind,
613 move_undo_meta_t *meta) {
614 if (!piece_lifted || !game_physical_undo_load_last(m, kind, meta)) {
615 return false;
616 }
617 if (!game_physical_undo_lift_matches(m, *kind, meta, lifted_piece_row,
618 lifted_piece_col, lifted_piece)) {
619 return false;
620 }
621 if ((*kind == MOVE_TYPE_CASTLE_KING || *kind == MOVE_TYPE_CASTLE_QUEEN) &&
622 lifted_piece_row == m->from_row &&
623 lifted_piece_col == meta->rook_castle_col) {
624 return to_row == m->from_row && to_col == meta->rook_home_col;
625 }
626 return to_row == m->from_row && to_col == m->from_col;
627}

629bool game_physical_undo_handle_pickup(const game_cmd_t *cmd,
630 uint8_t from_row, uint8_t from_col,
631 piece_t piece) {
632#if CONFIG_CHESS_MATRIX_INPUT_I2C_HALL
633 (void)cmd;
634 (void)from_row;
635 (void)from_col;
636 (void)piece;
637 return false;
638#else
639 (void)cmd;
640
641 if (s_undo.phase != PHYS_UNDO_IDLE && !game_physical_undo_enabled()) {
642 game_physical_undo_reset();
643 return false;
644 }
645
646 if (game_physical_undo_intercept_pickup(from_row, from_col)) {
647 return true;
648 }
649
650 if (s_undo.phase == PHYS_UNDO_PENDING) {
651 game_physical_undo_try_complete_engine();
652 if (s_undo.phase == PHYS_UNDO_PENDING) {
653 game_physical_undo_show_pending_leds();
654 }
655 return true;
656 }
657
658 if (!game_physical_undo_enabled() || game_physical_undo_mode_blocks_offer()) {
659 return false;
660 }
661
662 chess_move_t m;
663 move_type_t kind;
664 move_undo_meta_t meta;
665 if (!game_physical_undo_load_last(&m, &kind, &meta)) {
666 return false;
667 }
668
669 if (!game_physical_undo_lift_matches(&m, kind, &meta, from_row, from_col,
670 piece)) {
671 return false;
672 }
673
674 game_physical_undo_arm_from_last(&m, kind, &meta, from_row, from_col, piece);
675 game_physical_undo_show_armed_leds();
676 ESP_LOGI(TAG, "Takeback armed: lift %c%d blue return %c%d",
677 'a' + from_col, from_row + 1, 'a' + s_undo.blue_col,
678 s_undo.blue_row + 1);
679 return true;
680#endif
681}

683bool game_physical_undo_handle_drop(const game_cmd_t *cmd,
684 uint8_t to_row, uint8_t to_col) {
685#if CONFIG_CHESS_MATRIX_INPUT_I2C_HALL
686 (void)cmd;
687 (void)to_row;
688 (void)to_col;
689 return false;
690#else
691 (void)cmd;
692
693 if (s_undo.phase != PHYS_UNDO_IDLE && !game_physical_undo_enabled()) {
694 game_physical_undo_reset();
695 return false;
696 }
697
698 if (s_undo.phase == PHYS_UNDO_PENDING) {
699 game_physical_undo_try_complete_engine();
700 return true;
701 }
702
703 if (s_undo.phase == PHYS_UNDO_IDLE) {
704 if (!game_physical_undo_enabled() ||
705 game_physical_undo_mode_blocks_offer() || !piece_lifted) {
706 return false;
707 }
708 chess_move_t rev_m;
709 move_type_t rev_kind;
710 move_undo_meta_t rev_meta;
711 if (!game_physical_undo_drop_is_last_reverse(to_row, to_col, &rev_m,
712 &rev_kind, &rev_meta)) {
713 return false;
714 }
715 game_physical_undo_arm_from_last(&rev_m, rev_kind, &rev_meta,
716 lifted_piece_row, lifted_piece_col,
717 lifted_piece);
718 }
719
720 if (s_undo.phase != PHYS_UNDO_ARMED || !piece_lifted) {
721 return false;
722 }
723
724 if (to_row == s_undo.lift_row && to_col == s_undo.lift_col) {
725 ESP_LOGI(TAG, "Takeback cancelled — piece returned to lift square");
726 game_physical_undo_reset();
727 piece_lifted = false;
728 lifted_piece = PIECE_EMPTY;
729 lifted_piece_row = 0;
730 lifted_piece_col = 0;
731 game_led_clear_board();
732 game_policy_highlight_movable_if_enabled();
733#if CONFIG_CHESS_MATRIX_INPUT_I2C_HALL
734 game_hall_restore_guide_request_sync();
735#endif
736 return true;
737 }
738
739 if (to_row == s_undo.blue_row && to_col == s_undo.blue_col) {
740 s_undo.phase = PHYS_UNDO_PENDING;
741#if CONFIG_CHESS_MATRIX_INPUT_I2C_HALL
742 game_hall_restore_guide_stop_leds();
743#endif
744 game_physical_undo_build_target_board(s_undo.target_board, &s_undo.move,
745 s_undo.kind, &s_undo.meta);
746 piece_lifted = false;
747 lifted_piece = PIECE_EMPTY;
748 lifted_piece_row = 0;
749 lifted_piece_col = 0;
750 s_undo.pending_blink_phase = 0;
751
752 if (s_undo.kind == MOVE_TYPE_PROMOTION && (*game_state_promotion_state()).pending) {
753 if (xSemaphoreTakeRecursive((*game_state_promotion_mutex()), pdMS_TO_TICKS(100)) ==
754 pdTRUE) {
755 (*game_state_promotion_state()).pending = false;
756 (*game_state_promotion_state()).square_row = 0;
757 (*game_state_promotion_state()).square_col = 0;
758 xSemaphoreGiveRecursive((*game_state_promotion_mutex()));
759 }
760 }
761
762 ESP_LOGI(TAG, "Takeback pending — restore all pieces before engine rewind");
763 game_physical_undo_show_pending_leds();
764 game_physical_undo_try_complete_engine();
765 return true;
766 }
767
768 ESP_LOGI(TAG, "Takeback held — %c%d is not the previous square",
769 'a' + to_col, to_row + 1);
770 s_undo.stray_valid = true;
771 s_undo.stray_row = to_row;
772 s_undo.stray_col = to_col;
773 piece_lifted = false;
774 lifted_piece = PIECE_EMPTY;
775 lifted_piece_row = 0;
776 lifted_piece_col = 0;
777 game_physical_undo_show_armed_leds();
778 game_led_frame_sq(to_row, to_col, 255, 0, 0);
779 return true;
780#endif
781}
