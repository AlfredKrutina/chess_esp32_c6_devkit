54static uint8_t take_payload_slot(void) {

60/* Payload must be copied only after a free slot exists. Critical may wait 10 ms. */
61static bool queue_room(uint8_t kind) {
62 UBaseType_t need = event_is_critical(kind)
63 ? 1u
64 : (UBaseType_t)(GAME_EVENT_CRITICAL_RESERVE + 1u);
65 TickType_t wait = event_is_critical(kind) ? pdMS_TO_TICKS(10) : 0;
66 TickType_t start;
67
68 if (game_event_queue == NULL) {
69 return false;
70 }
71 start = xTaskGetTickCount();
72 for (;;) {
73 if (uxQueueSpacesAvailable(game_event_queue) >= need) {
74 return true;
75 }
76 if (wait == 0 || (TickType_t)(xTaskGetTickCount() - start) >= wait) {
77 ESP_LOGW(TAG, "event queue full kind=%d", (int)kind);
78 return false;
79 }
80 vTaskDelay(1);
81 }
82}

84static bool post_msg(const game_event_msg_t *msg) {
85 if (msg == NULL || game_event_queue == NULL) {
86 return false;
87 }
88 if (msg->slot == GAME_EVENT_NO_SLOT) {
89 bool critical = event_is_critical(msg->kind);
90 if (!critical && uxQueueSpacesAvailable(game_event_queue) <=
91 GAME_EVENT_CRITICAL_RESERVE) {
92 ESP_LOGW(TAG, "event queue full kind=%d", (int)msg->kind);
93 return false;
94 }
95 if (xQueueSend(game_event_queue, msg,
96 critical ? pdMS_TO_TICKS(10) : 0) != pdTRUE) {
97 ESP_LOGW(TAG, "event queue full kind=%d", (int)msg->kind);
98 return false;
99 }
100 return true;
101 }
102 if (xQueueSend(game_event_queue, msg, 0) != pdTRUE) {
103 s_slot_seq--;
104 ESP_LOGW(TAG, "event queue full kind=%d", (int)msg->kind);
105 return false;
106 }
107 return true;
108}

110void game_event_resolve(const game_event_msg_t *msg, game_event_t *out) {
111 if (msg == NULL || out == NULL) {
112 return;
113 }
114 memset(out, 0, sizeof(*out));
115 out->kind = (game_event_kind_t)msg->kind;
116 out->from = msg->from;
117 out->to = msg->to;
118 out->flags = msg->flags;
119
120 switch (msg->kind) {
121 case GAME_EVT_GAME_END:
122 out->end_result = msg->aux0;
123 break;
124 case GAME_EVT_RESTORE_BOARD:
125 out->restore_op = msg->aux0;
126 if (msg->slot < GAME_EVENT_QUEUE_DEPTH) {
127 const game_event_slot_t *sl = &s_slots[msg->slot];
128 if (msg->aux0 == GAME_EVT_RESTORE_OP_TAKEBACK) {
129 out->takeback_count = sl->u.takeback.takeback_count;
130 memcpy(out->takeback_squares, sl->u.takeback.takeback_squares,
131 sl->u.takeback.takeback_count);
132 } else {
133 out->restore = sl->u.restore;
134 }
135 }
136 break;
137 case GAME_EVT_HINT_SQUARES:
138 out->fade_ms = msg->fade_ms;
139 out->hint_lift_sq = msg->aux1;
140 out->square_count = msg->aux0;
141 if (msg->slot < GAME_EVENT_QUEUE_DEPTH && msg->aux0 > 0) {
142 const game_event_slot_t *sl = &s_slots[msg->slot];
143 memcpy(out->hints, sl->u.pixels.hints,
144 (size_t)msg->aux0 * sizeof(game_evt_hint_pixel_t));
145 if ((msg->flags & GAME_EVT_HINT_PATCH) == 0) {
146 out->breathe_count = msg->reserved;
147 if (out->breathe_count > GAME_EVT_BREATHE_MAX) {
148 out->breathe_count = GAME_EVT_BREATHE_MAX;
149 }
150 memcpy(out->breathes, sl->u.pixels.breathes,
151 (size_t)out->breathe_count * sizeof(game_evt_breathe_t));
152 }
153 }
154 break;
155 case GAME_EVT_LED_CTRL:
156 out->flags = msg->flags;
157 out->from = msg->from;
158 out->to = msg->to;
159 out->square_count = msg->aux0;
160 out->end_result = msg->aux1;
161 break;
162 default:
163 break;
164 }
165}

167bool game_event_post(const game_event_t *ev) {
168 if (ev == NULL) {
169 return false;
170 }
171 switch (ev->kind) {
172 case GAME_EVT_MOVE_ACCEPTED:
173 game_event_post_move_accepted(ev->from, ev->to);
174 return true;
175 case GAME_EVT_MOVE_REJECTED:
176 if (ev->flags & 2U) {
177 game_event_post_move_rejected_clear();
178 } else {
179 game_event_post_move_rejected(ev->from, (ev->flags & 1U) != 0);
180 }
181 return true;
182 case GAME_EVT_CHECK:
183 game_event_post_check(ev->from);
184 return true;
185 case GAME_EVT_GAME_END:
186 game_event_post_game_end(ev->from, ev->end_result);
187 return true;
188 case GAME_EVT_RESTORE_BOARD:
189 if (ev->restore_op == GAME_EVT_RESTORE_OP_TAKEBACK &&
190 ev->takeback_count > 0) {
191 game_event_post_takeback(ev->takeback_squares, ev->takeback_count);
192 } else {
193 game_event_post_restore(
194 ev->restore_op == GAME_EVT_RESTORE_OP_APPLY ? &ev->restore : NULL,
195 ev->restore_op);
196 }
197 return true;
198 default:
199 return false;
200 }
201}

299void game_event_post_pixels(const game_evt_hint_pixel_t *px, uint8_t count,
300 const game_evt_breathe_t *breathes,
301 uint8_t breathe_count, uint16_t fade_ms,
302 bool patch, uint8_t hint_lift_sq) {
303 uint8_t off = 0;
304 bool first = true;
305 if (breathe_count > GAME_EVT_BREATHE_MAX) {
306 breathe_count = GAME_EVT_BREATHE_MAX;
307 }
308 do {
309 game_event_msg_t msg = {
310 .kind = GAME_EVT_HINT_SQUARES,
311 .slot = GAME_EVENT_NO_SLOT,
312 };
313 uint8_t n = 0;
314 uint8_t i;
315 if (!first) {
316 msg.flags = GAME_EVT_HINT_PATCH;
317 msg.aux1 = 255U;
318 } else {
319 if (patch) {
320 msg.flags = GAME_EVT_HINT_PATCH;
321 }
322 msg.aux1 = hint_lift_sq;
323 msg.fade_ms = fade_ms;
324 msg.reserved = breathe_count;
325 }
326 if (!queue_room(GAME_EVT_HINT_SQUARES)) {
327 return;
328 }
329 uint8_t slot = take_payload_slot();
330 game_event_slot_t *sl = &s_slots[slot];
331 memset(sl->u.pixels.hints, 0, sizeof(sl->u.pixels.hints));
332 memset(sl->u.pixels.breathes, 0, sizeof(sl->u.pixels.breathes));
333 if (first && breathes != NULL) {
334 for (i = 0; i < breathe_count; i++) {
335 sl->u.pixels.breathes[i] = breathes[i];
336 }
337 }
338 while (off < count && n < GAME_EVT_HINT_MAX) {
339 if (px != NULL) {
340 sl->u.pixels.hints[n] = px[off];
341 }
342 n++;
343 off++;
344 }
345 msg.aux0 = n;
346 msg.slot = slot;
347 (void)post_msg(&msg);
348 first = false;
349 } while (off < count);
350}
