123static esp_err_t game_status_json_append(size_t *offset, char *buffer, size_t size,
124 const char *fmt, ...) {
125 va_list ap;
126 va_start(ap, fmt);
127 if (*offset >= size) {
128 va_end(ap);
129 ESP_LOGE(TAG, "game_get_status_json: offset invalid %zu", *offset);
130 return ESP_ERR_NO_MEM;
131 }
132 size_t rem = size - *offset;
133 if (rem <= 1) {
134 va_end(ap);
135 return ESP_ERR_NO_MEM;
136 }
137 int n = vsnprintf(buffer + *offset, rem, fmt, ap);
138 va_end(ap);
139 if (n < 0 || (size_t)n >= rem) {
140 ESP_LOGE(TAG,
141 "game_get_status_json: append overflow need=%d rem=%zu off=%zu", n,
142 rem, *offset);
143 return ESP_ERR_NO_MEM;
144 }
145 *offset += (size_t)n;
146 return ESP_OK;
147}

149static void game_json_escape(char *dst, size_t dst_len, const char *src) {
150 size_t j = 0;
151 if (dst == NULL || dst_len == 0) {
152 return;
153 }
154 if (src == NULL) {
155 dst[0] = '\0';
156 return;
157 }
158 for (size_t i = 0; src[i] != '\0' && j + 2 < dst_len; i++) {
159 unsigned char c = (unsigned char)src[i];
160 if (c < 0x20) {
161 continue;
162 }
163 if (c == '"' || c == '\\') {
164 dst[j++] = '\\';
165 }
166 dst[j++] = (char)c;
167 }
168 dst[j] = '\0';
169}
