285void ha_light_report_activity(const char *activity_type) {
286 if (!task_running) {
287 return;
288 }
289
290 // Keep this function lightweight: it may be called from input paths.
291 // Mode switching and LED refresh must happen in HA task context.
292 last_activity_time_ms = esp_timer_get_time() / 1000;
293
294 /* Rate limit: max 1 queued activity / 500 ms (saves the queue and MQTT).
295 * Exception: POST /api/light/game_mode — must always pass, otherwise the user
296 * will not bring the board back from lamp mode after a rapid series of other events. */
297 static uint32_t last_report_time = 0;
298 uint32_t current_time = esp_timer_get_time() / 1000;
299 const bool force_queue =
300 (activity_type != NULL && strcmp(activity_type, "web_game_mode") == 0);
301 if (!force_queue && current_time - last_report_time < 500) {
302 return;
303 }
304 last_report_time = current_time;
305
306 ha_light_command_t cmd;
307 cmd.type = HA_CMD_ACTIVITY;
308 cmd.u.data = (void *)activity_type;
309
310 if (ha_light_cmd_queue != NULL) {
311 xQueueSend(ha_light_cmd_queue, &cmd, 0);
312 }
313}

332static void ha_light_switch_to_ha_mode(void) {
333 ha_light_wdt_feed_if_own_task();
334 bool mode_changed = (current_mode != HA_MODE_HA);
335
336 if (mode_changed) {
337 ESP_LOGI(TAG, "Switching to HA MODE");
338 lamp_nvs_save_mode();
339 }
340 current_mode = HA_MODE_HA;
341 /* Game LED writes stop before the lamp color is applied. */
342 led_set_lamp_hold(true);
343
344 uint8_t r, g, b, brightness;
345 bool state;
346 if (ha_light_state_mutex != NULL &&
347 xSemaphoreTake(ha_light_state_mutex, pdMS_TO_TICKS(50)) == pdTRUE) {
348 r = ha_light_state.r;
349 g = ha_light_state.g;
350 b = ha_light_state.b;
351 brightness = ha_light_state.brightness;
352 state = ha_light_state.state;
353 xSemaphoreGive(ha_light_state_mutex);
354 } else {
355 r = g = b = 255;
356 brightness = 255;
357 state = true;
358 }
359
360 remember_lit_color(r, g, b, brightness);
361
362 /* Light off releases the board: a lift then continues the game, or the
363 * starting-position piece colors come back. */
364 if (state) {
365 led_set_ha_color(r, g, b, brightness);
366 } else {
367 led_set_ha_color(0, 0, 0, 0);
368 led_set_lamp_hold(false);
369 }
370 ha_light_wdt_feed_if_own_task();
371
372 ha_light_publish_state();
373 ha_light_wdt_feed_if_own_task();
374 if (mode_changed) {
375 ha_game_apply(HA_GAME_LAMP, HA_REASON_NONE);
376 }
377}

384static void ha_light_switch_to_game_mode(void) {
385 if (current_mode == HA_MODE_GAME) {
386 return; // Already in game mode
387 }

389 ESP_LOGI(TAG, "Switching to GAME MODE");
390 ha_light_wdt_feed_if_own_task();

429static esp_err_t mqtt_load_config_from_nvs(char *host, size_t host_len,
430 uint16_t *port, char *username,
431 size_t username_len, char *password,
432 size_t password_len) {
433 if (host == NULL || port == NULL || username == NULL || password == NULL ||
434 host_len == 0 || username_len == 0 || password_len == 0) {
435 ESP_LOGE(TAG, "Invalid parameters for mqtt_load_config_from_nvs");
436 return ESP_ERR_INVALID_ARG;
437 }
438
439 nvs_handle_t nvs_handle;
440 esp_err_t ret = nvs_open(MQTT_NVS_NAMESPACE, NVS_READONLY, &nvs_handle);
441 if (ret != ESP_OK) {
442 ESP_LOGD(TAG, "MQTT config not found in NVS, using defaults");
443 // Return default values
444 strncpy(host, MQTT_DEFAULT_HOST, host_len - 1);
445 host[host_len - 1] = '\0';
446 *port = MQTT_DEFAULT_PORT;
447 strncpy(username, MQTT_DEFAULT_USERNAME, username_len - 1);
448 username[username_len - 1] = '\0';
449 strncpy(password, MQTT_DEFAULT_PASSWORD, password_len - 1);
450 password[password_len - 1] = '\0';
451 return ESP_ERR_NOT_FOUND; // Not in NVS, but returned defaults
452 }
453
454 // Load guest
455 size_t required_size = host_len;
456 ret = nvs_get_str(nvs_handle, MQTT_NVS_KEY_HOST, host, &required_size);
457 if (ret != ESP_OK) {
458 ESP_LOGW(TAG, "Failed to get MQTT host from NVS, using default");
459 strncpy(host, MQTT_DEFAULT_HOST, host_len - 1);
460 host[host_len - 1] = '\0';
461 }
462
463 // Load the port
464 uint32_t port_val = MQTT_DEFAULT_PORT;
465 ret = nvs_get_u32(nvs_handle, MQTT_NVS_KEY_PORT, &port_val);
466 if (ret != ESP_OK || port_val == 0 || port_val > 65535) {
467 ESP_LOGW(TAG, "Failed to get MQTT port from NVS, using default");
468 port_val = MQTT_DEFAULT_PORT;
469 }
470 *port = (uint16_t)port_val;
471
472 // Load username (optional - can be empty)
473 required_size = username_len;
474 ret =
475 nvs_get_str(nvs_handle, MQTT_NVS_KEY_USERNAME, username, &required_size);
476 if (ret != ESP_OK) {
477 ESP_LOGD(TAG, "MQTT username not in NVS, using default");
478 strncpy(username, MQTT_DEFAULT_USERNAME, username_len - 1);
479 username[username_len - 1] = '\0';
480 }
481
482 // Load password (optional - can be empty)
483 required_size = password_len;
484 ret =
485 nvs_get_str(nvs_handle, MQTT_NVS_KEY_PASSWORD, password, &required_size);
486 if (ret != ESP_OK) {
487 ESP_LOGD(TAG, "MQTT password not in NVS, using default");
488 strncpy(password, MQTT_DEFAULT_PASSWORD, password_len - 1);
489 password[password_len - 1] = '\0';
490 }
491
492 nvs_close(nvs_handle);
493 ESP_LOGI(TAG, "MQTT config loaded from NVS: host=%s, port=%d, username=%s",
494 host, *port, username[0] ? username : "(empty)");
495
496 return ESP_OK;
497}

508esp_err_t mqtt_save_config_to_nvs(const char *host, uint16_t port,
509 const char *username, const char *password) {
510 if (host == NULL || port == 0) {
511 ESP_LOGE(TAG, "Invalid parameters: host is NULL or port is 0");
512 return ESP_ERR_INVALID_ARG;
513 }
514
515 size_t host_len = strlen(host);
516 if (host_len == 0 || host_len > 127) {
517 ESP_LOGE(TAG, "Invalid host length: %zu (must be 1-127)", host_len);
518 return ESP_ERR_INVALID_ARG;
519 }
520
521 // Validate username/password if set
522 if (username != NULL) {
523 size_t username_len = strlen(username);
524 if (username_len > 63) {
525 ESP_LOGE(TAG, "Invalid username length: %zu (max 63)", username_len);
526 return ESP_ERR_INVALID_ARG;
527 }
528 }
529
530 if (password != NULL) {
531 size_t password_len = strlen(password);
532 if (password_len > 63) {
533 ESP_LOGE(TAG, "Invalid password length: %zu (max 63)", password_len);
534 return ESP_ERR_INVALID_ARG;
535 }
536 }
537
538 nvs_handle_t nvs_handle;
539 esp_err_t ret = nvs_open(MQTT_NVS_NAMESPACE, NVS_READWRITE, &nvs_handle);
540 if (ret != ESP_OK) {
541 ESP_LOGE(TAG, "Failed to open NVS: %s", esp_err_to_name(ret));
542 return ret;
543 }
544
545 // Save guest
546 ret = nvs_set_str(nvs_handle, MQTT_NVS_KEY_HOST, host);
547 if (ret != ESP_OK) {
548 ESP_LOGE(TAG, "Failed to set MQTT host in NVS: %s", esp_err_to_name(ret));
549 nvs_close(nvs_handle);
550 return ret;
551 }
552
553 // Save port
554 ret = nvs_set_u32(nvs_handle, MQTT_NVS_KEY_PORT, (uint32_t)port);
555 if (ret != ESP_OK) {
556 ESP_LOGE(TAG, "Failed to set MQTT port in NVS: %s", esp_err_to_name(ret));
557 nvs_close(nvs_handle);
558 return ret;
559 }
560
561 // Save username (if set, otherwise save empty string)
562 const char *username_to_save =
563 (username != NULL && strlen(username) > 0) ? username : "";
564 ret = nvs_set_str(nvs_handle, MQTT_NVS_KEY_USERNAME, username_to_save);
565 if (ret != ESP_OK) {
566 ESP_LOGE(TAG, "Failed to set MQTT username in NVS: %s",
567 esp_err_to_name(ret));
568 nvs_close(nvs_handle);
569 return ret;
570 }
571
572 // Save password (if set, otherwise save empty string)
573 const char *password_to_save =
574 (password != NULL && strlen(password) > 0) ? password : "";
575 ret = nvs_set_str(nvs_handle, MQTT_NVS_KEY_PASSWORD, password_to_save);
576 if (ret != ESP_OK) {
577 ESP_LOGE(TAG, "Failed to set MQTT password in NVS: %s",
578 esp_err_to_name(ret));
579 nvs_close(nvs_handle);
580 return ret;
581 }
582
583 ret = nvs_commit(nvs_handle);
584 if (ret != ESP_OK) {
585 ESP_LOGE(TAG, "Failed to commit NVS: %s", esp_err_to_name(ret));
586 nvs_close(nvs_handle);
587 return ret;
588 }
589
590 nvs_close(nvs_handle);
591 ESP_LOGI(TAG, "MQTT config saved to NVS: host=%s, port=%d, username=%s", host,
592 port, username_to_save[0] ? username_to_save : "(empty)");
593
594 return ESP_OK;
595}

675static void ha_light_mqtt_event_handler(void *handler_args,

691 /* Discovery and state go out one message per loop, not in this callback. */
692 s_disc_step = 0;
693 break;
694
695 case MQTT_EVENT_DISCONNECTED:
696 ESP_LOGI(TAG, "MQTT Disconnected");
697 mqtt_connected = false;
698 s_mqtt_down = true;
699 s_disc_step = -1;
700 break;
701
702 case MQTT_EVENT_DATA:
703 ESP_LOGI(TAG, "MQTT DATA: topic=%.*s, data=%.*s", event->topic_len,
704 event->topic, event->data_len, event->data);
705
706 // Handle command
707 char topic[128] = {0};
708 char data[512] = {0};
709
710 if (event->topic != NULL && event->data != NULL &&
711 event->topic_len > 0 && event->topic_len < (int)sizeof(topic) &&
712 event->data_len > 0 && event->data_len < (int)sizeof(data)) {
713 memcpy(topic, event->topic, event->topic_len);
714 memcpy(data, event->data, event->data_len);
715 ha_light_handle_mqtt_command(topic, data, event->data_len);
716 }
717 break;
718
719 case MQTT_EVENT_ERROR:
720 mqtt_connected = false;
721 s_mqtt_down = true;
722 s_disc_step = -1;
723 if (event->error_handle != NULL) {
724 ESP_LOGW(TAG, "MQTT Error type=%d sock=%d tls=%s",
725 (int)event->error_handle->error_type,
726 (int)event->error_handle->esp_transport_sock_errno,
727 esp_err_to_name(event->error_handle->esp_tls_last_esp_err));
728 } else {
729 ESP_LOGW(TAG, "MQTT Error");
730 }
731 break;
732
733 default:
734 break;
735 }
736}

741static void ha_light_handle_mqtt_command(const char *topic, const char *data,
742 int data_len) {
743 if (strcmp(topic, HA_TOPIC_LIGHT_COMMAND) != 0) {
744 return; // Not our topic
745 }
746
747 // Parse JSON
748 cJSON *json = cJSON_ParseWithLength(data, data_len);
749 if (json == NULL) {
750 ESP_LOGE(TAG, "Failed to parse MQTT JSON");
751 return;
752 }
753
754 bool new_state = ha_light_state.state;
755 uint8_t new_r = ha_light_state.r, new_g = ha_light_state.g,
756 new_b = ha_light_state.b;
757 uint8_t new_brightness = ha_light_state.brightness;
758
759 cJSON *state_item = cJSON_GetObjectItem(json, "state");
760 if (cJSON_IsString(state_item)) {
761 if (strcasecmp(state_item->valuestring, "ON") == 0) {
762 new_state = true;
763 } else if (strcasecmp(state_item->valuestring, "OFF") == 0) {
764 new_state = false;
765 }
766 }
767
768 cJSON *brightness_item = cJSON_GetObjectItem(json, "brightness");
769 if (cJSON_IsNumber(brightness_item)) {
770 int brightness_val = (int)cJSON_GetNumberValue(brightness_item);
771 if (brightness_val < 0)
772 brightness_val = 0;
773 if (brightness_val > 255)
774 brightness_val = 255;
775 new_brightness = (uint8_t)brightness_val;
776 }
777
778 cJSON *color_item = cJSON_GetObjectItem(json, "color");
779 if (cJSON_IsObject(color_item)) {
780 cJSON *r = cJSON_GetObjectItem(color_item, "r");
781 cJSON *g = cJSON_GetObjectItem(color_item, "g");
782 cJSON *b = cJSON_GetObjectItem(color_item, "b");
783 if (cJSON_IsNumber(r) && cJSON_IsNumber(g) && cJSON_IsNumber(b)) {
784 new_r = (uint8_t)cJSON_GetNumberValue(r);
785 new_g = (uint8_t)cJSON_GetNumberValue(g);
786 new_b = (uint8_t)cJSON_GetNumberValue(b);
787 }
788 }
789
790 cJSON *rgb_item = cJSON_GetObjectItem(json, "rgb_color");
791 if (cJSON_IsArray(rgb_item) && cJSON_GetArraySize(rgb_item) >= 3) {
792 new_r = (uint8_t)cJSON_GetNumberValue(cJSON_GetArrayItem(rgb_item, 0));
793 new_g = (uint8_t)cJSON_GetNumberValue(cJSON_GetArrayItem(rgb_item, 1));
794 new_b = (uint8_t)cJSON_GetNumberValue(cJSON_GetArrayItem(rgb_item, 2));
795 }
796
797 cJSON_Delete(json);
798
799 if (ha_light_cmd_queue == NULL) {
800 return;
801 }
802
803 ha_light_command_t cmd = {0};
804 cmd.type = HA_CMD_MQTT_STATE;
805 cmd.u.web_lamp.state = new_state ? 1 : 0;
806 cmd.u.web_lamp.r = new_r;
807 cmd.u.web_lamp.g = new_g;
808 cmd.u.web_lamp.b = new_b;
809 cmd.u.web_lamp.brightness = new_brightness;
810 cmd.u.web_lamp.set_brightness = 1;
811 if (xQueueSend(ha_light_cmd_queue, &cmd, 0) != pdTRUE) {
812 ESP_LOGW(TAG, "MQTT state queue full");
813 }
814}

1080static void ha_discovery_pump(void) {
1081 int step = s_disc_step;
1082 if (step < 0 || !mqtt_connected || mqtt_client == NULL) {
1083 return;
1084 }
1085 /* Skip publishes once free heap is under the WiFi floor. */
1086 if (esp_get_free_heap_size() < HA_MQTT_STOP_HEAP_BYTES) {
1087 return;
1088 }
1089 static int64_t s_disc_next_us;
1090 int64_t now = esp_timer_get_time();
1091 if (now < s_disc_next_us) {
1092 return;
1093 }
1094 const int trigger0 = 4;
1095 const int trigger_n =
1096 (int)(sizeof(s_game_events) / sizeof(s_game_events[0]));
1097 const int state_step = trigger0 + trigger_n;
1098 bool sent = false;
1099 if (step == 0) {
1100 sent = ha_light_publish_discovery();
1101 } else if (step == 1) {
1102 /* One QoS 1 so Home Assistant keeps the online state. */
1103 sent = esp_mqtt_client_publish(mqtt_client, HA_TOPIC_LIGHT_AVAILABILITY,
1104 HA_MQTT_PAYLOAD_ONLINE, 0, 1, 1) >= 0;
1105 } else if (step == 2) {
1106 sent = ha_light_publish_state();
1107 } else if (step == 3) {
1108 sent = ha_game_publish_sensor();
1109 } else if (step < state_step) {
1110 sent = ha_game_publish_trigger((unsigned)(step - trigger0));
1111 } else {
1112 sent = ha_game_publish_state();
1113 if (sent && s_disc_step == step) {
1114 s_disc_step = -1;
1115 }
1116 s_disc_next_us = now + 100000;
1117 return;
1118 }
1119 s_disc_next_us = now + 100000;
1120 /* A failed send stays on this step. Disconnect must not be overwritten. */
1121 if (sent && s_disc_step == step) {
1122 s_disc_step = step + 1;
1123 }
1124}

1263 .credentials.client_id = client_id,

1631void ha_light_task_start(void *pvParameters) {
1632 ESP_LOGI(TAG, "Starting HA Light Task...");
1633 s_ha_light_task_hdl = xTaskGetCurrentTaskHandle();
1634
1635 // Register with WDT
1636 esp_err_t ret = esp_task_wdt_add(NULL);
1637 if (ret != ESP_OK && ret != ESP_ERR_INVALID_ARG) {
1638 ESP_LOGW(TAG, "Failed to register with WDT: %s", esp_err_to_name(ret));
1639 }
1640
1641 task_running = true;
1642 current_mode = HA_MODE_GAME;
1643 last_activity_time_ms = esp_timer_get_time() / 1000;
1644
1645 ha_light_state_mutex = xSemaphoreCreateMutex();
1646 if (ha_light_state_mutex == NULL) {
1647 ESP_LOGE(TAG, "Failed to create ha_light_state mutex");
1648 }
1649
1650 ha_light_cmd_queue = xQueueCreate(10, sizeof(ha_light_command_t));
1651 if (ha_light_cmd_queue == NULL) {
1652 ESP_LOGE(TAG, "Failed to create HA command queue");
1653 }
1654
1655 lamp_nvs_load();
1656
1657 // Main loop
1658 // TickType_t last_wake_time = xTaskGetTickCount(); // Unused in queue-based
1659 // loop
1660
1661 for (;;) {
1662 // =========================================================================
1663 // 🛑 BOOT ANIMATION PROTECTION - BLOCKS HA OPERATION DURING BOOT
1664 // =========================================================================
1665 // If the boot animation is running, the HA task must not send LED commands!
1666 // (e.g. "turn off the light" on a fast WiFi connection)
1667 if (led_is_booting()) {
1668 chess_task_wdt_reset_safe(TAG);
1669 vTaskDelay(pdMS_TO_TICKS(10)); // 10ms - quick wakeup after fade_out!
1670 continue;
1671 }
1672 if (!s_boot_mode_applied) {
1673 s_boot_mode_applied = true;
1674 if (s_nvs_mode_found && s_nvs_mode == HA_MODE_HA &&
1675 current_mode != HA_MODE_HA) {
1676 ESP_LOGI(TAG, "[STAGING] restore persisted lamp mode");
1677 ha_light_switch_to_ha_mode();
1678 }
1679 }
1680 // =========================================================================
1681
1682 // Reset WDT
1683 chess_task_wdt_reset_safe(TAG);
1684
1685 ha_discovery_pump();
1686
1687 // Process queue with timeout (100ms) - replaces vTaskDelay
1688 ha_light_command_t cmd;
1689 if (ha_light_cmd_queue &&
1690 xQueueReceive(ha_light_cmd_queue, &cmd, pdMS_TO_TICKS(100)) == pdTRUE) {
1691 if (cmd.type == HA_CMD_WEB_LAMP) {
1692 if (ha_light_state_mutex != NULL &&
1693 xSemaphoreTake(ha_light_state_mutex, pdMS_TO_TICKS(100)) ==
1694 pdTRUE) {
1695 remember_lit_color(ha_light_state.r, ha_light_state.g, ha_light_state.b,
1696 ha_light_state.brightness);
1697 ha_light_state.state = (cmd.u.web_lamp.state != 0);
1698 ha_light_state.r = cmd.u.web_lamp.r;
1699 ha_light_state.g = cmd.u.web_lamp.g;
1700 ha_light_state.b = cmd.u.web_lamp.b;
1701 if (cmd.u.web_lamp.set_brightness) {
1702 ha_light_state.brightness = cmd.u.web_lamp.brightness;
1703 }
1704 remember_lit_color(ha_light_state.r, ha_light_state.g, ha_light_state.b,
1705 ha_light_state.brightness);
1706 xSemaphoreGive(ha_light_state_mutex);
1707 }
1708 lamp_nvs_save();
1709 ha_light_switch_to_ha_mode();
1710 } else if (cmd.type == HA_CMD_MQTT_STATE) {
1711 bool was_on = ha_light_state.state;
1712 bool now_on_cmd = (cmd.u.web_lamp.state != 0);
1713 /* Capture the lit color before state becomes off, so the shrink wave
1714 * still has a disk to pull back into h1. */
1715 bool leaving_light =
1716 !now_on_cmd && was_on && (current_mode == HA_MODE_HA);
1717 if (leaving_light) {
1718 game_gesture_request_lamp_off_wave();
1719 }
1720 if (ha_light_state_mutex != NULL &&
1721 xSemaphoreTake(ha_light_state_mutex, pdMS_TO_TICKS(100)) ==
1722 pdTRUE) {
1723 was_on = ha_light_state.state;
1724 remember_lit_color(ha_light_state.r, ha_light_state.g, ha_light_state.b,
1725 ha_light_state.brightness);
1726 ha_light_state.state = (cmd.u.web_lamp.state != 0);
1727 ha_light_state.r = cmd.u.web_lamp.r;
1728 ha_light_state.g = cmd.u.web_lamp.g;
1729 ha_light_state.b = cmd.u.web_lamp.b;
1730 ha_light_state.brightness = cmd.u.web_lamp.brightness;
1731 remember_lit_color(ha_light_state.r, ha_light_state.g, ha_light_state.b,
1732 ha_light_state.brightness);
1733 xSemaphoreGive(ha_light_state_mutex);
1734 }
1735 lamp_nvs_save();
1736 bool now_on = (cmd.u.web_lamp.state != 0);
1737 /* Turning the lamp on leaves whatever is on the board, game included. */
1738 bool entering_light =
1739 now_on && (current_mode != HA_MODE_HA || !was_on);
1740 bool apply = true;
1741 if (!entering_light && current_mode == HA_MODE_GAME) {
1742 uint32_t now_ms = esp_timer_get_time() / 1000;
1743 uint32_t idle_ms = now_ms - last_activity_time_ms;
1744 apply = (idle_ms >= HA_ACTIVITY_TIMEOUT_COMMAND_MS ||
1745 game_get_move_count() == 0);
1746 if (!apply) {
1747 ESP_LOGI(TAG,
1748 "MQTT state stored (game active, moves=%lu, idle=%lu ms)",
1749 game_get_move_count(), idle_ms);
1750 }
1751 }
1752 if (apply && entering_light) {
1753 game_gesture_request_lamp_wave();
1754 } else if (leaving_light) {
1755 /* State is already off. The shrink wave owns the LEDs until it
1756 * commits HA mode, which then paints black. */
1757 ha_light_publish_state();
1758 } else if (apply) {
1759 ha_light_switch_to_ha_mode();
1760 } else {
1761 ha_light_publish_state();
1762 }
1763 } else if (cmd.type == HA_CMD_GAME_FACT) {
1764 ha_game_apply(cmd.u.web_lamp.state, cmd.u.web_lamp.r);
1765 } else if (cmd.type == HA_CMD_SET_MODE) {
1766 if (cmd.u.web_lamp.state == HA_MODE_HA) {
1767 /* Coming from the game, the lamp comes back in the last real color. */
1768 if (current_mode == HA_MODE_GAME && ha_light_state_mutex != NULL &&
1769 xSemaphoreTake(ha_light_state_mutex, pdMS_TO_TICKS(50)) ==
1770 pdTRUE) {
1771 if (!ha_light_state.state ||
1772 (ha_light_state.r | ha_light_state.g | ha_light_state.b) == 0) {
1773 ha_light_state.r = s_lit_r;
1774 ha_light_state.g = s_lit_g;
1775 ha_light_state.b = s_lit_b;
1776 ha_light_state.brightness = s_lit_br;
1777 ha_light_state.state = true;
1778 xSemaphoreGive(ha_light_state_mutex);
1779 lamp_nvs_save();
1780 } else {
1781 xSemaphoreGive(ha_light_state_mutex);
1782 }
1783 }
1784 ha_light_switch_to_ha_mode();
1785 } else {
1786 ha_light_switch_to_game_mode();
1787 }
1788 } else if (cmd.type == HA_CMD_ACTIVITY) {
1789 const char *activity_name = (const char *)cmd.u.data;
1790 // Reset activity timer in HA task context
1791 last_activity_time_ms = esp_timer_get_time() / 1000;
1792
1793 /* A bump that only renames a piece, or a piece put back on the same
1794 * square, is not a move. Only a real relocation leaves the lamp. */
1795 bool leave_lamp = false;
1796 if (activity_name != NULL &&
1797 (strcmp(activity_name, "move") == 0 ||
1798 strcmp(activity_name, "web_game_mode") == 0)) {
1799 leave_lamp = true;
1800 }
1801 if (current_mode == HA_MODE_HA && leave_lamp) {
1802 ESP_LOGI(TAG, "Switching from HA mode to GAME mode (activity: %s)",
1803 activity_name);
1804 ha_light_switch_to_game_mode();
1805 }
1806 if (activity_name != NULL && strcmp(activity_name, "move") == 0) {
1807 ha_game_apply(HA_GAME_MOVE, HA_REASON_NONE);
1808 }
1809
1810 // Check rate limit logic again here (in HA task context) to save MQTT
1811 // bandwidth
1812 static uint32_t last_mqtt_activity = 0;
1813 uint32_t now = esp_timer_get_time() / 1000;
1814
1815 if (now - last_mqtt_activity > 500) { // 500ms limit
1816 last_mqtt_activity = now;
1817
1818 if (mqtt_connected && mqtt_client != NULL &&
1819 ha_mqtt_set("{\"event\":\"%s\",\"timestamp_ms\":%lu}",
1820 activity_name ? activity_name : "unknown",
1821 (unsigned long)now)) {
1822 ha_mqtt_send(HA_TOPIC_GAME_ACTIVITY, 0);
1823 }
1824 }
1825 }
1826 chess_task_wdt_reset_safe(TAG);
1827 }
1828
1829 // POLL: Check WiFi STA status periodically (every 5 seconds)
1830 static uint32_t last_wifi_check = 0;
1831 static uint32_t last_mqtt_retry_ms = 0;
1832 uint32_t current_time_ms = esp_timer_get_time() / 1000;
1833 /* Stop within one loop (~100 ms). Start only after the gate holds. */
1834 ha_mqtt_stop_if_low();
1835 if (sta_connected && mqtt_client == NULL) {
1836 ha_mqtt_consider_start(current_time_ms);
