263void ha_light_report_activity(const char *activity_type) {
264 if (!task_running) {
265 return;
266 }
267
268 // Keep this function lightweight: it may be called from input paths.
269 // Mode switching and LED refresh must happen in HA task context.
270 last_activity_time_ms = esp_timer_get_time() / 1000;
271
272 /* Rate limit: max 1 queued activity / 500 ms (saves the queue and MQTT).
273 * Exception: POST /api/light/game_mode — must always pass, otherwise the user
274 * will not bring the board back from lamp mode after a rapid series of other events. */
275 static uint32_t last_report_time = 0;
276 uint32_t current_time = esp_timer_get_time() / 1000;
277 const bool force_queue =
278 (activity_type != NULL && strcmp(activity_type, "web_game_mode") == 0);
279 if (!force_queue && current_time - last_report_time < 500) {
280 return;
281 }
282 last_report_time = current_time;
283
284 ha_light_command_t cmd;
285 cmd.type = HA_CMD_ACTIVITY;
286 cmd.u.data = (void *)activity_type;
287
288 if (ha_light_cmd_queue != NULL) {
289 xQueueSend(ha_light_cmd_queue, &cmd, 0);
290 }
291}

310static void ha_light_switch_to_ha_mode(void) {
311 ha_light_wdt_feed_if_own_task();
312 bool mode_changed = (current_mode != HA_MODE_HA);
313
314 if (mode_changed) {
315 ESP_LOGI(TAG, "Switching to HA MODE");
316 lamp_nvs_save_mode();
317 }
318 current_mode = HA_MODE_HA;
319 /* Game LED writes stop before the lamp color is applied. */
320 led_set_lamp_hold(true);
321
322 uint8_t r, g, b, brightness;
323 bool state;
324 if (ha_light_state_mutex != NULL &&
325 xSemaphoreTake(ha_light_state_mutex, pdMS_TO_TICKS(50)) == pdTRUE) {
326 r = ha_light_state.r;
327 g = ha_light_state.g;
328 b = ha_light_state.b;
329 brightness = ha_light_state.brightness;
330 state = ha_light_state.state;
331 xSemaphoreGive(ha_light_state_mutex);
332 } else {
333 r = g = b = 255;
334 brightness = 255;
335 state = true;
336 }
337
338 remember_lit_color(r, g, b, brightness);
339
340 /* A drawing owns the squares. A solid lamp color is everything else.
341 * Light off releases the board: a lift then continues the game, or the
342 * starting-position piece colors come back. */
343 if (s_pat_on) {
344 ha_pattern_paint_all();
345 } else if (state) {
346 led_set_ha_color(r, g, b, brightness);
347 } else {
348 led_set_ha_color(0, 0, 0, 0);
349 led_set_lamp_hold(false);
350 }
351 ha_light_wdt_feed_if_own_task();
352
353 ha_light_publish_state();
354 ha_light_wdt_feed_if_own_task();
355 if (mode_changed) {
356 ha_game_apply(HA_GAME_LAMP, HA_REASON_NONE);
357 }
358}

410static esp_err_t mqtt_load_config_from_nvs(char *host, size_t host_len,
411 uint16_t *port, char *username,
412 size_t username_len, char *password,
413 size_t password_len) {
414 if (host == NULL || port == NULL || username == NULL || password == NULL ||
415 host_len == 0 || username_len == 0 || password_len == 0) {
416 ESP_LOGE(TAG, "Invalid parameters for mqtt_load_config_from_nvs");
417 return ESP_ERR_INVALID_ARG;
418 }
419
420 nvs_handle_t nvs_handle;
421 esp_err_t ret = nvs_open(MQTT_NVS_NAMESPACE, NVS_READONLY, &nvs_handle);
422 if (ret != ESP_OK) {
423 ESP_LOGD(TAG, "MQTT config not found in NVS, using defaults");
424 // Return default values
425 strncpy(host, MQTT_DEFAULT_HOST, host_len - 1);
426 host[host_len - 1] = '\0';
427 *port = MQTT_DEFAULT_PORT;
428 strncpy(username, MQTT_DEFAULT_USERNAME, username_len - 1);
429 username[username_len - 1] = '\0';
430 strncpy(password, MQTT_DEFAULT_PASSWORD, password_len - 1);
431 password[password_len - 1] = '\0';
432 return ESP_ERR_NOT_FOUND; // Not in NVS, but returned defaults
433 }
434
435 // Load guest
436 size_t required_size = host_len;
437 ret = nvs_get_str(nvs_handle, MQTT_NVS_KEY_HOST, host, &required_size);
438 if (ret != ESP_OK) {
439 ESP_LOGW(TAG, "Failed to get MQTT host from NVS, using default");
440 strncpy(host, MQTT_DEFAULT_HOST, host_len - 1);
441 host[host_len - 1] = '\0';
442 }
443
444 // Load the port
445 uint32_t port_val = MQTT_DEFAULT_PORT;
446 ret = nvs_get_u32(nvs_handle, MQTT_NVS_KEY_PORT, &port_val);
447 if (ret != ESP_OK || port_val == 0 || port_val > 65535) {
448 ESP_LOGW(TAG, "Failed to get MQTT port from NVS, using default");
449 port_val = MQTT_DEFAULT_PORT;
450 }
451 *port = (uint16_t)port_val;
452
453 // Load username (optional - can be empty)
454 required_size = username_len;
455 ret =
456 nvs_get_str(nvs_handle, MQTT_NVS_KEY_USERNAME, username, &required_size);
457 if (ret != ESP_OK) {
458 ESP_LOGD(TAG, "MQTT username not in NVS, using default");
459 strncpy(username, MQTT_DEFAULT_USERNAME, username_len - 1);
460 username[username_len - 1] = '\0';
461 }
462
463 // Load password (optional - can be empty)
464 required_size = password_len;
465 ret =
466 nvs_get_str(nvs_handle, MQTT_NVS_KEY_PASSWORD, password, &required_size);
467 if (ret != ESP_OK) {
468 ESP_LOGD(TAG, "MQTT password not in NVS, using default");
469 strncpy(password, MQTT_DEFAULT_PASSWORD, password_len - 1);
470 password[password_len - 1] = '\0';
471 }
472
473 nvs_close(nvs_handle);
474 ESP_LOGI(TAG, "MQTT config loaded from NVS: host=%s, port=%d, username=%s",
475 host, *port, username[0] ? username : "(empty)");
476
477 return ESP_OK;
478}

489esp_err_t mqtt_save_config_to_nvs(const char *host, uint16_t port,
490 const char *username, const char *password) {
491 if (host == NULL || port == 0) {
492 ESP_LOGE(TAG, "Invalid parameters: host is NULL or port is 0");
493 return ESP_ERR_INVALID_ARG;
494 }
495
496 size_t host_len = strlen(host);
497 if (host_len == 0 || host_len > 127) {
498 ESP_LOGE(TAG, "Invalid host length: %zu (must be 1-127)", host_len);
499 return ESP_ERR_INVALID_ARG;
500 }
501
502 // Validate username/password if set
503 if (username != NULL) {
504 size_t username_len = strlen(username);
505 if (username_len > 63) {
506 ESP_LOGE(TAG, "Invalid username length: %zu (max 63)", username_len);
507 return ESP_ERR_INVALID_ARG;
508 }
509 }
510
511 if (password != NULL) {
512 size_t password_len = strlen(password);
513 if (password_len > 63) {
514 ESP_LOGE(TAG, "Invalid password length: %zu (max 63)", password_len);
515 return ESP_ERR_INVALID_ARG;
516 }
517 }
518
519 nvs_handle_t nvs_handle;
520 esp_err_t ret = nvs_open(MQTT_NVS_NAMESPACE, NVS_READWRITE, &nvs_handle);
521 if (ret != ESP_OK) {
522 ESP_LOGE(TAG, "Failed to open NVS: %s", esp_err_to_name(ret));
523 return ret;
524 }
525
526 // Save guest
527 ret = nvs_set_str(nvs_handle, MQTT_NVS_KEY_HOST, host);
528 if (ret != ESP_OK) {
529 ESP_LOGE(TAG, "Failed to set MQTT host in NVS: %s", esp_err_to_name(ret));
530 nvs_close(nvs_handle);
531 return ret;
532 }
533
534 // Save port
535 ret = nvs_set_u32(nvs_handle, MQTT_NVS_KEY_PORT, (uint32_t)port);
536 if (ret != ESP_OK) {
537 ESP_LOGE(TAG, "Failed to set MQTT port in NVS: %s", esp_err_to_name(ret));
538 nvs_close(nvs_handle);
539 return ret;
540 }
541
542 // Save username (if set, otherwise save empty string)
543 const char *username_to_save =
544 (username != NULL && strlen(username) > 0) ? username : "";
545 ret = nvs_set_str(nvs_handle, MQTT_NVS_KEY_USERNAME, username_to_save);
546 if (ret != ESP_OK) {
547 ESP_LOGE(TAG, "Failed to set MQTT username in NVS: %s",
548 esp_err_to_name(ret));
549 nvs_close(nvs_handle);
550 return ret;
551 }
552
553 // Save password (if set, otherwise save empty string)

571 nvs_close(nvs_handle);
572 ESP_LOGI(TAG, "MQTT config saved to NVS: host=%s, port=%d, username=%s", host,
573 port, username_to_save[0] ? username_to_save : "(empty)");

575 return ESP_OK;
576}

656static void ha_light_mqtt_event_handler(void *handler_args,
657 esp_event_base_t base, int32_t event_id,
658 void *event_data) {
659 esp_mqtt_event_handle_t event = (esp_mqtt_event_handle_t)event_data;
660 esp_mqtt_client_handle_t client = event->client;
661
662 switch ((esp_mqtt_event_id_t)event_id) {
663 case MQTT_EVENT_CONNECTED:
664 ESP_LOGI(TAG, "MQTT Connected");
665 mqtt_connected = true;
666 s_mqtt_down = false;
667
668 // Subscribe to light command topic
669 esp_mqtt_client_subscribe(client, HA_TOPIC_LIGHT_COMMAND, 0);
670 ESP_LOGI(TAG, "Subscribed to %s", HA_TOPIC_LIGHT_COMMAND);
671
672 /* Discovery and state go out one message per loop, not in this callback. */
673 s_disc_step = 0;
674 break;
675
676 case MQTT_EVENT_DISCONNECTED:
677 ESP_LOGI(TAG, "MQTT Disconnected");
678 mqtt_connected = false;
679 s_mqtt_down = true;
680 s_disc_step = -1;
681 break;
682
683 case MQTT_EVENT_DATA:
684 ESP_LOGI(TAG, "MQTT DATA: topic=%.*s, data=%.*s", event->topic_len,
685 event->topic, event->data_len, event->data);
686
687 // Handle command
688 char topic[128] = {0};
689 char data[512] = {0};
690
691 if (event->topic != NULL && event->data != NULL &&
692 event->topic_len > 0 && event->topic_len < (int)sizeof(topic) &&
693 event->data_len > 0 && event->data_len < (int)sizeof(data)) {
694 memcpy(topic, event->topic, event->topic_len);
695 memcpy(data, event->data, event->data_len);
696 ha_light_handle_mqtt_command(topic, data, event->data_len);
697 }
698 break;
699
700 case MQTT_EVENT_ERROR:

722static void ha_light_handle_mqtt_command(const char *topic, const char *data,
723 int data_len) {
724 if (strcmp(topic, HA_TOPIC_LIGHT_COMMAND) != 0) {
725 return; // Not our topic
726 }
727
728 // Parse JSON
729 cJSON *json = cJSON_ParseWithLength(data, data_len);
730 if (json == NULL) {
731 ESP_LOGE(TAG, "Failed to parse MQTT JSON");
732 return;
733 }
734
735 bool new_state = ha_light_state.state;
736 uint8_t new_r = ha_light_state.r, new_g = ha_light_state.g,
737 new_b = ha_light_state.b;
738 uint8_t new_brightness = ha_light_state.brightness;
739
740 cJSON *state_item = cJSON_GetObjectItem(json, "state");
741 if (cJSON_IsString(state_item)) {
742 if (strcasecmp(state_item->valuestring, "ON") == 0) {
743 new_state = true;
744 } else if (strcasecmp(state_item->valuestring, "OFF") == 0) {
745 new_state = false;
746 }
747 }
748
749 cJSON *brightness_item = cJSON_GetObjectItem(json, "brightness");
750 if (cJSON_IsNumber(brightness_item)) {
751 int brightness_val = (int)cJSON_GetNumberValue(brightness_item);
752 if (brightness_val < 0)
753 brightness_val = 0;
754 if (brightness_val > 255)
755 brightness_val = 255;
756 new_brightness = (uint8_t)brightness_val;
757 }
758
759 cJSON *color_item = cJSON_GetObjectItem(json, "color");
760 if (cJSON_IsObject(color_item)) {
761 cJSON *r = cJSON_GetObjectItem(color_item, "r");
762 cJSON *g = cJSON_GetObjectItem(color_item, "g");
763 cJSON *b = cJSON_GetObjectItem(color_item, "b");
764 if (cJSON_IsNumber(r) && cJSON_IsNumber(g) && cJSON_IsNumber(b)) {
765 new_r = (uint8_t)cJSON_GetNumberValue(r);
766 new_g = (uint8_t)cJSON_GetNumberValue(g);
767 new_b = (uint8_t)cJSON_GetNumberValue(b);
768 }
769 }
770
771 cJSON *rgb_item = cJSON_GetObjectItem(json, "rgb_color");
772 if (cJSON_IsArray(rgb_item) && cJSON_GetArraySize(rgb_item) >= 3) {
773 new_r = (uint8_t)cJSON_GetNumberValue(cJSON_GetArrayItem(rgb_item, 0));
774 new_g = (uint8_t)cJSON_GetNumberValue(cJSON_GetArrayItem(rgb_item, 1));
775 new_b = (uint8_t)cJSON_GetNumberValue(cJSON_GetArrayItem(rgb_item, 2));
776 }
777
778 cJSON_Delete(json);
779
780 if (ha_light_cmd_queue == NULL) {
781 return;
782 }
783
784 ha_light_command_t cmd = {0};
785 cmd.type = HA_CMD_MQTT_STATE;
786 cmd.u.web_lamp.state = new_state ? 1 : 0;
787 cmd.u.web_lamp.r = new_r;
788 cmd.u.web_lamp.g = new_g;
789 cmd.u.web_lamp.b = new_b;
790 cmd.u.web_lamp.brightness = new_brightness;
791 cmd.u.web_lamp.set_brightness = 1;
792 if (xQueueSend(ha_light_cmd_queue, &cmd, 0) != pdTRUE) {
793 ESP_LOGW(TAG, "MQTT state queue full");
794 }
795}

1061static void ha_discovery_pump(void) {
1062 int step = s_disc_step;
1063 if (step < 0 || !mqtt_connected || mqtt_client == NULL) {
1064 return;
1065 }
1066 if (esp_get_free_heap_size() < HA_MQTT_MIN_FREE_HEAP_BYTES) {
1067 return;
1068 }
1069 static int64_t s_disc_next_us;
1070 int64_t now = esp_timer_get_time();
1071 if (now < s_disc_next_us) {
1072 return;
1073 }
1074 const int trigger0 = 4;
1075 const int trigger_n =
1076 (int)(sizeof(s_game_events) / sizeof(s_game_events[0]));
1077 const int state_step = trigger0 + trigger_n;
1078 bool sent = false;
1079 if (step == 0) {
1080 sent = ha_light_publish_discovery();
1081 } else if (step == 1) {
1082 /* One QoS 1 so Home Assistant keeps the online state. */
1083 sent = esp_mqtt_client_publish(mqtt_client, HA_TOPIC_LIGHT_AVAILABILITY,
1084 HA_MQTT_PAYLOAD_ONLINE, 0, 1, 1) >= 0;
1085 } else if (step == 2) {
1086 sent = ha_light_publish_state();
1087 } else if (step == 3) {
1088 sent = ha_game_publish_sensor();
1089 } else if (step < state_step) {
1090 sent = ha_game_publish_trigger((unsigned)(step - trigger0));
1091 } else {
1092 sent = ha_game_publish_state();
1093 if (sent && s_disc_step == step) {
1094 s_disc_step = -1;
1095 }
1096 s_disc_next_us = now + 100000;
1097 return;
1098 }
1099 s_disc_next_us = now + 100000;
1100 /* A failed send stays on this step. Disconnect must not be overwritten. */
1101 if (sent && s_disc_step == step) {
1102 s_disc_step = step + 1;
1103 }
1104}

1146static esp_err_t ha_light_init_mqtt(void) {
1147 // Prevent multiple clients/memory leaks
1148 if (mqtt_client != NULL) {
1149 ESP_LOGI(TAG, "MQTT client already initialized, skipping re-init");
1150 return ESP_OK;
1151 }
1152
1153 size_t free_heap = esp_get_free_heap_size();
1154 if (free_heap < HA_MQTT_MIN_FREE_HEAP_BYTES) {
1155 ESP_LOGW(TAG,
1156 "MQTT deferred: free heap %zu B < %d B (need RAM for mqtt task)",
1157 free_heap, HA_MQTT_MIN_FREE_HEAP_BYTES);
1158 return ESP_ERR_NO_MEM;
1159 }
1160
1161 // Ensure config is loaded from NVS (first time only)
1162 mqtt_ensure_config_loaded();
1163
1164 // Build MQTT broker URI (buffer size 512 to avoid format-truncation warning)
1165 char mqtt_broker_uri[512];
1166 snprintf(mqtt_broker_uri, sizeof(mqtt_broker_uri), "mqtt://%s:%d",
1167 mqtt_config.host, mqtt_config.port);
1168
1169 // Generate unique Client ID combined from Base + MAC
1170 uint8_t mac_raw[6];
1171 esp_read_mac(mac_raw, ESP_MAC_WIFI_STA);
1172 char client_id[64];
1173 snprintf(client_id, sizeof(client_id), "%s-%02x%02x%02x%02x%02x%02x",
1174 HA_MQTT_CLIENT_ID, mac_raw[0], mac_raw[1], mac_raw[2], mac_raw[3],
1175 mac_raw[4], mac_raw[5]);
1176
1177 ESP_LOGI(TAG, "Initializing MQTT client: %s (ID: %s)", mqtt_broker_uri,
1178 client_id);
1179
1180 esp_mqtt_client_config_t mqtt_cfg = {
1181 .broker.address.uri = mqtt_broker_uri,
1182 .credentials.client_id = client_id,
1183 .credentials.username =
1184 mqtt_config.username[0] != '\0' ? mqtt_config.username : NULL,
1185 .credentials.authentication.password =
1186 mqtt_config.password[0] != '\0' ? mqtt_config.password : NULL,
1187 .session.last_will.topic = HA_TOPIC_LIGHT_AVAILABILITY,
1188 .session.last_will.msg = HA_MQTT_PAYLOAD_OFFLINE,
1189 .session.last_will.qos = 1,
1190 .session.last_will.retain = true,
1191 .session.keepalive = 120, // Increase keepalive to 120s for stability
1192 .network.timeout_ms = 4000,
1193 .network.disable_auto_reconnect = false,
1194 /* Incoming commands are small. Outgoing discovery fits in 768. */
1195 .buffer.size = 512,
1196 .buffer.out_size = HA_MQTT_PAYLOAD_MAX,
1197 };
1198
1199 mqtt_client = esp_mqtt_client_init(&mqtt_cfg);
1200 if (mqtt_client == NULL) {
1201 ESP_LOGE(TAG, "Failed to initialize MQTT client");
1202 return ESP_FAIL;
1203 }
1204
1205 esp_mqtt_client_register_event(mqtt_client, ESP_EVENT_ANY_ID,
1206 ha_light_mqtt_event_handler, NULL);
1207
1208 esp_err_t ret = esp_mqtt_client_start(mqtt_client);
1209 if (ret != ESP_OK) {
1210 ESP_LOGE(TAG, "Failed to start MQTT client: %s", esp_err_to_name(ret));
1211 esp_mqtt_client_destroy(mqtt_client);
1212 mqtt_client = NULL;
1213 return ret;
1214 }
1215
1216 ESP_LOGI(TAG, "MQTT client initialized and started");
1217 return ESP_OK;
1218}

1652void ha_light_task_start(void *pvParameters) {
1653 ESP_LOGI(TAG, "Starting HA Light Task...");
1654 s_ha_light_task_hdl = xTaskGetCurrentTaskHandle();
1655
1656 // Register with WDT
1657 esp_err_t ret = esp_task_wdt_add(NULL);
1658 if (ret != ESP_OK && ret != ESP_ERR_INVALID_ARG) {
1659 ESP_LOGW(TAG, "Failed to register with WDT: %s", esp_err_to_name(ret));
1660 }
1661
1662 task_running = true;
1663 current_mode = HA_MODE_GAME;
1664 last_activity_time_ms = esp_timer_get_time() / 1000;
1665
1666 ha_light_state_mutex = xSemaphoreCreateMutex();
1667 if (ha_light_state_mutex == NULL) {
1668 ESP_LOGE(TAG, "Failed to create ha_light_state mutex");
1669 }
1670
1671 ha_light_cmd_queue = xQueueCreate(10, sizeof(ha_light_command_t));
1672 if (ha_light_cmd_queue == NULL) {
1673 ESP_LOGE(TAG, "Failed to create HA command queue");
1674 }
1675
1676 lamp_nvs_load();
1677
1678 // Main loop
1679 // TickType_t last_wake_time = xTaskGetTickCount(); // Unused in queue-based
1680 // loop
1681
1682 for (;;) {
1683 // =========================================================================
1684 // 🛑 BOOT ANIMATION PROTECTION - BLOCKS HA OPERATION DURING BOOT
1685 // =========================================================================
1686 // If the boot animation is running, the HA task must not send LED commands!
1687 // (e.g. "turn off the light" on a fast WiFi connection)
1688 if (led_is_booting()) {
1689 chess_task_wdt_reset_safe(TAG);
1690 vTaskDelay(pdMS_TO_TICKS(10)); // 10ms - quick wakeup after fade_out!
1691 continue;
1692 }
1693 if (!s_boot_mode_applied) {
1694 s_boot_mode_applied = true;
1695 if (s_nvs_mode_found && s_nvs_mode == HA_MODE_HA &&
1696 current_mode != HA_MODE_HA) {
1697 ESP_LOGI(TAG, "[STAGING] restore persisted lamp mode");
1698 ha_light_switch_to_ha_mode();
1699 }
1700 }
1701 // =========================================================================
1702
1703 // Reset WDT
1704 chess_task_wdt_reset_safe(TAG);
1705
1706 ha_discovery_pump();
1707
1708 // Process queue with timeout (100ms) - replaces vTaskDelay
1709 ha_light_command_t cmd;
1710 if (ha_light_cmd_queue &&
1711 xQueueReceive(ha_light_cmd_queue, &cmd, pdMS_TO_TICKS(100)) == pdTRUE) {
1712 if (cmd.type == HA_CMD_WEB_LAMP) {
1713 s_pat_on = false;
1714 if (ha_light_state_mutex != NULL &&
1715 xSemaphoreTake(ha_light_state_mutex, pdMS_TO_TICKS(100)) ==
1716 pdTRUE) {
1717 remember_lit_color(ha_light_state.r, ha_light_state.g, ha_light_state.b,
1718 ha_light_state.brightness);
1719 ha_light_state.state = (cmd.u.web_lamp.state != 0);
1720 ha_light_state.r = cmd.u.web_lamp.r;
1721 ha_light_state.g = cmd.u.web_lamp.g;
1722 ha_light_state.b = cmd.u.web_lamp.b;
1723 if (cmd.u.web_lamp.set_brightness) {
1724 ha_light_state.brightness = cmd.u.web_lamp.brightness;
1725 }
1726 remember_lit_color(ha_light_state.r, ha_light_state.g, ha_light_state.b,
1727 ha_light_state.brightness);
1728 xSemaphoreGive(ha_light_state_mutex);
1729 }
1730 lamp_nvs_save();
1731 ha_light_switch_to_ha_mode();
1732 } else if (cmd.type == HA_CMD_PATTERN) {
1733 ha_pattern_apply(&cmd);
1734 } else if (cmd.type == HA_CMD_MQTT_STATE) {
1735 s_pat_on = false;
1736 bool was_on = ha_light_state.state;
1737 bool now_on_cmd = (cmd.u.web_lamp.state != 0);
1738 /* Capture the lit color before state becomes off, so the shrink wave
1739 * still has a disk to pull back into h1. */
1740 bool leaving_light =
1741 !now_on_cmd && was_on && (current_mode == HA_MODE_HA);
1742 if (leaving_light) {
1743 game_gesture_request_lamp_off_wave();
1744 }
1745 if (ha_light_state_mutex != NULL &&
1746 xSemaphoreTake(ha_light_state_mutex, pdMS_TO_TICKS(100)) ==
1747 pdTRUE) {
1748 was_on = ha_light_state.state;
1749 remember_lit_color(ha_light_state.r, ha_light_state.g, ha_light_state.b,
1750 ha_light_state.brightness);
1751 ha_light_state.state = (cmd.u.web_lamp.state != 0);
1752 ha_light_state.r = cmd.u.web_lamp.r;
1753 ha_light_state.g = cmd.u.web_lamp.g;
1754 ha_light_state.b = cmd.u.web_lamp.b;
1755 ha_light_state.brightness = cmd.u.web_lamp.brightness;
1756 remember_lit_color(ha_light_state.r, ha_light_state.g, ha_light_state.b,
1757 ha_light_state.brightness);
1758 xSemaphoreGive(ha_light_state_mutex);
1759 }
1760 lamp_nvs_save();
1761 bool now_on = (cmd.u.web_lamp.state != 0);
1762 /* Turning the lamp on leaves whatever is on the board, game included. */
1763 bool entering_light =
1764 now_on && (current_mode != HA_MODE_HA || !was_on);
1765 bool apply = true;
1766 if (!entering_light && current_mode == HA_MODE_GAME) {
1767 uint32_t now_ms = esp_timer_get_time() / 1000;
1768 uint32_t idle_ms = now_ms - last_activity_time_ms;
1769 apply = (idle_ms >= HA_ACTIVITY_TIMEOUT_COMMAND_MS ||
1770 game_get_move_count() == 0);
1771 if (!apply) {
1772 ESP_LOGI(TAG,
1773 "MQTT state stored (game active, moves=%lu, idle=%lu ms)",
1774 game_get_move_count(), idle_ms);
1775 }
1776 }
1777 if (apply && entering_light) {
1778 game_gesture_request_lamp_wave();
1779 } else if (leaving_light) {
1780 /* State is already off. The shrink wave owns the LEDs until it
1781 * commits HA mode, which then paints black. */
1782 ha_light_publish_state();
1783 } else if (apply) {
1784 ha_light_switch_to_ha_mode();
1785 } else {
1786 ha_light_publish_state();
1787 }
1788 } else if (cmd.type == HA_CMD_GAME_FACT) {
1789 ha_game_apply(cmd.u.web_lamp.state, cmd.u.web_lamp.r);
1790 } else if (cmd.type == HA_CMD_SET_MODE) {
1791 if (cmd.u.web_lamp.state == HA_MODE_HA) {
1792 /* Coming from the game, the lamp comes back in the last real color. */
1793 if (current_mode == HA_MODE_GAME && ha_light_state_mutex != NULL &&
1794 xSemaphoreTake(ha_light_state_mutex, pdMS_TO_TICKS(50)) ==
1795 pdTRUE) {
1796 if (!ha_light_state.state ||
1797 (ha_light_state.r | ha_light_state.g | ha_light_state.b) == 0) {
1798 ha_light_state.r = s_lit_r;
1799 ha_light_state.g = s_lit_g;
1800 ha_light_state.b = s_lit_b;
1801 ha_light_state.brightness = s_lit_br;
1802 ha_light_state.state = true;
1803 xSemaphoreGive(ha_light_state_mutex);
1804 lamp_nvs_save();
1805 } else {
1806 xSemaphoreGive(ha_light_state_mutex);
1807 }
1808 }
1809 ha_light_switch_to_ha_mode();
1810 } else {
1811 ha_light_switch_to_game_mode();
1812 }
1813 } else if (cmd.type == HA_CMD_ACTIVITY) {
1814 const char *activity_name = (const char *)cmd.u.data;
1815 // Reset activity timer in HA task context
1816 last_activity_time_ms = esp_timer_get_time() / 1000;
1817
1818 /* A bump that only renames a piece, or a piece put back on the same
1819 * square, is not a move. Only a real relocation leaves the lamp. */
1820 bool leave_lamp = false;
1821 if (activity_name != NULL &&
1822 (strcmp(activity_name, "move") == 0 ||
1823 strcmp(activity_name, "web_game_mode") == 0)) {
1824 leave_lamp = true;
1825 }
1826 if (current_mode == HA_MODE_HA && leave_lamp) {
1827 ESP_LOGI(TAG, "Switching from HA mode to GAME mode (activity: %s)",
1828 activity_name);
1829 ha_light_switch_to_game_mode();
1830 }
1831 if (activity_name != NULL && strcmp(activity_name, "move") == 0) {
1832 ha_game_apply(HA_GAME_MOVE, HA_REASON_NONE);
1833 }
1834
1835 // Check rate limit logic again here (in HA task context) to save MQTT
1836 // bandwidth
1837 static uint32_t last_mqtt_activity = 0;
1838 uint32_t now = esp_timer_get_time() / 1000;
1839
1840 if (now - last_mqtt_activity > 500) { // 500ms limit
1841 last_mqtt_activity = now;
1842
1843 if (mqtt_connected && mqtt_client != NULL &&
1844 ha_mqtt_set("{\"event\":\"%s\",\"timestamp_ms\":%lu}",
1845 activity_name ? activity_name : "unknown",
1846 (unsigned long)now)) {
1847 ha_mqtt_send(HA_TOPIC_GAME_ACTIVITY, 0);
1848 }
1849 }
1850 }
1851 chess_task_wdt_reset_safe(TAG);
1852 }
1853
1854 // POLL: Check WiFi STA status periodically (every 5 seconds)
1855 static uint32_t last_wifi_check = 0;
