127static esp_err_t journal_open(nvs_open_mode_t mode, nvs_handle_t *h) {
128 esp_err_t e = nvs_open(OTA_JOURNAL_NS, mode, h);
129 if (e == ESP_ERR_NVS_NOT_INITIALIZED) {
130 /* Never erase here: NVS holds Wi‑Fi, tokens and game settings. */
131 if (nvs_flash_init() == ESP_OK) {
132 e = nvs_open(OTA_JOURNAL_NS, mode, h);
133 }
134 }
135 return e;

138static bool journal_load(ota_journal_t *j) {
139 nvs_handle_t h;
140 if (journal_open(NVS_READONLY, &h) != ESP_OK) {
141 return false;
142 }
143 size_t len = sizeof(*j);
144 esp_err_t e = nvs_get_blob(h, OTA_JOURNAL_KEY, j, &len);
145 nvs_close(h);
146 if (e != ESP_OK || len != sizeof(*j) || j->ver != OTA_JOURNAL_VER) {
147 return false;
148 }
149 j->transport[sizeof(j->transport) - 1] = '\0';
150 j->target_slot[sizeof(j->target_slot) - 1] = '\0';
151 j->target_version[sizeof(j->target_version) - 1] = '\0';
152 j->reason[sizeof(j->reason) - 1] = '\0';
153 return true;
154}

156static void journal_save(void) {
157 ota_journal_t j;
158 memset(&j, 0, sizeof(j));
159 j.ver = OTA_JOURNAL_VER;
160 taskENTER_CRITICAL(&s_mux);
161 j.phase = (uint8_t)s_st.state;
162 j.seq = s_st.seq;
163 memcpy(j.transport, s_st.transport, sizeof(j.transport));
164 memcpy(j.target_slot, s_st.target_slot, sizeof(j.target_slot));
165 memcpy(j.target_version, s_st.target_version, sizeof(j.target_version));
166 memcpy(j.reason, s_st.reason, sizeof(j.reason));
167 taskEXIT_CRITICAL(&s_mux);
168
169 nvs_handle_t h;
170 esp_err_t e = journal_open(NVS_READWRITE, &h);
171 if (e == ESP_OK) {
172 e = nvs_set_blob(h, OTA_JOURNAL_KEY, &j, sizeof(j));
173 if (e == ESP_OK) {
174 e = nvs_commit(h);
175 }
176 nvs_close(h);
177 }
178 if (e != ESP_OK) {
179 ESP_LOGW(TAG, "OTA journal write failed: %s", esp_err_to_name(e));
180 }
181}

261static void deadline_cb(void *arg) {
262 (void)arg;
263 if (s_confirm_done) {
264 return;
265 }
266 ESP_LOGE(TAG,
267 "OTA: image not confirmed within %u s — marking invalid, rollback",
268 (unsigned)(OTA_GUARD_CONFIRM_DEADLINE_MS / 1000));
269 set_state(OTA_GUARD_ROLLED_BACK, "confirm_timeout");
270 journal_save();
271 staging_log_state("deadline");
272 esp_err_t e = esp_ota_mark_app_invalid_rollback_and_reboot();
273 ESP_LOGE(TAG, "rollback_and_reboot failed: %s — plain restart",
274 esp_err_to_name(e));
275 /* Reset while PENDING_VERIFY → bootloader aborts this image anyway. */
276 esp_restart();

279void ota_guard_boot_early(void) {
280 esp_reset_reason_t rr = esp_reset_reason();
281 if (s_rtc_magic != OTA_GUARD_RTC_MAGIC || rr == ESP_RST_POWERON) {
282 s_rtc_magic = OTA_GUARD_RTC_MAGIC;
283 s_rtc_crash_streak = 0;
284 s_rtc_fallback_done = 0;
285 } else if (is_crash_reset(rr)) {
286 s_rtc_crash_streak++;
287 } else if (rr == ESP_RST_BROWNOUT || rr == ESP_RST_EXT) {
288 /* Battery / external reset is not a firmware crash. */
289 s_rtc_crash_streak = 0;
290 }
291
292 memset(&s_st, 0, sizeof(s_st));
293 s_st.state = OTA_GUARD_IDLE;
294 s_st.confirmed = true;
295
296 const esp_partition_t *running = esp_ota_get_running_partition();
297 const esp_app_desc_t *app = esp_app_get_description();
298 if (running != NULL) {
299 safe_copy(s_st.running_slot, sizeof(s_st.running_slot), running->label);
300 }
301 if (app != NULL) {
302 safe_copy(s_st.running_version, sizeof(s_st.running_version), app->version);
303 }
304
305 esp_ota_img_states_t ist = ESP_OTA_IMG_UNDEFINED;
306 if (running != NULL &&
307 esp_ota_get_state_partition(running, &ist) == ESP_OK &&
308 ist == ESP_OTA_IMG_PENDING_VERIFY) {
309 s_st.rollback_active = true;
310 s_st.confirmed = false;
311 const esp_timer_create_args_t targs = {.callback = deadline_cb,
312 .name = "ota_confirm_dl"};
313 if (esp_timer_create(&targs, &s_deadline_timer) == ESP_OK) {
314 esp_timer_start_once(s_deadline_timer,
315 (uint64_t)OTA_GUARD_CONFIRM_DEADLINE_MS * 1000ULL);
316 }
317 ESP_LOGW(TAG,
318 "OTA: running %s is PENDING_VERIFY — confirm after %u s stable, "
319 "rollback after %u s",
320 s_st.running_slot, (unsigned)(OTA_GUARD_CONFIRM_STABLE_MS / 1000),
321 (unsigned)(OTA_GUARD_CONFIRM_DEADLINE_MS / 1000));
322 } else if (ist == ESP_OTA_IMG_NEW) {
323 /* Bootloader without rollback support never moves NEW → PENDING_VERIFY. */
324 s_st.confirmed = false;
325 ESP_LOGW(TAG, "OTA: running %s in state NEW — bootloader built without "
326 "APP_ROLLBACK, no automatic rollback on this board",
327 s_st.running_slot);
328 }
329 ESP_LOGI(TAG, "OTA guard: reset=%d crash_streak=%u fallback_done=%u",
330 (int)rr, (unsigned)s_rtc_crash_streak, (unsigned)s_rtc_fallback_done);
331}

334static bool try_crash_loop_fallback(const char *reason) {
335 if (s_rtc_fallback_done) {
336 ESP_LOGW(TAG, "OTA: crash loop, fallback already used this power cycle");
337 return false;
338 }
339 if (!esp_ota_check_rollback_is_possible()) {
340 ESP_LOGW(TAG, "OTA: crash loop, no valid image in the other slot");
341 return false;
342 }
343 const esp_partition_t *other = esp_ota_get_next_update_partition(NULL);
344 esp_app_desc_t d;
345 if (other == NULL || esp_ota_get_partition_description(other, &d) != ESP_OK) {
346 return false;
347 }
348 char why[48];
349 if (ota_guard_validate_desc(d.project_name, NULL, why, sizeof(why)) != ESP_OK) {
350 ESP_LOGW(TAG, "OTA: crash loop, other slot rejected: %s", why);
351 return false;
352 }
353 ESP_LOGE(TAG, "OTA: %u crash resets in a row — switching to %s (%s)",
354 (unsigned)s_rtc_crash_streak, other->label, d.version);
355 set_target(s_st.running_slot, s_st.running_version);
356 set_state(OTA_GUARD_ROLLED_BACK, reason);
357 journal_save();
358 s_rtc_fallback_done = 1;
359 s_rtc_crash_streak = 0;
360 esp_err_t e = esp_ota_set_boot_partition(other);
361 if (e != ESP_OK) {
362 ESP_LOGE(TAG, "OTA: set_boot_partition(%s) failed: %s", other->label,
363 esp_err_to_name(e));
364 return false;
365 }
366 staging_log_state("crash_loop_fallback");
367 vTaskDelay(pdMS_TO_TICKS(200));
368 esp_restart();
369 return true;
370}

372void ota_guard_init(void) {
373 ota_journal_t j;
374 bool have = journal_load(&j);
375 const esp_partition_t *running = esp_ota_get_running_partition();
376
377 if (!have) {
378 const esp_partition_t *inv = esp_ota_get_last_invalid_partition();
379 if (inv != NULL && inv != running) {
380 esp_app_desc_t d;
381 memset(&d, 0, sizeof(d));
382 (void)esp_ota_get_partition_description(inv, &d);
383 set_target(inv->label, d.version);
384 set_state(OTA_GUARD_ROLLED_BACK, "bootloader_rollback");
385 }
386 } else {
387 taskENTER_CRITICAL(&s_mux);
388 s_st.seq = j.seq;
389 memcpy(s_st.transport, j.transport, sizeof(s_st.transport));
390 memcpy(s_st.target_slot, j.target_slot, sizeof(s_st.target_slot));
391 memcpy(s_st.target_version, j.target_version, sizeof(s_st.target_version));
392 memcpy(s_st.reason, j.reason, sizeof(s_st.reason));
393 taskEXIT_CRITICAL(&s_mux);
394
395 bool on_target = running != NULL && j.target_slot[0] != '\0' &&
396 strcmp(running->label, j.target_slot) == 0;
397 switch ((ota_guard_state_t)j.phase) {
398 case OTA_GUARD_RECEIVING:
399 set_state(OTA_GUARD_ABORTED, "reset_during_transfer");
400 journal_save();
401 break;
402 case OTA_GUARD_VERIFYING:
403 case OTA_GUARD_REBOOTING:
404 if (on_target) {
405 taskENTER_CRITICAL(&s_mux);
406 s_st.awaiting_confirm = true;
407 s_st.percent = 100;
408 taskEXIT_CRITICAL(&s_mux);
409 set_state(OTA_GUARD_VERIFYING, "awaiting_confirm");
410 } else {
411 const esp_partition_t *t = esp_partition_find_first(
412 ESP_PARTITION_TYPE_APP, ESP_PARTITION_SUBTYPE_ANY, j.target_slot);
413 esp_ota_img_states_t ts = ESP_OTA_IMG_UNDEFINED;
414 bool failed_boot = t != NULL &&
415 esp_ota_get_state_partition(t, &ts) == ESP_OK &&
416 (ts == ESP_OTA_IMG_ABORTED || ts == ESP_OTA_IMG_INVALID);
417 if (failed_boot) {
418 set_state(OTA_GUARD_ROLLED_BACK, "reset_before_confirm");
419 } else {
420 set_state(OTA_GUARD_ABORTED, "not_activated");
421 }
422 journal_save();
423 }
424 break;
425 case OTA_GUARD_SUCCESS:
426 if (on_target) {
427 set_state(OTA_GUARD_SUCCESS, NULL);
428 } else {
429 set_state(OTA_GUARD_IDLE, "running_other_slot");
430 }
431 break;
432 case OTA_GUARD_ABORTED:
433 case OTA_GUARD_ROLLED_BACK:
434 case OTA_GUARD_ERROR:
435 set_state((ota_guard_state_t)j.phase, NULL);
436 break;
437 default:
438 set_state(OTA_GUARD_IDLE, NULL);
439 break;
440 }
441 }
442
443 staging_log_state("boot");
444
445 if (s_rtc_crash_streak >= OTA_GUARD_CRASH_LOOP_LIMIT && !s_st.rollback_active) {
446 (void)try_crash_loop_fallback("crash_loop");
447 }
448}

450bool ota_guard_confirm_if_healthy(bool transport_ready) {
451 if (s_confirm_done) {
452 return true;
453 }
454 int64_t up_ms = esp_timer_get_time() / 1000;
455 if (up_ms < OTA_GUARD_CONFIRM_STABLE_MS || !transport_ready) {
456 return false;
457 }
458
459 ota_guard_status_t s;
460 ota_guard_get_status(&s);
461 if (s.rollback_active || s.awaiting_confirm || !s.confirmed) {
462 esp_err_t e = esp_ota_mark_app_valid_cancel_rollback();
463 if (e != ESP_OK) {
464 ESP_LOGE(TAG, "OTA: mark_app_valid failed: %s (retry)", esp_err_to_name(e));
465 return false;
466 }
467 }
468 if (s_deadline_timer != NULL) {
469 esp_timer_stop(s_deadline_timer);
470 }
471 s_rtc_crash_streak = 0;
472 s_rtc_fallback_done = 0;
473 s_confirm_done = true;
474
475 bool report_success = false;
476 taskENTER_CRITICAL(&s_mux);
477 s_st.confirmed = true;
478 if (s_st.awaiting_confirm && !s_live && s_st.state == OTA_GUARD_VERIFYING) {
479 s_st.awaiting_confirm = false;
480 s_st.state = OTA_GUARD_SUCCESS;
481 safe_copy(s_st.reason, sizeof(s_st.reason), "confirmed");
482 safe_copy(s_st.target_version, sizeof(s_st.target_version),
483 s_st.running_version);
484 report_success = true;
485 }
486 taskEXIT_CRITICAL(&s_mux);
487
488 if (report_success) {
489 journal_save();
490 ESP_LOGI(TAG, "OTA: new image %s (%s) confirmed after %lld ms — success",
491 s.running_slot, s.running_version, (long long)up_ms);
492 } else {
493 ESP_LOGI(TAG, "OTA guard: stable start confirmed (%lld ms)", (long long)up_ms);
494 }
495 staging_log_state("confirm");
496 return true;
497}

499void ota_guard_on_init_failure(const char *what) {
500 ESP_LOGE(TAG, "OTA guard: init failure (%s)", what != NULL ? what : "?");
501 ota_guard_status_t s;
502 ota_guard_get_status(&s);
503 if (s.rollback_active && !s_confirm_done) {
504 set_state(OTA_GUARD_ROLLED_BACK, "init_failed");
505 journal_save();
506 staging_log_state("init_failed");
507 esp_err_t e = esp_ota_mark_app_invalid_rollback_and_reboot();
508 ESP_LOGE(TAG, "rollback_and_reboot failed: %s — plain restart",
509 esp_err_to_name(e));
510 esp_restart();
511 }
512 if (s_rtc_fallback_done || !esp_ota_check_rollback_is_possible()) {
513 return;
514 }
515 s_rtc_crash_streak++;
516 if (s_rtc_crash_streak >= OTA_GUARD_CRASH_LOOP_LIMIT) {
517 (void)try_crash_loop_fallback("crash_loop_init_failed");
518 return;
519 }
520 ESP_LOGW(TAG, "OTA guard: restart %u/%u before falling back to other slot",
521 (unsigned)s_rtc_crash_streak, (unsigned)OTA_GUARD_CRASH_LOOP_LIMIT);
522 vTaskDelay(pdMS_TO_TICKS(3000));
523 esp_restart();
524}
