40move_error_t game_is_valid_move(const chess_move_t *move) {
41 if (!move) {
42 return MOVE_ERROR_INVALID_MOVE_STRUCTURE;
43 }
44
45 if (!game_is_valid_position(move->from_row, move->from_col) ||
46 !game_is_valid_position(move->to_row, move->to_col)) {
47 return MOVE_ERROR_OUT_OF_BOUNDS;
48 }
49
50 if (!game_active) {
51 return MOVE_ERROR_GAME_NOT_ACTIVE;
52 }
53
54 // Check if source position has a piece
55 piece_t source_piece = game_get_piece(move->from_row, move->from_col);
56
57 // If source_piece is empty (the king is raised during the resignation timer),
58 // use move->piece instead of source_piece
59 if (source_piece == PIECE_EMPTY) {
60 // If resignation timer is active and move->piece is king, use
61 // move->piece
62 if (resignation_state.active && move->piece != PIECE_EMPTY &&
63 (move->piece == PIECE_WHITE_KING || move->piece == PIECE_BLACK_KING) &&
64 resignation_state.king_row == move->from_row &&
65 resignation_state.king_col == move->from_col) {
66 source_piece = move->piece;
67 ESP_LOGI(TAG,
68 "🏰 game_is_valid_move: Using move->piece (%d) because king is "
69 "lifted (resignation timer active)",
70 move->piece);
71 } else {
72 return MOVE_ERROR_NO_PIECE;
73 }
74 }
75
76 // Check if source piece belongs to current player
77 if ((current_player == PLAYER_WHITE && !game_is_white_piece(source_piece)) ||
78 (current_player == PLAYER_BLACK && !game_is_black_piece(source_piece))) {
79 return MOVE_ERROR_WRONG_COLOR;
80 }
81
82 // Check if destination is occupied by own piece
83 piece_t dest_piece = game_get_piece(move->to_row, move->to_col);
84 if (dest_piece != PIECE_EMPTY &&
85 game_is_same_color(source_piece, dest_piece)) {
86 return MOVE_ERROR_DESTINATION_OCCUPIED;
87 }
88
89 // CHECKING THE ROLL IN PROGRESS - BEFORE validating the figure's movement
90 // If it's a correct cast rook move, skip the standard move validation
91 // pieces (which would otherwise find the path blocked by the king)
92 bool is_castling_rook_move = false;
93 if (castling_state.in_progress) {
94 ESP_LOGI(TAG, "🏰 game_is_valid_move: Castling in progress - checking if "
95 "this is the expected rook move");
96 ESP_LOGI(TAG, "🏰 Expected: from %c%d to %c%d, actual: from %c%d to %c%d",
97 'a' + castling_state.rook_from_col,
98 castling_state.rook_from_row + 1, 'a' + castling_state.rook_to_col,
99 castling_state.rook_to_row + 1, 'a' + move->from_col,
100 move->from_row + 1, 'a' + move->to_col, move->to_row + 1);
101
102 // We expect a rook move
103 if (move->from_row != castling_state.rook_from_row ||
104 move->from_col != castling_state.rook_from_col) {
105 ESP_LOGE(TAG, "❌ Castling in progress - expected rook move from %c%d",
106 'a' + castling_state.rook_from_col,
107 castling_state.rook_from_row + 1);
108 return MOVE_ERROR_CASTLING_BLOCKED; // We will use the existing error code
109 }
110
111 if (move->to_row != castling_state.rook_to_row ||
112 move->to_col != castling_state.rook_to_col) {
113 ESP_LOGE(
114 TAG, "❌ Incorrect rook destination during castling - expected %c%d",
115 'a' + castling_state.rook_to_col, castling_state.rook_to_row + 1);
116 return MOVE_ERROR_CASTLING_BLOCKED; // We will use the existing error code
117 }
118
119 // This is a proper cast rook move - skip the standard move validation
120 // figurines
121 is_castling_rook_move = true;
122 ESP_LOGI(TAG, "✅ Correct rook move for castling completion - skipping "
123 "standard piece validation (path would be blocked by king)");
124 }
125
126 // Validate move based on piece type (ONLY if casting is NOT tower move)
127 if (!is_castling_rook_move) {
128 move_error_t piece_error =
129 game_validate_piece_move_enhanced(move, source_piece);
130 if (piece_error != MOVE_ERROR_NONE) {
131 return piece_error;
132 }
133 } else {
134 ESP_LOGI(TAG,
135 "🏰 Skipping standard piece validation for castling rook move");
136 }
137
138 // Check if move would leave king in check
139 if (game_would_move_leave_king_in_check(move)) {
140 return MOVE_ERROR_KING_IN_CHECK;
141 }
142
143 return MOVE_ERROR_NONE;
144}

217move_error_t game_validate_pawn_move_enhanced(const chess_move_t *move,
218 piece_t piece) {
219 int row_diff = move->to_row - move->from_row;
220 int col_diff = move->to_col - move->from_col;
221 int abs_col_diff = abs(col_diff);
222
223 bool is_white = game_is_white_piece(piece);
224 int direction = is_white ? 1 : -1;
225 int start_row = is_white ? 1 : 6;
226
227 // Logovani validace pesce
228 char from_sq[3], to_sq[3];
229 game_coords_to_square(move->from_row, move->from_col, from_sq);
230 game_coords_to_square(move->to_row, move->to_col, to_sq);
231 piece_t dest_piece = game_get_piece(move->to_row, move->to_col);
232
233 // Forward move
234 if (col_diff == 0) {
235 // Check if moving backwards (invalid for pawns)
236 if (row_diff * direction < 0) {
237 ESP_LOGD(TAG, "❌ Pawn %s→%s: cannot move backwards", from_sq, to_sq);
238 return MOVE_ERROR_INVALID_PATTERN;
239 }
240
241 // Single square move
242 if (row_diff == direction) {
243 if (game_is_empty(move->to_row, move->to_col)) {
244 return MOVE_ERROR_NONE;
245 } else {
246 ESP_LOGD(TAG, "🔍 Pawn %s->%s: blocked by piece", from_sq, to_sq);
247 return MOVE_ERROR_BLOCKED_PATH;
248 }
249 }
250
251 // Double square move from starting position
252 if (row_diff == 2 * direction && move->from_row == start_row) {
253 if (game_is_empty(move->from_row + direction, move->from_col) &&
254 game_is_empty(move->to_row, move->to_col)) {
255 return MOVE_ERROR_NONE;
256 } else {
257 ESP_LOGD(TAG, "🔍 Pawn %s->%s: blocked path double move", from_sq,
258 to_sq);
259 return MOVE_ERROR_BLOCKED_PATH;
260 }
261 }
262
263 ESP_LOGD(TAG, "🔍 Pawn %s->%s: invalid forward distance", from_sq, to_sq);
264 return MOVE_ERROR_INVALID_PATTERN;
265 }
266
267 // Capture move (diagonal)
268 if (abs_col_diff == 1 && row_diff == direction) {
269 if (dest_piece != PIECE_EMPTY && !game_is_same_color(piece, dest_piece)) {
270 ESP_LOGD(TAG, "✅ Pawn %s→%s: valid capture of %s", from_sq, to_sq,
271 game_get_piece_name(dest_piece));
272 return MOVE_ERROR_NONE;
273 }
274
275 // Check for en passant
276 if (game_is_en_passant_possible(move)) {
277 ESP_LOGD(TAG, "✅ Pawn %s→%s: valid en passant", from_sq, to_sq);
278 return MOVE_ERROR_NONE;
279 }
280
281 if (dest_piece == PIECE_EMPTY) {
282 ESP_LOGD(TAG, "❌ Pawn %s→%s: diagonal to empty square (not en passant)",
283 from_sq, to_sq);
284 } else {
285 ESP_LOGD(TAG, "❌ Pawn %s→%s: trying to capture own piece", from_sq,
286 to_sq);
287 }
288 return MOVE_ERROR_INVALID_PATTERN;
289 }
290
291 ESP_LOGD(TAG,
292 "❌ Pawn %s→%s: invalid move pattern (col_diff=%d, row_diff=%d)",
293 from_sq, to_sq, col_diff, row_diff);
294 return MOVE_ERROR_INVALID_PATTERN;
295}

340move_error_t game_validate_bishop_move_enhanced(const chess_move_t *move) {
341 int row_diff = move->to_row - move->from_row;
342 int col_diff = move->to_col - move->from_col;
343 int abs_row_diff = abs(row_diff);
344 int abs_col_diff = abs(col_diff);
345
346 // Bishop moves diagonally
347 if (abs_row_diff != abs_col_diff) {
348 return MOVE_ERROR_INVALID_PATTERN;
349 }
350
351 // Check if path is blocked
352 // Originally: while (...&& ...) - terminated prematurely
353 // New: while (...|| ...) - checks until the destination
354 int row_step = (row_diff > 0) ? 1 : -1;
355 int col_step = (col_diff > 0) ? 1 : -1;
356
357 int current_row = move->from_row + row_step;
358 int current_col = move->from_col + col_step;
359
360 // SAFETY: Prevent infinite loop (max 8 squares on diagonal)
361 int steps = 0;
362 while ((current_row != move->to_row || current_col != move->to_col) &&
363 steps < 8) {
364 if (!game_is_empty(current_row, current_col)) {
365 ESP_LOGD(TAG, "🔍 Bishop blocked at %c%d", 'a' + current_col,
366 current_row + 1);
367 return MOVE_ERROR_BLOCKED_PATH;
368 }
369 current_row += row_step;
370 current_col += col_step;
371 steps++;
372 }
373
374 // Safety check: if loop ended without reaching target, something is wrong
375 if (steps >= 8 &&
376 (current_row != move->to_row || current_col != move->to_col)) {
377 ESP_LOGE(TAG, "🔍 Bishop validation error: loop exceeded max steps");
378 return MOVE_ERROR_INVALID_PATTERN;
379 }
380
381 return MOVE_ERROR_NONE;
382}

389move_error_t game_validate_rook_move_enhanced(const chess_move_t *move) {
390 int row_diff = move->to_row - move->from_row;
391 int col_diff = move->to_col - move->from_col;
392
393 // Rook moves horizontally or vertically
394 if (row_diff != 0 && col_diff != 0) {
395 return MOVE_ERROR_INVALID_PATTERN;
396 }
397
398 // Check if path is blocked
399 if (row_diff == 0) {
400 // Horizontal move
401 int col_step = (col_diff > 0) ? 1 : -1;
402 int current_col = move->from_col + col_step;
403
404 while (current_col != move->to_col) {
405 if (!game_is_empty(move->from_row, current_col)) {
406 return MOVE_ERROR_BLOCKED_PATH;
407 }
408 current_col += col_step;
409 }
410 } else {
411 // Vertical move
412 int row_step = (row_diff > 0) ? 1 : -1;
413 int current_row = move->from_row + row_step;
414
415 while (current_row != move->to_row) {
416 if (!game_is_empty(current_row, move->from_col)) {
417 return MOVE_ERROR_BLOCKED_PATH;
418 }
419 current_row += row_step;
420 }
421 }
422
423 return MOVE_ERROR_NONE;
424}

514bool game_would_move_leave_king_in_check(const chess_move_t *move) {
515 if (!move)
516 return true; // Safety check
517
518 // Save original board state
519 piece_t original_from_piece = board[move->from_row][move->from_col];
520 piece_t original_to_piece = board[move->to_row][move->to_col];
521
522 // If from_piece is empty (the king is raised during the resignation timer),
523 // use move->piece instead of original_from_piece
524 if (original_from_piece == PIECE_EMPTY && move->piece != PIECE_EMPTY) {
525 original_from_piece = move->piece;
526 ESP_LOGD(TAG,
527 "🏰 game_would_move_leave_king_in_check: Using move->piece (%d) "
528 "because from square is empty (king lifted)",
529 move->piece);
530 }
531
532 // Check if this is an en passant move
533 bool is_en_passant = game_is_en_passant_possible(move);
534 piece_t original_en_passant_piece = PIECE_EMPTY;
535 uint8_t en_passant_victim_row_local = 0;
536 uint8_t en_passant_victim_col_local = 0;
537
538 bool is_castling = game_is_king_castling_move(move, original_from_piece);
539 piece_t original_rook = PIECE_EMPTY;
540 int rook_row = move->from_row;
541 int rook_from_col = -1;
542 int rook_to_col = -1;
543
544 // If en passant, save the victim pawn position
545 if (is_en_passant) {
546 original_en_passant_piece =
547 board[en_passant_victim_row][en_passant_victim_col];
548 en_passant_victim_row_local = en_passant_victim_row;
549 en_passant_victim_col_local = en_passant_victim_col;
550 // Remove the victim pawn for simulation
551 board[en_passant_victim_row][en_passant_victim_col] = PIECE_EMPTY;
552 }
553
554 if (is_castling) {
555 bool kingside = move->to_col > move->from_col;
556 rook_from_col = kingside ? 7 : 0;
557 rook_to_col = kingside ? 5 : 3;
558 original_rook = board[rook_row][rook_from_col];
559 board[rook_row][rook_from_col] = PIECE_EMPTY;
560 board[rook_row][rook_to_col] = original_rook;
561 }
562
563 piece_t placed = original_from_piece;
564 if (original_from_piece == PIECE_WHITE_PAWN && move->to_row == 7) {
565 placed = game_check_sim_promoted_piece(original_from_piece);
566 } else if (original_from_piece == PIECE_BLACK_PAWN && move->to_row == 0) {
567 placed = game_check_sim_promoted_piece(original_from_piece);
568 }
569
570 // Make the move temporarily
571 board[move->to_row][move->to_col] = placed;
572 board[move->from_row][move->from_col] = PIECE_EMPTY;
573
574 // Determine which player's king to check
575 player_t player =
576 (game_is_white_piece(original_from_piece)) ? PLAYER_WHITE : PLAYER_BLACK;
577
578 // Check if king is in check after this move
579 bool king_in_check = game_is_king_in_check(player);
580
581 // Undo the move
582 board[move->from_row][move->from_col] = original_from_piece;
583 board[move->to_row][move->to_col] = original_to_piece;
584
585 // Restore en passant victim if it was removed
586 if (is_en_passant) {
587 board[en_passant_victim_row_local][en_passant_victim_col_local] =
588 original_en_passant_piece;
589 }
590
591 if (is_castling) {
592 board[rook_row][rook_to_col] = PIECE_EMPTY;
593 board[rook_row][rook_from_col] = original_rook;
594 }
595
596 return king_in_check;
597}

613bool game_is_en_passant_possible(const chess_move_t *move) {
614 if (!has_last_move) {
615 ESP_LOGD(TAG, "🔍 En passant: no last move");
616 return false;
617 }
618
619 // Check if the last move was a pawn moving two squares
620 piece_t last_piece = board[last_move_to_row][last_move_to_col];
621 bool last_was_pawn =
622 (last_piece == PIECE_WHITE_PAWN || last_piece == PIECE_BLACK_PAWN);
623
624 if (!last_was_pawn) {
625 ESP_LOGD(TAG, "🔍 En passant: last move was not pawn");
626 return false;
627 }
628
629 // Check if last move was a double square move
630 int last_row_diff = abs((int)last_move_to_row - (int)last_move_from_row);
631 if (last_row_diff != 2) {
632 ESP_LOGD(TAG, "🔍 En passant: last move was not 2 squares (diff=%d)",
633 last_row_diff);
634 return false;
635 }
636
637 // Check if the attacking pawn is on the same row as the current one
638 // moved pawn
639 if (move->from_row != last_move_to_row) {
640 ESP_LOGD(TAG,
641 "🔍 En passant: attacking pawn not on same row (from_row=%d, "
642 "last_to_row=%d)",
643 move->from_row, last_move_to_row);
644 return false;
645 }
646
647 // Check if the attacking pawn is exactly next to the currently moved one
648 // on foot
649 int col_diff = abs((int)move->from_col - (int)last_move_to_col);
650 if (col_diff != 1) {
651 ESP_LOGD(TAG, "🔍 En passant: pawn not adjacent (col_diff=%d)", col_diff);
652 return false;
653 }
654
655 // En passant field is always midway between from and to (simple mean)
656 int en_passant_target_row = (last_move_from_row + last_move_to_row) / 2;
657
658 ESP_LOGD(TAG,
659 "🔍 En passant check: last=%c%d→%c%d, target=%c%d, move=%c%d→%c%d",
660 'a' + last_move_from_col, last_move_from_row + 1,
661 'a' + last_move_to_col, last_move_to_row + 1, 'a' + last_move_to_col,
662 en_passant_target_row + 1, 'a' + move->from_col, move->from_row + 1,
663 'a' + move->to_col, move->to_row + 1);
664
665 if (move->to_row == en_passant_target_row &&
666 move->to_col == last_move_to_col) {
667 ESP_LOGD(TAG, "✅ En passant VALID!");
668 return true;
669 }
670
671 ESP_LOGD(TAG,
672 "❌ En passant: target mismatch (to_row=%d vs target=%d, to_col=%d "
673 "vs last_col=%d)",
674 move->to_row, en_passant_target_row, move->to_col, last_move_to_col);
675 return false;
676}

700move_error_t game_validate_castling(const chess_move_t *move) {
701 if (!move) {
702 return MOVE_ERROR_INVALID_MOVE_STRUCTURE;
703 }
704
705 piece_t piece = move->piece;
706 bool is_white = game_is_white_piece(piece);
707
708 // Check if it's a king
709 if (piece != PIECE_WHITE_KING && piece != PIECE_BLACK_KING) {
710 return MOVE_ERROR_INVALID_PATTERN;
711 }
712
713 // Check if king has moved
714 if ((is_white && white_king_moved) || (!is_white && black_king_moved)) {
715 ESP_LOGD(TAG, "❌ Castling: king has moved");
716 return MOVE_ERROR_CASTLING_BLOCKED;
717 }
718
719 // If the resignation timer is active and the king is raised,
720 // use the saved king position from resignation_state instead of checking the board
721 int king_row = is_white ? 0 : 7;
722 int expected_king_col = 4; // e-file
723
724 // If the resignation timer is active and the king belongs to the current player,
725 // use saved king position from resignation_state
726 if (resignation_state.active &&
727 resignation_state.player == (is_white ? PLAYER_WHITE : PLAYER_BLACK)) {
728 // King is raised - use saved position from resignation_state
729 king_row = resignation_state.king_row;
730 expected_king_col = resignation_state.king_col;
731 ESP_LOGI(TAG,
732 "🏰 Castling validation: Using stored king position from "
733 "resignation_state: %c%d",
734 'a' + expected_king_col, king_row + 1);
735 }
736
737 // Checking that from_row/from_col matches the expected position of the king
738 // (either from board or resignation_state)
739 if (move->from_row != king_row || move->from_col != expected_king_col) {
740 ESP_LOGD(
741 TAG,
742 "❌ Castling: king not in starting position (from %c%d, expected %c%d)",
743 'a' + move->from_col, move->from_row + 1, 'a' + expected_king_col,
744 king_row + 1);
745 return MOVE_ERROR_CASTLING_BLOCKED;
746 }
747
748 // Determine castling direction
749 bool is_kingside = (move->to_col == 6);
750 bool is_queenside = (move->to_col == 2);
751
752 if (!is_kingside && !is_queenside) {
753 return MOVE_ERROR_INVALID_PATTERN;
754 }
755
756 // Determine rook position
757 int rook_col = is_kingside ? 7 : 0;
758
759 // Check if rook exists at expected position
760 piece_t expected_rook = is_white ? PIECE_WHITE_ROOK : PIECE_BLACK_ROOK;
761 piece_t rook_piece = board[king_row][rook_col];
762 if (rook_piece != expected_rook) {
763 ESP_LOGD(TAG, "❌ Castling: rook not found at expected position %c%d",
764 'a' + rook_col, king_row + 1);
765 return MOVE_ERROR_CASTLING_BLOCKED;
766 }
767
768 // Check if corresponding rook has moved
769 if (is_white) {
770 if (is_kingside && white_rook_h_moved) {
771 ESP_LOGD(TAG, "❌ Castling: kingside rook has moved");
772 return MOVE_ERROR_CASTLING_BLOCKED;
773 }
774 if (is_queenside && white_rook_a_moved) {
775 ESP_LOGD(TAG, "❌ Castling: queenside rook has moved");
776 return MOVE_ERROR_CASTLING_BLOCKED;
777 }
778 } else {
779 if (is_kingside && black_rook_h_moved) {
780 ESP_LOGD(TAG, "❌ Castling: kingside rook has moved");
781 return MOVE_ERROR_CASTLING_BLOCKED;
782 }
783 if (is_queenside && black_rook_a_moved) {
784 ESP_LOGD(TAG, "❌ Castling: queenside rook has moved");
785 return MOVE_ERROR_CASTLING_BLOCKED;
786 }
787 }
788
789 // Check if squares between king and rook are empty
790 // rook_col already defined above
791 int start_col =
792 (move->from_col < rook_col) ? move->from_col + 1 : rook_col + 1;
793 int end_col = (move->from_col < rook_col) ? rook_col : move->from_col;
794
795 for (int col = start_col; col < end_col; col++) {
796 if (!game_is_empty(king_row, col)) {
797 return MOVE_ERROR_CASTLING_BLOCKED;
798 }
799 }
800
801 // Check if king is currently in check
802 if (game_is_king_in_check(is_white ? PLAYER_WHITE : PLAYER_BLACK)) {
803 return MOVE_ERROR_CASTLING_BLOCKED;
804 }
805
806 // Check if king would move through check or end in check
807 int step = is_kingside ? 1 : -1;
808 for (int col = move->from_col; col != move->to_col + step; col += step) {
809 if (col == move->from_col)
810 continue; // Skip starting position
811
812 // Create temporary move to test this square
813 chess_move_t temp_move = {.from_row = move->from_row,
814 .from_col = move->from_col,
815 .to_row = move->to_row,
816 .to_col = col,
817 .piece = piece,
818 .captured_piece = PIECE_EMPTY,
819 .timestamp = 0};
820
821 if (game_would_move_leave_king_in_check(&temp_move)) {
822 return MOVE_ERROR_CASTLING_BLOCKED;
823 }
824 }
825
826 return MOVE_ERROR_NONE;
827}
