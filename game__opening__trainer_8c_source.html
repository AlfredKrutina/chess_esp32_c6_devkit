114static bool opening_parse_uci(const char *uci, char *from, char *to) {
115 if (uci == NULL || from == NULL || to == NULL) {
116 return false;
117 }
118 size_t len = strlen(uci);
119 if (len < 4) {
120 return false;
121 }
122 from[0] = uci[0];
123 from[1] = uci[1];
124 from[2] = '\0';
125 to[0] = uci[2];
126 to[1] = uci[3];
127 to[2] = '\0';
128 return true;
129}

164static void opening_show_hint_led(void) {

166 opening_state.awaiting_opponent_physical) {

182 opening_state.mode == OPENING_MODE_TIMED || guidance == 1);
183 uint8_t from_led = 64;
184 if (!to_only && guidance >= 2 &&
185 game_notation_to_coords(opening_state.expected_from, &from_row, &from_col)) {
186 from_led = chess_pos_to_led_index(from_row, from_col);

189 game_event_post_board_anim(GAME_EVT_ANIM_HINT_PAIR, from_led, to_led);

214bool game_opening_apply_uci(const char *uci) {
215 if (uci == NULL) {
216 return false;
217 }
218 char from[3] = {0};
219 char to[3] = {0};
220 if (!opening_parse_uci(uci, from, to)) {
221 return false;
222 }
223 uint8_t from_row = 0, from_col = 0, to_row = 0, to_col = 0;
224 if (!game_notation_to_coords(from, &from_row, &from_col) ||
225 !game_notation_to_coords(to, &to_row, &to_col)) {
226 return false;
227 }
228 piece_t piece = game_get_piece(from_row, from_col);
229 if (piece == PIECE_EMPTY) {
230 return false;
231 }
232 chess_move_t move = {
233 .from_row = from_row,
234 .from_col = from_col,
235 .to_row = to_row,
236 .to_col = to_col,
237 .piece = piece,
238 .captured_piece = game_get_piece(to_row, to_col),
239 .timestamp = esp_timer_get_time() / 1000,
240 };
241 if (game_is_valid_move(&move) != MOVE_ERROR_NONE) {
242 return false;
243 }
244 return game_execute_move(&move);
245}

261static bool opening_begin_opponent_physical_turn(void) {
262 if (opening_state.ply_index >= opening_state.line_uci_count) {
263 return false;
264 }
265 const char *uci = opening_state.line_uci[opening_state.ply_index];
266 opening_parse_uci(uci, opening_state.expected_from, opening_state.expected_to);
267 strncpy(opening_state.last_opponent_uci, uci,
268 sizeof(opening_state.last_opponent_uci) - 1);
269 opening_state.awaiting_opponent_physical = true;
270 opening_state.feedback = OPENING_FEEDBACK_OPPONENT_TURN;
271 opening_show_opponent_physical_led(false);
272 STAGING_LOGI(TAG, "opponent physical ply=%u uci=%s", (unsigned)opening_state.ply_index,
273 uci);
274 return true;

277static void opening_run_until_player_turn(void) {
278 while (opening_state.active &&
279 opening_state.ply_index < opening_state.line_uci_count) {
280 if (opening_is_player_ply(opening_state.ply_index)) {
281 opening_prepare_player_turn();
282 return;
283 }
284
285 if (opening_state.opponent_mode == OPENING_OPPONENT_PHYSICAL) {
286 opening_begin_opponent_physical_turn();
287 return;
288 }
289
290 const char *uci = opening_state.line_uci[opening_state.ply_index];
291 char from[3], to[3];
292 opening_parse_uci(uci, from, to);
293 if (game_opening_apply_uci(uci)) {
294 strncpy(opening_state.last_opponent_uci, uci,
295 sizeof(opening_state.last_opponent_uci) - 1);
296 opening_play_move_led(from, to);
297 STAGING_LOGI(TAG, "virtual reply ply=%u uci=%s", (unsigned)opening_state.ply_index,
298 uci);
299 } else {
300 ESP_LOGE(TAG, "virtual uci failed: %s", uci);
301 break;
302 }
303 opening_state.ply_index++;
304 if (opening_should_checkpoint_after_ply(opening_state.ply_index - 1)) {
305 opening_state.awaiting_checkpoint_ack = true;
306 opening_state.feedback = OPENING_FEEDBACK_CHECKPOINT;
307 return;
308 }
309 }
310
311 if (opening_state.ply_index >= opening_state.line_uci_count) {
312 opening_state.active = false;
313 opening_state.feedback = OPENING_FEEDBACK_COMPLETE;
314 game_led_flush();
315 game_event_post_game_end(0, 0);
316 STAGING_LOGI(TAG, "line complete id=%s", opening_state.line_id);
317 }
318}

366bool game_opening_validate_checkpoint_physical(void) {
367 uint8_t matrix[64] = {0};
368 squares_get_state(matrix);
369 for (uint8_t i = 0; i < 64; i++) {
370 uint8_t row = i / 8;

380bool game_opening_load_config(const char *line_id, const char *start_fen,
381 const char line_uci[][6], uint8_t line_uci_count,
382 const uint8_t *player_ply_indices,
383 uint8_t player_ply_count,
384 const uint8_t *checkpoint_ply_indices,
385 uint8_t checkpoint_count, uint8_t mode,
386 uint8_t player_side_white,
387 uint8_t opponent_mode) {
388 if (line_id == NULL || start_fen == NULL || line_uci == NULL ||
389 line_uci_count == 0 || line_uci_count > OPENING_MAX_PLIES ||
390 player_ply_count == 0 || player_ply_count > OPENING_MAX_PLAYER_PLIES) {
391 return false;
392 }
393 game_opening_cancel();
394 memset(&opening_state, 0, sizeof(opening_state));
395 strncpy(opening_state.line_id, line_id, sizeof(opening_state.line_id) - 1);
396 strncpy(opening_state.start_fen, start_fen, sizeof(opening_state.start_fen) - 1);
397 opening_state.line_uci_count = line_uci_count;
398 for (uint8_t i = 0; i < line_uci_count; i++) {
399 strncpy(opening_state.line_uci[i], line_uci[i], sizeof(opening_state.line_uci[i]) - 1);
400 }
401 opening_state.player_ply_count = player_ply_count;
402 for (uint8_t i = 0; i < player_ply_count; i++) {
403 opening_state.player_ply_indices[i] = player_ply_indices[i];
404 }
405 opening_state.checkpoint_count =
406 checkpoint_count > OPENING_MAX_CHECKPOINTS ? OPENING_MAX_CHECKPOINTS : checkpoint_count;
407 for (uint8_t i = 0; i < opening_state.checkpoint_count; i++) {
408 opening_state.checkpoint_ply_indices[i] = checkpoint_ply_indices[i];
409 }
410 opening_state.mode = (opening_mode_t)mode;
411 opening_state.opponent_mode =
412 opponent_mode == 1 ? OPENING_OPPONENT_PHYSICAL : OPENING_OPPONENT_VIRTUAL;
413 opening_state.player_side = player_side_white ? PLAYER_WHITE : PLAYER_BLACK;
414 opening_state.feedback = OPENING_FEEDBACK_NONE;
415 opening_state.wrong_move_count = 0;
416 opening_state.last_wrong_uci[0] = '\0';
417 opening_state.awaiting_opponent_physical = false;
418 opening_state.setup_phase = false;
419 STAGING_LOGI(TAG, "load line=%s plies=%u opponent=%s", line_id,
420 (unsigned)line_uci_count,
421 opening_state.opponent_mode == OPENING_OPPONENT_PHYSICAL
422 ? "physical"
423 : "virtual");
424 return true;
425}

520 uint8_t to_row, uint8_t to_col) {

539 }
540 return (strcmp(from_notation, opening_state.expected_from) == 0 &&
541 strcmp(to_notation, opening_state.expected_to) == 0);

863void game_opening_service_hall_move(void) {
864#if CONFIG_CHESS_MATRIX_INPUT_I2C_HALL
865 static uint32_t s_since;
866 static bool s_armed;
867 uint8_t occ[64];
868 uint8_t fr = 0;
869 uint8_t fc = 0;
870 uint8_t tr = 0;
871 uint8_t tc = 0;
872 uint32_t now;
873 piece_t mover;
874 chess_move_t move;
875 bool castle;
876 bool ep;
877 uint8_t rook_from = 0;
878 uint8_t rook_to = 0;
879 uint8_t sq;
880 if (!opening_state.active || opening_state.awaiting_checkpoint_ack ||
881 opening_state.expected_from[0] == '\0' ||
882 !hall_i2c_squares_readings_usable()) {
883 s_armed = false;
884 return;
885 }
886 if (!game_notation_to_coords(opening_state.expected_from, &fr, &fc) ||
887 !game_notation_to_coords(opening_state.expected_to, &tr, &tc)) {
888 s_armed = false;
889 return;
890 }
891 if (!game_opening_validate_expected_move(fr, fc, tr, tc)) {
892 s_armed = false;
893 return;
894 }
895 mover = game_get_piece(fr, fc);
896 if (mover == PIECE_EMPTY) {
897 s_armed = false;
898 return;
899 }
900 castle = (mover == PIECE_WHITE_KING || mover == PIECE_BLACK_KING) && fr == tr &&
901 (fc > tc ? fc - tc : tc - fc) == 2;
902 ep = en_passant_available && tr == en_passant_target_row &&
903 tc == en_passant_target_col &&
904 (mover == PIECE_WHITE_PAWN || mover == PIECE_BLACK_PAWN);
905 if (castle) {
906 rook_from = (uint8_t)(fr * 8 + (tc > fc ? 7 : 0));
907 rook_to = (uint8_t)(fr * 8 + (tc > fc ? 5 : 3));
908 }
909 squares_get_state(occ);
910 for (sq = 0; sq < 64; sq++) {
911 bool want = game_get_piece(sq / 8, sq % 8) != PIECE_EMPTY;
912 if (sq == (uint8_t)(fr * 8 + fc)) {
913 want = false;
914 }
915 if (sq == (uint8_t)(tr * 8 + tc)) {
916 want = true;
917 }
918 if (castle && sq == rook_from) {
919 want = false;
920 }
921 if (castle && sq == rook_to) {
922 want = true;
923 }
924 if (ep &&
925 sq == (uint8_t)(en_passant_victim_row * 8 + en_passant_victim_col)) {
926 want = false;
927 }
928 if ((occ[sq] != 0) != want) {
929 s_armed = false;
930 return;
931 }
932 }
933 now = (uint32_t)(esp_timer_get_time() / 1000);
934 if (!s_armed) {
935 s_armed = true;
936 s_since = now;
937 return;
938 }
939 if ((uint32_t)(now - s_since) < 150U) {
940 return;
941 }
942 s_armed = false;
943 memset(&move, 0, sizeof(move));
944 move.from_row = fr;
945 move.from_col = fc;
946 move.to_row = tr;
947 move.to_col = tc;
948 move.piece = mover;
949 move.captured_piece = game_get_piece(tr, tc);
950 {
951 board_machine_out_t bout = game_board_on_drop_squares(fr, fc, tr, tc, true);
952 if ((bout.kind != BM_OUT_MOVE && bout.kind != BM_OUT_NEED_PROMO) ||
953 !game_board_apply_machine_move(&bout, &move)) {
954 return;
955 }
956 }
957 if (castle) {
958 board_machine_reset(game_board_machine());
959 }
960 game_cmd_reply("Opening move accepted.", false, NULL);
961#endif
962}
