127static int try_castle(position_t *p, int fr, int fc, int tr, int tc) {
128 int white = p->side == 0;
129 int home = white ? 0 : 7;
130 int kingside = tc == 6;
131 int rook_from = kingside ? 7 : 0;
132 int rook_to = kingside ? 5 : 3;
133 uint8_t need = 0;
134 uint8_t rook = white ? POS_WHITE_ROOK : POS_BLACK_ROOK;
135 uint8_t king = white ? POS_WHITE_KING : POS_BLACK_KING;
136 int by_white = white ? 0 : 1;
137 int pass_col = kingside ? 5 : 3;
138
139 if (fr != home || tr != home || fc != 4) {
140 return 0;
141 }
142 if (!kingside && tc != 2) {
143 return 0;
144 }
145 need = white ? (kingside ? POS_CASTLE_WK : POS_CASTLE_WQ)
146 : (kingside ? POS_CASTLE_BK : POS_CASTLE_BQ);
147 if ((p->castling & need) == 0) {
148 return 0;
149 }
150 if (p->board[home][fc] != king || p->board[home][rook_from] != rook) {
151 return 0;
152 }
153 if (kingside) {
154 if (p->board[home][5] != POS_EMPTY || p->board[home][6] != POS_EMPTY) {
155 return 0;
156 }
157 } else if (p->board[home][1] != POS_EMPTY || p->board[home][2] != POS_EMPTY ||
158 p->board[home][3] != POS_EMPTY) {
159 return 0;
160 }
161 if (position_sq_attacked(p->board, home, 4, by_white) ||
162 position_sq_attacked(p->board, home, pass_col, by_white) ||
163 position_sq_attacked(p->board, home, tc, by_white)) {
164 return 0;
165 }
166
167 p->board[home][fc] = POS_EMPTY;
168 p->board[home][rook_from] = POS_EMPTY;
169 p->board[home][tc] = king;
170 p->board[home][rook_to] = rook;
171 if (white) {
172 p->castling &= (uint8_t)~(POS_CASTLE_WK | POS_CASTLE_WQ);
173 } else {
174 p->castling &= (uint8_t)~(POS_CASTLE_BK | POS_CASTLE_BQ);
175 }
176 p->ep_square = POSITION_SQUARE_NONE;
177 if (p->halfmove < 65535) {
178 p->halfmove++;
179 }
180 switch_side(p);
181 return 1;
182}

184static int slider_ok(const position_t *p, int fr, int fc, int tr, int tc, int bishop,
185 int rook) {
186 int dr = tr - fr;
187 int dc = tc - fc;
188 int step_r = 0;
189 int step_c = 0;
190 int row;
191 int col;
192
193 if (dr != 0 && dc != 0 && (dr > 0 ? dr : -dr) == (dc > 0 ? dc : -dc)) {
194 if (!bishop) {
195 return 0;
196 }
197 } else if (dr == 0 || dc == 0) {
198 if (!rook) {
199 return 0;
200 }
201 } else {
202 return 0;
203 }
204 if (dr != 0) {
205 step_r = dr > 0 ? 1 : -1;
206 }
207 if (dc != 0) {
208 step_c = dc > 0 ? 1 : -1;
209 }
210 row = fr + step_r;
211 col = fc + step_c;
212 while (row != tr || col != tc) {
213 if (p->board[row][col] != POS_EMPTY) {
214 return 0;
215 }
216 row += step_r;
217 col += step_c;
218 }
219 return 1;

436 p->board[tr][tc] = POS_EMPTY;
437 p->board[fr][fc] = (h.side_before == 0) ? POS_WHITE_PAWN : POS_BLACK_PAWN;
438 if (vr >= 0 && vr < 8) {
439 p->board[vr][tc] = cap;
440 }
441 } else {
442 if (h.mv.promo != POS_PROMO_NONE) {
443 moved = (h.side_before == 0) ? POS_WHITE_PAWN : POS_BLACK_PAWN;
444 } else if ((moved == POS_WHITE_PAWN && tr == 7) || (moved == POS_BLACK_PAWN && tr == 0)) {
445 moved = (h.side_before == 0) ? POS_WHITE_PAWN : POS_BLACK_PAWN;
446 }
447 p->board[tr][tc] = cap;
448 p->board[fr][fc] = moved;
449 }

452 p->ep_square = h.ep_before;
453 p->side = h.side_before;
454 p->halfmove = h.halfmove_before;
455 p->fullmove = h.fullmove_before;
456 return true;
457}

535static uint32_t pos_hash(const position_t *p) {
536 uint32_t hash = 0;
537 int row;
538 int col;
539
540 for (row = 0; row < 8; row++) {
541 for (col = 0; col < 8; col++) {
542 uint8_t piece = p->board[row][col];
543 if (piece != POS_EMPTY) {
544 uint32_t piece_hash = ((uint32_t)piece << 16) | ((uint32_t)row << 8) | (uint32_t)col;
545 hash = ((hash << 5) + hash) ^ piece_hash;
546 }
547 }
548 }
549 hash = ((hash << 5) + hash) ^ ((uint32_t)p->side << 24);
550 hash = ((hash << 5) + hash) ^ p->castling;
551 if (p->ep_square < 64) {
552 hash = ((hash << 5) + hash) ^ p->ep_square;
553 }
554 return hash;
555}
