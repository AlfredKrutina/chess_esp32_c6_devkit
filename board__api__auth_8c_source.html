34 uint8_t d = 0;

41static int hex_nibble(char c) {
42 if (c >= '0' && c <= '9') {
43 return (int)(c - '0');
44 }
45 if (c >= 'a' && c <= 'f') {
46 return (int)(c - 'a') + 10;
47 }
48 if (c >= 'A' && c <= 'F') {
49 return (int)(c - 'A') + 10;
50 }
51 return -1;
52}

54static bool parse_hex32(const char *hex, uint8_t *out32) {
55 if (hex == NULL) {
56 return false;
57 }
58 size_t len = strlen(hex);
59 if (len != 64) {
60 return false;
61 }
62 for (size_t i = 0; i < 32; i++) {
63 int hi = hex_nibble(hex[i * 2]);
64 int lo = hex_nibble(hex[i * 2 + 1]);
65 if (hi < 0 || lo < 0) {
66 return false;
67 }
68 out32[i] = (uint8_t)((unsigned)hi << 4 | (unsigned)lo);
69 }
70 return true;
71}

88 nvs_close(h);
89 return e;
90 }
91 e = nvs_commit(h);
92 if (e != ESP_OK) {
93 ESP_LOGE(TAG, "nvs_commit: %s", esp_err_to_name(e));
94 nvs_close(h);
95 return e;
96 }
97#if CHESS_DEBUG_MODE
98 ESP_LOGW(TAG, "[STAGING] Generated new board API token (use UART API_TOKEN)");
99#endif
100 } else if (e != ESP_OK) {
101 ESP_LOGE(TAG, "nvs_get_blob token: %s", esp_err_to_name(e));
102 nvs_close(h);
103 return e;
104 }
105
106 nvs_close(h);
107 s_loaded = true;
108 (void)board_pair_slots_init();

301static esp_err_t board_pair_slots_init(void) {
302 board_pair_book_init(&s_pair_book);
303 if (s_pair_mu == NULL) {
304 s_pair_mu = xSemaphoreCreateMutex();
305 if (s_pair_mu == NULL) {
306 return ESP_ERR_NO_MEM;
307 }
308 }
309 nvs_handle_t h;
310 esp_err_t e = nvs_open(PAIR_NVS_NS, NVS_READONLY, &h);
311 if (e == ESP_OK) {
312 for (int i = 0; i < BOARD_PAIR_SLOT_COUNT; i++) {
313 char key[4];
314 snprintf(key, sizeof(key), "s%d", i);
315 uint8_t blob[44];
316 size_t sz = sizeof(blob);
317 if (nvs_get_blob(h, key, blob, &sz) == ESP_OK && sz == sizeof(blob)) {
318 slot_load(&s_pair_book.slots[i], blob);
319 if (s_pair_book.slots[i].seq > s_pair_book.next_seq) {
320 s_pair_book.next_seq = s_pair_book.slots[i].seq;
321 }
322 }
323 }
324 uint32_t seq = 0;
325 if (nvs_get_u32(h, "pseq", &seq) == ESP_OK && seq > s_pair_book.next_seq) {
326 s_pair_book.next_seq = seq;
327 }
328 nvs_close(h);
329 }
330 s_pair_ready = true;
331 return ESP_OK;
332}

414bool board_pair_http_command_denied(httpd_req_t *req) {
415 uint8_t token[BOARD_PAIR_TOKEN_LEN];
416 bool have = read_bearer(req, token);
417 if (!s_pair_ready || !pair_lock()) {
418 send_auth_error(req, "api_token_required",
419 "Board command token required.");
420 return true;
421 }
422 expire_wifi_lease_locked();
423 int verdict = have ? board_pair_book_command(&s_pair_book, token, true)
424 : BOARD_PAIR_CMD_NEED_TOKEN;
425 if (verdict == BOARD_PAIR_CMD_ALLOW) {
426 arm_wifi_lease_locked();
427#if CHESS_DEBUG_MODE
428 ESP_LOGI(TAG, "[STAGING] command token accepted (wifi seat %d)",
429 s_pair_book.wifi_slot);
430#endif
431 }
432 pair_unlock();
433 if (verdict == BOARD_PAIR_CMD_ALLOW) {
434 return false;
435 }
436 if (verdict == BOARD_PAIR_CMD_BUSY) {
437#if CHESS_DEBUG_MODE
438 ESP_LOGW(TAG, "[STAGING] command refused: board busy");
439#endif
440 send_auth_error(req, "board_busy",
441 "Board is connected to another device.");
442 return true;
443 }
444#if CHESS_DEBUG_MODE
445 ESP_LOGW(TAG, "[STAGING] command refused: token required");
446#endif
447 send_auth_error(req, "api_token_required", "Board command token required.");
448 return true;
449}

492esp_err_t board_pair_http_handler(httpd_req_t *req) {
493 /* Wi-Fi hosts have no GATT. The phone path is the cmd_ack notify
494 * {"code":"paired","api_token":"..."} after SMP. This POST is the same
495 * token, and only while the 60 s window is still open. The first success
496 * claims the window so a second device does not get another token. */
497 if (!ble_task_claim_pairing_success()) {
498 send_auth_error(req, "not_pairing",
499 "Board is not in pairing mode.");
500 return ESP_OK;
501 }
502 char hex[65];
503 esp_err_t e = board_pair_issue_wifi(hex, sizeof(hex));
504 ble_task_finish_pairing_window();
505 if (e != ESP_OK) {
506 httpd_resp_set_status(req, "500 Internal Server Error");
507 httpd_resp_set_type(req, "application/json");
508 (void)httpd_resp_send(req, "{\"success\":false,\"error\":\"token\"}",
509 HTTPD_RESP_USE_STRLEN);
510 return ESP_OK;
511 }
512#if CHESS_DEBUG_MODE
513 ESP_LOGI(TAG, "[STAGING] Wi-Fi pair token issued, window closed");
514#endif
515 char body[128];
516 snprintf(body, sizeof(body),
517 "{\"success\":true,\"api_token\":\"%s\"}", hex);
518 httpd_resp_set_type(req, "application/json");
519 (void)httpd_resp_send(req, body, HTTPD_RESP_USE_STRLEN);
520 return ESP_OK;
521}
