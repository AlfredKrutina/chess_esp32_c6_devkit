144static void timeout_abandon(void) {
145 ESP_LOGW(TAG, "mandatory step idle timeout — entering restore");
146 if (capture_in_progress) {
147 capture_in_progress = false;
148 capture_target_row = 0;
149 capture_target_col = 0;
150 capture_removed_piece = PIECE_EMPTY;
151 }
152 if (guided_capture_state.active) {
153 game_reset_guided_capture_state();
154 }
155 if (castling_state.in_progress) {
156 memset(&castling_state, 0, sizeof(castling_state));
157 }
158 if (game_ep_capture_waiting()) {
159 game_ep_capture_reset();
160 }
161 game_board_set_piece_lifted(false, 0, 0, PIECE_EMPTY);
162 game_required_step_clear();
163#if CONFIG_CHESS_MATRIX_INPUT_I2C_HALL
164 /* In-play plan against the committed board. Lamp hold returns false;
165 * yield on release redraws later. Do not clear s_opening_done. */
166 (void)game_hall_restore_guide_show_play();
167#endif
168 s_timeout_armed = false;
169}

171void game_required_step_service(void) {
172 if (!game_required_step_blocks_restore()) {
173 if (s_timeout_armed) {
174 game_required_step_clear();
175 }
176 return;
177 }
178
179 uint32_t now_ms = (uint32_t)(esp_timer_get_time() / 1000);
180 if (s_timeout_armed && (*game_state_promotion_state()).pending) {
181 /* Promotion choice may take longer; do not drop the pending latch. */
182 } else if (s_timeout_armed &&
183 (uint32_t)(now_ms - s_step_since_ms) >= REQUIRED_STEP_IDLE_MS) {
184 timeout_abandon();
185 return;
186 }
187
188 if (s_last_repost_ms == 0 ||
189 (uint32_t)(now_ms - s_last_repost_ms) >= REQUIRED_STEP_REPOST_MS) {
190 s_last_repost_ms = now_ms;
191 game_required_step_refresh();
192 }
193}
