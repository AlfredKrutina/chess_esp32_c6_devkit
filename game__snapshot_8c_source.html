178static uint32_t game_crc32_simple(const uint8_t *data, size_t len) {
179 uint32_t h = 2166136261u;
180 if (data == NULL) {
181 return h;
182 }
183 for (size_t i = 0; i < len; i++) {
184 h ^= data[i];
185 h *= 16777619u;
186 }
187 return h;
188}

474static esp_err_t game_save_snapshot_to_nvs(void) {
475 /* Static: three v2 blobs do not fit on the 6 KB game task stack. */
476 static game_snapshot_v2_t slot_a;
477 static game_snapshot_v2_t slot_b;
478 static game_snapshot_v2_t snap;
479 bool have_a =
480 game_snapshot_load_v2_key(CONFIG_NVS_KEY_GAME_SNAPSHOT_FULL, &slot_a);
481 bool have_b = game_snapshot_load_v2_key(CONFIG_NVS_KEY_GAME_SNAPSHOT_B, &slot_b);
482 uint32_t best_seq = 0;
483 int best_slot = 0;
484 if (have_a && slot_a.seq >= best_seq) {
485 best_seq = slot_a.seq;
486 best_slot = 1;
487 }
488 if (have_b && slot_b.seq >= best_seq) {
489 best_seq = slot_b.seq;
490 best_slot = 2;
491 }
492
493 game_snapshot_fill_v2(&snap, best_seq + 1u);
494 /* Write the slot that is not the current good copy. A cut mid-write leaves it. */
495 const char *key = (best_slot == 1) ? CONFIG_NVS_KEY_GAME_SNAPSHOT_B
496 : CONFIG_NVS_KEY_GAME_SNAPSHOT_FULL;
497 esp_err_t ret = config_save_blob_to_nvs(key, &snap, sizeof(snap));
498 if (ret == ESP_OK) {
499 snapshot_fallback_used = false;
500 snapshot_save_failed = false;
501 STAGING_LOGI(TAG, "snapshot v2 seq=%lu -> %s", (unsigned long)snap.seq, key);
502 return ESP_OK;
503 }
504
505 game_snapshot_min_t min = {0};
506 min.version = GAME_SNAPSHOT_VERSION;
507 min.format = 1;
508 memcpy(min.board, snap.board, sizeof(min.board));
509 min.current_player = snap.current_player;
510 min.game_state = snap.game_state;
511 min.move_count = snap.move_count;
512 min.white_king_moved = snap.white_king_moved;
513 min.white_rook_a_moved = snap.white_rook_a_moved;
514 min.white_rook_h_moved = snap.white_rook_h_moved;
515 min.black_king_moved = snap.black_king_moved;
516 min.black_rook_a_moved = snap.black_rook_a_moved;

684static bool game_boot_tracker_should_force_new_game(void) {
685 time_t now = time(NULL);
686 const bool wall_time_ok = (now >= 1700000000);
687 const uint32_t now_s = wall_time_ok ? (uint32_t)now : 0u;
688
689 game_boot_tracker_t trk = {0};
690 size_t len = sizeof(trk);
691 bool loaded =
692 (config_load_blob_from_nvs(CONFIG_NVS_KEY_BOOT_TRACKER, &trk, &len) == ESP_OK &&
693 len == sizeof(trk) && trk.version == GAME_SNAPSHOT_VERSION &&
694 trk.crc32 == game_crc32_simple(((const uint8_t *)&trk) + 8, sizeof(trk) - 8));
695
696 bool in_window = false;
697 if (loaded) {
698 if (wall_time_ok) {
699 in_window = (now_s >= trk.first_boot_epoch_s &&
700 (now_s - trk.first_boot_epoch_s) <= BOOT_WINDOW_SECONDS);
701 } else {
702 in_window = true;
703 }
704 }
705
706 bool force_new_game = false;
707 if (wall_time_ok && loaded && trk.moved_since_boot == 0u && in_window &&
708 trk.boot_counter_window == 1u) {
709 force_new_game = true;
710 }
711
712 game_boot_tracker_t next = {0};
713 next.version = GAME_SNAPSHOT_VERSION;
714 next.moved_since_boot = 0u;
715 if (loaded && in_window) {
716 next.first_boot_epoch_s = trk.first_boot_epoch_s;
717 next.boot_counter_window = trk.boot_counter_window + 1u;
718 } else if (wall_time_ok) {
719 next.first_boot_epoch_s = now_s;
720 next.boot_counter_window = 1u;
721 } else {
722 next.first_boot_epoch_s = loaded ? trk.first_boot_epoch_s : 0u;
723 next.boot_counter_window = 1u;
724 }
725 next.crc32 = game_crc32_simple(((const uint8_t *)&next) + 8, sizeof(next) - 8);
726 (void)config_save_blob_to_nvs(CONFIG_NVS_KEY_BOOT_TRACKER, &next, sizeof(next));
727
728 return force_new_game;
729}
