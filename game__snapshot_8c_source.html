314static uint32_t game_crc32_simple(const uint8_t *data, size_t len) {
315 uint32_t h = 2166136261u;
316 if (data == NULL) {
317 return h;
318 }
319 for (size_t i = 0; i < len; i++) {
320 h ^= data[i];

635 /* Static: three v2 blobs do not fit on the 6 KB game task stack. */
636 static game_snapshot_v2_t slot_a;

638 static game_snapshot_v2_t snap;

662 return ESP_OK;

819static bool game_boot_tracker_should_force_new_game(void) {
820 time_t now = time(NULL);
821 const bool wall_time_ok = (now >= 1700000000);
822 const uint32_t now_s = wall_time_ok ? (uint32_t)now : 0u;
823
824 game_boot_tracker_t trk = {0};
825 size_t len = sizeof(trk);

827 (config_load_blob_from_nvs(CONFIG_NVS_KEY_BOOT_TRACKER, &trk, &len) == ESP_OK &&
828 len == sizeof(trk) && trk.version == GAME_SNAPSHOT_VERSION &&

830
831 bool in_window = false;
832 if (loaded) {

834 in_window = (now_s >= trk.first_boot_epoch_s &&
835 (now_s - trk.first_boot_epoch_s) <= BOOT_WINDOW_SECONDS);
836 } else {
837 in_window = true;
838 }
839 }
840
841 bool force_new_game = false;
842 if (wall_time_ok && loaded && trk.moved_since_boot == 0u && in_window &&
843 trk.boot_counter_window == 1u) {
844 force_new_game = true;
845 }
846
847 game_boot_tracker_t next = {0};
848 next.version = GAME_SNAPSHOT_VERSION;
849 next.moved_since_boot = 0u;
850 if (loaded && in_window) {
851 next.first_boot_epoch_s = trk.first_boot_epoch_s;
852 next.boot_counter_window = trk.boot_counter_window + 1u;
853 } else if (wall_time_ok) {
854 next.first_boot_epoch_s = now_s;
855 next.boot_counter_window = 1u;
856 } else {
857 next.first_boot_epoch_s = loaded ? trk.first_boot_epoch_s : 0u;
858 next.boot_counter_window = 1u;

861 (void)config_save_blob_to_nvs(CONFIG_NVS_KEY_BOOT_TRACKER, &next, sizeof(next));
862
863 return force_new_game;
864}
