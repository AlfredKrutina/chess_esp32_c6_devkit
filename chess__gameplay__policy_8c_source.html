203void chess_policy_refresh_idle_board_leds(void) {
204#if CONFIG_CHESS_MATRIX_INPUT_I2C_HALL
205 if (game_hall_restore_guide_is_active()) {
206 return;
207 }
208 if (game_hall_position_blocks_idle_leds()) {
209 return;
210 }
211#endif
212 if (piece_lifted || castling_state.in_progress ||
213 promotion_state.pending || game_ep_capture_waiting() ||
214 game_is_castle_animation_active() || game_is_castling_expected() ||
215 game_puzzle_suppress_move_hints()) {
216 return;
217 }
218 if (current_game_state != GAME_STATE_ACTIVE &&
219 current_game_state != GAME_STATE_IDLE) {
220 return;
221 }
222
223 led_target_begin_fade(HINT_FADE_MS);
224 chess_policy_highlight_movable_if_enabled();
225
226 if (!game_is_king_in_check(current_player)) {
227 led_target_end();
228 return;
229 }
230 int king_row = -1;
231 int king_col = -1;
232 piece_t king_piece = (current_player == PLAYER_WHITE) ? PIECE_WHITE_KING
233 : PIECE_BLACK_KING;
234 for (int row = 0; row < 8; row++) {
235 for (int col = 0; col < 8; col++) {
236 if (board[row][col] == king_piece) {
237 king_row = row;
238 king_col = col;
239 break;
240 }
241 }
242 if (king_row != -1) {
243 break;
244 }
245 }
246 if (king_row == -1 || king_col == -1 ||
247 !game_led_guidance_show_check_anim()) {
248 led_target_end();
249 return;
250 }
251 led_target_set(chess_pos_to_led_index((uint8_t)king_row, (uint8_t)king_col),
252 255, 192, 203, LED_VIS_BREATHE);
253 led_target_end();
254}
