37static bool demo_is_castling_move(uint8_t from_row, uint8_t from_col,
38 uint8_t to_row, uint8_t to_col,
39 piece_t board[8][8]) {
40 // Must be on same rank
41 if (from_row != to_row) {
42 return false;
43 }
44
45 // Check if piece is a king
46 piece_t piece = board[from_row][from_col];
47 if (piece != PIECE_WHITE_KING && piece != PIECE_BLACK_KING) {
48 return false;
49 }
50
51 // Check if king moves 2 squares (castling signature)
52 int col_diff = abs((int)to_col - (int)from_col);
53 return (col_diff == 2);
54}

101static bool demo_is_promotion_move(uint8_t from_row, uint8_t from_col,
102 uint8_t to_row, piece_t board[8][8]) {
103 piece_t piece = board[from_row][from_col];
104
105 // Check if piece is a pawn
106 if (piece != PIECE_WHITE_PAWN && piece != PIECE_BLACK_PAWN) {
107 return false;
108 }
109
110 // White pawn reaching rank 8 (row 7)
111 if (piece == PIECE_WHITE_PAWN && to_row == 7) {
112 return true;
113 }
114
115 // Black pawn reaching rank 1 (row 0)
116 if (piece == PIECE_BLACK_PAWN && to_row == 0) {
117 return true;
118 }
119
120 return false;
121}
