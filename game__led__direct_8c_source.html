39void game_coach_hint_paint(void) {
40 uint8_t row;
41 uint8_t col;
42 piece_t dest;
43 bool opponent;
44
45 if (!s_coach_on || !s_frame_open) {
46 return;
47 }
48 if (s_coach_from < 64) {
49 game_led_frame_sq((uint8_t)(s_coach_from / 8u), (uint8_t)(s_coach_from % 8u),
50 255, 255, 0);
51 }
52 if (s_coach_to >= 64) {
53 return;
54 }
55 row = (uint8_t)(s_coach_to / 8u);
56 col = (uint8_t)(s_coach_to % 8u);
57 dest = game_get_piece(row, col);
58 opponent = (current_player == PLAYER_WHITE && dest >= PIECE_BLACK_PAWN &&
59 dest <= PIECE_BLACK_KING) ||
60 (current_player == PLAYER_BLACK && dest >= PIECE_WHITE_PAWN &&
61 dest <= PIECE_WHITE_KING);
62 if (opponent) {
63 game_led_frame_sq(row, col, 255, 165, 0);
64 } else {
65 game_led_frame_sq(row, col, 0, 255, 0);
66 }
67}

90void game_led_frame_sq(uint8_t row, uint8_t col, uint8_t r, uint8_t g,
91 uint8_t b) {
92 if (row >= 8 || col >= 8) {
93 return;
94 }
95 if (!s_frame_open) {
96 if (s_acc_n >= 64) {
97 game_led_flush();
98 }
99 s_acc[s_acc_n].square = (uint8_t)(row * 8u + col);
100 s_acc[s_acc_n].r = r;
101 s_acc[s_acc_n].g = g;
102 s_acc[s_acc_n].b = b;
103 s_acc_n++;
104 return;
105 }
106 if (s_frame_n >= 64) {
107 return;
108 }
109 s_frame_px[s_frame_n].square = (uint8_t)(row * 8u + col);
110 s_frame_px[s_frame_n].r = r;
111 s_frame_px[s_frame_n].g = g;
112 s_frame_px[s_frame_n].b = b;
113 s_frame_n++;

116void game_led_frame_breathe(uint8_t row, uint8_t col, uint8_t r, uint8_t g,

136 return;

223void game_process_web_hint(const game_cmd_t *cmd) {
224 if (cmd == NULL) {
225 return;
226 }
227 if (cmd->u.hint.clear) {
228 game_coach_hint_clear();
229 } else if (cmd->u.hint.from_sq >= 64 && cmd->u.hint.to_sq >= 64) {
230 game_coach_hint_clear();
231 } else {
232 s_coach_on = true;
233 s_coach_from = cmd->u.hint.from_sq;
234 s_coach_to = cmd->u.hint.to_sq;
235 }
236 if (game_is_puzzle_active() && !game_is_puzzle_setup_active() &&
237 !game_puzzle_sync_hold()) {
238 game_puzzle_refresh_play_leds();
239 return;
240 }
241 if (game_is_puzzle_setup_active()) {
242 return;
243 }
244 game_policy_refresh_idle_board_leds();
245}
